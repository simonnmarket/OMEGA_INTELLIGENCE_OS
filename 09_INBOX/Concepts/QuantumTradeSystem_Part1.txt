        handleADX = iADX(Symbol(), Period(), InpADXPeriod);

        handleATR = iATR(Symbol(), Period(), 14);

        handleRSI = iRSI(Symbol(), Period(), 14, PRICE_CLOSE);

    MARKET_REGIME DetermineMarketRegime() {
        double adx[], atr[], bb_upper[], bb_lower[];
        ArraySetAsSeries(adx, true);
        ArraySetAsSeries(atr, true);
        ArraySetAsSeries(bb_upper, true);
        ArraySetAsSeries(bb_lower, true);
        
        CopyBuffer(handleADX, 0, 0, 2, adx);
        CopyBuffer(handleATR, 0, 0, 2, atr);
        CopyBuffer(handleBB, 1, 0, 1, bb_upper);
        CopyBuffer(handleBB, 2, 0, 1, bb_lower);
        
        double trend_strength = adx[0];
        double volatility = atr[0] / atr[1];
        double range_width = (bb_upper[0] - bb_lower[0]) / Point();
        
        if(trend_strength > 30) return TRENDING_STRONG;
        if(trend_strength > 20) return TRENDING_WEAK;
        if(volatility > 2.0) return VOLATILE_BULLISH;
        if(volatility > 1.5) return VOLATILE_BEARISH;
        if(range_width < 50) return RANGING_NARROW;
        return RANGING_WIDE;
    }
