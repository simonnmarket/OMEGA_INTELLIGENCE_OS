@staticmethod
def _calculate_runtime_checksum() -> str:
    """Calculate SHA3-256 checksum of runtime environment"""
    runtime_info = f"{sys.version}_{sys.platform}_{os.getpid()}_{datetime.now(timezone.utc).timestamp()}"
    return hashlib.sha3_256(runtime_info.encode()).hexdigest()[:16].upper()

        """Classe com encapsulamento."""
        
        def __init__(self, titular: str, saldo: float = 0.0):
            self._titular = titular  # Protegido
            self.__saldo = saldo     # Privado
        
        @property
        def titular(self) -> str:
            """Getter para titular."""
            return self._titular
        
        @titular.setter
        def titular(self, valor: str):
            """Setter para titular."""
            if not valor:
                raise ValueError("Titular não pode ser vazio")
            self._titular = valor
        
        @property
        def saldo(self) -> float:
            """Getter para saldo."""
            return self.__saldo
        
        def depositar(self, valor: float):
            """Deposita valor na conta."""
            if valor <= 0:
                raise ValueError("Valor deve ser positivo")
            self.__saldo += valor
        
        def sacar(self, valor: float) -> bool:
            """Saca valor da conta."""
            if 0 < valor <= self.__saldo:
                self.__saldo -= valor
                return True
            return False

    from abc import ABC, abstractmethod
    
    class FormaGeometrica(ABC):
        """Classe abstrata para formas geométricas."""

        def area(self) -> float:
            pass

        def perimetro(self) -> float:
            pass

            self.largura = largura
            self.altura = altura
        
        def area(self) -> float:
            return self.largura * self.altura
        
        def perimetro(self) -> float:
            return 2 * (self.largura + self.altura)

        def area(self) -> float:
            return math.pi * self.raio ** 2
        
        def perimetro(self) -> float:
            return 2 * math.pi * self.raio

def demonstrar_concorrencia(self):
    """Demonstrar programação concorrente"""
    print("\n" + "="*80)
    print("XI. PROGRAMAÇÃO CONCORRENTE E PARALELA")
    print("="*80)
    
    print("\n1. THREADING:")
    
    def tarefa_thread(nome: str, delay: float):
        """Tarefa executada em thread."""
        time.sleep(delay)
        print(f"  Thread {nome}: completada após {delay}s")
    
    threads = []
    for i in range(3):
        t = threading.Thread(target=tarefa_thread, args=(f"T{i}", 0.5 * (i+1)))
        threads.append(t)
        t.start()
    
    for t in threads:
        t.join()
    print("  Todas as threads completaram")
    
    print("\n2. MULTIPROCESSING:")
    
    def tarefa_processo(nome: str, numero: int):
        """Tarefa executada em processo."""
        resultado = numero * 2
        print(f"  Processo {nome}: {numero} * 2 = {resultado}")
        return resultado
    
    with multiprocessing.Pool(processes=3) as pool:
        resultados = pool.starmap(tarefa_processo, [("A", 1), ("B", 2), ("C", 3)])
    
    print(f"  Resultados: {resultados}")
    
    print("\n3. ASYNCIO:")
    
    async def tarefa_async(nome: str, delay: float):
        """Tarefa assíncrona."""
        await asyncio.sleep(delay)
        return f"{nome}:{delay}"
    
    async def main_async():
        """Função principal assíncrona."""
        tarefas = [
            tarefa_async("A", 0.3),
            tarefa_async("B", 0.1),
            tarefa_async("C", 0.2)
        ]
        
        # Executar concorrentemente
        resultados = await asyncio.gather(*tarefas)
        print(f"  Async resultados: {resultados}")
        
        # Executar com timeout
        try:
            resultado = await asyncio.wait_for(tarefa_async("D", 2.0), timeout=1.0)
        except asyncio.TimeoutError:
            print("  ❌ Timeout: Tarefa D demorou muito")
    
    asyncio.run(main_async())
    
    print("\n4. CONCURRENT.FUTURES:")
    
    def calcular_quadrado(n: int) -> int:
        """Calcula quadrado com delay simulado."""
        time.sleep(0.1)
        return n ** 2
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as executor:
        futuros = {executor.submit(calcular_quadrado, i): i for i in range(1, 6)}
        
        for futuro in concurrent.futures.as_completed(futuros):
            numero = futuros[futuro]
            try:
                resultado = futuro.result()
                print(f"  {numero}² = {resultado}")
            except Exception as e:
                print(f"  ❌ Erro para {numero}: {e}")
    
    print("\n5. LOCK E SINCRONIZAÇÃO:")
    
    class ContadorSeguro:
        """Contador thread-safe."""
        
        def __init__(self):
            self.valor = 0
            self.lock = threading.Lock()
        
        def incrementar(self):
            """Incrementa valor de forma segura."""
            with self.lock:
                self.valor += 1
                return self.valor
    
    contador = ContadorSeguro()
    
    def worker():
        for _ in range(1000):
            contador.incrementar()
    
    threads = []
    for _ in range(10):
        t = threading.Thread(target=worker)
        threads.append(t)
        t.start()
    
    for t in threads:
        t.join()
    
    print(f"  Contador final: {contador.valor} (esperado: 10000)")
    
    print("\n6. QUEUES PARA COMUNICAÇÃO:")
    
    def produtor(fila: multiprocessing.Queue, itens: List[str]):
        """Produtor que coloca itens na fila."""
        for item in itens:
            time.sleep(0.1)
            fila.put(item)
            print(f"  Produtor: enviou '{item}'")
        fila.put(None)  # Sinal de término
    
    def consumidor(fila: multiprocessing.Queue):
        """Consumidor que processa itens da fila."""
        while True:
            item = fila.get()
            if item is None:
                break
            print(f"  Consumidor: recebeu '{item}'")
            time.sleep(0.2)
    
    fila = multiprocessing.Queue()
    
    p = multiprocessing.Process(target=produtor, args=(fila, ["A", "B", "C", "D"]))
    c = multiprocessing.Process(target=consumidor, args=(fila,))
    
    p.start()
    c.start()
    
    p.join()
    c.join()
    
    print("  Produtor/Consumidor completaram")
