//| Classe para execução de ordens                                     |
//+------------------------------------------------------------------+
class CTradeExecutor
{
private:
    string m_symbol;      // Símbolo para operar
    int m_magic;         // Número mágico para identificar ordens
    CLogger* m_logger;   // Objeto para logging
    CTrade m_trade;      // Objeto para execução de trades
    
public:
    CTradeExecutor(CLogger* logger)
    {
        m_logger = logger;
    }
    
    bool Init(string symbol, int magic)
    {
        m_symbol = symbol;
        m_magic = magic;
        
        // Configura objeto de trade
        m_trade.SetExpertMagicNumber(magic);
        m_trade.SetMarginMode();
        m_trade.SetTypeFillingBySymbol(symbol);
        m_trade.SetDeviationInPoints(10);
        
        if(m_logger != NULL)
            m_logger.LogInfo("Trade executor inicializado");
        return true;
    }
    
    bool IsMarketOpen()
    {
        MqlDateTime dt;
        TimeCurrent(dt);
        
        // Verifica se é fim de semana
        if(dt.day_of_week == 0 || dt.day_of_week == 6)
        {
            return false;
        }
        
        // Verifica horário de mercado
        datetime time = TimeCurrent();
        int hour = dt.hour;
        
        // Mercado fechado entre 22:00 e 00:00
        if(hour >= 22)
        {
            return false;
        }
        
        return true;
    }
    
    bool OpenPosition(bool isLong, double volume, double stopLoss, double takeProfit)
    {
        // Verifica volume
        if(volume <= 0)
        {
            if(m_logger != NULL)
                m_logger.LogError("Volume inválido");
            return false;
        }
        
        // Obtém preços
        double price = isLong ? SymbolInfoDouble(m_symbol, SYMBOL_ASK) : SymbolInfoDouble(m_symbol, SYMBOL_BID);
        
        // Executa ordem
        bool result = isLong ?
            m_trade.Buy(volume, m_symbol, price, stopLoss, takeProfit) :
            m_trade.Sell(volume, m_symbol, price, stopLoss, takeProfit);
            
        if(!result)
        {
            if(m_logger != NULL)
                m_logger.LogError("Falha ao abrir posição: " + IntegerToString(m_trade.ResultRetcode()));
            return false;
        }
        
        string message = StringFormat("Posição aberta: %s, Volume: %.2f, SL: %.5f, TP: %.5f",
            isLong ? "COMPRA" : "VENDA", volume, stopLoss, takeProfit);
        if(m_logger != NULL)
            m_logger.LogInfo(message);
        
        return true;
    }
    
    bool ModifyPosition(ulong ticket, double stopLoss, double takeProfit)
    {
        // Seleciona posição
        if(!PositionSelectByTicket(ticket))
        {
            if(m_logger != NULL)
                m_logger.LogError("Posição não encontrada: " + IntegerToString(ticket));
            return false;
        }
        
        // Modifica posição
        if(!m_trade.PositionModify(ticket, stopLoss, takeProfit))
        {
            if(m_logger != NULL)
                m_logger.LogError("Falha ao modificar posição: " + IntegerToString(m_trade.ResultRetcode()));
            return false;
        }
        
        string message = StringFormat("Posição modificada: Ticket=%d, SL=%.5f, TP=%.5f",
            ticket, stopLoss, takeProfit);
        if(m_logger != NULL)
            m_logger.LogInfo(message);
        
        return true;
    }
    
    bool ClosePosition(ulong ticket)
    {
        // Seleciona posição
        if(!PositionSelectByTicket(ticket))
        {
            if(m_logger != NULL)
                m_logger.LogError("Posição não encontrada: " + IntegerToString(ticket));
            return false;
        }
        
        // Fecha posição
        if(!m_trade.PositionClose(ticket))
        {
            if(m_logger != NULL)
                m_logger.LogError("Falha ao fechar posição: " + IntegerToString(m_trade.ResultRetcode()));
            return false;
        }
        
        if(m_logger != NULL)
            m_logger.LogInfo("Posição fechada: " + IntegerToString(ticket));
        return true;
    }
};

//| Classe para análise de mercado                                     |

class CMarketAnalyzer
{
private:
    string m_symbol;      // Símbolo para análise
    int m_ma20Handle;    // Handle da média móvel de 20 períodos
    int m_ma50Handle;    // Handle da média móvel de 50 períodos
    int m_ma200Handle;   // Handle da média móvel de 200 períodos
    int m_rsiHandle;     // Handle do RSI
    int m_atrHandle;     // Handle do ATR
    
    // Buffers dos indicadores
    double m_maBuffer20[];   // Buffer da média móvel de 20 períodos
    double m_maBuffer50[];   // Buffer da média móvel de 50 períodos
    double m_maBuffer200[];  // Buffer da média móvel de 200 períodos
    double m_rsiBuffer[];    // Buffer do RSI
    double m_atrBuffer[];    // Buffer do ATR
    double m_volumeBuffer[]; // Buffer de volume
    
    CLogger* m_logger;   // Objeto para logging
    
public:
    CMarketAnalyzer(CLogger* logger)
    {
        m_logger = logger;
    }
    
    bool Init(string symbol)
    {
        m_symbol = symbol;
        
        // Inicializa handles dos indicadores
        m_ma20Handle = iMA(symbol, PERIOD_CURRENT, 20, 0, MODE_SMA, PRICE_CLOSE);
        m_ma50Handle = iMA(symbol, PERIOD_CURRENT, 50, 0, MODE_SMA, PRICE_CLOSE);
        m_ma200Handle = iMA(symbol, PERIOD_CURRENT, 200, 0, MODE_SMA, PRICE_CLOSE);
        m_rsiHandle = iRSI(symbol, PERIOD_CURRENT, 14, PRICE_CLOSE);
        m_atrHandle = iATR(symbol, PERIOD_CURRENT, 14);
        
        // Verifica se os handles foram criados corretamente
        if(m_ma20Handle == INVALID_HANDLE || 
           m_ma50Handle == INVALID_HANDLE || 
           m_ma200Handle == INVALID_HANDLE ||
           m_rsiHandle == INVALID_HANDLE ||
           m_atrHandle == INVALID_HANDLE)
        {
            if(m_logger != NULL)
                m_logger.LogError("Falha ao criar handles dos indicadores");
            return false;
        }
        
        // Inicializa buffers
        ArraySetAsSeries(m_maBuffer20, true);
        ArraySetAsSeries(m_maBuffer50, true);
        ArraySetAsSeries(m_maBuffer200, true);
        ArraySetAsSeries(m_rsiBuffer, true);
        ArraySetAsSeries(m_atrBuffer, true);
        ArraySetAsSeries(m_volumeBuffer, true);
        
        if(m_logger != NULL)
            m_logger.LogInfo("Market analyzer inicializado");
        return true;
    }
    
    void Update()
    {
        // Copia dados dos indicadores para os buffers
        if(CopyBuffer(m_ma20Handle, 0, 0, 1000, m_maBuffer20) <= 0 ||
           CopyBuffer(m_ma50Handle, 0, 0, 1000, m_maBuffer50) <= 0 ||
           CopyBuffer(m_ma200Handle, 0, 0, 1000, m_maBuffer200) <= 0 ||
           CopyBuffer(m_rsiHandle, 0, 0, 1000, m_rsiBuffer) <= 0 ||
           CopyBuffer(m_atrHandle, 0, 0, 1000, m_atrBuffer) <= 0)
        {
            if(m_logger != NULL)
                m_logger.LogError("Falha ao copiar dados dos indicadores");
            return;
        }
        
        // Copia dados de volume
        long volumeBuffer[];
        ArraySetAsSeries(volumeBuffer, true);
        if(CopyTickVolume(m_symbol, PERIOD_CURRENT, 0, 1000, volumeBuffer) <= 0)
        {
            if(m_logger != NULL)
                m_logger.LogError("Falha ao copiar dados de volume");
            return;
        }
    }
    
    double GetTrend()
    {
        // Calcula tendência usando médias móveis
        double ma20 = m_maBuffer20[0];
        double ma50 = m_maBuffer50[0];
        double ma200 = m_maBuffer200[0];
        
        // Verifica alinhamento das médias
        if(ma20 > ma50 && ma50 > ma200)
        {
            return 1.0; // Tendência de alta
        }
        else if(ma20 < ma50 && ma50 < ma200)
        {
            return -1.0; // Tendência de baixa
        }
        
        return 0.5; // Tendência lateral
    }
    
    double GetStrength()
    {
        // Retorna valor do RSI
        return m_rsiBuffer[0];
    }
    
    double GetVolatility()
    {
        // Retorna valor do ATR
        return m_atrBuffer[0];
    }
    
    double GetATR()
    {
        // Retorna valor do ATR
        return m_atrBuffer[0];
    }
    
    double GetVolume()
    {
        // Calcula média de volume dos últimos 20 períodos
        double avgVolume = 0;
        for(int i = 0; i < 20; i++)
        {
            avgVolume += m_volumeBuffer[i];
        }
        avgVolume /= 20;
        
        return avgVolume;
    }
    
    ~CMarketAnalyzer()
    {
        // Libera handles dos indicadores
        if(m_ma20Handle != INVALID_HANDLE) IndicatorRelease(m_ma20Handle);
        if(m_ma50Handle != INVALID_HANDLE) IndicatorRelease(m_ma50Handle);
        if(m_ma200Handle != INVALID_HANDLE) IndicatorRelease(m_ma200Handle);
        if(m_rsiHandle != INVALID_HANDLE) IndicatorRelease(m_rsiHandle);
        if(m_atrHandle != INVALID_HANDLE) IndicatorRelease(m_atrHandle);
    }
};

class CRiskManager
{
private:
    double m_maxRiskPercent;
    double m_maxDrawdown;
    double m_initialBalance;
    
public:
    CRiskManager(double maxRiskPercent = 1.0, double maxDrawdown = 20.0)
    {
        m_maxRiskPercent = maxRiskPercent;
        m_maxDrawdown = maxDrawdown;
        m_initialBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    }
    
    bool CheckRisk(double volume, double stopLoss)
    {
        double potentialLoss = volume * stopLoss * _Point;
        double accountBalance = AccountInfoDouble(ACCOUNT_BALANCE);
        double riskPercent = (potentialLoss / accountBalance) * 100;
        
        return (riskPercent <= m_maxRiskPercent);
    }
    
    bool CheckDrawdown()
    {
        double currentBalance = AccountInfoDouble(ACCOUNT_BALANCE);
        double drawdown = ((m_initialBalance - currentBalance) / m_initialBalance) * 100;
        
        return (drawdown <= m_maxDrawdown);
    }
    
    double GetSafeVolume(double stopLoss)
    {
        double accountBalance = AccountInfoDouble(ACCOUNT_BALANCE);
        double maxRiskAmount = accountBalance * (m_maxRiskPercent / 100);
        double pointValue = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
        
        return NormalizeDouble(maxRiskAmount / (stopLoss * pointValue), 2);
    }
};

    // Log final
    Print("CRaioX destruído com sucesso");
}

double CRaioX::CalculateSignal()
{
    if(m_marketAnalyzer == NULL)
        return 0.0;
        
    // Obtém indicadores técnicos
    double trend = m_marketAnalyzer.GetTrend();
    double strength = m_marketAnalyzer.GetStrength();
    double volatility = m_marketAnalyzer.GetVolatility();
    double volume = m_marketAnalyzer.GetVolume();
    
    // Calcula sinal baseado na física do mercado
    double physicsSignal = m_gravitationalForce * m_rotationFactor / 100.0;
    
    // Calcula força do volume
    double volumeSignal = (volume - VolumeMinimo) / (VolumeMaximo - VolumeMinimo);
    volumeSignal = MathMax(MathMin(volumeSignal, 1.0), 0.0);
    
    // Combina sinais com pesos otimizados
    double signal = (trend * TrendWeight) + 
                   (physicsSignal * 0.3) + 
                   (strength * MomentumWeight) + 
                   (volumeSignal * VolumeWeight) + 
                   (volatility * VolatilityWeight);
    
    // Normaliza sinal entre -1 e 1
    signal = MathMax(MathMin(signal, 1.0), -1.0);
    
    // Log do sinal calculado
    string message = StringFormat("Sinal calculado: %.4f (Trend: %.2f, Physics: %.2f, Strength: %.2f, Volume: %.2f, Volatility: %.2f)",
        signal, trend, physicsSignal, strength, volumeSignal, volatility);
    if(m_logger != NULL)
        m_logger.LogInfo(message);
    
    return signal;
}

double CRaioX::CalculatePositionSize()
{
    // Obtém saldo da conta
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    
    // Calcula tamanho base da posição
    double baseSize = balance * RiskPercent / 100.0;
    
    // Obtém indicadores para ajuste dinâmico
    double volatility = m_marketAnalyzer.GetVolatility();
    double trend = m_marketAnalyzer.GetTrend();
    double volume = m_marketAnalyzer.GetVolume();
    
    // Ajusta tamanho baseado na volatilidade
    double volatilityAdjustment = 1.0 - (volatility / 100.0);
    volatilityAdjustment = MathMax(volatilityAdjustment, VolatilityFactor);
    
    // Ajusta tamanho baseado no volume
    double volumeAdjustment = (volume - VolumeMinimo) / (VolumeMaximo - VolumeMinimo);
    volumeAdjustment = MathMax(MathMin(volumeAdjustment, 1.0), 0.5);
    
    // Ajusta tamanho baseado na força gravitacional
    double gravityAdjustment = 1.0;
    if(m_gravitationalForce > MinGravitationalForce)
    {
        gravityAdjustment = 1.0 + (m_gravitationalForce / MinGravitationalForce) * 0.5;
    }
    
    // Calcula tamanho final
    double adjustedSize = baseSize * volatilityAdjustment * volumeAdjustment * gravityAdjustment;
    
    // Converte para lotes
    double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
    double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
    double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
    
    double lots = NormalizeDouble(adjustedSize / (balance * lotStep), 2);
    lots = MathMax(MathMin(lots, maxLot), minLot);
    
    // Log do cálculo do tamanho da posição
    string message = StringFormat("Tamanho da posição calculado: %.2f (Base: %.2f, Vol: %.2f, VolAdj: %.2f, VolAdj: %.2f, GravityAdj: %.2f)",
        lots, baseSize, volatility, volatilityAdjustment, volumeAdjustment, gravityAdjustment);
    if(m_logger != NULL)
        m_logger.LogInfo(message);
    
    return lots;
}

void CRaioX::ExecuteTsunamiStrategy()
{
    if(m_marketAnalyzer == NULL)
    {
        m_logger.LogError("Market analyzer não inicializado");
        return;
    }
    
    // Obtém indicadores técnicos
    double trend = m_marketAnalyzer.GetTrend();
    double strength = m_marketAnalyzer.GetStrength();
    double volatility = m_marketAnalyzer.GetVolatility();
    double volume = m_marketAnalyzer.GetVolume();
    
    // Verifica condições para tsunami
    bool isTsunami = (m_gravitationalForce > MinGravitationalForce) && 
                    (MathAbs(m_rotationFactor) > MinRotationFactor) &&
                    (volatility > MinVolatility) &&
                    (volume > VolumeMinimo);
    
    if(!isTsunami)
    {
        m_logger.LogInfo("Condições de tsunami não atendidas");
        return;
    }
    
    // Calcula tamanho da posição baseado na força do tsunami
    double positionSize = CalculatePositionSize() * (m_gravitationalForce / MinGravitationalForce);
    
    // Determina direção baseado no trend e força gravitacional
    bool isLong = (trend > 0 && m_rotationFactor > 0) || (trend < 0 && m_rotationFactor < 0);
    
    // Calcula níveis de stop loss e take profit
    double atr = m_marketAnalyzer.GetATR();
    double currentPrice = isLong ? SymbolInfoDouble(_Symbol, SYMBOL_ASK) : SymbolInfoDouble(_Symbol, SYMBOL_BID);
    
    // Ajusta stop loss e take profit baseado na volatilidade
    double stopLossMultiplier = 1.5 + (volatility / 100.0);
    double takeProfitMultiplier = 3.0 + (volatility / 100.0);
    
    double stopLoss = isLong ? 
        currentPrice - (atr * stopLossMultiplier) : 
        currentPrice + (atr * stopLossMultiplier);
    
    double takeProfit = isLong ? 
        currentPrice + (atr * takeProfitMultiplier) : 
        currentPrice - (atr * takeProfitMultiplier);
    
    // Log dos parâmetros da estratégia
    string message = StringFormat(
        "Executando estratégia de tsunami:\n" +
        "Direção: %s\n" +
        "Tamanho: %.2f\n" +
        "Preço: %.5f\n" +
        "SL: %.5f\n" +
        "TP: %.5f\n" +
        "GF: %.2f\n" +
        "RF: %.2f\n" +
        "Vol: %.2f",
        isLong ? "COMPRA" : "VENDA",
        positionSize,
        currentPrice,
        stopLoss,
        takeProfit,
        m_gravitationalForce,
        m_rotationFactor,
        volume
    );
    m_logger.LogInfo(message);
    
    // Executa ordem
    if(!m_tradeExecutor.OpenPosition(isLong, positionSize, stopLoss, takeProfit))
    {
        m_logger.LogError("Falha ao executar estratégia de tsunami");
    }
    else
    {
        m_logger.LogInfo("Estratégia de tsunami executada com sucesso");
    }
}

void CRaioX::ExecuteNormalStrategy()
{
    if(m_marketAnalyzer == NULL)
    {
        m_logger.LogError("Market analyzer não inicializado");
        return;
    }
    
    // Obtém indicadores técnicos
    double trend = m_marketAnalyzer.GetTrend();
    double strength = m_marketAnalyzer.GetStrength();
    double volatility = m_marketAnalyzer.GetVolatility();
    
    // Calcula sinal de trading
    double signal = CalculateSignal();
    
    // Verifica força do sinal
    if(MathAbs(signal) < MinSignalStrength)
    {
        m_logger.LogInfo("Sinal muito fraco para executar estratégia normal");
        return;
    }
    
    // Determina direção baseado no sinal
    bool isLong = signal > 0;
    
    // Calcula tamanho da posição
    double positionSize = CalculatePositionSize();
    
    // Verifica se o tamanho da posição é válido
    if(positionSize <= 0)
    {
        m_logger.LogError("Tamanho da posição inválido");
        return;
    }
    
    // Calcula níveis de stop loss e take profit
    double atr = m_marketAnalyzer.GetATR();
    double currentPrice = isLong ? SymbolInfoDouble(_Symbol, SYMBOL_ASK) : SymbolInfoDouble(_Symbol, SYMBOL_BID);
    
    // Verifica se os preços são válidos
    if(currentPrice <= 0 || atr <= 0)
    {
        m_logger.LogError("Preços ou ATR inválidos");
        return;
    }
    
    double stopLoss = isLong ? currentPrice - (atr * 1.5) : currentPrice + (atr * 1.5);
    double takeProfit = isLong ? currentPrice + (atr * 3) : currentPrice - (atr * 3);
    
    // Executa ordem
    if(!m_tradeExecutor.OpenPosition(isLong, positionSize, stopLoss, takeProfit))
    {
        m_logger.LogError("Falha ao executar estratégia normal");
    }
    else
    {
        m_logger.LogInfo("Estratégia normal executada com sucesso");
    }
}

ENUM_MARKET_CONDITION CRaioX::AnalyzeMarket()
{
    // Obtém dados atuais
    double currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    double currentVolume = m_volumeBuffer[0];
    
    // Calcula médias de volume
    double avgVolume = 0;
    for(int i = 0; i < 20; i++)
    {
        avgVolume += m_volumeBuffer[i];
    }
    avgVolume /= 20;
    
    // Calcula força do volume
    double localVolumeForce = currentVolume / avgVolume;
    
    // Verifica força gravitacional
    if(m_gravitationalForce > MinGravitationalForce && localVolumeForce > VolumeForceThreshold)
    {
        // Verifica se o fator de rotação é significativo
        if(MathAbs(m_rotationFactor) > MinRotationFactor)
        {
            // Verifica alinhamento das médias móveis
            double ma20 = m_maBuffer20[0];
            double ma50 = m_maBuffer50[0];
            double ma200 = m_maBuffer200[0];
            
            // Verifica se as médias estão alinhadas
            bool maAligned = (ma20 > ma50 && ma50 > ma200) || (ma20 < ma50 && ma50 < ma200);
            
            if(maAligned)
            {
                m_logger.LogInfo("Condições de tsunami detectadas: GF=" + DoubleToString(m_gravitationalForce, 2) + 
                               ", RF=" + DoubleToString(m_rotationFactor, 2) + 
                               ", VF=" + DoubleToString(localVolumeForce, 2));
                return MARKET_TSUNAMI;
            }
        }
    }
    
    // Verifica volatilidade
    double atr = m_atrBuffer[0];
    double normalizedATR = atr / currentPrice * 100.0;
    
    // Verifica RSI
    double rsi = m_rsiBuffer[0];
    
    // Verifica médias móveis
    double ma20 = m_maBuffer20[0];
    double ma50 = m_maBuffer50[0];
    double ma200 = m_maBuffer200[0];
    
    // Calcula distância entre médias
    double distance1 = MathAbs(ma20 - ma50) / ma50 * 100.0;
    double distance2 = MathAbs(ma50 - ma200) / ma200 * 100.0;
    
    // Verifica consolidação
    if(normalizedATR < MinVolatility && 
       distance1 < MinMADistance && 
       distance2 < MinMADistance &&
       rsi > 40 && rsi < 60)
    {
        m_logger.LogInfo("Mercado em consolidação: ATR=" + DoubleToString(normalizedATR, 2) + 
                        "%, RSI=" + DoubleToString(rsi, 2));
        return MARKET_CONSOLIDATION;
    }
    
    // Condição normal de mercado
    m_logger.LogInfo("Condição normal de mercado: ATR=" + DoubleToString(normalizedATR, 2) + 
                    "%, RSI=" + DoubleToString(rsi, 2));
    return MARKET_NORMAL;
}

void CRaioX::UpdateAnalysis()
{
    // Copia dados dos indicadores para os buffers
    if(CopyBuffer(m_ma20Handle, 0, 0, 1000, m_maBuffer20) <= 0 ||
       CopyBuffer(m_ma50Handle, 0, 0, 1000, m_maBuffer50) <= 0 ||
       CopyBuffer(m_ma200Handle, 0, 0, 1000, m_maBuffer200) <= 0 ||
       CopyBuffer(m_rsiHandle, 0, 0, 1000, m_rsiBuffer) <= 0 ||
       CopyBuffer(m_atrHandle, 0, 0, 1000, m_atrBuffer) <= 0)
    {
        if(m_logger != NULL)
            m_logger.LogError("Falha ao copiar dados dos indicadores");
        return;
    }
    
    // Inicializa e copia dados de volume de forma segura
    long volumeBuffer[];
    ArraySetAsSeries(volumeBuffer, true);
    ArrayResize(volumeBuffer, 1000);
    
    if(CopyTickVolume(_Symbol, PERIOD_CURRENT, 0, 1000, volumeBuffer) <= 0)
    {
        if(m_logger != NULL)
            m_logger.LogError("Falha ao copiar dados de volume");
        return;
    }
    
    // Garante que o buffer de volume está inicializado
    if(ArraySize(m_volumeBuffer) != 1000)
    {
        ArrayResize(m_volumeBuffer, 1000);
    }
    
    // Copia dados do volumeBuffer para m_volumeBuffer
    for(int i = 0; i < 1000; i++)
    {
        m_volumeBuffer[i] = (double)volumeBuffer[i];
    }
    
    // Atualiza força gravitacional
    m_gravitationalForce = CalculateGravitationalForce();
    
    // Atualiza fator de rotação
    m_rotationFactor = CalculateRotationFactor();
    
    // Log dos valores atualizados
    string message = StringFormat("Análise atualizada: GF=%.2f, RF=%.2f", 
        m_gravitationalForce, m_rotationFactor);
    if(m_logger != NULL)
        m_logger.LogInfo(message);
}

void CRaioX::UpdatePositions()
{
    // Obtém todas as posições abertas
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        ulong ticket = PositionGetTicket(i);
        if(ticket <= 0)
        {
            m_logger.LogError("Falha ao obter ticket da posição");
            continue;
        }
        
        // Verifica se a posição é do símbolo atual
        if(PositionGetString(POSITION_SYMBOL) != _Symbol)
        {
            continue;
        }
        
        // Obtém dados da posição
        double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
        double currentPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
        double stopLoss = PositionGetDouble(POSITION_SL);
        double takeProfit = PositionGetDouble(POSITION_TP);
        double profit = PositionGetDouble(POSITION_PROFIT);
        bool isLong = PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY;
        
        // Verifica trailing stop
        if(UseTrailingStop && profit > 0)
        {
            double atr = m_atrBuffer[0];
            double newStopLoss = isLong ? 
                currentPrice - (atr * TrailingStopATRMultiplier) :
                currentPrice + (atr * TrailingStopATRMultiplier);
                
            // Verifica se o novo stop loss é melhor que o atual
            if(isLong && newStopLoss > stopLoss)
            {
                m_tradeExecutor.ModifyPosition(ticket, newStopLoss, takeProfit);
            }
            else if(!isLong && newStopLoss < stopLoss)
            {
                m_tradeExecutor.ModifyPosition(ticket, newStopLoss, takeProfit);
            }
        }
        
        // Atualiza estatísticas
        m_marketStats.totalProfit += profit;
    }
}

double CRaioX::CalculateGravitationalForce()
{
    // Obtém dados de preço e volume
    double currentPrice = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    double currentVolume = m_volumeBuffer[0];
    
    // Calcula médias de volume em diferentes períodos
    double avgVolume20 = 0, avgVolume50 = 0, avgVolume200 = 0;
    for(int i = 0; i < 200; i++)
    {
        if(i < 20) avgVolume20 += m_volumeBuffer[i];
        if(i < 50) avgVolume50 += m_volumeBuffer[i];
        avgVolume200 += m_volumeBuffer[i];
    }
    avgVolume20 /= 20;
    avgVolume50 /= 50;
    avgVolume200 /= 200;
    
    // Calcula força do volume em diferentes períodos
    double volumeForce20 = currentVolume / avgVolume20;
    double volumeForce50 = currentVolume / avgVolume50;
    double volumeForce200 = currentVolume / avgVolume200;
    
    // Calcula média ponderada da força do volume
    double localVolumeForce = (volumeForce20 * 0.5) + (volumeForce50 * 0.3) + (volumeForce200 * 0.2);
    
    // Obtém dados de volatilidade
    double atr = m_atrBuffer[0];
    double normalizedATR = atr / currentPrice * 100.0;
    
    // Calcula força gravitacional baseada no volume e volatilidade
    double gravityForce = localVolumeForce * normalizedATR * GravityMultiplier;
    
    // Aplica filtro de suavização
    static double lastGravityForce = 0;
    if(lastGravityForce == 0)
    {
        lastGravityForce = gravityForce;
    }
    else
    {
        gravityForce = (gravityForce * 0.3) + (lastGravityForce * 0.7);
        lastGravityForce = gravityForce;
    }
    
    // Normaliza entre 0 e 100
    gravityForce = MathMin(MathMax(gravityForce, 0.0), 100.0);
    
    // Log do cálculo
    string message = StringFormat("Força gravitacional calculada: %.2f (VF20: %.2f, VF50: %.2f, VF200: %.2f, ATR: %.2f%%)",
        gravityForce, volumeForce20, volumeForce50, volumeForce200, normalizedATR);
    if(m_logger != NULL)
        m_logger.LogInfo(message);
    
    return gravityForce;
}

double CRaioX::CalculateRotationFactor()
{
    // Obtém dados das médias móveis
    double ma20 = m_maBuffer20[0];
    double ma50 = m_maBuffer50[0];
    double ma200 = m_maBuffer200[0];
    
    // Calcula ângulos entre as médias
    double angle1 = MathArctan((ma20 - ma50) / ma50) * 180.0 / M_PI;
    double angle2 = MathArctan((ma50 - ma200) / ma200) * 180.0 / M_PI;
    
    // Calcula fator de rotação baseado nos ângulos
    double rotationFactorValue = (angle1 + angle2) / 2.0;
    
    // Normaliza entre -100 e 100
    rotationFactorValue = MathMin(MathMax(rotationFactorValue, -100.0), 100.0);
    
    return rotationFactorValue;
}

bool CRaioX::CheckSafetyConditions()
{
    // Verifica spread
    double spread = SymbolInfoInteger(_Symbol, SYMBOL_SPREAD) * _Point;
    if(spread > MAX_SPREAD_POINTS)
    {
        m_logger.LogWarning("Spread muito alto: " + DoubleToString(spread, 2));
        return false;
    }
    
    // Verifica saldo da conta
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    if(balance < MIN_ACCOUNT_BALANCE)
    {
        m_logger.LogError("Saldo da conta muito baixo: " + DoubleToString(balance, 2));
        return false;
    }
    
    // Verifica perda diária
    if(m_dailyProfit < -MAX_DAILY_LOSS_PERCENT)
    {
        m_logger.LogError("Perda diária excedida: " + DoubleToString(m_dailyProfit, 2) + "%");
        return false;
    }
    
    // Verifica número de posições
    if(m_openPositions >= MAX_OPEN_POSITIONS)
    {
        m_logger.LogWarning("Número máximo de posições atingido: " + IntegerToString(m_openPositions));
        return false;
    }
    
    return true;
}

void CRaioX::UpdateDailyStats()
{
    MqlDateTime dt;
    TimeCurrent(dt);
    
    // Reseta estatísticas diárias se necessário
    MqlDateTime lastReset;
    TimeToStruct(m_lastDailyReset, lastReset);
    if(m_lastDailyReset == 0 || dt.day != lastReset.day)
    {
        m_dailyProfit = 0;
        m_lastDailyReset = TimeCurrent();
        m_logger.LogInfo("Estatísticas diárias resetadas");
    }
}

void CRaioX::UpdateDrawdownProtection()
{
    double currentBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    
    // Atualiza saldo máximo
    if(currentBalance > m_peakBalance)
    {
        m_peakBalance = currentBalance;
    }
    
    // Calcula drawdown atual
    if(m_peakBalance > 0)
    {
        m_currentDrawdown = ((m_peakBalance - currentBalance) / m_peakBalance) * 100.0;
        
        // Verifica se drawdown excedeu o limite
        if(m_currentDrawdown > m_maxAllowedDrawdown)
        {
            m_logger.LogError("Drawdown excessivo detectado: " + DoubleToString(m_currentDrawdown, 2) + "%");
            
            // Fecha todas as posições
            for(int i = PositionsTotal() - 1; i >= 0; i--)
            {
                ulong ticket = PositionGetTicket(i);
                if(ticket > 0)
                {
                    m_tradeExecutor.ClosePosition(ticket);
                }
            }
        }
    }
}

bool CRaioX::CheckNewsProtection()
{
    // Verifica se deve checar notícias
    if(TimeCurrent() - m_lastNewsCheck < 300) // Verifica a cada 5 minutos
        return true;
            
    m_lastNewsCheck = TimeCurrent();
    
    // Verifica se o filtro de notícias está ativo
    if(!UseNewsFilter)
    {
        m_newsProtectionActive = false;
        return true;
    }
    
    // Verifica horário para proteção contra notícias importantes
    MqlDateTime dt;
    TimeCurrent(dt);
    
    // Lista de horários importantes (NFP, FOMC, etc.)
    bool isImportantTime = false;
    
    // Verifica se é quarta-feira (FOMC)
    if(dt.day_of_week == 3 && dt.hour >= 13 && dt.hour <= 15)
    {
        isImportantTime = true;
    }
    
    // Verifica se é primeira sexta-feira do mês (NFP)
    if(dt.day_of_week == 5 && dt.day <= 7 && dt.hour >= 7 && dt.hour <= 9)
    {
        isImportantTime = true;
    }
    
    // Verifica se é dia de CPI
    if(dt.day == 10 && dt.hour >= 7 && dt.hour <= 9)
    {
        isImportantTime = true;
    }
    
    if(isImportantTime)
    {
        m_newsProtectionActive = true;
        m_logger.LogWarning("Período de notícias importantes detectado");
        return false;
    }
    
    m_newsProtectionActive = false;
    return true;
}

void CRaioX::OnTick()
{
    // Atualiza estatísticas diárias
    UpdateDailyStats();
    
    // Verifica condições de segurança
    if(!CheckSafetyConditions())
    {
        m_logger.LogWarning("Condições de segurança não atendidas - aguardando");
        return;
    }
    
    // Verifica se o mercado está aberto
    if(!m_tradeExecutor.IsMarketOpen())
    {
        m_logger.LogInfo("Mercado fechado");
        return;
    }
    
    // Verifica proteção contra notícias
    if(!CheckNewsProtection())
    {
        m_logger.LogInfo("Proteção contra notícias ativa - aguardando");
        return;
    }
    
    // Atualiza análise de mercado
    UpdateAnalysis();
    
    // Verifica condições de mercado
    ENUM_MARKET_CONDITION marketCondition = AnalyzeMarket();
    
    // Executa estratégia apropriada
    switch(marketCondition)
    {
        case MARKET_TSUNAMI:
            m_logger.LogInfo("Condição de tsunami detectada");
            ExecuteTsunamiStrategy();
            break;
            
        case MARKET_NORMAL:
            m_logger.LogInfo("Condição normal de mercado");
            ExecuteNormalStrategy();
            break;
            
        case MARKET_CONSOLIDATION:
            m_logger.LogInfo("Mercado em consolidação - aguardando");
            break;
            
        default:
            m_logger.LogWarning("Condição de mercado não reconhecida");
            break;
    }
    
    // Atualiza posições existentes
    UpdatePositions();
    
    // Atualiza proteção contra drawdown
    UpdateDrawdownProtection();
}

bool CRaioX::Init()
{
    // Verifica ambiente
    if(!MQLInfoInteger(MQL_TRADE_ALLOWED))
    {
        if(m_logger != NULL)
            m_logger.LogError("Trading não permitido");
        return false;
    }
    
    // Verifica símbolo
    if(!SymbolSelect(_Symbol, true))
    {
        if(m_logger != NULL)
            m_logger.LogError("Falha ao selecionar símbolo");
        return false;
    }
    
    // Inicializa market analyzer
    if(m_marketAnalyzer == NULL)
    {
        m_marketAnalyzer = new CMarketAnalyzer(m_logger);
    }
    
    if(!m_marketAnalyzer.Init(_Symbol))
    {
        if(m_logger != NULL)
            m_logger.LogError("Falha ao inicializar market analyzer");
        return false;
    }
    
    // Inicializa buffers dos indicadores
    if(!InitializeBuffers())
    {
        if(m_logger != NULL)
            m_logger.LogError("Falha ao inicializar buffers");
        return false;
    }
    
    // Configura trade executor
    if(!m_tradeExecutor.Init(_Symbol, Magic))
    {
        if(m_logger != NULL)
            m_logger.LogError("Falha ao inicializar trade executor");
        return false;
    }
    
    // Log dos parâmetros
    string params = StringFormat(
        "EA inicializado com sucesso:\n" +
        "Símbolo: %s\n" +
        "Magic: %d\n" +
        "Risk%%: %.2f\n" +
        "UseTrailingStop: %s\n" +
        "MinGravitationalForce: %.2f\n" +
        "MinRotationFactor: %.2f",
        _Symbol,
        Magic,
        RiskPercent,
        UseTrailingStop ? "true" : "false",
        MinGravitationalForce,
        MinRotationFactor
    );
    if(m_logger != NULL)
        m_logger.LogInfo(params);
    
    return true;
}

//+------------------------------------------------------------------+
//| Expert initialization function                                     |
//+------------------------------------------------------------------+
int OnInit()
{
    // Cria instância do EA
    g_raioX = new CRaioX();
    
    // Inicializa o EA
    if(!g_raioX.Init())
    {
        Print("Falha ao inicializar EA");
        return INIT_FAILED;
    }
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                   |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    // Deleta instância do EA
    if(g_raioX != NULL)
    {
        delete g_raioX;
        g_raioX = NULL;
    }
}

//+------------------------------------------------------------------+
//| Expert tick function                                              |
//+------------------------------------------------------------------+
void OnTick()
{
    // Verifica se o EA está inicializado
    if(g_raioX == NULL)
    {
        Print("EA não inicializado");
        return;
    }
    
    // Processa tick
    g_raioX.OnTick();
}
