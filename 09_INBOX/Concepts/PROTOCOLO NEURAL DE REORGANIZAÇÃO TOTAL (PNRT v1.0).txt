# 🔍 FASE 1: DIAGNÓSTICO NEURAL COMPLETO
echo "[PNRT] Iniciando diagnóstico neural do sistema..."
find MQL5/ -type f -name "*.mqh" -o -name "*.mq5" | xargs grep -l "#include <include/" > /tmp/includes_errados.txt
echo "[PNRT] Arquivos com #include <include/ encontrados: $(wc -l < /tmp/includes_errados.txt)"

# 🧬 FASE 2: REESTRUTURAÇÃO DE PASTAS (PADRÃO TIER-0)
echo "[PNRT] Validando estrutura de pastas..."
mkdir -p MQL5/{CORE,AGENTS,UTILS,CONFIG,INCLUDE/{Analysis,ExecutionLogic,DecisionEngine,Intelligence,Integration,Visuals,Modules,Tools,Types},AUDITOR,EXPERT,SCRIPTS}

# 🔧 FASE 3: CORREÇÃO AUTOMATIZADA DE INCLUDES
echo "[PNRT] Corrigindo includes com padrão TIER-0..."
while IFS= read -r file; do
    echo "[PNRT] Corrigindo: $file"
    sed -i.bak 's|#include <include/\(.*\)>|#include <\1>|g' "$file"
    sed -i.bak 's|#include "include/\\\(.*\\\)"|#include "\\\1"|g' "$file"
    rm "$file.bak" 2>/dev/null || true
done < /tmp/includes_errados.txt

# 🧩 FASE 4: VALIDAÇÃO DE DEPENDÊNCIAS CRÍTICAS
echo "[PNRT] Validando dependências críticas..."
declare -A critical_files=(
    ["utils/logger_institutional.mqh"]="MQL5/INCLUDE/utils/logger_institutional.mqh"
    ["security/quantum_firewall.mqh"]="MQL5/INCLUDE/security/quantum_firewall.mqh"
    ["risk/risk_profile.mqh"]="MQL5/INCLUDE/risk/risk_profile.mqh"
    ["analysis/market_regime_detector.mqh"]="MQL5/INCLUDE/analysis/market_regime_detector.mqh"
    ["intelligence/anomaly_detector_ai.mqh"]="MQL5/INCLUDE/intelligence/anomaly_detector_ai.mqh"
    ["visuals/log_panel.mqh"]="MQL5/INCLUDE/visuals/log_panel.mqh"
    ["quantum/quantum_processor.mqh"]="MQL5/INCLUDE/quantum/quantum_processor.mqh"
    ["CORE/core_brain_manager.mqh"]="MQL5/CORE/core_brain_manager.mqh"
)

for key in "${!critical_files[@]}"; do
    if [[ ! -f "${critical_files[$key]}" ]]; then
        echo "[PNRT] ERRO: Arquivo crítico ausente: ${critical_files[$key]}"
        echo "[PNRT] SOLUÇÃO: Crie o arquivo ou corrija o caminho"
    else
        echo "[PNRT] OK: ${key}"
    fi
done

# 🧠 FASE 5: VALIDAÇÃO DO NÚCLEO CEREBRAL (CORE)
echo "[PNRT] Validando CoreBrainManager..."
if grep -q "CoreBrainManager" MQL5/EXPERT/NumeiaEA.mq5; then
    echo "[PNRT] CoreBrainManager encontrado"
    if grep -A 10 "CoreBrainManager" MQL5/EXPERT/NumeiaEA.mq5 | grep -q "IsReady()"; then
        echo "[PNRT] CoreBrainManager está sendo validado no OnInit()"
    else
        echo "[PNRT] ALERTA: CoreBrainManager não está sendo validado"
    fi
else
    echo "[PNRT] ERRO: CoreBrainManager não está sendo instanciado"
fi

# 🛡 FASE 6: APLICAÇÃO DE BLINDAGEM TIER-0
echo "[PNRT] Aplicando blindagem TIER-0..."
cat << 'EOF' > MQL5/INCLUDE/utils/tier0_protection.mqh
//+------------------------------------------------------------------+
//| tier0_protection.mqh - Blindagem Institucional TIER-0            |
//| Projeto: QuantumOmegaGodMode / EA Numeia                        |
//+------------------------------------------------------------------+
#ifndef __TIER0_PROTECTION_MQH__
#define __TIER0_PROTECTION_MQH__

bool ValidateContext()
{
   if(!TerminalInfoInteger(TERMINAL_CONNECTED))
   {
      Print("[TIER0] Sem conexão com o servidor");
      return false;
   }
   
   if(!IsDllsAllowed())
   {
      Print("[TIER0] DLLs não permitidas");
      return false;
   }
   
   if(!IsTradeAllowed())
   {
      Print("[TIER0] Negociação bloqueada");
      return false;
   }
   
   return true;
}

#endif // __TIER0_PROTECTION_MQH__
EOF

# 🔁 FASE 7: INTEGRAÇÃO COM NumeiaEA.mq5
echo "[PNRT] Integrando blindagem no NumeiaEA.mq5..."
if ! grep -q "#include <tier0_protection.mqh>" MQL5/EXPERT/NumeiaEA.mq5; then
    sed -i '1i#include <tier0_protection.mqh>' MQL5/EXPERT/NumeiaEA.mq5
    sed -i '/OnInit()/a\   if(!ValidateContext()) ExpertRemove();' MQL5/EXPERT/NumeiaEA.mq5
    echo "[PNRT] Blindagem TIER-0 integrada"
fi
