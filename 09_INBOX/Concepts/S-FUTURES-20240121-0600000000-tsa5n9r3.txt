    async def _calculate_quantum_theoretical_spread(self, front: TermStructurePoint, 
                                                  back: TermStructurePoint,

        Calcula spread teórico com modelo quântico de cost-of-carry
        """
        time_diff = Decimal(back.days_to_expiry - front.days_to_expiry) / Decimal('365')
        
        # Modelo quântico aprimorado com fatores de mercado
        risk_free_rate = await self._get_quantum_risk_free_rate(market_analysis)
        storage_costs = await self._get_quantum_storage_costs(front.contract.split()[0])
        convenience_yield = await self._get_convenience_yield(front.contract.split()[0], market_analysis)
        
        # Fórmula quântica: F = S * e^((r + s - c)*t)
        adjusted_rate = risk_free_rate + storage_costs - convenience_yield
        theoretical_spread = front.price * ((adjusted_rate * time_diff).exp() - Decimal('1'))
        
        return theoretical_spread
