python# HJB PRÁTICA COM LBFGS (scipy.optimize, grid-free approx)
from scipy.optimize import minimize

def hjb_control(state):  # state = [price, vol, regime, liquidity]
    def hjb_obj(u):  # u = [size, hold_time]
        size, hold_time = u
        # Approx: ∂V/∂t + μ ∂V/∂x + 0.5 σ² ∂²V/∂x² - c(size) ≈ 0 (finite diff central)
        mu_dV = 0.001 * (state[0] * size)  # Drift approx
        sigma_d2V = 0.5 * state[1]**2 * size**2  # Vol diffusion
        cost = 0.1 * np.sqrt(hold_time / 3600) + 0.3 * state[1]**2 * hold_time  # Almgren-like
        residual = abs(mu_dV + sigma_d2V - cost)  # Min residual
        return residual
    
    # Bounds: size 0-0.02, hold 900-7200s
    res = minimize(hjb_obj, [0.01, 1800], bounds=[(0,0.02),(900,7200)], method='L-BFGS-B')
    return res.x if res.success else [0.007875, 1800]  # Fallback example: 0.7875% size, 30min hold
