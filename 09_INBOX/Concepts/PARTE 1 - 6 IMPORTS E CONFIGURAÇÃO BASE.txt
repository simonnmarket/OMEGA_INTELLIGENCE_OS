class OmegaProtocol:
    """Protocolo Omega TIER-0 - Blindagem Institucional"""
    
    def __init__(self):
        self.protocol_version = "8.0.0"
        self.activation_timestamp = datetime.now(timezone.utc)
        self.security_level = "TIER-0"
        self.checksums = {}
        
    def activate(self):
        """Ativar protocolo Omega completo"""
        logger.info("⚡ PROTOCOLO OMEGA ATIVADO", 
                   version=self.protocol_version,
                   security_level=self.security_level)
        
        # Validar checksums críticos
        self._validate_critical_checksums()
        
        # Ativar blindagem
        self._activate_shielding()
        
        return {
            "status": "ACTIVE",
            "timestamp": self.activation_timestamp.isoformat(),
            "protocol": "OMEGA_TIER-0",
            "checksum": self._generate_system_checksum()
        }
    
    def _validate_critical_checksums(self):
        """Validar checksums de componentes críticos"""
        critical_components = [
            "agent_council",
            "risk_engine", 
            "consensus_engine",
            "execution_layer"
        ]
        
        for component in critical_components:
            checksum = self._calculate_component_checksum(component)
            self.checksums[component] = checksum
            logger.debug(f"Checksum validado: {component}", checksum=checksum[:16])
    
    def _activate_shielding(self):
        """Ativar camadas de blindagem"""
        shield_layers = [
            "LAYER_1: Integrity Protection",
            "LAYER_2: Runtime Validation",
            "LAYER_3: Audit Trail",
            "LAYER_4: Emergency Protocols"
        ]
        
        for layer in shield_layers:
            logger.info(f"🛡️ Shield ativado: {layer}")
    
    def _generate_system_checksum(self) -> str:
        """Gerar checksum do sistema completo"""
        system_state = {
            "protocol_version": self.protocol_version,
            "timestamp": self.activation_timestamp.isoformat(),
            "components": list(self.checksums.keys())
        }
        return hashlib.sha3_256(json.dumps(system_state, sort_keys=True).encode()).hexdigest()
    
    def _calculate_component_checksum(self, component: str) -> str:
        """Calcular checksum de componente"""
        return hashlib.sha3_256(f"{component}_{self.activation_timestamp.isoformat()}".encode()).hexdigest()

class MarketData(BaseModel):
    """Dados de mercado padronizados"""
    symbol: str
    timestamp: datetime = Field(default_factory=lambda: datetime.now(timezone.utc))
    price: float
    spread: float
    volume: float
    ma5: Optional[float] = None
    ma20: Optional[float] = None
    ma_signal: Optional[str] = Field(None, pattern="^(BULLISH|BEARISH|NEUTRAL)$")
    volatility_24h: float = 0.0
    
    @validator('ma_signal', pre=True, always=True)
    def calculate_ma_signal(cls, v, values):
        """Calcular sinal MA automaticamente"""
        if 'ma5' in values and 'ma20' in values:
            if values['ma5'] is not None and values['ma20'] is not None:
                diff = values['ma5'] - values['ma20']
                if diff > 0.0001:  # Threshold pequeno
                    return "BULLISH"
                elif diff < -0.0001:
                    return "BEARISH"
                else:
                    return "NEUTRAL"
        return v or "NEUTRAL"

class AgentVote(BaseModel):
    """Voto estruturado de agente com NO_VOTE support"""
    agent_name: str
    decision: str = Field(..., pattern="^(BUY|SELL|HOLD|NO_VOTE)$")
    confidence: float = Field(..., ge=0.0, le=1.0)
    reason: str
    technical_error: bool = False
    timestamp: datetime = Field(default_factory=lambda: datetime.now(timezone.utc))
    weight: float = 1.0
    
    def should_count_in_consensus(self) -> bool:
        """Verificar se voto deve contar no consenso"""
        return not (self.technical_error or self.decision == "NO_VOTE")

class TradeSignal(BaseModel):
    """Sinal de trading completo"""
    symbol: str
    action: str = Field(..., pattern="^(BUY|SELL|HOLD)$")
    confidence: float = Field(..., ge=0.0, le=1.0)
    agents_votes: List[AgentVote]
    timestamp: datetime = Field(default_factory=lambda: datetime.now(timezone.utc))
    risk_score: Optional[float] = None
    position_size: Optional[float] = None
    
    @property
    def weighted_confidence(self) -> float:
        """Confiança ponderada pelos pesos dos agentes"""
        if not self.agents_votes:
            return 0.0
        
        valid_votes = [v for v in self.agents_votes if v.should_count_in_consensus()]
        if not valid_votes:
            return 0.0
            
        total_weight = sum(v.weight for v in valid_votes)
        weighted_sum = sum(v.confidence * v.weight for v in valid_votes)
        
        return weighted_sum / total_weight if total_weight > 0 else 0.0

class ZeroMQBus:
    """Backbone de comunicação ZeroMQ para µs latency"""
    
    def __init__(self):
        self.context = zmq.asyncio.Context()
        self.sockets = {}
        self.channels = {
            "agent_signals": {"type": zmq.PUB, "address": "inproc://agent_signals"},
            "execution_orders": {"type": zmq.PUSH, "address": "inproc://execution_orders"},
            "health_check": {"type": zmq.REP, "address": "inproc://health"},
            "emergency": {"type": zmq.PUSH, "address": "inproc://emergency"},
            "market_data": {"type": zmq.SUB, "address": "inproc://market_data"}
        }
        
    async def initialize(self):
        """Inicializar todos os sockets"""
        for name, config in self.channels.items():
            socket_type = config["type"]
            address = config["address"]
            
            if socket_type in [zmq.PUB, zmq.PUSH]:
                socket = self.context.socket(socket_type)
                socket.bind(address)
            else:  # SUB, REP, etc.
                socket = self.context.socket(socket_type)
                socket.connect(address)
                if socket_type == zmq.SUB:
                    socket.setsockopt_string(zmq.SUBSCRIBE, "")
            
            self.sockets[name] = socket
        
        logger.info("✅ ZeroMQ Bus inicializado", channels=list(self.channels.keys()))
    
    async def publish_signal(self, signal: TradeSignal):
        """Publicar sinal para todos os agentes"""
        message = signal.model_dump_json()
        await self.sockets["agent_signals"].send_string(message)
        logger.debug("📤 Sinal publicado", symbol=signal.symbol, action=signal.action)
    
    async def receive_signal(self, timeout_ms: int = 100) -> Optional[TradeSignal]:
        """Receber sinal com timeout"""
        try:
            message = await self.sockets["agent_signals"].recv_string(flags=zmq.NOBLOCK)
            data = json.loads(message)
            # Converter timestamp string para datetime
            if 'timestamp' in data and isinstance(data['timestamp'], str):
                data['timestamp'] = datetime.fromisoformat(data['timestamp'].replace('Z', '+00:00'))
            return TradeSignal(**data)
        except zmq.Again:
            return None
        except Exception as e:
            logger.error("❌ Erro ao receber sinal", error=str(e))
            return None
    
    async def send_order(self, order: Dict[str, Any]):
        """Enviar ordem para execução"""
        await self.sockets["execution_orders"].send_json(order)
        logger.debug("📨 Ordem enviada para execução", order=order.get('symbol'))
    
    async def health_check(self) -> bool:
        """Verificar saúde do bus"""
        try:
            # Enviar ping
            ping_socket = self.context.socket(zmq.REQ)
            ping_socket.setsockopt(zmq.LINGER, 0)
            ping_socket.connect("inproc://health")
            await ping_socket.send_string("ping")
            
            # Receber pong com timeout
            try:
                response = await asyncio.wait_for(ping_socket.recv_string(), timeout=1.0)
                ping_socket.close()
                return response == "pong"
            except asyncio.TimeoutError:
                ping_socket.close()
                return False
                
        except Exception as e:
            logger.error("❌ Health check falhou", error=str(e))
            return False
    
    async def close(self):
        """Fechar todos os sockets"""
        for socket in self.sockets.values():
            socket.close()
        self.context.term()
        logger.info("🔒 ZeroMQ Bus fechado")
