    async def _calculate_composite_sentiment_v7(self, bank: CentralBank,

        CALCULA SCORE COMPOSTO DE SENTIMENT v7.0

        # Scores individuais
        speech_score = speech_sentiment['sentiment_score']
        meeting_score = meeting_sentiment['sentiment_score']
        
        # Confianças individuais
        speech_confidence = speech_sentiment['confidence']
        meeting_confidence = meeting_sentiment['confidence']
        
        # Pesos dinâmicos
        speech_weight = bank_params['speech_weight']
        meeting_weight = bank_params['meeting_weight']
        
        # Score composto ponderado
        composite_score = (
            speech_score * speech_weight * speech_confidence +
            meeting_score * meeting_weight * meeting_confidence
        ) / (speech_weight + meeting_weight)
        
        # Confiança composta
        composite_confidence = (speech_confidence + meeting_confidence) / Decimal('2')
        
        # Intensidade composta
        intensity = (speech_sentiment['intensity'] + meeting_sentiment.get('impact', Decimal('0'))) / Decimal('2')
        
        return {
            'sentiment_score': composite_score,
            'confidence': composite_confidence,
            'intensity': intensity,
            'key_phrases': speech_sentiment.get('key_phrases', [])

    async def _calculate_sentiment_differential_v7(self, base_signal: CentralBankSentimentSignal,
                                                 quote_signal: CentralBankSentimentSignal) -> Decimal:
        """
        CALCULA DIFFERENTIAL DE SENTIMENT ENTRE BANCOS CENTRAIS v7.0
        """
        base_params = self.central_banks[base_signal.bank]
        quote_params = self.central_banks[quote_signal.bank]
        
        # Scores normalizados por importância
        base_score = base_signal.sentiment_score * base_params['importance']
        quote_score = quote_signal.sentiment_score * quote_params['importance']
        
        # Differential base
        raw_differential = base_score - quote_score
        
        # Ajuste por confiança
        confidence_adjustment = (base_signal.confidence + quote_signal.confidence) / Decimal('2')
        
        # Ajuste por intensidade
        intensity_adjustment = (base_signal.speech_intensity + quote_signal.speech_intensity) / Decimal('2')
        
        differential = raw_differential * confidence_adjustment * intensity_adjustment
        
        return differential

    async def _calculate_forex_position_size_v7(self, sentiment_differential: Decimal,
                                              base_signal: CentralBankSentimentSignal,
                                              quote_signal: CentralBankSentimentSignal,
                                              allocated_capital: Decimal) -> Decimal:
        """
        CALCULA TAMANHO DA POSIÇÃO FOREX v7.0
        """
        # Base size adaptativo
        base_size = allocated_capital * Decimal('0.06')  # 6% base para Forex
        
        # Ajuste por força do differential
        strength_adjustment = min(
            Decimal('2.0'), 
            abs(sentiment_differential) * Decimal('10.0')  # Multiplicador para Forex
        )
        
        # Ajuste por volatilidade
        avg_volatility = (base_signal.expected_volatility + quote_signal.expected_volatility) / Decimal('2')
        vol_adjustment = self.sentiment_params['volatility_adjustment'] / avg_volatility
        
        # Ajuste por probabilidade de execução
        prob_adjustment = (base_signal.execution_probability + quote_signal.execution_probability) / Decimal('2')
        
        adjusted_size = base_size * strength_adjustment * vol_adjustment * prob_adjustment
        
        # Limites de segurança Forex
        max_size = allocated_capital * self.sentiment_params['max_position_size']
        min_size = Decimal('5000')  # Mínimo €5,000 para Forex
        
        return max(min(adjusted_size, max_size), min_size)

    async def _calculate_forex_risk_score_v7(self, pair: str, sentiment_differential: Decimal) -> Decimal:
        """
        CALCULA SCORE DE RISCO PARA FOREX v7.0
        """
        # Risco base para Forex
        base_risk = Decimal('0.3')
        
        # Ajustes baseados no pair e differential
        if 'JPY' in pair:
            base_risk += Decimal('0.1')  # JPY pairs mais voláteis
        
        if abs(sentiment_differential) > Decimal('0.25'):
            base_risk += Decimal('0.1')  # Differential muito alto = mais risco
            
        return min(base_risk, Decimal('0.6'))
