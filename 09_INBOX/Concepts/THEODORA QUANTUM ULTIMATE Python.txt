def _build_advanced_model(self) -> Model:
"""Híbrido LSTM-Transformer com atenção multi-head."""
main_input = Input(shape=self.input_shape, name='main_input')
LSTM para sequências temporais
lstm1 = LSTM(256, return_sequences=True)(main_input)
lstm2 = LSTM(128, return_sequences=True)(lstm1)
Transformer com Multi-Head Attention
attention_output = MultiHeadAttention(num_heads=16, key_dim=128)(lstm2, lstm2)
attention_output = Dropout(0.1)(attention_output)
attention_output = LayerNormalization(epsilon=1e-6)(attention_output)
ff = Dense(256, activation='relu')(attention_output)
ff = Dense(128)(ff)
transformer_output = GlobalAveragePooling1D()(ff)
CNN para padrões locais
conv1 = Conv1D(128, 3, activation='relu', padding='same')(main_input)
pool1 = MaxPooling1D(pool_size=2)(conv1)
flatten = Flatten()(pool1)
Combinação com DRL output (simulado)
combined = tf.keras.layers.concatenate([transformer_output, flatten])
dense1 = Dense(256, activation='relu', kernel_regularizer=l2(0.005))(combined)
dropout1 = Dropout(0.2)(dense1)
Multi-output para horizontes
output_short = Dense(1, activation='sigmoid', name='short_term')(dropout1)
output_medium = Dense(1, activation='sigmoid', name='medium_term')(dropout1)
output_long = Dense(1, activation='sigmoid', name='long_term')(dropout1)
model = Model(inputs=main_input, outputs=[output_short, output_medium, output_long])
model.compile(
optimizer=Adam(learning_rate=5e-5),
loss='binary_crossentropy',
metrics=['accuracy']
)

profit = forward - 1 - 0.0015  # Fees ajustados
if profit > 0:

def statistical_arbitrage(self, pair1: str, pair2: str, data1: pd.Series, data2: pd.Series, window: int = 50) -> Optional[float]:
"""Arbitragem estatística com cointegration e ML."""
if len(data1) < window:
return None
spread = data1 - data2
zscore = (spread - spread.mean()) / spread.std()
if abs(zscore.iloc[-1]) > 2.5:  # Threshold ajustado
score, pvalue, _ = coint(data1, data2)
if pvalue < 0.01:
return zscore.iloc[-1]
return None

async def execute_order(self, order_data: Dict):

risk_score = await self._compute_risk(order_data)
if risk_score > 0.05:  # Threshold CVaR

return

"action": mt5.TRADE_ACTION_DEAL,
"symbol": order_data['symbol'],
"volume": order_data['volume'],
"type": mt5.ORDER_TYPE_BUY if order_data['side'] == 'buy' else mt5.ORDER_TYPE_SELL,
}
mt5.order_send(request)
elif platform in self.exchange_clients:
exchange = self.exchange_clients[platform]
await exchange.create_order(order_data['symbol'], 'market', order_data['side'], order_data['amount'])
