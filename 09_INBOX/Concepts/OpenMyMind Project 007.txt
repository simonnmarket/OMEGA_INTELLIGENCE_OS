# 1) Tweets (snscrape) - coleta pública
def scrape_tweets(query, max_tweets=500):
    """
    Retorna DataFrame com tweets públicos que batem no query.
    Exemplo query: '"whale transfer" BTC lang:en' ou 'BTC lang:en'
    """
    rows = []
    if "snscrape" not in sys.modules:
        logging.warning("snscrape não disponível; pule coleta de tweets.")
        return pd.DataFrame(rows)
    for i, t in enumerate(sntwitter.TwitterSearchScraper(query).get_items()):
        if i >= max_tweets:
            break
        rows.append({
            "id": t.id,
            "date": t.date.to_pydatetime(),
            "user": t.user.username,
            "displayname": t.user.displayname,
            "followers": t.user.followersCount,
            "retweets": t.retweetCount,
            "content": t.content,
            "url": t.url
        })
    df = pd.DataFrame(rows)
    if not df.empty:
        fn = DATA_DIR / f"tweets_{datetime.utcnow().strftime('%Y%m%d%H%M%S')}.parquet"
        df.to_parquet(fn, index=False)
        logging.info("Tweets salvos em %s (rows=%d)", fn, len(df))
    else:
        logging.info("Nenhum tweet coletado para query: %s", query)
    return df

def extract_eth_addresses(text):
    # regex para endereços Ethereum (0x + 40 hex)
    return re.findall(r'0x[a-fA-F0-9]{40}', text or "")

# 2) Orderbook via CCXT (public)
def fetch_orderbook_snapshot(symbol=ASSET_SYMBOL, exchange_name="binance"):
    if "ccxt" not in sys.modules:
        logging.warning("ccxt não disponível; pule coleta de orderbook.")
        return None
    try:
        exc = getattr(ccxt, exchange_name)()
        ob = exc.fetch_order_book(symbol, limit=50)
        ts = datetime.utcnow()
        top_bid = ob['bids'][0] if ob['bids'] else [None, None]
        top_ask = ob['asks'][0] if ob['asks'] else [None, None]
        bid_depth = sum([b[1] for b in ob['bids'][:10]]) if ob['bids'] else 0.0
        ask_depth = sum([a[1] for a in ob['asks'][:10]]) if ob['asks'] else 0.0
        out = {
            "ts": ts,
            "exchange": exchange_name,
            "symbol": symbol,
            "bid": top_bid[0],
            "bid_size": top_bid[1],
            "ask": top_ask[0],
            "ask_size": top_ask[1],
            "bid_depth_top10": bid_depth,
            "ask_depth_top10": ask_depth
        }
        # persist
        df = pd.DataFrame([out])
        fn = DATA_DIR / f"orderbook_{exchange_name}_{symbol.replace('/','')}_{ts.strftime('%Y%m%d%H%M%S')}.parquet"
        df.to_parquet(fn, index=False)
        logging.info("Orderbook salvo em %s", fn)
        return out
    except Exception as e:
        logging.exception("Erro ao buscar orderbook: %s", e)
        return None

def orderbook_imbalance_ratio(snapshot):
    if not snapshot:
        return None
    b = snapshot.get("bid_depth_top10", 0.0) or 1e-9
    a = snapshot.get("ask_depth_top10", 0.0) or 1e-9
    if b >= a:
        return float(b / a)
    else:
        return float(a / b)

ETHERSCAN_API = "https://api.etherscan.io/api"
def get_etherscan_txs(address="", startblock=0, endblock=99999999, page=1, offset=100):
    if not ETHERSCAN_API_KEY:
        logging.info("ETHERSCAN_API_KEY não fornecido; pule coleta on-chain via Etherscan.")
        return pd.DataFrame()
    params = {
        "module": "account",
        "action": "txlist",
        "address": address,
        "startblock": startblock,
        "endblock": endblock,
        "page": page,
        "offset": offset,
        "sort": "desc",
        "apikey": ETHERSCAN_API_KEY
    }
    try:
        r = requests.get(ETHERSCAN_API, params=params, timeout=20)
        data = r.json()
        if data.get("status") == "1":
            df = pd.DataFrame(data["result"])
            if not df.empty:
                df['timeStamp'] = pd.to_datetime(df['timeStamp'].astype(int), unit='s')
                fn = DATA_DIR / f"etherscan_txs_{datetime.utcnow().strftime('%Y%m%d%H%M%S')}.parquet"
                df.to_parquet(fn, index=False)
                logging.info("Etherscan txs salvos em %s", fn)
            return df
        else:
            logging.info("Etherscan sem resultados: %s", data.get("message"))
            return pd.DataFrame()
    except Exception as e:
        logging.exception("Erro Etherscan: %s", e)
        return pd.DataFrame()

# Validator / Cross-validate
# --------------------
def is_large_onchain(tx_row):
    # heurística simples: procura por 'amount' ou 'value' e compara com threshold
    try:
        val = None
        if isinstance(tx_row, dict):
            if 'amount' in tx_row:
                val = float(tx_row.get('amount', 0))
            elif 'value' in tx_row:
                # WhaleAlert reports 'amount' in crypto units for many events; Etherscan reports 'value' in wei
                val = float(tx_row.get('value', 0))
            elif 'tokenInfo' in tx_row and 'amount' in tx_row['tokenInfo']:
                val = float(tx_row['tokenInfo']['amount'])
        else:
            # pandas Series
            if 'amount' in tx_row.index:
                val = float(tx_row['amount'])
            elif 'value' in tx_row.index:
                val = float(tx_row['value'])
        if val is None:
            return False
        # If value probably in wei (Etherscan), normalize if very large
        if val > 1e18:  # likely wei
            val = val / 1e18
        return val >= ONCHAIN_LARGE_BTC
    except Exception:
        return False

def tweet_credible(row):
    try:
        f = int(row.get("followers", 0))
        rts = int(row.get("retweets", 0))
        return (f >= TWEET_MIN_FOLLOWERS) or (rts >= TWEET_MIN_RETWEETS)
    except Exception:
        return False

def orderbook_implied_execution(snapshot):
    try:
        ratio = orderbook_imbalance_ratio(snapshot)
        if not ratio:
            return False, None
        return (ratio >= ORDERBOOK_IMBALANCE_RATIO), ratio
    except Exception:
        return False, None

def cross_validate_events(onchain_df, whale_df, tweets_df, orderbook_list):
    """
    Para cada onchain event, procura fontes corroborantes dentro de janela temporal.
    Requer pelo menos 2 fontes diferentes para validar (onchain + whale OR onchain + tweet credível, etc.)
    Retorna lista de dicts validados.
    """
    validated = []
    oc = onchain_df if (onchain_df is not None and not onchain_df.empty) else pd.DataFrame()
    wa = whale_df if (whale_df is not None and not whale_df.empty) else pd.DataFrame()
    tw = tweets_df if (tweets_df is not None and not tweets_df.empty) else pd.DataFrame()
    obs = orderbook_list or []

    window = timedelta(seconds=ONCHAIN_TIME_WINDOW_SEC)

    if oc.empty:
        logging.info("Sem eventos onchain para validar.")
        return validated

    # normalize tweet dates
    if not tw.empty and 'date' in tw.columns:
        tw = tw.copy()
        tw['date'] = pd.to_datetime(tw['date'])

    for idx, row in oc.iterrows():
        ts = None
        if 'timeStamp' in row:
            ts = pd.to_datetime(row['timeStamp'])
        elif 'timestamp' in row:
            ts = pd.to_datetime(row['timestamp'])
        else:
            continue
        sources = []
        score = 0.0
        # onchain large?
        if is_large_onchain(row):
            sources.append('onchain')
            score += CRED_WEIGHTS.get('onchain', 0)
        # whale alerts near ts
        near_wa = pd.DataFrame()
        if not wa.empty and 'timestamp' in wa.columns:
            near_wa = wa[(wa['timestamp'] >= (ts - window)) & (wa['timestamp'] <= (ts + window))]
            if not near_wa.empty:
                sources.append('whale_alert')
                score += CRED_WEIGHTS.get('whale_alert', 0)
        # credible tweets near ts
        credible_tweet_found = False
        near_tw = pd.DataFrame()
        if not tw.empty:
            near_tw = tw[(tw['date'] >= (ts - window)) & (tw['date'] <= (ts + window))]
            if not near_tw.empty:
                # any tweet credible?
                if any(near_tw.apply(tweet_credible, axis=1)):
                    credible_tweet_found = True
                    sources.append('tweet')
                    score += CRED_WEIGHTS.get('tweet', 0)
        # orderbook matches
        matched_ob = None
        if obs:
            for ob in obs:
                ob_ts = ob.get('ts')
                if ob_ts and abs((ob_ts - ts).total_seconds()) <= ONCHAIN_TIME_WINDOW_SEC:
                    implied, ratio = orderbook_implied_execution(ob)
                    if implied:
                        sources.append('orderbook')
                        score += CRED_WEIGHTS.get('orderbook', 0)
                        matched_ob = ob
                        break
        # require at least 2 distinct sources
        if len(set(sources)) >= 2:
            validated.append({
                "ts": ts.isoformat(),
                "onchain": row.to_dict() if hasattr(row, "to_dict") else dict(row),
                "matched_whalealerts": near_wa.to_dict('records') if not near_wa.empty else [],
                "matched_tweets": near_tw.to_dict('records') if not near_tw.empty else [],
                "matched_orderbook": matched_ob,
                "sources": list(set(sources)),
                "score": float(score)
            })
    logging.info("Cross-validation concluiu: %d eventos validados", len(validated))
    return validated

def persist_signal_to_db(strategy, ts_iso, payload, score):
    if engine is None:
        logging.warning("DB não inicializada; persisto sinal como JSON em data/")
        fn = DATA_DIR / f"signal_{strategy}_{datetime.utcnow().strftime('%Y%m%d%H%M%S')}.json"
        with open(fn, "w", encoding="utf-8") as f:
            json.dump({"strategy": strategy, "ts": ts_iso, "payload": payload, "score": score}, f, default=str, indent=2)
        logging.info("Sinal persistido em %s", fn)
        return
    meta = MetaData(bind=engine)
    signals = Table('signals', meta, autoload_with=engine)
    conn = engine.connect()
    ins = signals.insert().values(strategy=strategy, ts=pd.to_datetime(ts_iso), payload=payload, score=score)
    conn.execute(ins)
    conn.close()
    logging.info("Sinal persistido na DB: strategy=%s ts=%s score=%.3f", strategy, ts_iso, score)

# --------------------
# Backtest (simples)
# --------------------
def simple_backtest_on_signals(signal_times_iso):
    """
    Carrega close price historical (1h) para BTC-USD via yfinance e faz backtest simples:
    - Buy at next bar after signal
    - Hold fixed horizon (24h) or until end
    - Fees e slippage simplificados
    Usa vectorbt se instalado; caso contrário, usa simulação pandas.
    """
    if not BACKTEST_CAN_RUN:
        logging.warning("Dependências para backtest (yfinance) ausentes. Pule backtest.")
        return None, {}
    # carregar preço hourly
    try:
        logging.info("Baixando histórico via yfinance (BTC-USD 1h 365d)...")
        df = yf.download("BTC-USD", period="365d", interval="1h", progress=False)
        price = df['Close'].tz_localize(None)
    except Exception as e:
        logging.exception("Erro ao baixar preço: %s", e)
        return None, {}

    # construir series de sinais
    sig = pd.Series(0, index=price.index)
    for iso in signal_times_iso:
        try:
            t = pd.to_datetime(iso)
            idx = price.index.searchsorted(t)
            if idx < len(price):
                sig.iloc[idx] = 1
        except Exception:
            continue

    if vbt is not None:
        # vectorbt backtest
        logging.info("Rodando backtest com vectorbt (se disponível)...")
        pf = vbt.Portfolio.from_signals(price, entries=sig==1, exits=None, init_cash=BACKTEST_INITIAL_CAPITAL, fees=0.0005)
        stats = pf.stats().to_dict()
        # persist equity
        equity = pf.total_equity()
        fn = DATA_DIR / f"equity_curve_{datetime.utcnow().strftime('%Y%m%d%H%M%S')}.parquet"
        equity.reset_index().to_parquet(fn, index=False)
        logging.info("Equity curve salva em %s", fn)
        return pf, stats
    else:
        # fallback simples
        logging.info("Rodando backtest simplificado sem vectorbt...")
        cash = BACKTEST_INITIAL_CAPITAL
        pos = 0.0
        records = []
        for i in range(len(price)):
            if sig.iloc[i] == 1 and pos == 0:
                # buy with all cash at next bar price
                p = float(price.iloc[i])
                if p <= 0:
                    continue
                pos = cash / p
                cash = 0.0
                records.append({"idx": price.index[i], "action": "buy", "price": p, "pos": pos})
        # compute equity series
        equity = (price * pos).fillna(0) + cash
        fn = DATA_DIR / f"equity_curve_simple_{datetime.utcnow().strftime('%Y%m%d%H%M%S')}.parquet"
        equity.reset_index(name="equity").to_parquet(fn, index=False)
        logging.info("Equity simple salva em %s", fn)
        # basic stats
        total_return = (equity.iloc[-1] / equity.iloc[0] - 1) if equity.iloc[0] != 0 else None
        stats = {"total_return": total_return}
        return equity, stats

def run_pipeline_once():
    logging.info("Iniciando pipeline OpenMyMind - single run")
    # init DB
    init_db()

    # 1) tweets
    logging.info("Coletando tweets públicos (whale mentions)...")
    tweets_df = scrape_tweets('("whale" OR "whales" OR "large transfer" OR "large transfer") BTC lang:en', max_tweets=500)

    # 2) orderbook snapshot
    logging.info("Coletando snapshot do orderbook via CCXT...")
    ob = fetch_orderbook_snapshot(ASSET_SYMBOL, exchange_name="binance")
    obs_list = [ob] if ob else []

    # 3) on-chain (Etherscan) - se chave disponível (nota: Etherscan é ETH; para BTC use outra fonte)
    logging.info("Coletando transações on-chain (Etherscan) se chave presente...")
    onchain_df = get_etherscan_txs(address="", offset=100)  # você pode filtrar por address se desejar

    # 4) Whale Alert (opcional)
    logging.info("Coletando Whale Alert (se chave presente)...")
    wa_df = fetch_whale_alerts(min_value=500_000)

    # 5) validação cruzada
    logging.info("Validando eventos cruzando fontes (require >=2 sources)...")
    validated = cross_validate_events(onchain_df, wa_df, tweets_df, obs_list)

    # persist signals
    for v in validated:
        persist_signal_to_db("whale_onchain_cross", v["ts"], v, v["score"])

    # 6) backtest quick
    if validated:
        logging.info("Rodando backtest simples com os sinais validados...")
        signal_times = [v["ts"] for v in validated]
        pf, stats = simple_backtest_on_signals(signal_times)
        logging.info("Backtest stats: %s", json.dumps(stats, default=str, indent=2))
    else:
        logging.info("Nenhum evento validado para backtest.")

    logging.info("Pipeline finalizado.")

7) Notas importantes sobre validação "melhor entre as melhores"

O pipeline só considera um evento high-confidence se >= 2 fontes independentes confirmarem (on-chain + WhaleAlert, ou on-chain + tweet credível, ou on-chain + orderbook imbalance).

Ajuste ONCHAIN_LARGE_BTC, TWEET_MIN_FOLLOWERS, e ORDERBOOK_IMBALANCE_RATIO para elevar o filtro e reduzir ruído.

Para melhor qualidade adquira dados pagos (Glassnode, Kaiko, Coin Metrics) e integre no pipeline; isso melhora drastically sinais "high-confidence".

Sempre faça uma checagem humana final (human-in-the-loop) antes de executar ordens reais; este pipeline produz candidatos validados, não ordens automáticas.
