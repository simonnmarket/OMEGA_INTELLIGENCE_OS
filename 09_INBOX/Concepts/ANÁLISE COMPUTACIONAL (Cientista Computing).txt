import chromadb
import numpy as np
from datetime import datetime

class EpisodicMemorySystem:
    def __init__(self):
        self.client = chromadb.PersistentClient(path="./chroma_db")
        self.collection = self.client.get_or_create_collection(
            name="market_experiences",
            metadata={"hnsw:space": "cosine"}
        )
    
    def create_market_embedding(self, market_state):
        """Embedding vetorial do estado de mercado"""
        ohlcv_features = self.normalize_features(market_state['ohlcv'])
        orderbook_features = self.extract_orderbook_metrics(market_state['orderbook'])
        sentiment_features = self.get_sentiment_embedding(market_state['news'])
        
        # Concatenação final: 512 dimensões
        full_embedding = np.concatenate([
            ohlcv_features,      # 200 dim
            orderbook_features,  # 200 dim  
            sentiment_features   # 112 dim
        ])
        return full_embedding.tolist()
    
    def store_experience(self, market_state, action, result):
        """Armazenar experiência completa"""
        embedding = self.create_market_embedding(market_state)
        
        self.collection.add(
            embeddings=[embedding],
            documents=[f"action:{action}_result:{result}"],
            metadatas=[{
                "timestamp": datetime.utcnow().isoformat(),
                "regime": market_state['regime'],
                "assets": market_state['assets'],
                "pnl": result['pnl'],
                "var_95": result['risk_metrics']['var_95'],
                "action_type": action['type']
            }],
            ids=[f"exp_{datetime.utcnow().strftime('%Y%m%d%H%M%S%f')}"]
        )
    
    def retrieve_similar_experiences(self, current_state, k=5):
        """Recuperar experiências similares"""
        query_embedding = self.create_market_embedding(current_state)
        
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=k,
            include=["metadatas", "documents", "distances"]
        )
        return self.parse_retrieved_experiences(results)

        self.dilithium_private_key = None
        self.generate_key_pair()
    
    def generate_key_pair(self):
        # Kyber para criptografia
        self.kyber_private_key = kyber.KyberPrivateKey.generate()
        
        # Dilithium para assinatura
        self.dilithium_private_key = dilithium.DilithiumPrivateKey.generate()
    
    def encrypt_message(self, message):
        public_key = self.kyber_private_key.public_key()
        ciphertext, shared_secret = public_key.encrypt(message)
        
        # Derivação de chave para AES-GCM
        aes_key = self.derive_aes_key(shared_secret)
        
        return {
            'ciphertext': ciphertext,
            'aes_key': aes_key,
            'signature': self.sign_message(ciphertext)
        }
    
    def sign_message(self, message):
        return self.dilithium_private_key.sign(message)

class CognitiveBiasDetector:
    def __init__(self):
        self.decision_history = []
        self.bias_patterns = self.load_bias_patterns()
    
    def analyze_decision_quality(self, decision_context, chosen_action):
        bias_score = 0.0
        
        # Detecção de overconfidence
        if self.detect_overconfidence(decision_context):
            bias_score += 0.3
        
        # Detecção de loss aversion  
        if self.detect_loss_aversion(decision_context):
            bias_score += 0.25
            
        # Detecção de confirmation bias
        if self.detect_confirmation_bias(decision_context):
            bias_score += 0.2
            
        return {
            'bias_score': bias_score,
            'recommended_adjustment': self.calculate_bias_adjustment(bias_score),
            'confidence_multiplier': 1.0 - (bias_score * 0.5)
        }
    
    def detect_overconfidence(self, context):
        win_streak = context.get('consecutive_wins', 0)
        position_size = context.get('position_size', 0)
        avg_position = context.get('avg_position_size', 0)
        
        return position_size > (avg_position * (1 + win_streak * 0.1))
