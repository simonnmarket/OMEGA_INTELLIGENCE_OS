        for venue_name, venue_data in liquidity_data.items():

                    if pair in venue_data:

                            venue=venue,

    async def _detect_liquidity_opportunities_v7(self) -> List[LiquidityArbitrageOpportunity]:
        """
        DETECTA OPORTUNIDADES DE ARBITRAGEM DE LIQUIDEZ v7.0
        """
        opportunities = []
        current_time = time.time()
        
        for pair in self.target_pairs:
            if pair not in self.liquidity_map:
                continue
                
            venue_nodes = self.liquidity_map[pair]
            
            # Encontrar melhores preços de compra e venda
            best_ask = await self._find_best_ask_v7(venue_nodes)
            best_bid = await self._find_best_bid_v7(venue_nodes)
            
            if best_ask and best_bid and best_ask.venue != best_bid.venue:
                # Calcular lucro teórico
                theoretical_profit = best_bid.bid_price - best_ask.ask_price
                
                # Deduzir custos de transação
                total_costs = (
                    self.liquidity_venues[best_ask.venue]['cost_per_trade'] +
                    self.liquidity_venues[best_bid.venue]['cost_per_trade'] +
                    self.liquidity_params['cross_venue_cost']
                )
                
                net_profit = theoretical_profit - total_costs
                
                if net_profit > self.liquidity_params['min_profit_threshold']:
                    # Calcular métricas da oportunidade
                    execution_probability = await self._calculate_execution_probability_v7(
                        best_ask, best_bid
                    )
                    
                    liquidity_score = await self._calculate_opportunity_liquidity_score_v7(
                        best_ask, best_bid
                    )
                    
                    latency_advantage = await self._calculate_latency_advantage_v7(
                        best_ask, best_bid
                    )
                    
                    risk_adjusted_score = await self._calculate_risk_adjusted_score_v7(
                        net_profit, execution_probability, liquidity_score, latency_advantage
                    )
                    
                    # Verificar segurança
                    security_clearance = await self._security_check_liquidity_arbitrage_v7(
                        best_ask, best_bid
                    )
                    
                    opportunity = LiquidityArbitrageOpportunity(
                        buy_venue=best_ask.venue,
                        sell_venue=best_bid.venue,
                        pair=pair,
                        theoretical_profit=net_profit,
                        execution_probability=execution_probability,
                        liquidity_score=liquidity_score,
                        latency_advantage=latency_advantage,
                        risk_adjusted_score=risk_adjusted_score,
                        security_clearance=security_clearance
                    )
                    
                    opportunities.append(opportunity)
        
        return opportunities

    async def _filter_liquidity_opportunities_v7(self, opportunities: List[LiquidityArbitrageOpportunity]) -> List[LiquidityArbitrageOpportunity]:
        """
        FILTRA OPORTUNIDADES DE LIQUIDITY MINING v7.0
        """
        filtered = []
        
        for opportunity in opportunities:
            # Critérios de filtragem
            if (opportunity.theoretical_profit > self.liquidity_params['min_profit_threshold'] and
                opportunity.execution_probability > Decimal('0.70') and
                opportunity.liquidity_score > self.liquidity_params['min_liquidity_score'] and
                opportunity.latency_advantage < self.liquidity_params['max_latency_difference'] and
                opportunity.security_clearance):
                
                filtered.append(opportunity)
        
        # Ordenar por score ajustado ao risco
        filtered.sort(key=lambda x: x.risk_adjusted_score, reverse=True)
        return filtered

    async def _calculate_liquidity_position_size_v7(self, opportunity: LiquidityArbitrageOpportunity,
                                                  allocated_capital: Decimal) -> Decimal:
        """
        CALCULA TAMANHO DA POSIÇÃO PARA LIQUIDITY MINING v7.0
        """
        # Base size adaptativo para liquidity mining
        base_size = allocated_capital * Decimal('0.04')  # 4% base
        
        # Ajuste por profit opportunity
        profit_adjustment = min(
            Decimal('2.5'),
            opportunity.theoretical_profit * Decimal('15000')  # Multiplicador Forex
        )
        
        # Ajuste por liquidez
        liquidity_adjustment = opportunity.liquidity_score
        
        # Ajuste por probabilidade
        probability_adjustment = opportunity.execution_probability
        
        # Ajuste por latency advantage
        latency_adjustment = Decimal('1.5') - (opportunity.latency_advantage / Decimal('4.0'))
        
        adjusted_size = base_size * profit_adjustment * liquidity_adjustment * probability_adjustment * latency_adjustment
        
        # Limites de segurança
        max_size = allocated_capital * self.liquidity_params['max_position_size']
        min_size = Decimal('8000')  # Mínimo €8,000 para liquidity mining
        
        return max(min(adjusted_size, max_size), min_size)

    async def _calculate_liquidity_risk_score_v7(self, opportunity: LiquidityArbitrageOpportunity) -> Decimal:
        """
        CALCULA SCORE DE RISCO PARA LIQUIDITY MINING v7.0
        """
        # Risco base para arbitragem de liquidez
        base_risk = Decimal('0.4')
        
        # Ajustes baseados na oportunidade
        if opportunity.latency_advantage > Decimal('1.5'):
            base_risk += Decimal('0.1')  # Alta diferença de latency = mais risco
            
        if opportunity.liquidity_score < Decimal('0.75'):
            base_risk += Decimal('0.1')  # Baixa liquidez = mais risco
            
        return min(base_risk, Decimal('0.7'))

    # MÉTODOS AUXILIARES DE CÁLCULO
    async def _calculate_node_confidence_v7(self, venue: LiquidityVenue, pair_data: Dict) -> Decimal:
        """Calcula confiança do nó de liquidez"""
        venue_params = self.liquidity_venues[venue]
        
        factors = [
            venue_params['reliability'],
            min(Decimal('1.0'), pair_data.get('depth_quality', Decimal('1.0'))),
            Decimal('1.0') - (pair_data.get('spread', Decimal('0')) * Decimal('100'))
        ]
        
        return sum(factors) / Decimal(str(len(factors)))

    async def _find_best_ask_v7(self, venue_nodes: Dict[LiquidityVenue, LiquidityNodeV7]) -> Optional[LiquidityNodeV7]:
        """Encontra melhor preço de ask entre venues"""
        best_ask = None
        for node in venue_nodes.values():
            if (best_ask is None or node.ask_price < best_ask.ask_price) and node.confidence > Decimal('0.6'):
                best_ask = node
        return best_ask

    async def _find_best_bid_v7(self, venue_nodes: Dict[LiquidityVenue, LiquidityNodeV7]) -> Optional[LiquidityNodeV7]:
        """Encontra melhor preço de bid entre venues"""
        best_bid = None
        for node in venue_nodes.values():
            if (best_bid is None or node.bid_price > best_bid.bid_price) and node.confidence > Decimal('0.6'):
                best_bid = node
        return best_bid

    async def _calculate_execution_probability_v7(self, ask_node: LiquidityNodeV7, bid_node: LiquidityNodeV7) -> Decimal:
        """Calcula probabilidade de execução"""
        factors = [
            ask_node.confidence,
            bid_node.confidence,
            min(Decimal('1.0'), ask_node.ask_depth / Decimal('1000000')),
            min(Decimal('1.0'), bid_node.bid_depth / Decimal('1000000')),
            Decimal('1.0') - ((ask_node.latency + bid_node.latency) / Decimal('10.0'))
        ]
        return sum(factors) / Decimal(str(len(factors)))

    async def _calculate_opportunity_liquidity_score_v7(self, ask_node: LiquidityNodeV7, bid_node: LiquidityNodeV7) -> Decimal:
        """Calcula score de liquidez da oportunidade"""
        return (ask_node.confidence + bid_node.confidence) / Decimal('2')

    async def _calculate_latency_advantage_v7(self, ask_node: LiquidityNodeV7, bid_node: LiquidityNodeV7) -> Decimal:
        """Calcula vantagem de latency"""
        return abs(ask_node.latency - bid_node.latency)

    async def _calculate_risk_adjusted_score_v7(self, profit: Decimal, execution_prob: Decimal,
                                              liquidity_score: Decimal, latency_advantage: Decimal) -> Decimal:
        """Calcula score ajustado ao risco"""
        profit_factor = min(Decimal('2.0'), profit * Decimal('10000'))
        latency_factor = Decimal('1.5') - (latency_advantage / Decimal('4.0'))
        
        return (profit_factor * Decimal('0.3') +
                execution_prob * Decimal('0.3') +
                liquidity_score * Decimal('0.2') +
                latency_factor * Decimal('0.2'))

    async def _calculate_expected_duration_v7(self, opportunity: LiquidityArbitrageOpportunity) -> Decimal:
        """Calcula duração esperada do trade"""
        return (self.liquidity_venues[opportunity.buy_venue]['latency'] +
                self.liquidity_venues[opportunity.sell_venue]['latency'] +
                Decimal('0.5'))  # 500ms adicional

    async def _calculate_venue_costs_v7(self, opportunity: LiquidityArbitrageOpportunity) -> Decimal:
        """Calcula custos totais das venues"""
        return (self.liquidity_venues[opportunity.buy_venue]['cost_per_trade'] +
                self.liquidity_venues[opportunity.sell_venue]['cost_per_trade'] +
                self.liquidity_params['cross_venue_cost'])

    async def _get_depth_analysis_v7(self, opportunity: LiquidityArbitrageOpportunity) -> Dict[str, float]:
        """Obtém análise de depth da oportunidade"""
        buy_node = self.liquidity_map[opportunity.pair][opportunity.buy_venue]
        sell_node = self.liquidity_map[opportunity.pair][opportunity.sell_venue]
        
        return {
            'buy_venue_depth': float(buy_node.ask_depth),
            'sell_venue_depth': float(sell_node.bid_depth),
            'buy_venue_spread': float(buy_node.spread),
            'sell_venue_spread': float(sell_node.spread)
        }

    async def _security_check_liquidity_arbitrage_v7(self, ask_node: LiquidityNodeV7, bid_node: LiquidityNodeV7) -> bool:
        """Verificações de segurança para arbitragem de liquidez"""
        return (ask_node.venue_reliability > self.liquidity_params['min_venue_reliability'] and
                bid_node.venue_reliability > self.liquidity_params['min_venue_reliability'])
