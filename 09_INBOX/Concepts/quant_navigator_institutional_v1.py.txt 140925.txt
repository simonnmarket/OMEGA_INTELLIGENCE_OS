# === 1. INICIALIZAÇÃO INSTITUCIONAL ===
class SystemInitializer:
    def __init__(self):
        self.exchanges = {}
        self.redis = redis.Redis(host='redis', port=6379, decode_responses=True)
        self.zmq_context = zmq.Context()
        self.zmq_sub = self.zmq_context.socket(zmq.SUB)
        self.zmq_pub = self.zmq_context.socket(zmq.PUB)
        self.performance = {'latency': [], 'memory_mb': 0, 'cpu_percent': 0}

    async def connect_exchanges(self):
        configs = [
            ('binance', {'apiKey': 'YOUR_API_KEY', 'secret': 'YOUR_SECRET', 'enableRateLimit': True}),
            ('kraken', {'apiKey': 'YOUR_API_KEY', 'secret': 'YOUR_SECRET', 'enableRateLimit': True}),
        ]
        for name, config in configs:
            try:
                exchange = getattr(ccxt, name)(config)
                await exchange.load_markets()
                self.exchanges[name] = exchange
                self.redis.set(f'exchange_status:{name}', 'connected')
            except Exception as e:
                self.redis.set(f'exchange_status:{name}', f'failed: {str(e)}')
        return len(self.exchanges) > 0

    async def list_all_symbols(self, asset_type: Optional[str] = None, min_volume: float = 1_000_000) -> List[str]:
        all_symbols = []
        for exchange_name, exchange in self.exchanges.items():
            try:
                cache_key = f'symbols:{exchange_name}'
                cached = self.redis.get(cache_key)
                if cached:
                    symbols = eval(cached)
                else:
                    markets = await exchange.load_markets()
                    symbols = []
                    for symbol, market in markets.items():
                        if asset_type and market.get('type') != asset_type:
                            continue
                        if 'info' in market and 'quoteVolume' in market['info'] and float(market['info']['quoteVolume']) < min_volume:
                            continue
                        symbols.append(symbol)
                    self.redis.setex(cache_key, 3600, str(symbols))
                all_symbols.extend(symbols)
            except Exception as e:
                self.redis.set(f'symbols_error:{exchange_name}', str(e))
        return list(set(all_symbols))

    def monitor_performance(self):
        self.performance['memory_mb'] = psutil.Process().memory_info().rss / 1024 / 1024
        self.performance['cpu_percent'] = psutil.cpu_percent()
        self.redis.set('system_performance', str(self.performance))

# === 2. DETECTOR DE BIG PLAYERS ===
class BigPlayerDetector:
    def __init__(self, volume_period: int = 20, alert_level: float = 2.0):
        self.ml_model = joblib.load('bigplayer_model.joblib') if joblib.os.path.exists('bigplayer_model.joblib') else xgb.XGBClassifier()
        self.lstm_model = self._build_lstm()
        self.hmm_model = GaussianHMM(n_components=4, covariance_type='diag', n_iter=1000)
        self.volume_period = volume_period
        self.alert_level = alert_level
        self.swing_detector = InstitutionalSwingDetector()
        self.optimize_model()

    def _build_lstm(self):
        model = Sequential([
            LSTM(128, input_shape=(20, 8), return_sequences=False),
            Dense(64, activation='relu'),
            Dense(1, activation='sigmoid')
        ])
        model.compile(optimizer='adam', loss='binary_crossentropy')
        return model

    def optimize_model(self):
        def objective(trial):
            params = {
                'max_depth': trial.suggest_int('max_depth', 3, 12),
                'learning_rate': trial.suggest_float('learning_rate', 0.01, 0.3),
                'n_estimators': trial.suggest_int('n_estimators', 100, 1000)
            }
            self.ml_model = xgb.XGBClassifier(**params)
            X, y = np.random.rand(1000, 8), np.random.randint(0, 2, 1000)  # Substitua por tick data
            self.ml_model.fit(X, y)
            return self.ml_model.score(X, y)
        study = optuna.create_study(direction='maximize')
        study.optimize(objective, n_trials=50)
        self.ml_model = xgb.XGBClassifier(**study.best_params)
        joblib.dump(self.ml_model, 'bigplayer_model.joblib')

    @staticmethod
    @njit(parallel=True, fastmath=True)
    def calculate_volume_metrics(volumes: np.ndarray, period: int) -> Tuple[np.ndarray, np.ndarray]:
        n = len(volumes)
        zscores = np.zeros(n, dtype=np.float64)
        avg_volumes = np.zeros(n, dtype=np.float64)
        for i in prange(period, n):
            window = volumes[i-period:i]
            avg = np.mean(window)
            std = np.std(window)
            avg_volumes[i] = avg
            zscores[i] = (volumes[i] - avg) / std if std > 0 else 0.0
        return zscores, avg_volumes

    @staticmethod
    @njit(fastmath=True)
    def detect_institutional_bars(opens: np.ndarray, closes: np.ndarray, volumes: np.ndarray, zscores: np.ndarray, alert_level: float) -> np.ndarray:
        n = len(volumes)
        institutional_bars = np.zeros(n, dtype=np.bool_)
        for i in range(n):
            high_volume = zscores[i] >= alert_level
            delta = closes[i] - opens[i]
            positive_delta = delta > 0 and volumes[i] > 0
            institutional_bars[i] = high_volume and positive_delta
        return institutional_bars

    @njit
    def analyze_volume(self, volumes: np.ndarray, prices: np.ndarray) -> np.ndarray:
        if len(volumes) < 20:
            return np.array([1.0, 0.0, 0.0, 0.0])
        avg_vol = np.mean(volumes[-20:-1])
        volume_ratio = volumes[-1] / avg_vol if avg_vol > 0 else 1.0
        vwap = np.sum(prices * volumes) / np.sum(volumes) if np.sum(volumes) > 0 else prices[-1]
        vwap_deviation = (prices[-1] - vwap) / vwap if vwap > 0 else 0.0
        cumulative_delta = np.sum(np.where(prices[-20:] > prices[-21:-1], volumes[-20:], -volumes[-20:]))
        return np.array([volume_ratio, vwap_deviation, cumulative_delta, np.std(prices[-20:]) / prices[-1]])

    async def analyze_order_flow(self, exchange, symbol) -> Dict:
        try:
            orderbook = await exchange.fetch_order_book(symbol, limit=100)
            bids = sum([level[1] for level in orderbook['bids']])
            asks = sum([level[1] for level in orderbook['asks']])
            large_trades = sum([vol for price, vol in orderbook['bids'] + orderbook['asks'] if vol > 100])
            imbalance = bids / asks if asks > 0 else 1.0
            return {'imbalance': imbalance, 'large_trades': large_trades}
        except:
            return {'imbalance': 1.0, 'large_trades': 0.0}

    async def detect_big_players(self, df: pd.DataFrame, exchange, symbol) -> Dict:
        zscores, _ = self.calculate_volume_metrics(df['volume'].values, self.volume_period)
        institutional_bars = self.detect_institutional_bars(df['open'].values, df['close'].values, df['volume'].values, zscores, self.alert_level)
        df['institutional'] = institutional_bars
        swings = self.swing_detector.detect_swings(df)
        significant_swings = self.swing_detector.filter_significant_swings(swings)
        volume_features = self.analyze_volume(df['volume'].values, df['close'].values)
        order_flow = await self.analyze_order_flow(exchange, symbol)
        hurst = self._calculate_hurst(df['close'].values)
        swing_strength = sum(s.strength for s in significant_swings[-3:]) / 3 if significant_swings else 0.0
        features = np.concatenate([volume_features, [order_flow['imbalance'], order_flow['large_trades'], hurst, swing_strength, zscores[-1]])
        prob_xgb = self.ml_model.predict_proba(features.reshape(1, -1))[0, 1]
        lstm_input = np.array([features[-20:]]).reshape(1, 20, 8) if len(features) >= 20 else np.zeros((1, 20, 8))
        prob_lstm = self.lstm_model.predict(lstm_input, verbose=0)[0, 0]
        prob = (prob_xgb + prob_lstm) / 2
        return {
            'is_big_player': prob > 0.75,
            'confidence': prob,
            'volume_ratio': volume_features[0],
            'vwap_deviation': volume_features[1],
            'cumulative_delta': volume_features[2],
            'order_imbalance': order_flow['imbalance'],
            'large_trades': order_flow['large_trades'],
            'hurst': hurst,
            'swing_strength': swing_strength,
            'zscore': zscores[-1]
        }

    @njit
    def _calculate_hurst(self, prices: np.ndarray) -> float:
        if len(prices) < 100:
            return 0.5
        lags = np.arange(1, 20)
        variances = np.array([np.var(np.diff(prices, n=int(lag))) for lag in lags])
        log_vars = np.log(variances)
        log_lags = np.log(lags)
        return np.polyfit(log_lags, log_vars, 1)[0] / 2

# === 3. DETECTOR DE SWINGS INSTITUCIONAIS ===
class InstitutionalSwingDetector:
    def __init__(self, ticks_per_swing: int = 15, min_swing_candles: int = 3, ma_period: int = 20, volume_factor: float = 1.3):
        self.ticks_per_swing = ticks_per_swing
        self.min_swing_candles = min_swing_candles
        self.ma_period = ma_period
        self.volume_factor = volume_factor
        self.swing_history: List[MarketSwing] = []
        self.max_history_size = 1000

    def detect_swings(self, df: pd.DataFrame) -> List[MarketSwing]:
        swings = []
        highs = df['high'].values
        lows = df['low'].values
        closes = df['close'].values
        volumes = df['volume'].values
        institutional = df.get('institutional', np.zeros(len(df))).astype(bool)
        n = len(closes)
        swing_dir = 0
        swing_start_idx = 0
        swing_start_price = closes[0]
        swing_volume = 0.0
        swing_candle_count = 0
        institutional_volume = 0.0
        for i in range(1, n):
            price_delta = closes[i] - swing_start_price
            swing_volume += volumes[i]
            if institutional[i]:
                institutional_volume += volumes[i]
            if swing_dir == 0:
                swing_dir = 1 if price_delta >= 0 else -1
                swing_start_idx = i - 1
                swing_start_price = closes[i - 1]
                swing_volume = volumes[i]
                swing_candle_count = 1
                institutional_volume = volumes[i] if institutional[i] else 0.0
            else:
                swing_candle_count += 1
                reverse_condition = False
                if swing_dir == 1:
                    reverse_condition = price_delta <= -self.ticks_per_swing * 0.0001
                else:
                    reverse_condition = price_delta >= self.ticks_per_swing * 0.0001
                if reverse_condition and swing_candle_count >= self.min_swing_candles:
                    price_change_pct = abs(closes[i] - swing_start_price) / swing_start_price
                    institutional_participation = institutional_volume / swing_volume if swing_volume > 0 else 0
                    swing = MarketSwing(
                        start_index=swing_start_idx,
                        end_index=i,
                        start_price=swing_start_price,
                        end_price=closes[i],
                        volume=swing_volume,
                        is_uptrend=swing_dir == 1,
                        duration=swing_candle_count,
                        strength=price_change_pct * swing_volume,
                        institutional_participation=institutional_participation
                    )
                    swings.append(swing)
                    self._add_to_history(swing)
                    swing_dir = -swing_dir
                    swing_start_idx = i
                    swing_start_price = closes[i]
                    swing_volume = volumes[i]
                    swing_candle_count = 1
                    institutional_volume = volumes[i] if institutional[i] else 0.0
        return swings

    def _add_to_history(self, swing: MarketSwing):
        self.swing_history.append(swing)
        if len(self.swing_history) > self.max_history_size:
            self.swing_history.pop(0)

    def calculate_ma_volume(self) -> float:
        if not self.swing_history:
            return 0.0
        recent_swings = self.swing_history[-self.ma_period:]
        total_volume = sum(swing.volume for swing in recent_swings)
        return total_volume / len(recent_swings)

    def filter_significant_swings(self, swings: List[MarketSwing]) -> List[MarketSwing]:
        ma_volume = self.calculate_ma_volume()
        significant_swings = []
        for swing in swings:
            volume_ok = swing.volume >= ma_volume * self.volume_factor
            institutional_ok = swing.institutional_participation >= 0.3
            if volume_ok and institutional_ok:
                significant_swings.append(swing)
        return significant_swings

# === 4. GERADOR DE SINAIS ALPHA ===
class AlphaSignalGenerator:
    def __init__(self):
        self.flow_processor = BigPlayerDetector()
        self.swing_detector = InstitutionalSwingDetector()

    async def generate_signals(self, df: pd.DataFrame, symbol: str) -> List[AlphaSignal]:
        with ThreadPoolExecutor(max_workers=4) as executor:
            volume_task = executor.submit(
                self.flow_processor.calculate_volume_metrics,
                df['volume'].values, 20
            )
            swing_task = executor.submit(
                self.flow_processor.calculate_swing_strength,
                df['high'].values, df['low'].values, df['close'].values, df['volume'].values
            )
            zscores, _ = volume_task.result()
            swing_up, swing_down = swing_task.result()
        institutional_bars = self.flow_processor.detect_institutional_bars(
            df['open'].values, df['close'].values, df['volume'].values, zscores, 2.0
        )
        df = df.copy()
        df['zscore'] = zscores
        df['institutional'] = institutional_bars
        df['swing_up_strength'] = swing_up
        df['swing_down_strength'] = swing_down
        swings = self.swing_detector.detect_swings(df)
        significant_swings = self.swing_detector.filter_significant_swings(swings)
        signals = []
        n = len(df)
        for i in range(2, n):
            current_bar = df.iloc[i]
            prev_bar = df.iloc[i-1]
            two_bars_ago = df.iloc[i-2]
            if (current_bar['institutional'] and
                current_bar['close'] > current_bar['open'] and
                current_bar['close'] > prev_bar['high'] and
                current_bar['volume'] > prev_bar['volume'] * 1.5):
                signal = AlphaSignal(
                    timestamp=current_bar.name,
                    symbol=symbol,
                    signal_type="BREAKOUT",
                    direction=1,
                    confidence=min(0.95, current_bar['zscore'] / 5.0),
                    entry_price=current_bar['close'],
                    stop_loss=current_bar['low'],
                    take_profit=current_bar['close'] + (current_bar['close'] - current_bar['low']) * 2,
                    volume_confidence=current_bar['zscore'] / 3.0,
                    swing_confirmation=current_bar['swing_up_strength'] > 0,
                    institutional_flow=True
                )
                signals.append(signal)
            elif (current_bar['institutional'] and
                  prev_bar['institutional'] and
                  ((current_bar['close'] < current_bar['open'] and two_bars_ago['close'] > two_bars_ago['open']) or
                   (current_bar['close'] > current_bar['open'] and two_bars_ago['close'] < two_bars_ago['open']))):
                direction = 1 if current_bar['close'] > current_bar['open'] else -1
                signal = AlphaSignal(
                    timestamp=current_bar.name,
                    symbol=symbol,
                    signal_type="REVERSAL",
                    direction=direction,
                    confidence=0.7,
                    entry_price=current_bar['close'],
                    stop_loss=current_bar['low'] if direction == 1 else current_bar['high'],
                    take_profit=current_bar['close'] + (current_bar['close'] - current_bar['low']) * 1.5 * direction,
                    volume_confidence=(current_bar['zscore'] + prev_bar['zscore']) / 6.0,
                    swing_confirmation=True,
                    institutional_flow=True
                )
                signals.append(signal)
        return signals

# === 5. MARKET PROFILE ===
class MarketProfile:
    def calculate_profile(self, df: pd.DataFrame) -> Dict:
        prices, volumes = df['close'].values, df['volume'].values
        price_volume = {}
        for p, v in zip(prices, volumes):
            price_volume[p] = price_volume.get(p, 0) + v
        sorted_prices = sorted(price_volume.keys())
        total_volume = sum(price_volume.values())
        target_volume = total_volume * 0.7
        current_volume, value_prices = 0, []
        for p in sorted_prices:
            current_volume += price_volume[p]
            value_prices.append(p)
            if current_volume >= target_volume:
                break
        return {
            'poc': max(price_volume, key=price_volume.get),
            'value_area_low': min(value_prices),
            'value_area_high': max(value_prices)
        }

# === 6. RISK MANAGEMENT ===
class RiskManager:
    def __init__(self, max_risk: float = 0.01):
        self.max_risk = max_risk
        self.max_open_trades = 3

    def calculate_lots(self, equity: float, df: pd.DataFrame, sl_points: float) -> float:
        vol = df['close'].pct_change().std() * np.sqrt(252) if len(df) > 1 else 0.01
        cvar = self._compute_cvar(df['close'].pct_change().dropna())
        allocation = self.max_risk / vol
        lots = (equity * allocation) / (sl_points * df['close'].iloc[-1])
        return max(0.01, lots * (1 - abs(cvar)))

    def _compute_cvar(self, returns: pd.Series, alpha: float = 0.95) -> float:
        if len(returns) == 0:
            return 0.0
        var = np.percentile(returns, 100 * (1 - alpha))
        return returns[returns <= var].mean() if len(returns[returns <= var]) > 0 else var

# === 7. EXECUÇÃO E AUDIT ===
class ExecutionEngine:
    def __init__(self, initializer: SystemInitializer):
        self.exchanges = initializer.exchanges
        self.logger = AuditLogger()

    async def execute_order(self, symbol: str, side: str, lots: float) -> Dict:
        start = time.time_ns()
        for name, exchange in self.exchanges.items():
            try:
                order = await exchange.create_market_order(symbol, side, lots)
                latency = (time.time_ns() - start) / 1e9
                result = {'status': 'filled', 'order_id': order['id'], 'latency': latency}
                self.logger.log(result)
                return result
            except Exception as e:
                self.logger.log({'error': str(e), 'exchange': name})
        return {'status': 'failed'}

class AuditLogger:
    def __init__(self):
        self.chain = []
        self.redis = redis.Redis(host='redis', port=6379, decode_responses=True)

    def log(self, data: Dict):
        previous_hash = self.chain[-1]['hash'] if self.chain else 'genesis'
        block = {'data': data, 'timestamp': time.time()}
        block['hash'] = hashlib.sha256(f"{str(block)}{previous_hash}".encode()).hexdigest()
        self.chain.append(block)
        self.redis.lpush('audit_log', str(block))
        pd.DataFrame([block]).to_csv('audit_log.csv', mode='a', index=False)

# === 8. BACKTESTING ===
class Backtester:
    def backtest(self, df: pd.DataFrame, strategy_func) -> Dict:
        if df.empty:
            return {'sharpe': 0, 'max_dd': 0, 'total_return': 0}
        pf = vbt.Portfolio.from_signals(
            df['close'], entries=strategy_func(df) == 'BUY', exits=strategy_func(df) == 'SELL',
            freq='1min', init_cash=100000, fees=0.001, slippage=0.0005
        )
        return {
            'sharpe': pf.sharpe_ratio(),
            'max_dd': pf.max_drawdown(),
            'total_return': pf.total_return()
        }

class QuantNavigatorInstitutional:

        self.performance_metrics = {
            'signals_generated': 0,
            'signals_triggered': 0,
            'winning_signals': 0,
            'losing_signals': 0
        }

    async def initialize(self):
        await self.initializer.connect_exchanges()
        self.initializer.zmq_sub.connect("tcp://localhost:5555")
        self.initializer.zmq_sub.setsockopt_string(zmq.SUBSCRIBE, "")
        self.initializer.zmq_pub.bind("tcp://*:5556")
        threading.Thread(target=self._monitor, daemon=True).start()
        print("🚀 Quant Navigator Institutional v1.0 - Produção")

    async def fetch_data(self, symbol: str, timeframe: str = '1m', limit: int = 1000) -> pd.DataFrame:
        for exchange in self.initializer.exchanges.values():
            try:
                ohlcv = await exchange.fetch_ohlcv(symbol, timeframe, limit=limit)
                df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
                df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
                return df
            except:
                continue
        return pd.DataFrame()

    async def scan_big_players(self, symbols: List[str], asset_type: Optional[str] = None, min_volume: float = 1_000_000) -> Dict:
        if symbols == ['all']:
            symbols = await self.initializer.list_all_symbols(asset_type, min_volume)
        results = {}
        vix = self.macro.fetch_fred_data()
        batch_size = 100
        for i in range(0, len(symbols), batch_size):
            batch = symbols[i:i + batch_size]
            tasks = [self._process_symbol(symbol, vix) for symbol in batch]
            batch_results = await asyncio.gather(*tasks, return_exceptions=True)
            for symbol, result in zip(batch, batch_results):
                if isinstance(result, dict):
                    results[symbol] = result
                else:
                    results[symbol] = {'status': 'error', 'message': str(result)}
        return results

    async def _process_symbol(self, symbol: str, vix: float) -> Dict:
        df = await self.fetch_data(symbol)
        if df.empty:
            return {'status': 'no_data'}
        exchange = list(self.initializer.exchanges.values())[0]
        detection = await self.detector.detect_big_players(df, exchange, symbol)
        detection['vix_adjusted'] = detection['confidence'] * (1 if vix < 20 else 0.8)
        signals = await self.signal_generator.generate_signals(df, symbol)
        high_confidence_signals = [s for s in signals if s.confidence > 0.6 and s.volume_confidence > 0.5]
        self.performance_metrics['signals_generated'] += len(signals)
        self.performance_metrics['signals_triggered'] += len(high_confidence_signals)
        profile = self.profile.calculate_profile(df)
        lots = self.risk.calculate_lots(100000, df, 100)
        signal = "BUY" if detection['is_big_player'] and df['close'].iloc[-1] > profile['value_area_high'] else \
                 "SELL" if detection['is_big_player'] and df['close'].iloc[-1] < profile['value_area_low'] else "HOLD"
        if signal != "HOLD" and self.risk.max_open_trades > len(self._get_open_trades()):

def require_auth(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        token = request.headers.get('Authorization')
        if not token:
            return jsonify({'error': 'Token required'}), 401
        try:
            jwt.decode(token.replace('Bearer ', ''), JWT_SECRET, algorithms=['HS256'])
        except:
            return jsonify({'error': 'Invalid token'}), 401
        return f(*args, **kwargs)
    return decorated

@app.route('/scan_bigplayers', methods=['GET'])
@require_auth
async def scan_endpoint():
    symbols = request.args.get('symbols', 'BTC/USDT,ETH/USDT').split(',')
