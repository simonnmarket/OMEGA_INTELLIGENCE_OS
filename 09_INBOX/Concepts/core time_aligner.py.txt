    def _safe_shutdown(self):
        """Safe shutdown of MT5 connection"""
        if self.mt5_connected:
            try:
                mt5.shutdown()
                self.mt5_connected = False
                logger.info("MT5 connection safely shutdown")
            except Exception as e:
                logger.error(f"Error shutting down MT5: {e}")
    
    def _connect_mt5_with_retry(self) -> bool:
        """
        Connect to MT5 with exponential backoff retry logic.
        
        Returns:
            bool: True if connection successful, False otherwise
        """
        import time
        
        for attempt in range(self.max_mt5_retries + 1):
            try:
                if mt5.initialize():
                    self.mt5_connected = True
                    self.connection_attempts = 0
                    self.last_connection_time = datetime.now(timezone.utc)
                    
                    # Verify MT5 version
                    version = mt5.version()
                    logger.info(f"MT5 connected successfully | Version: {version} | Attempt: {attempt + 1}")
                    
                    # Test connection with simple operation
                    symbols = mt5.symbols_get()
                    if symbols:
                        logger.info(f"MT5 connection verified | Available symbols: {len(symbols)}")
                    
                    return True
                
            except Exception as e:
                logger.warning(f"MT5 connection attempt {attempt + 1} failed: {e}")
            
            # Exponential backoff
            if attempt < self.max_mt5_retries:
                wait_time = 2 ** attempt  # 1, 2, 4, 8 seconds
                time.sleep(wait_time)
        
        # All attempts failed
        self.mt5_connected = False
        self.connection_attempts = self.max_mt5_retries
        logger.error(f"MT5 connection failed after {self.max_mt5_retries} retries")
        
        return False
    
    def _ensure_mt5_connection(self) -> bool:
        """
        Ensure MT5 connection is active, reconnect if necessary.
        
        Returns:
            bool: True if connected, False otherwise
        """
        if not self.mt5_connected:
            logger.warning("MT5 not connected, attempting reconnection")
            return self._connect_mt5_with_retry()
        
        # Quick health check
        try:
            # Simple MT5 operation to verify connection
            _ = mt5.terminal_info()
            return True
        except Exception:
            logger.warning("MT5 connection appears stale, reconnecting")
            self.mt5_connected = False
            return self._connect_mt5_with_retry()

    def align_timeframes_safe(

            target_tf: Target timeframe for alignment
            max_wait_ms: Maximum wait time for candle close (not used in strict mode)

                - aligned_data: Aligned DataFrame (empty if failed)

        self.metrics['total_alignments'] += 1

        # PHASE 1: GENESIS LAYER 1 FAIL-SAFE CHECK
        # ========================================================================

                'status': 'GENESIS_FAIL_SAFE',
                'reason': safe_reason,
                'timestamp': start_time.isoformat(),
                'genesis_checksum': GENESIS.runtime_checksum
            }
        
        # ========================================================================
        # PHASE 2: SINGLE TIMESTAMP CAPTURE (CR-01 FIXED)
        # ========================================================================
        # CRITICAL: Single timestamp for ALL temporal calculations

        # PHASE 3: MT5 CONNECTION VERIFICATION
        # ========================================================================
        if not self._ensure_mt5_connection():
            self.metrics['failed_alignments'] += 1
            return False, pd.DataFrame(), {
                'status': 'MT5_CONNECTION_FAILED',
                'reason': 'MT5 connection unavailable',
                'timestamp': reference_time.isoformat(),
                'last_error': mt5.last_error() if hasattr(mt5, 'last_error') else 'Unknown'
            }

        # PHASE 4: TIMEFRAME MAPPING AND VALIDATION
        # ========================================================================
        try:
            tf_map = self._get_timeframe_mapping()
            src_config = tf_map[source_tf]
            tgt_config = tf_map[target_tf]
            
            src_minutes = src_config['minutes']
            tgt_minutes = tgt_config['minutes']
            src_mt5 = src_config['mt5']
            
        except KeyError as e:
            self.metrics['failed_alignments'] += 1
            return False, pd.DataFrame(), {
                'status': 'TIMEFRAME_ERROR',
                'reason': f'Invalid timeframe: {str(e)}',
                'timestamp': reference_time.isoformat(),
                'valid_timeframes': list(tf_map.keys())
            }
        except Exception as e:
            self.metrics['failed_alignments'] += 1
            return False, pd.DataFrame(), {
                'status': 'TIMEFRAME_MAPPING_ERROR',
                'reason': str(e),
                'timestamp': reference_time.isoformat()
            }
        
        # ========================================================================
        # PHASE 5: CANDLE CLOSE VERIFICATION (NO WAIT — STRICT NO-REPAINT)
        # ========================================================================
        next_candle_close = self._calculate_candle_close(reference_time, src_minutes)
        time_to_close = (next_candle_close - reference_time).total_seconds()
        
        # Mathematical Guarantee: Only use fully closed candles
        if time_to_close > 0.0:
            # Candle not yet closed — strict no-repaint enforcement
            self.metrics['failed_alignments'] += 1
            return False, pd.DataFrame(), {
                'status': 'CANDLE_NOT_CLOSED',
                'time_to_close_ms': time_to_close * 1000,
                'next_close_time': next_candle_close.isoformat(),
                'current_time': reference_time.isoformat(),
                'reason': f'Source candle ({source_tf}) not yet closed (closes in {time_to_close:.3f}s)',
                'timestamp': reference_time.isoformat()
            }

        # PHASE 6: FETCH CLOSED CANDLES ONLY
        # ========================================================================
        closed_data = self._fetch_closed_candles_only(symbol, src_mt5, src_minutes, reference_time)

            self.metrics['failed_alignments'] += 1
            return False, pd.DataFrame(), {
                'status': 'NO_CLOSED_DATA',
                'reason': 'No closed candles available',
                'timestamp': reference_time.isoformat(),
                'symbol': symbol,
                'timeframe': source_tf
            }
        
        # ========================================================================
        # PHASE 7: ALIGNMENT TO TARGET TIMEFRAME
        # ========================================================================

            # Update LRU cache
            self.alignment_validation[cache_key] = {
                'timestamp': reference_time,
                'validation': validation,
                'aligned_data_shape': aligned_data.shape
            }
            
            # Maintain max cache size
            if len(self.alignment_validation) > self.max_cache_size:
                self.alignment_validation.popitem(last=False)
            
            self.metrics['cache_misses'] += 1
        else:
            self.metrics['cache_hits'] += 1

        self.metrics['successful_alignments'] += 1
        self.metrics['average_latency_ms'] = (
            (self.metrics['average_latency_ms'] * (self.metrics['successful_alignments'] - 1) + latency_ms)

        )
        self.metrics['max_latency_ms'] = max(self.metrics['max_latency_ms'], latency_ms)
        
        # Add performance metrics to validation
        validation['performance'] = {
            'alignment_latency_ms': round(latency_ms, 2),
            'total_alignments': self.metrics['total_alignments'],
            'success_rate': round(self.metrics['successful_alignments'] / self.metrics['total_alignments'], 4),
            'cache_hits': self.metrics['cache_hits'],
            'cache_misses': self.metrics['cache_misses'],
            'cache_hit_ratio': round(self.metrics['cache_hits'] / max(1, self.metrics['cache_hits'] + self.metrics['cache_misses']), 4)
        }
        
        logger.debug(f"Alignment completed: {symbol} {source_tf}→{target_tf} | "
                    f"Latency: {latency_ms:.1f}ms | Validated: {validation.get('validated', False)}")
        
        return True, aligned_data, validation

    def _calculate_candle_close(self, reference_time: datetime, minutes: int) -> datetime:
        """
        Calculate exact candle close time with nanosecond precision.
        
        Mathematical Proof:
        Let t be reference_time, m be minutes
        Let s = seconds since epoch = int(t.timestamp())
        Let c = candle_duration_seconds = m * 60
        Then seconds_until_close = c - (s % c)
        If seconds_until_close == c, candle just closed (return t)
        Else return t + timedelta(seconds=seconds_until_close)
        
        Args:
            reference_time: Reference datetime (must be timezone-aware)
            minutes: Candle duration in minutes
        
        Returns:
            datetime: Exact candle close time
        """
        # Ensure UTC timezone
        if reference_time.tzinfo is None:
            reference_time = reference_time.replace(tzinfo=timezone.utc)
        elif reference_time.tzinfo != timezone.utc:
            reference_time = reference_time.astimezone(timezone.utc)
        
        seconds_since_epoch = int(reference_time.timestamp())
        candle_duration_seconds = minutes * 60
        
        # Seconds until next candle close
        seconds_until_close = candle_duration_seconds - (seconds_since_epoch % candle_duration_seconds)
        
        # Edge case: candle just closed (seconds_until_close == candle_duration_seconds)
        if seconds_until_close == candle_duration_seconds:
            return reference_time
        
        return reference_time + timedelta(seconds=seconds_until_close)
    
    def _fetch_closed_candles_only(
        self,
        symbol: str,
        mt5_tf: int,
        minutes: int,
        reference_time: datetime
    ) -> pd.DataFrame:
        """
        Fetch only fully closed candles (strict no-repaint).
        
        Mathematical Guarantee:
        1. Fetches n+1 candles
        2. Drops last candle (incomplete)
        3. Validates candle duration
        4. Ensures all candles closed before reference_time
        
        Args:
            symbol: Trading symbol
            mt5_tf: MT5 timeframe code
            minutes: Timeframe minutes for validation
            reference_time: Reference time for closure check
        
        Returns:
            pd.DataFrame: DataFrame with only closed candles
        """
        # Fetch more candles than needed
        n_candles = 101  # 100 closed + 1 current (to drop)
        
        try:
            rates = mt5.copy_rates_from_pos(symbol, mt5_tf, 1, n_candles)
            
            if rates is None or len(rates) < 2:
                logger.warning(f"No rates returned for {symbol} TF{mt5_tf}")
                return pd.DataFrame()
            
            # Convert to DataFrame
            df = pd.DataFrame(rates)
            
            # Convert timestamp to datetime with UTC
            df['time'] = pd.to_datetime(df['time'], unit='s', utc=True)
            df.set_index('time', inplace=True)
            
            # Drop the most recent candle (incomplete)
            df = df.iloc[:-1] if len(df) > 1 else df
            
            if df.empty:
                return pd.DataFrame()
            
            # ====================================================================
            # CRITICAL VALIDATION: Ensure all candles are fully closed
            # ====================================================================
            
            # 1. Calculate candle durations
            expected_seconds = minutes * 60
            time_diffs = df.index.to_series().diff().dt.total_seconds()
            
            # First candle has no previous, fill with expected
            time_diffs.iloc[0] = expected_seconds
            
            # 2. Filter valid candles (within ±10% tolerance)
            tolerance = 0.1  # 10% tolerance
            valid_mask = (
                (time_diffs >= expected_seconds * (1 - tolerance)) &
                (time_diffs <= expected_seconds * (1 + tolerance))
            )
            
            valid_candles = df[valid_mask].copy()
            
            # 3. Ensure all candles closed before reference_time
            # Last candle close time = candle open + duration
            last_candle_close = valid_candles.index[-1] + timedelta(seconds=expected_seconds)
            
            if last_candle_close > reference_time:
                logger.warning(f"Last candle closes after reference time: "
                             f"{last_candle_close} > {reference_time}")
                # Drop candles that close after reference
                mask = valid_candles.index + timedelta(seconds=expected_seconds) <= reference_time
                valid_candles = valid_candles[mask]
            
            # 4. Final integrity check
            if len(valid_candles) < 2:
                logger.warning(f"Insufficient valid candles for {symbol}: {len(valid_candles)}")
                return pd.DataFrame()
            
            # Add metadata
            valid_candles.attrs.update({
                'symbol': symbol,
                'timeframe_minutes': minutes,
                'n_candles': len(valid_candles),
                'first_candle': valid_candles.index[0].isoformat(),
                'last_candle': valid_candles.index[-1].isoformat(),
                'last_candle_close': (valid_candles.index[-1] + timedelta(seconds=expected_seconds)).isoformat(),
                'reference_time': reference_time.isoformat()
            })
            
            return valid_candles
            
        except Exception as e:
            logger.error(f"Error fetching candles for {symbol}: {e}")
            return pd.DataFrame()
    
    def _align_to_target(self, source_data: pd.DataFrame, target_minutes: int) -> pd.DataFrame:
        """
        Align higher timeframe data to lower execution timeframe.
        
        Mathematical Method:
        1. Forward fill with last observation carried forward (LOCF)
        2. Align to target timeframe grid
        3. Ensure no future data creation
        
        Args:
            source_data: Source DataFrame with closed candles
            target_minutes: Target timeframe in minutes
        
        Returns:
            pd.DataFrame: Aligned data on target timeframe
        """
        if source_data.empty:
            return pd.DataFrame()
        
        # Calculate source timeframe from data
        if len(source_data) > 1:
            source_freq = source_data.index.to_series().diff().median()
            source_minutes = source_freq.total_seconds() / 60
        else:
            source_minutes = 60  # Default assumption
        
        # If target is same or higher timeframe, return source
        if target_minutes >= source_minutes:
            return source_data.copy()
        
        # ========================================================================
        # ALIGNMENT: Create target timeframe grid
        # ========================================================================
        
        # Define alignment grid
        start_time = source_data.index[0].floor(f'{target_minutes}T')
        end_time = source_data.index[-1].ceil(f'{target_minutes}T')
        
        try:
            # Create target index
            target_index = pd.date_range(
                start=start_time,
                end=end_time,
                freq=f'{target_minutes}T',
                tz='UTC'
            )
            
            # Reindex with forward fill (LOCF)
            aligned_data = source_data.reindex(target_index, method='ffill')
            
            # Remove any rows that extend beyond source data
            max_valid_time = source_data.index[-1] + pd.Timedelta(minutes=source_minutes)
            aligned_data = aligned_data[aligned_data.index <= max_valid_time]
            
            # Ensure timezone consistency
            aligned_data.index = aligned_data.index.tz_convert('UTC')
            
            return aligned_data
            
        except Exception as e:
            logger.error(f"Alignment error: {e}")
            return pd.DataFrame()
    
    # ============================================================================

    # ============================================================================
    
    def _validate_no_repainting(
        self,
        symbol: str,
        source_tf: str,
        target_tf: str,
        aligned_data: pd.DataFrame,
        reference_time: datetime
    ) -> Dict[str, Any]:
        """
        Comprehensive statistical validation of no re-painting.
        
        Statistical Tests:
        1. Future Data Check: No data points after reference_time
        2. Ljung-Box Test: No significant autocorrelation (re-painting indicator)
        3. ADF Test: Stationarity verification
        4. Distribution Analysis: Return distribution properties
        
        Args:
            symbol: Trading symbol
            source_tf: Source timeframe
            target_tf: Target timeframe
            aligned_data: Aligned DataFrame to validate
            reference_time: Reference time for future data check
        
        Returns:
            Dict: Comprehensive validation results
        """
        validation = {
            'symbol': symbol,
            'source_tf': source_tf,
            'target_tf': target_tf,
            'validation_timestamp': reference_time.isoformat(),
            'genesis_checksum': GENESIS.runtime_checksum,
            'n_data_points': len(aligned_data),
            'data_range': {
                'start': aligned_data.index[0].isoformat() if not aligned_data.empty else None,
                'end': aligned_data.index[-1].isoformat() if not aligned_data.empty else None
            }
        }
        
        # ========================================================================
        # CHECK 1: FUTURE DATA DETECTION (CRITICAL)
        # ========================================================================
        future_mask = aligned_data.index > reference_time
        future_count = int(future_mask.sum())
        
        validation['future_data_check'] = {
            'future_data_points': future_count,
            'future_data_percentage': future_count / len(aligned_data) if len(aligned_data) > 0 else 0.0,
            'no_future_data': future_count == 0,
            'reference_time': reference_time.isoformat(),
            'latest_data_point': aligned_data.index[-1].isoformat() if not aligned_data.empty else None
        }
        
        # ========================================================================
        # CHECK 2: DATA SUFFICIENCY
        # ========================================================================
        if len(aligned_data) < 20:
            validation['data_sufficiency'] = {
                'sufficient': False,
                'n_points': len(aligned_data),
                'minimum_required': 20,
                'reason': 'Insufficient data for statistical tests'
            }
            validation['validated'] = validation['future_data_check']['no_future_data']
            return validation
        
        validation['data_sufficiency'] = {
            'sufficient': True,
            'n_points': len(aligned_data)
        }
        
        # ========================================================================
        # CHECK 3: RETURN CALCULATION (WITH ROBUST ERROR HANDLING)
        # ========================================================================
        try:
            # Calculate returns with robust error handling
            price_series = aligned_data['close'].astype(float)
            
            # Replace infinities with NaN
            price_series = price_series.replace([np.inf, -np.inf], np.nan)
            
            # Forward fill then backfill any remaining NaN
            price_series = price_series.ffill().bfill()
            
            # Calculate percentage returns
            returns = price_series.pct_change()
            
            # Remove NaN and infinite values
            returns = returns.replace([np.inf, -np.inf], np.nan).dropna()
            
            if len(returns) < 10:
                raise ValueError(f"Insufficient valid returns: {len(returns)}")
            
            validation['returns_analysis'] = {
                'n_valid_returns': len(returns),
                'return_mean': float(returns.mean()),
                'return_std': float(returns.std()),
                'return_skew': float(stats.skew(returns.dropna())),
                'return_kurtosis': float(stats.kurtosis(returns.dropna())),
                'shapiro_wilk_p': float(stats.shapiro(returns.dropna())[1]) if len(returns) <= 5000 else None
            }
            
        except Exception as e:
            validation['returns_error'] = str(e)
            validation['validated'] = validation['future_data_check']['no_future_data']
            return validation
        
        # ========================================================================
        # CHECK 4: LJUNG-BOX TEST FOR AUTOCORRELATION (RE-PAINTING DETECTION)
        # ========================================================================
        try:
            # Calculate ACF
            acf_vals = acf(returns, nlags=5, fft=True, missing='drop')
            
            # Ljung-Box test (exclude lag 0 which is always 1.0)
            if len(acf_vals) > 1 and len(returns) > 10:
                q_stat_result = q_stat(acf_vals[1:], len(returns))
                
                if isinstance(q_stat_result, tuple) and len(q_stat_result) >= 2:
                    q_stats, p_values = q_stat_result
                    
                    validation['ljung_box_test'] = {
                        'lag1_q_stat': float(q_stats[0]) if len(q_stats) > 0 else None,
                        'lag1_p_value': float(p_values[0]) if len(p_values) > 0 else 1.0,
                        'lag5_q_stat': float(q_stats[-1]) if len(q_stats) > 4 else None,
                        'lag5_p_value': float(p_values[-1]) if len(p_values) > 4 else 1.0,
                        'no_significant_autocorr': p_values[0] > 0.05 if len(p_values) > 0 else True,
                        'test_passed': p_values[0] > 0.05 if len(p_values) > 0 else True
                    }
                else:
                    validation['ljung_box_error'] = 'Unexpected q_stat result format'
                    validation['ljung_box_test'] = {'test_passed': True}  # Conservative
            else:
                validation['ljung_box_error'] = 'Insufficient data for Ljung-Box test'
                validation['ljung_box_test'] = {'test_passed': True}  # Conservative
                
        except Exception as e:
            validation['ljung_box_error'] = str(e)
            validation['ljung_box_test'] = {'test_passed': True}  # Conservative on error
        
        # ========================================================================
        # CHECK 5: AUGMENTED DICKEY-FULLER TEST FOR STATIONARITY
        # ========================================================================
        try:
            adf_result = adfuller(returns, autolag='AIC')
            
            validation['adf_test'] = {
                'adf_statistic': float(adf_result[0]),
                'p_value': float(adf_result[1]),
                'critical_values': {k: float(v) for k, v in adf_result[4].items()},
                'is_stationary': adf_result[1] < 0.05,
                'test_passed': adf_result[1] < 0.05
            }
            
        except Exception as e:
            validation['adf_error'] = str(e)
            validation['adf_test'] = {'test_passed': True}  # Conservative
        
        # ========================================================================
        # CHECK 6: DISTRIBUTION ANALYSIS
        # ========================================================================
        try:
            # Jarque-Bera test for normality
            jb_stat, jb_pvalue = stats.jarque_bera(returns)
            
            validation['distribution_analysis'] = {
                'jarque_bera_stat': float(jb_stat),
                'jarque_bera_p': float(jb_pvalue),
                'is_normal': jb_pvalue > 0.05,
                'mean_return': float(returns.mean()),
                'std_return': float(returns.std()),
                'sharpe_ratio': float(returns.mean() / returns.std() * np.sqrt(252)) if returns.std() > 0 else 0.0,
                'min_return': float(returns.min()),
                'max_return': float(returns.max())
            }
            
        except Exception as e:
            validation['distribution_error'] = str(e)
        
        # ========================================================================
        # FINAL VALIDATION DECISION
        # ========================================================================
        
        # Collect all test results
        test_results = [
            validation['future_data_check']['no_future_data'],
            validation.get('ljung_box_test', {}).get('test_passed', True),
            validation.get('adf_test', {}).get('test_passed', True)
        ]
        
        # Calculate overall score
        valid_tests = sum(test_results)
        total_tests = len(test_results)
        
        validation['overall_validation_score'] = valid_tests / total_tests
        validation['validated'] = validation['overall_validation_score'] >= 0.75
        
        validation['validation_summary'] = {
            'tests_performed': total_tests,
            'tests_passed': valid_tests,
            'success_rate': validation['overall_validation_score'],
            'meets_threshold': validation['validated'],
            'threshold': 0.75
        }
        
        return validation
