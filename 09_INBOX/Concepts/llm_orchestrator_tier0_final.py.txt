class GenesisLayer1:
    """Genesis Layer 1 para verificação de integridade e fail-safe"""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super(GenesisLayer1, cls).__new__(cls)
        return cls._instance
    
    def __init__(self):
        self.runtime_checksum = self._calculate_runtime_checksum()
        self.fail_safe_triggered = False
        self.start_time = datetime.now(timezone.utc)
        
    @staticmethod
    def _calculate_runtime_checksum() -> str:
        """Calcula checksum do ambiente de execução"""
        runtime_info = f"{sys.version}_{sys.platform}_{os.getpid()}_{datetime.now(timezone.utc).timestamp()}"
        return hashlib.sha3_256(runtime_info.encode()).hexdigest()[:16].upper()
    
    def fail_safe_check(self, metrics: Dict) -> Tuple[bool, str]:
        """Verificação de fail-safe baseada em métricas"""
        # Verificar latência
        if metrics.get('avg_latency', 0) > 10.0:  # > 10 segundos
            self.fail_safe_triggered = True
            return False, f"Latência muito alta: {metrics['avg_latency']:.1f}s"
        
        # Verificar sucesso
        if metrics.get('success_rate', 1.0) < 0.8:  # < 80%
            self.fail_safe_triggered = True
            return False, f"Taxa de sucesso muito baixa: {metrics['success_rate']:.1%}"
        
        # Verificar memória
        try:
            memory_percent = psutil.virtual_memory().percent
            if memory_percent > 90:
                return False, f"Uso de memória crítico: {memory_percent}%"
        except:
            pass
        
        return True, "Genesis Layer 1: Sistema estável"
    
    def execute_fail_safe(self, reason: str):
        """Executa procedimento de fail-safe"""
        logger.critical(f"FAIL-SAFE ATIVADO: {reason}")
        
        # 1. Fechar posições (simulação)
        logger.info("Fechando todas as posições abertas...")
        
        # 2. Notificar operador
        logger.info("Notificando operador via CometAPI...")
        
        # 3. Salvar estado
        logger.info("Salvando estado atual no backup...")
        
        # 4. Aguardar recuperação
        logger.info("Aguardando recuperação do sistema...")
        
        self.fail_safe_triggered = True

@dataclass
class VerificationEvidence:
    """Evidência de verificação conforme IEEE 1012"""
    evidence_type: str
    content: Any
    timestamp: str
    checksum: str
    source: str
    
    def validate_checksum(self) -> bool:
        """Valida integridade da evidência"""
        content_str = json.dumps(self.content, sort_keys=True)
        calculated = hashlib.sha3_256(content_str.encode()).hexdigest()
        return calculated == self.checksum
    
    def to_dict(self) -> Dict:
        """Converte para dicionário"""
        return {
            "evidence_type": self.evidence_type,
            "timestamp": self.timestamp,
            "checksum": self.checksum,
            "source": self.source,
            "content": self.content
        }

    async def _get_concurrent_votes(self, asset: str) -> List[AgentVote]:
        """
        Obtém votos de todos os agentes concorrentemente
        """
        tasks = []
        
        for agent, config in AGENTS_CONFIG.items():
            task = self._get_agent_vote_async(agent, config, asset)
            tasks.append(task)
        
        # Executar todas as tarefas concorrentemente
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # Processar resultados
        votes = []
        for i, result in enumerate(results):
            agent = list(AGENTS_CONFIG.keys())[i]
            
            if isinstance(result, Exception):
                logger.error(f"Agente {agent} falhou: {result}")
                # Criar voto de abstinência em caso de falha
                votes.append(self._create_abstain_vote(agent))
            elif result:
                votes.append(result)
        
        return votes
    
    async def _get_agent_vote_async(self, agent: str, config: Dict, asset: str) -> Optional[AgentVote]:
        """
        Obtém voto de um agente de forma assíncrona
        """
        start_time = time.time()
        
        try:
            # Gerar prompt especializado
            prompt = self._generate_agent_prompt(agent, config["role"], asset)
            
            # Verificar cache primeiro
            cache_key = hashlib.sha256(prompt.encode()).hexdigest()[:16]
            cached_response = self.cache["responses"].get(cache_key)
            
            if cached_response and (time.time() - cached_response["timestamp"]) < self.cache["ttl"]:
                logger.debug(f"Cache hit para {agent}")
                response_text = cached_response["response"]
                provider = "cache"
            else:
                # Chamar LLM com fallback
                response_text, provider = await self._call_llm_with_fallback(
                    agent=agent,
                    model=config["model"],
                    prompt=prompt,
                    timeout=config["timeout"]
                )
                
                # Atualizar cache
                self.cache["responses"][cache_key] = {
                    "response": response_text,
                    "timestamp": time.time(),
                    "agent": agent
                }
            
            # Parsear resposta
            vote_data = self._parse_llm_response(response_text, agent, config["role"])
            
            # Calcular tempo de resposta
            response_time = time.time() - start_time
            
            # Criar objeto de voto
            vote = AgentVote(
                agent=agent,
                decision=vote_data["decision"],
                confidence=vote_data["confidence"],
                reason=vote_data["reason"],
                weight=config["weight"],
                model_used=config["model"],
                provider=provider,
                response_time=response_time,
                timestamp=datetime.now(timezone.utc).isoformat()
            )
            
            return vote
            
        except Exception as e:
            logger.error(f"Erro no agente {agent}: {e}")
            raise

                if provider_config["name"] == "ollama":

                        timeout=timeout

                        model=provider_config["model"],

                return response, provider_config["name"]
                
            except Exception as e:
                logger.debug(f"Provider {provider_name} falhou: {e}")
                continue

        raise ConnectionError("Todos os provedores LLM falharam")

        json_start = cleaned_text.find('{')
        json_end = cleaned_text.rfind('}') + 1
        
        if json_start != -1 and json_end > json_start:
            try:
                json_str = cleaned_text[json_start:json_end]
                data = json.loads(json_str)
                
                decision = data.get("decision", "").upper()
                if decision not in ["BULLISH", "BEARISH", "NEUTRAL", "ABSTAIN"]:
                    decision = "ABSTAIN"
                
                confidence = float(data.get("confidence", 0.5))
                confidence = max(0.0, min(1.0, confidence))  # Clamp 0-1
                
                reason = str(data.get("reason", f"Análise {role}")).strip()
                reason = reason[:200]  # Limitar tamanho
                
                return {
                    "decision": decision,
                    "confidence": confidence,
                    "reason": reason
                }
                
            except (json.JSONDecodeError, ValueError) as e:
                logger.warning(f"JSON parse error for {agent}: {e}")
        
        # Fallback: análise heurística do texto
        return self._parse_heuristic(cleaned_text, agent)
    
    def _parse_heuristic(self, text: str, agent: str) -> Dict:
        """Parse heurístico se JSON falhar"""
        text_upper = text.upper()
        
        if "BULLISH" in text_upper or "COMPRA" in text_upper or "BUY" in text_upper:
            decision = "BULLISH"
        elif "BEARISH" in text_upper or "VENDA" in text_upper or "SELL" in text_upper:
            decision = "BEARISH"
        elif "NEUTRAL" in text_upper or "NEUTRO" in text_upper:
            decision = "NEUTRAL"
        else:
            decision = "ABSTAIN"
        
        # Tentar extrair confiança do texto
        confidence = 0.5
        import re
        conf_match = re.search(r'(\d+(?:\.\d+)?)%', text)
        if conf_match:
            try:
                confidence = float(conf_match.group(1)) / 100.0
            except:
                pass
        
        return {
            "decision": decision,
            "confidence": confidence,
            "reason": f"Parse heurístico para {agent}"
        }
    
    def _create_abstain_vote(self, agent: str) -> AgentVote:
        """Cria voto de abstinência para falhas"""
        return AgentVote(
            agent=agent,
            decision="ABSTAIN",
            confidence=0.0,
            reason="Falha na análise",
            weight=AGENTS_CONFIG.get(agent, {}).get("weight", 1.0),
            model_used="N/A",
            provider="error",
            response_time=0.0,
            timestamp=datetime.now(timezone.utc).isoformat()
        )
    
    def _calculate_weighted_consensus(self, votes: List[AgentVote]) -> Dict:
        """Calcula consenso ponderado"""
        if not votes:
            return {"decision": "ABSTAIN", "confidence": 0.0, "votes": 0}
        
        # Filtrar votos válidos
        valid_votes = [v for v in votes if v.decision != "ABSTAIN"]
        
        if not valid_votes:
            return {"decision": "ABSTAIN", "confidence": 0.0, "votes": 0}
        
        # Calcular scores ponderados
        scores = {"BULLISH": 0.0, "BEARISH": 0.0, "NEUTRAL": 0.0}
        total_weight = 0.0
        
        for vote in valid_votes:
            weight = vote.weight * vote.confidence
            
            if vote.decision == "BULLISH":
                scores["BULLISH"] += weight
            elif vote.decision == "BEARISH":
                scores["BEARISH"] += weight
            elif vote.decision == "NEUTRAL":
                scores["NEUTRAL"] += weight
            
            total_weight += weight
        
        if total_weight == 0:
            return {"decision": "ABSTAIN", "confidence": 0.0, "votes": len(valid_votes)}
        
        # Normalizar scores
        for key in scores:
            scores[key] /= total_weight
        
        # Determinar decisão
        if scores["BULLISH"] >= 0.6:  # 60%+ para Bullish
            decision = "BULLISH"
            confidence = scores["BULLISH"]
        elif scores["BEARISH"] >= 0.6:  # 60%+ para Bearish
            decision = "BEARISH"
            confidence = scores["BEARISH"]
        elif abs(scores["BULLISH"] - scores["BEARISH"]) < 0.2:  # Diferença < 20%
            decision = "NEUTRAL"
            confidence = 1.0 - abs(scores["BULLISH"] - scores["BEARISH"])
        elif scores["BULLISH"] > scores["BEARISH"]:
            decision = "BULLISH"
            confidence = scores["BULLISH"] / (scores["BULLISH"] + scores["BEARISH"])
        else:
            decision = "BEARISH"
            confidence = scores["BEARISH"] / (scores["BULLISH"] + scores["BEARISH"])
        
        return {
            "decision": decision,
            "confidence": float(confidence),
            "votes": len(valid_votes),
            "score_breakdown": scores
        }
    
    def _create_verification_evidence(self, asset: str, votes: List[AgentVote], 
                                    consensus: Dict, cycle_time: float) -> VerificationEvidence:
        """Cria evidência de verificação IEEE 1012"""
        
        content = {
            "asset": asset,
            "consensus": consensus,
            "cycle_time": cycle_time,
            "agent_count": len(votes),
            "valid_votes": len([v for v in votes if v.decision != "ABSTAIN"]),
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "genesis_checksum": GENESIS.runtime_checksum
        }
        
        checksum = hashlib.sha3_256(
            json.dumps(content, sort_keys=True).encode()
        ).hexdigest()
        
        return VerificationEvidence(
            evidence_type="LLM_Cycle_Analysis_TIER0",
            content=content,
            timestamp=datetime.now(timezone.utc).isoformat(),
            checksum=checksum,
            source="aurora_llm_orchestrator"
        )
    
    def _update_metrics(self, result: CycleResult):
        """Atualiza métricas do sistema"""
        self.metrics["total_cycles"] += 1
        
        if result.consensus_decision != "ABSTAIN":
            self.metrics["successful_cycles"] += 1
        else:
            self.metrics["failed_cycles"] += 1
        
        # Calcular taxa de sucesso
        self.metrics["success_rate"] = (
            self.metrics["successful_cycles"] / max(self.metrics["total_cycles"], 1)
        )
        
        # Atualizar tempo médio de resposta
        if result.agent_votes:
            avg_response = np.mean([v.response_time for v in result.agent_votes])
            self.metrics["avg_response_time"] = (
                self.metrics["avg_response_time"] * 0.8 + avg_response * 0.2
            )
    
    def _prioritize_assets(self) -> List[str]:
        """Retorna lista de ativos prioritizada"""
        # Ordenar por prioridade e volatilidade
        prioritized = sorted(
            ASSETS_CONFIG.items(),
            key=lambda x: (x[1]["priority"].value, -x[1]["volatility"])
        )
        return [asset for asset, _ in prioritized]
    
    async def run_full_scan(self):
        """Executa varredura completa de todos os ativos"""
        logger.info("=" * 70)

        logger.info("=" * 70)
        
        start_time = time.time()
        all_results = []
        
        # Obter ativos prioritizados
        prioritized_assets = self._prioritize_assets()
        
        for i, asset in enumerate(prioritized_assets):
            logger.info(f"\n📊 [{i+1}/{len(prioritized_assets)}] Analisando {asset}...")
            
            try:
                result = await self.run_cycle(asset)
                all_results.append(result)
                
                # Calcular métricas intermediárias
                elapsed = time.time() - start_time
                avg_time = elapsed / (i + 1)
                remaining = len(prioritized_assets) - (i + 1)
                eta = avg_time * remaining
                
                logger.info(f"   Progresso: {i+1}/{len(prioritized_assets)}")
                logger.info(f"   Tempo decorrido: {elapsed:.1f}s")
                logger.info(f"   ETA: {eta:.1f}s ({eta/60:.1f} min)")
                
                # Intervalo entre ciclos
                if i < len(prioritized_assets) - 1:
                    wait_time = max(0, CYCLE_CONFIG["cycle_interval"] - result.cycle_time)
                    if wait_time > 0:
                        logger.info(f"   ⏳ Aguardando {wait_time:.1f}s...")
                        await asyncio.sleep(wait_time)
                        
            except Exception as e:
                logger.error(f"❌ Erro analisando {asset}: {e}")
                continue
        
        # Calcular estatísticas finais
        total_time = time.time() - start_time
        successful = len([r for r in all_results if r.consensus_decision != "ABSTAIN"])
        
        logger.info("\n" + "=" * 70)

        logger.info(f"   Ciclos completos: {len(all_results)}")
        logger.info(f"   Análises bem-sucedidas: {successful}")
        logger.info(f"   Taxa de sucesso: {successful/max(len(all_results),1):.1%}")
        logger.info(f"   Dentro do alvo? {'✅ SIM' if total_time <= CYCLE_CONFIG['full_scan_target'] else '❌ NÃO'}")

        return {

        }

class HealthMonitor:
    """Monitor de saúde do sistema"""
    
    def __init__(self):
        self.metrics_history = []
        self.alert_thresholds = {
            "success_rate": 0.8,
            "avg_latency": 10.0,
            "memory_usage": 0.9
        }
    
    def check_system_health(self) -> Dict:
        """Verifica saúde do sistema"""
        health = {
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "checks": {}
        }
        
        # Verificar memória
        try:
            memory = psutil.virtual_memory()
            health["checks"]["memory"] = {
                "percent": memory.percent,
                "status": "OK" if memory.percent < 90 else "WARNING"
            }
        except:
            health["checks"]["memory"] = {"status": "UNKNOWN"}
        
        # Verificar CPU
        try:
            cpu = psutil.cpu_percent(interval=0.1)
            health["checks"]["cpu"] = {
                "percent": cpu,
                "status": "OK" if cpu < 80 else "WARNING"
            }
        except:
            health["checks"]["cpu"] = {"status": "UNKNOWN"}
        
        # Verificar disco
        try:
            disk = psutil.disk_usage('/')
            health["checks"]["disk"] = {
                "free_gb": disk.free / (1024**3),
                "status": "OK" if disk.percent < 90 else "WARNING"
            }
        except:
            health["checks"]["disk"] = {"status": "UNKNOWN"}
        
        # Determinar status geral
        all_ok = all(check["status"] == "OK" for check in health["checks"].values())
        health["overall_status"] = "HEALTHY" if all_ok else "DEGRADED"
        
        return health
