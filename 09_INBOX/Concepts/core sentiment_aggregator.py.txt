@staticmethod
def _calculate_runtime_checksum() -> str:
    """Calculate SHA3-256 checksum of runtime environment"""
    import sys
    import platform
    import os
    
    runtime_info = f"{platform.python_version()}_{sys.platform}_{os.getpid()}_{datetime.now(timezone.utc).timestamp()}"
    return hashlib.sha3_256(runtime_info.encode()).hexdigest()[:16].upper()

def _create_lexicon_model(self) -> Any:
    """
    Create lexicon-based sentiment model as fallback.
    """
    class LexiconSentimentModel:
        def __init__(self):
            # Financial sentiment lexicon
            self.positive_words = {
                'bullish', 'rally', 'surge', 'gain', 'profit', 'growth', 'strong',
                'positive', 'optimistic', 'buy', 'long', 'outperform', 'beat',
                'exceed', 'upgrade', 'target', 'recommend', 'strong buy', 'bull',
                'breakout', 'momentum', 'support', 'resistance broken', 'uptrend',
                'accumulation', 'oversold bounce', 'earning surprise', 'guidance raise',
                'dividend increase', 'share buyback', 'merger', 'acquisition'
            }
            
            self.negative_words = {
                'bearish', 'plunge', 'drop', 'loss', 'decline', 'weak', 'negative',
                'pessimistic', 'sell', 'short', 'underperform', 'miss', 'below',
                'downgrade', 'cut', 'reduce', 'sell', 'risk', 'bear', 'breakdown',
                'distribution', 'overbought', 'downtrend', 'resistance holds',
                'support broken', 'earning miss', 'guidance cut', 'dividend cut',
                'layoff', 'bankruptcy', 'fraud', 'investigation', 'lawsuit'
            }
            
            self.intensifiers = {
                'very': 1.5, 'extremely': 2.0, 'highly': 1.8, 'strongly': 1.7,
                'slightly': 0.7, 'somewhat': 0.8, 'moderately': 1.0,
                'significantly': 1.6, 'dramatically': 1.9, 'substantially': 1.7
            }
            
            self.negators = {'not', 'no', 'never', 'none', 'neither', 'nor'}
            
            # Market-specific phrases
            self.bullish_phrases = {
                'beats estimates', 'raises guidance', 'strong earnings',
                'positive outlook', 'buy rating', 'upgraded to buy',
                'price target increased', 'dividend hike', 'share repurchase',
                'merger approved', 'regulatory approval', 'clinical trial success'
            }
            
            self.bearish_phrases = {
                'misses estimates', 'cuts guidance', 'weak earnings',
                'negative outlook', 'sell rating', 'downgraded to sell',
                'price target cut', 'dividend cut', 'layoffs announced',
                'merger blocked', 'regulatory issues', 'clinical trial failure'
            }
        
        def analyze(self, text: str) -> Dict[str, Any]:
            """
            Analyze sentiment using lexicon-based approach.
            
            Returns:
                Dict: Sentiment analysis results
            """
            text_lower = text.lower()
            
            # Check for phrases first
            phrase_score = 0
            for phrase in self.bullish_phrases:
                if phrase in text_lower:
                    phrase_score += 1.0
            
            for phrase in self.bearish_phrases:
                if phrase in text_lower:
                    phrase_score -= 1.0
            
            # Tokenize
            tokens = re.findall(r'\b\w+\b', text_lower)
            
            # Calculate word scores
            positive_score = 0
            negative_score = 0
            
            for i, token in enumerate(tokens):
                intensity = 1.0
                
                # Check for intensifiers
                if i > 0 and tokens[i-1] in self.intensifiers:
                    intensity = self.intensifiers[tokens[i-1]]
                
                # Check for negators
                is_negated = False
                if i > 0 and tokens[i-1] in self.negators:
                    is_negated = True
                
                if token in self.positive_words:
                    if is_negated:
                        negative_score += intensity
                    else:
                        positive_score += intensity
                
                if token in self.negative_words:
                    if is_negated:
                        positive_score += intensity
                    else:
                        negative_score += intensity
            
            # Combine scores
            total_score = positive_score - negative_score + phrase_score
            
            # Normalize
            total_tokens = len(tokens) + len(self.bullish_phrases) + len(self.bearish_phrases)
            normalized = total_score / max(total_tokens, 1)
            
            # Clamp to [-1, 1]
            normalized = max(-1.0, min(1.0, normalized))
            
            # Calculate confidence
            total_matches = positive_score + negative_score + abs(phrase_score)
            confidence = min(0.8, total_matches / max(len(tokens), 1))
            
            # Determine label
            if normalized > 0.1:
                label = 'POSITIVE'
            elif normalized < -0.1:
                label = 'NEGATIVE'
            else:
                label = 'NEUTRAL'
            
            return {
                'label': label,
                'score': abs(normalized),
                'normalized_sentiment': normalized,
                'confidence': confidence,
                'positive_score': positive_score,
                'negative_score': negative_score,
                'phrase_score': phrase_score
            }
    
    return LexiconSentimentModel()

def _create_ensemble_model(self, models: Dict[str, Any]) -> Any:
    """
    Create ensemble model combining multiple sentiment models.
    """
    class EnsembleModel:
        def __init__(self, models, weights):
            self.models = models
            self.weights = weights
        
        def analyze(self, text: str) -> Dict[str, Any]:
            results = []
            weights_used = []
            
            for model_name, model in self.models.items():
                if model_name == SentimentModel.ENSEMBLE.value:
                    continue
                
                try:
                    if hasattr(model, 'analyze'):
                        result = model.analyze(text)
                    elif hasattr(model, '__call__'):
                        result = model(text[:512], truncation=True)
                        # Convert to standard format
                        if isinstance(result, list):
                            result = self._convert_transformers_result(result)
                    else:
                        continue
                    
                    results.append(result)
                    weights_used.append(self.weights.get(model_name, 0.1))
                except Exception as e:
                    logger.debug(f"Model {model_name} failed: {e}")
                    continue
            
            if not results:
                return {
                    'label': 'NEUTRAL',
                    'score': 0.0,
                    'normalized_sentiment': 0.0,
                    'confidence': 0.0,
                    'ensemble_size': 0
                }
            
            # Weighted aggregation
            total_sentiment = 0.0
            total_confidence = 0.0
            total_weight = sum(weights_used)
            
            for result, weight in zip(results, weights_used):
                sentiment = result.get('normalized_sentiment', 
                                      result.get('score', 0.0) * (1 if result.get('label', '').upper() == 'POSITIVE' else -1))
                
                confidence = result.get('confidence', 0.5)
                
                total_sentiment += sentiment * weight * confidence
                total_confidence += confidence * weight
            
            # Normalize
            if total_weight > 0 and total_confidence > 0:
                final_sentiment = total_sentiment / total_confidence
                final_confidence = total_confidence / total_weight
            else:
                final_sentiment = 0.0
                final_confidence = 0.0
            
            # Clamp
            final_sentiment = max(-1.0, min(1.0, final_sentiment))
            
            # Determine label
            if final_sentiment > 0.1:
                label = 'POSITIVE'
            elif final_sentiment < -0.1:
                label = 'NEGATIVE'
            else:
                label = 'NEUTRAL'
            
            return {
                'label': label,
                'score': abs(final_sentiment),
                'normalized_sentiment': final_sentiment,
                'confidence': final_confidence,
                'ensemble_size': len(results),
                'model_contributions': weights_used
            }
        
        def _convert_transformers_result(self, result: List[Dict]) -> Dict[str, Any]:
            """Convert transformers pipeline result to standard format"""
            if not result:
                return {'label': 'NEUTRAL', 'score': 0.0, 'normalized_sentiment': 0.0, 'confidence': 0.0}
            
            # Find positive and negative scores
            pos_score = 0.0
            neg_score = 0.0
            
            for item in result:
                label = item.get('label', '').upper()
                score = item.get('score', 0.0)
                
                if 'POS' in label or label == 'POSITIVE':
                    pos_score = score
                elif 'NEG' in label or label == 'NEGATIVE':
                    neg_score = score
                elif label == 'NEUTRAL':
                    # Neutral contributes to both
                    pos_score += score * 0.1
                    neg_score += score * 0.1
            
            # Calculate normalized sentiment
            normalized = pos_score - neg_score
            
            # Determine label
            if normalized > 0.1:
                label = 'POSITIVE'
            elif normalized < -0.1:
                label = 'NEGATIVE'
            else:
                label = 'NEUTRAL'
            
            return {
                'label': label,
                'score': abs(normalized),
                'normalized_sentiment': normalized,
                'confidence': max(pos_score, neg_score)
            }
    
    return EnsembleModel(models, self.model_weights)

    """

def analyze_sentiment(
    self,
    text: str,
    source: str = 'news',
    asset: str = None,
    model_preference: str = 'auto',
    use_cache: bool = True,
    max_length: int = 512
) -> Dict[str, Any]:
    """
    Analyze sentiment with institutional-grade validation.
    
    Args:
        text: Text to analyze
        source: Data source (news, twitter, reddit, etc.)
        asset: Financial asset symbol
        model_preference: Preferred model ('auto', 'finbert', 'ensemble', etc.)
        use_cache: Use caching
        max_length: Maximum text length for analysis
    
    Returns:
        Dict: Comprehensive sentiment analysis
    """
    start_time = datetime.now(timezone.utc)
    self.metrics['total_analyses'] += 1
    
    # ========================================================================
    # PHASE 1: INPUT VALIDATION AND PREPARATION
    # ========================================================================
    
    if not text or not text.strip():
        result = {
            'status': 'ERROR',
            'reason': 'Empty text provided',
            'timestamp': start_time.isoformat(),
            'genesis_checksum': GENESIS.runtime_checksum
        }
        self.metrics['failed_analyses'] += 1
        return result
    
    # Generate cache key
    cache_key = None
    if use_cache:
        cache_key = hashlib.sha3_256(
            f"{source}_{asset}_{text[:100]}_{model_preference}".encode()
        ).hexdigest()[:32]
        
        if cache_key in self._cache:
            cache_entry = self._cache[cache_key]
            
            # Check TTL
            if datetime.now(timezone.utc) < cache_entry['expiry']:
                self.metrics['cache_hits'] += 1
                
                # Update LRU order
                self._cache.move_to_end(cache_key)
                
                result = cache_entry['data'].copy()
                result['model_used'] += '_CACHED'
                result['cache_hit'] = True
                
                return result
    
    self.metrics['cache_misses'] += 1
    
    # ========================================================================
    # PHASE 2: TEXT PREPROCESSING
    # ========================================================================
    
    processed_text = self._preprocess_text(text, source, max_length)
    
    # ========================================================================
    # PHASE 3: MODEL SELECTION
    # ========================================================================
    
    model_name, model = self._select_model(model_preference, source, processed_text)
    
    # ========================================================================
    # PHASE 4: SENTIMENT ANALYSIS
    # ========================================================================
    
    try:
        if hasattr(model, 'analyze'):
            raw_result = model.analyze(processed_text)
        elif hasattr(model, '__call__'):
            raw_result = model(processed_text, truncation=True)
            # Convert transformers format
            if isinstance(raw_result, list):
                raw_result = self._convert_transformers_result(raw_result)
        else:
            raw_result = {
                'label': 'NEUTRAL',
                'score': 0.0,
                'normalized_sentiment': 0.0,
                'confidence': 0.0
            }
    except Exception as e:
        logger.error(f"Sentiment analysis failed: {e}")
        
        # Use lexicon as fallback
        if SentimentModel.LEXICON.value in self.models:
            model_name = SentimentModel.LEXICON.value
            model = self.models[SentimentModel.LEXICON.value]
            raw_result = model.analyze(processed_text)
        else:
            raw_result = {
                'label': 'NEUTRAL',
                'score': 0.0,
                'normalized_sentiment': 0.0,
                'confidence': 0.0,
                'error': str(e)
            }
    
    # ========================================================================
    # PHASE 5: RESULT PROCESSING
    # ========================================================================
    
    # Extract sentiment score
    sentiment_score = raw_result.get('normalized_sentiment', 0.0)
    if 'normalized_sentiment' not in raw_result:
        # Calculate from label and score
        label = raw_result.get('label', 'NEUTRAL').upper()
        score = raw_result.get('score', 0.0)
        
        if 'POS' in label or label == 'POSITIVE':
            sentiment_score = score
        elif 'NEG' in label or label == 'NEGATIVE':
            sentiment_score = -score
        else:
            sentiment_score = 0.0
    
    # Source-specific normalization
    normalized_score = self._normalize_for_source(sentiment_score, source)
    
    # Confidence calculation
    confidence = raw_result.get('confidence', 0.5)
    source_confidence = self.source_weights.get(source, 0.3)
    final_confidence = confidence * source_confidence
    
    # ========================================================================
    # PHASE 6: GENESIS LAYER 1 FAIL-SAFE
    # ========================================================================
    
    # Calculate sentiment consistency (placeholder)
    sentiment_consistency = 0.7  # Would be calculated from multiple analyses
    
    is_safe, safe_reason = GENESIS.fail_safe_check(
        confidence=final_confidence,
        volatility=abs(normalized_score) * 2,  # Proxy for volatility
        sentiment_consistency=sentiment_consistency
    )
    
    if not is_safe:
        result = {
            'status': 'GENESIS_FAIL_SAFE',
            'reason': safe_reason,
            'timestamp': start_time.isoformat(),
            'genesis_checksum': GENESIS.runtime_checksum,
            'confidence': float(final_confidence),
            'sentiment_score': float(sentiment_score)
        }
        self.metrics['failed_analyses'] += 1
        return result
    
    # ========================================================================
    # PHASE 7: RESULT COMPILATION
    # ========================================================================
    
    end_time = datetime.now(timezone.utc)
    latency_ms = (end_time - start_time).total_seconds() * 1000
    
    # Update performance metrics
    self.metrics['average_latency_ms'] = (
        (self.metrics['average_latency_ms'] * (self.metrics['total_analyses'] - self.metrics['failed_analyses'] - 1) + latency_ms) 
        / (self.metrics['total_analyses'] - self.metrics['failed_analyses'])
    )
    
    # Update model distribution
    if model_name not in self.metrics['model_distribution']:
        self.metrics['model_distribution'][model_name] = 0
    self.metrics['model_distribution'][model_name] += 1
    
    result = {
        'status': 'SUCCESS',
        'text': text[:500] + '...' if len(text) > 500 else text,
        'text_length': len(text),
        'processed_length': len(processed_text),
        'source': source,
        'asset': asset or 'N/A',
        'model_used': model_name,
        'sentiment_score': float(sentiment_score),
        'normalized_score': float(normalized_score),
        'confidence': float(final_confidence),
        'timestamp': start_time.isoformat(),
        'genesis_checksum': GENESIS.runtime_checksum,
        
        # Raw model output
        'raw_model_output': raw_result,
        
        # Interpretation
        'interpretation': self._interpret_sentiment(normalized_score, final_confidence),
        
        # Performance
        'performance': {
            'analysis_latency_ms': round(latency_ms, 2),
            'cache_used': cache_key is not None
        }
    }
    
    # ========================================================================
    # PHASE 8: CACHE MANAGEMENT
    # ========================================================================
    
    if cache_key:
        self._cache[cache_key] = {
            'timestamp': start_time,
            'expiry': start_time + self.cache_ttl,
            'data': result
        }
        
        # Move to end (LRU)
        self._cache.move_to_end(cache_key)
        
        # Trim cache if needed
        if len(self._cache) > self.max_cache_size:
            oldest_key = next(iter(self._cache))
            del self._cache[oldest_key]
        
        # Clean expired entries
        self._clean_expired_cache()
    
    logger.debug(f"Sentiment analysis: score={normalized_score:.3f}, confidence={final_confidence:.3f}, "
                f"model={model_name}, latency={latency_ms:.1f}ms")
    
    return result

def _preprocess_text(self, text: str, source: str, max_length: int) -> str:
    """
    Preprocess text for sentiment analysis.
    """
    if not text:
        return ""
    
    # Remove URLs
    text = re.sub(r'http\S+|www\S+|https\S+', '', text)
    
    # Remove special characters but keep financial symbols
    text = re.sub(r'[^\w\s$%€£¥@#.,!?\'"-]', ' ', text)
    
    # Handle source-specific preprocessing
    if source == SentimentSource.TWITTER.value:
        # Remove RT and mentions
        text = re.sub(r'RT @\w+:', '', text)
        text = re.sub(r'@\w+', '', text)
        # Convert hashtags to words
        text = re.sub(r'#(\w+)', r'\1', text)
    
    elif source == SentimentSource.REDDIT.value:
        # Remove Reddit-specific markup
        text = re.sub(r'\[.*?\]\(.*?\)', '', text)  # Links
        text = re.sub(r'\*\*(.*?)\*\*', r'\1', text)  # Bold
        text = re.sub(r'\*(.*?)\*', r'\1', text)  # Italic
    
    # Normalize whitespace
    text = re.sub(r'\s+', ' ', text).strip()
    
    # Truncate if needed
    if len(text) > max_length:
        text = text[:max_length].rsplit(' ', 1)[0] + '...'
    
    return text

def _select_model(self, preference: str, source: str, text: str) -> Tuple[str, Any]:
    """
    Select appropriate model for sentiment analysis.
    """
    # Check preference
    if preference != 'auto' and preference in self.models:
        return preference, self.models[preference]
    
    # Source-based selection
    if source in [SentimentSource.NEWS.value, SentimentSource.BLOOMBERG.value, 
                 SentimentSource.REUTERS.value, SentimentSource.YAHOO.value]:
        # Financial news - prefer FinBERT
        if SentimentModel.FINBERT.value in self.models:
            return SentimentModel.FINBERT.value, self.models[SentimentModel.FINBERT.value]
    
    elif source in [SentimentSource.TWITTER.value, SentimentSource.REDDIT.value,
                   SentimentSource.STOCKTWITS.value]:
        # Social media - prefer ensemble or VADER
        if SentimentModel.ENSEMBLE.value in self.models:
            return SentimentModel.ENSEMBLE.value, self.models[SentimentModel.ENSEMBLE.value]
        elif SentimentModel.VADER.value in self.models:
            return SentimentModel.VADER.value, self.models[SentimentModel.VADER.value]
    
    # Default to ensemble if available
    if SentimentModel.ENSEMBLE.value in self.models:
        return SentimentModel.ENSEMBLE.value, self.models[SentimentModel.ENSEMBLE.value]
    
    # Fallback to lexicon
    return SentimentModel.LEXICON.value, self.models[SentimentModel.LEXICON.value]

def _convert_transformers_result(self, result: List[Dict]) -> Dict[str, Any]:
    """Convert transformers pipeline result to standard format"""
    if not result:
        return {'label': 'NEUTRAL', 'score': 0.0, 'normalized_sentiment': 0.0, 'confidence': 0.0}
    
    # Extract sentiment from results
    pos_score = 0.0
    neg_score = 0.0
    neu_score = 0.0
    
    for item in result:
        label = item.get('label', '').upper()
        score = item.get('score', 0.0)
        
        if 'POS' in label or label == 'POSITIVE' or 'POS' in label:
            pos_score = score
        elif 'NEG' in label or label == 'NEGATIVE' or 'NEG' in label:
            neg_score = score
        elif 'NEU' in label or label == 'NEUTRAL':
            neu_score = score
    
    # Calculate normalized sentiment with neutral adjustment
    total = pos_score + neg_score + neu_score
    if total > 0:
        normalized = (pos_score - neg_score) / total
        confidence = max(pos_score, neg_score, neu_score)
    else:
        normalized = 0.0
        confidence = 0.0
    
    # Determine label
    if normalized > 0.1:
        label = 'POSITIVE'
    elif normalized < -0.1:
        label = 'NEGATIVE'
    else:
        label = 'NEUTRAL'
    
    return {
        'label': label,
        'score': abs(normalized),
        'normalized_sentiment': normalized,
        'confidence': confidence,
        'pos_score': pos_score,
        'neg_score': neg_score,
        'neu_score': neu_score
    }

def _normalize_for_source(self, score: float, source: str) -> float:
    """
    Normalize sentiment score based on source characteristics.
    """
    base_score = score
    
    # Source-specific adjustments
    adjustments = {
        SentimentSource.TWITTER.value: 0.8,      # Twitter tends to be more extreme
        SentimentSource.REDDIT.value: 0.7,       # Reddit can be very polarized
        SentimentSource.STOCKTWITS.value: 0.9,   # StockTwits is financial-focused
        SentimentSource.NEWS.value: 1.0,         # News is relatively balanced
        SentimentSource.BLOOMBERG.value: 1.0,
        SentimentSource.REUTERS.value: 1.0,
        SentimentSource.YAHOO.value: 0.9,
        SentimentSource.FOREX_FACTORY.value: 0.9,
        SentimentSource.TRADING_VIEW.value: 0.8
    }
    
    adjustment = adjustments.get(source, 0.8)
    normalized = base_score * adjustment
    
    # Clamp to [-1, 1]
    return max(-1.0, min(1.0, normalized))

def _interpret_sentiment(self, score: float, confidence: float) -> Dict[str, Any]:
    """
    Interpret sentiment score and confidence.
    """
    # Determine sentiment direction
    if score > 0.3:
        direction = 'STRONGLY_BULLISH'
        color = 'green'
        action = 'Strong buy signal'
        strength = 'VERY_STRONG'
    elif score > 0.15:
        direction = 'BULLISH'
        color = 'light_green'
        action = 'Consider long positions'
        strength = 'STRONG'
    elif score > 0.05:
        direction = 'SLIGHTLY_BULLISH'
        color = 'light_green'
        action = 'Watch for confirmation'
        strength = 'MODERATE'
    elif score > -0.05:
        direction = 'NEUTRAL'
        color = 'gray'
        action = 'No clear direction'
        strength = 'WEAK'
    elif score > -0.15:
        direction = 'SLIGHTLY_BEARISH'
        color = 'light_red'
        action = 'Watch for breakdown'
        strength = 'MODERATE'
    elif score > -0.3:
        direction = 'BEARISH'
        color = 'red'
        action = 'Consider short positions'
        strength = 'STRONG'
    else:
        direction = 'STRONGLY_BEARISH'
        color = 'dark_red'
        action = 'Strong sell signal'
        strength = 'VERY_STRONG'
    
    # Determine confidence level
    if confidence > 0.7:
        confidence_level = 'HIGH'
    elif confidence > 0.5:
        confidence_level = 'MEDIUM'
    else:
        confidence_level = 'LOW'
        action += ' (Low confidence)'
    
    # Market impact assessment
    abs_score = abs(score)
    if abs_score > 0.25 and confidence > 0.6:
        market_impact = 'HIGH'
    elif abs_score > 0.15 and confidence > 0.5:
        market_impact = 'MEDIUM'
    else:
        market_impact = 'LOW'
    
    return {
        'direction': direction,
        'color': color,
        'recommended_action': action,
        'sentiment_strength': strength,
        'confidence_level': confidence_level,
        'market_impact': market_impact,
        'numerical_score': float(score),
        'confidence_score': float(confidence)
    }

def _clean_expired_cache(self):
    """Remove expired cache entries"""
    now = datetime.now(timezone.utc)
    expired_keys = [k for k, v in self._cache.items() if v['expiry'] < now]
    
    for key in expired_keys:
        del self._cache[key]

def aggregate_sentiment(
    self,
    asset: str,
    sources_data: Dict[str, List[str]],
    weights: Optional[Dict[str, float]] = None,
    time_decay_factor: float = 0.1,
    min_samples: int = 3
) -> Dict[str, Any]:
    """
    Aggregate sentiment from multiple sources with time decay.
    
    Args:
        asset: Financial asset symbol
        sources_data: Dict of source -> list of texts
        weights: Source weights (default: predefined)
        time_decay_factor: Decay factor for older samples (0-1)
        min_samples: Minimum samples for reliable aggregation
    
    Returns:
        Dict: Aggregated sentiment analysis
    """
    start_time = datetime.now(timezone.utc)
    
    # Use default weights if not provided
    if weights is None:
        weights = self.source_weights
    
    # Validate inputs
    total_samples = sum(len(texts) for texts in sources_data.values())
    if total_samples < min_samples:
        return {
            'status': 'INSUFFICIENT_DATA',
            'reason': f'Not enough samples (have {total_samples}, need {min_samples})',
            'asset': asset,
            'timestamp': start_time.isoformat(),
            'genesis_checksum': GENESIS.runtime_checksum
        }
    
    # Analyze each text
    all_results = []
    source_breakdown = {}
    
    for source, texts in sources_data.items():
        if not texts:
            continue
        
        source_weight = weights.get(source, 0.1)
        source_results = []
        
        for text in texts[:20]:  # Limit per source
            result = self.analyze_sentiment(
                text=text,
                source=source,
                asset=asset,
                use_cache=True
            )
            
            if result.get('status') == 'SUCCESS':
                # Apply time decay (simulated - in production would use actual timestamps)
                decay = 1.0 - time_decay_factor
                
                weighted_score = result['normalized_score'] * source_weight * decay
                weighted_confidence = result['confidence'] * source_weight * decay
                
                source_results.append({
                    'score': result['normalized_score'],
                    'weighted_score': weighted_score,
                    'confidence': result['confidence'],
                    'weighted_confidence': weighted_confidence,
                    'model': result['model_used']
                })
                
                all_results.append({
                    'score': result['normalized_score'],
                    'weight': source_weight,
                    'confidence': result['confidence'],
                    'source': source,
                    'weighted_score': weighted_score
                })
        
        if source_results:
            scores = [r['score'] for r in source_results]
            weighted_scores = [r['weighted_score'] for r in source_results]
            confidences = [r['confidence'] for r in source_results]
            
            source_breakdown[source] = {
                'n_samples': len(source_results),
                'average_score': float(np.mean(scores)),
                'weighted_average': float(np.mean(weighted_scores)),
                'std_dev': float(np.std(scores)) if len(scores) > 1 else 0.0,
                'average_confidence': float(np.mean(confidences)),
                'source_weight': source_weight,
                'sentiment_distribution': {
                    'positive': sum(1 for s in scores if s > 0.1),
                    'negative': sum(1 for s in scores if s < -0.1),
                    'neutral': sum(1 for s in scores if -0.1 <= s <= 0.1)
                }
            }
    
    if not all_results:
        return {
            'status': 'NO_VALID_DATA',
            'asset': asset,
            'timestamp': start_time.isoformat(),
            'genesis_checksum': GENESIS.runtime_checksum
        }
    
    # ========================================================================
    # CALCULATE AGGREGATE METRICS
    # ========================================================================
    
    # Extract weighted scores and weights
    weighted_scores = [r['weighted_score'] for r in all_results]
    weights_list = [r['weight'] for r in all_results]
    confidences = [r['confidence'] for r in all_results]
    raw_scores = [r['score'] for r in all_results]
    
    # Calculate aggregate sentiment
    total_weight = sum(weights_list)
    if total_weight > 0:
        aggregate_score = sum(weighted_scores) / total_weight
    else:
        aggregate_score = np.mean(raw_scores) if raw_scores else 0.0
    
    # Calculate volatility
    volatility = float(np.std(raw_scores)) if len(raw_scores) > 1 else 0.0
    
    # Calculate confidence-weighted aggregate
    confidence_weights = [c * w for c, w in zip(confidences, weights_list)]
    total_confidence_weight = sum(confidence_weights)
    
    if total_confidence_weight > 0:
        confidence_weighted_score = sum(s * c * w for s, c, w in zip(raw_scores, confidences, weights_list)) / total_confidence_weight
        aggregate_confidence = total_confidence_weight / total_weight if total_weight > 0 else np.mean(confidences)
    else:
        confidence_weighted_score = aggregate_score
        aggregate_confidence = np.mean(confidences) if confidences else 0.0
    
    # Calculate sentiment consistency
    if len(raw_scores) > 1:
        # Consistency = 1 - (coefficient of variation)
        mean_abs = np.mean([abs(s) for s in raw_scores])
        if mean_abs > 0:
            consistency = 1.0 - (volatility / mean_abs)
            consistency = max(0.0, min(1.0, consistency))
        else:
            consistency = 0.0
    else:
        consistency = 0.0
    
    # ========================================================================
    # GENESIS LAYER 1 FAIL-SAFE
    # ========================================================================
    
    is_safe, safe_reason = GENESIS.fail_safe_check(
        confidence=aggregate_confidence,
        volatility=volatility,
        sentiment_consistency=consistency
    )
    
    if not is_safe:
        return {
            'status': 'GENESIS_FAIL_SAFE',
            'reason': safe_reason,
            'asset': asset,
            'timestamp': start_time.isoformat(),
            'genesis_checksum': GENESIS.runtime_checksum,
            'aggregate_score': float(aggregate_score),
            'volatility': float(volatility),
            'consistency': float(consistency)
        }
    
    # ========================================================================
    # COMPILE RESULTS
    # ========================================================================
    
    end_time = datetime.now(timezone.utc)
    latency_ms = (end_time - start_time).total_seconds() * 1000
    
    result = {
        'status': 'SUCCESS',
        'asset': asset,
        'timestamp': start_time.isoformat(),
        'genesis_checksum': GENESIS.runtime_checksum,
        
        # Aggregation results
        'aggregation_metrics': {
            'aggregate_sentiment': float(aggregate_score),
            'confidence_weighted_sentiment': float(confidence_weighted_score),
            'sentiment_volatility': float(volatility),
            'aggregate_confidence': float(aggregate_confidence),
            'sentiment_consistency': float(consistency),
            'n_total_samples': len(all_results),
            'n_sources': len(source_breakdown)
        },
        
        # Source breakdown
        'source_breakdown': source_breakdown,
        
        # Statistical distribution
        'sentiment_distribution': {
            'mean': float(np.mean(raw_scores)),
            'median': float(np.median(raw_scores)),
            'std': float(np.std(raw_scores)) if len(raw_scores) > 1 else 0.0,
            'min': float(np.min(raw_scores)) if raw_scores else 0.0,
            'max': float(np.max(raw_scores)) if raw_scores else 0.0,
            'percentile_25': float(np.percentile(raw_scores, 25)) if len(raw_scores) > 3 else 0.0,
            'percentile_75': float(np.percentile(raw_scores, 75)) if len(raw_scores) > 3 else 0.0,
            'skewness': float(stats.skew(raw_scores)) if len(raw_scores) > 2 else 0.0,
            'kurtosis': float(stats.kurtosis(raw_scores)) if len(raw_scores) > 3 else 0.0
        },
        
        # Interpretation
        'interpretation': self._interpret_aggregate(
            confidence_weighted_score, 
            volatility, 
            aggregate_confidence,
            consistency
        ),
        
        # Performance
        'performance': {
            'aggregation_latency_ms': round(latency_ms, 2),
            'samples_per_second': round(len(all_results) / (latency_ms / 1000), 1) if latency_ms > 0 else 0
        }
    }
    
    return result

def _interpret_aggregate(
    self, 
    score: float, 
    volatility: float, 
    confidence: float,
    consistency: float
) -> Dict[str, Any]:
    """
    Interpret aggregate sentiment metrics.
    """
    # Sentiment direction
    if score > 0.25:
        direction = 'STRONGLY_BULLISH'
        market_bias = 'BULLISH'
    elif score > 0.1:
        direction = 'BULLISH'
        market_bias = 'SLIGHTLY_BULLISH'
    elif score > -0.1:
        direction = 'NEUTRAL'
        market_bias = 'NEUTRAL'
    elif score > -0.25:
        direction = 'BEARISH'
        market_bias = 'SLIGHTLY_BEARISH'
    else:
        direction = 'STRONGLY_BEARISH'
        market_bias = 'BEARISH'
    
    # Volatility assessment
    if volatility > 0.3:
        volatility_level = 'HIGH'
        market_state = 'VOLATILE'
    elif volatility > 0.15:
        volatility_level = 'MEDIUM'
        market_state = 'MODERATELY_VOLATILE'
    else:
        volatility_level = 'LOW'
        market_state = 'STABLE'
    
    # Confidence assessment
    if confidence > 0.7:
        confidence_level = 'HIGH'
        reliability = 'RELIABLE'
    elif confidence > 0.5:
        confidence_level = 'MEDIUM'
        reliability = 'MODERATELY_RELIABLE'
    else:
        confidence_level = 'LOW'
        reliability = 'UNRELIABLE'
    
    # Consistency assessment
    if consistency > 0.7:
        consistency_level = 'HIGH'
        agreement = 'STRONG_AGREEMENT'
    elif consistency > 0.5:
        consistency_level = 'MEDIUM'
        agreement = 'MODERATE_AGREEMENT'
    else:
        consistency_level = 'LOW'
        agreement = 'WEAK_AGREEMENT'
    
    # Trading recommendation
    if direction == 'STRONGLY_BULLISH' and confidence_level == 'HIGH' and volatility_level == 'LOW':
        recommendation = 'STRONG_BUY'
        risk_level = 'LOW'
    elif direction == 'BULLISH' and confidence_level in ['HIGH', 'MEDIUM']:
        recommendation = 'BUY'
        risk_level = 'LOW_TO_MEDIUM'
    elif direction == 'NEUTRAL' or confidence_level == 'LOW':
        recommendation = 'HOLD'
        risk_level = 'MEDIUM'
    elif direction == 'BEARISH' and confidence_level in ['HIGH', 'MEDIUM']:
        recommendation = 'SELL'
        risk_level = 'MEDIUM_TO_HIGH'
    elif direction == 'STRONGLY_BEARISH' and confidence_level == 'HIGH':
        recommendation = 'STRONG_SELL'
        risk_level = 'HIGH'
    else:
        recommendation = 'NO_CLEAR_SIGNAL'
        risk_level = 'HIGH'
    
    return {
        'sentiment_direction': direction,
        'market_bias': market_bias,
        'volatility_level': volatility_level,
        'market_state': market_state,
        'confidence_level': confidence_level,
        'reliability': reliability,
        'consistency_level': consistency_level,
        'source_agreement': agreement,
        'trading_recommendation': recommendation,
        'risk_level': risk_level,
        'summary': f"{direction} sentiment with {confidence_level} confidence and {volatility_level} volatility"
    }
