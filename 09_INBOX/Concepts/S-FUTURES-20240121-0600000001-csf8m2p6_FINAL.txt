@dataclass
class CalendarSpread:
    """Estrutura para calendar spreads"""
    front_contract: str
    back_contract: str
    spread_price: Decimal
    z_score: Decimal
    roll_yield: Decimal
    historical_percentile: Decimal

@dataclass
class SpreadOpportunity:
    """Oportunidade de calendar spread"""
    spread: CalendarSpread
    expected_return: Decimal
    risk_adjusted_score: Decimal
    execution_priority: int

class CalendarSpreadFutures:
    """

        self.version = "3.0-FORTIFIED"
        
        # Contratos para calendar spreads
        self.spread_universe = {
            'INDEX': ['ES', 'NQ', 'RTY'],  # Equity indices
            'COMMODITY': ['GC', 'SI'],     # Metals
            'ENERGY': ['CL', 'NG'],        # Energy
            'RATES': ['ZB', 'ZN']          # Interest rates

        self.spread_params = {
            'zscore_threshold': Decimal('2.0'),
            'min_roll_yield': Decimal('0.005'),  # 0.5%
            'historical_percentile_min': Decimal('0.80'),

    async def scan_calendar_spreads(self) -> List[CalendarSpread]:

        Escaneia todos os calendar spreads possíveis

        for asset_class, contracts in self.spread_universe.items():
            for contract in contracts:

                all_spreads.extend(spreads)

    async def _analyze_contract_calendar_spreads(self, contract: str) -> List[CalendarSpread]:
        """
        Analisa calendar spreads para um contrato específico
        """
        spreads = []
        
        try:
            # Obter chain de futuros para o contrato
            futures_chain = await self._get_futures_chain(contract)

                spread_price = await self._calculate_spread_price(

                    # Calcular Z-score histórico
                    z_score = await self._calculate_spread_zscore(
                        contract, front_contract, back_contract, spread_price

                    roll_yield = await self._calculate_roll_yield(
                        contract, front_contract, back_contract

    async def identify_spread_opportunities(self, spreads: List[CalendarSpread]) -> List[SpreadOpportunity]:
        """
        Identifica oportunidades de calendar spread baseadas em múltiplos fatores
        """
        opportunities = []
        
        for spread in spreads:
            # Aplicar filtros de qualidade
            if self._passes_quality_filters(spread):
                # Calcular retorno esperado
                expected_return = self._calculate_expected_return(spread)
                
                # Calcular score ajustado ao risco
                risk_adjusted_score = self._calculate_risk_adjusted_score(spread, expected_return)
                
                if risk_adjusted_score > Decimal('0.75'):
                    opportunity = SpreadOpportunity(
                        spread=spread,
                        expected_return=expected_return,
                        risk_adjusted_score=risk_adjusted_score,
                        execution_priority=self._determine_execution_priority(spread)
                    )
                    opportunities.append(opportunity)
                    
        # Ordenar por score ajustado ao risco
        opportunities.sort(key=lambda x: x.risk_adjusted_score, reverse=True)
        return opportunities

    def _passes_quality_filters(self, spread: CalendarSpread) -> bool:
        """
        Aplica filtros de qualidade ao spread
        """
        filters = [
            abs(spread.z_score) >= self.spread_params['zscore_threshold'],
            abs(spread.roll_yield) >= self.spread_params['min_roll_yield'],
            spread.historical_percentile >= self.spread_params['historical_percentile_min'],
            self._is_liquid_spread(spread.front_contract, spread.back_contract)
        ]
        
        return all(filters)

    def _calculate_expected_return(self, spread: CalendarSpread) -> Decimal:
        """
        Calcula retorno esperado do calendar spread
        """
        # Componente de mean reversion
        mean_reversion_return = abs(spread.z_score) * Decimal('0.15')
        
        # Componente de roll yield
        roll_yield_component = spread.roll_yield * Decimal('2.0')
        
        # Componente de historical percentile
        percentile_component = (spread.historical_percentile - Decimal('0.5')) * Decimal('0.10')
        
        expected_return = mean_reversion_return + roll_yield_component + percentile_component
        
        return min(expected_return, Decimal('0.25'))  # Máximo 25%

    async def execute_calendar_spread(self, opportunity: SpreadOpportunity):
        """
        Executa calendar spread trade
        """
        try:
            spread = opportunity.spread
            
            # Determinar direção baseada no Z-score
            if spread.z_score > Decimal('0'):
                # Spread caro - vender spread (vender front, comprar back)
                front_action, back_action = 'SELL', 'BUY'
            else:
                # Spread barato - comprar spread (comprar front, vender back)
                front_action, back_action = 'BUY', 'SELL'
            
            # Calcular tamanho da posição
            position_size = self._calculate_spread_position_size(opportunity)
            
            # Executar spread trade
            execution_result = await self._execute_spread_trade(
                leg1_contract=spread.front_contract,
                leg1_action=front_action,
                leg2_contract=spread.back_contract,
                leg2_action=back_action,
                quantity=position_size
            )
            
            if execution_result['success']:
                await self._log_calendar_spread(opportunity, execution_result)
                logging.info(f"Calendar spread executed: {spread.front_contract}-"
                           f"{spread.back_contract} | Z-score: {spread.z_score:.2f}")
                           
        except Exception as e:
            logging.error(f"Calendar spread execution error: {e}")

    def _calculate_spread_position_size(self, opportunity: SpreadOpportunity) -> int:
        """
        Calcula tamanho da posição para calendar spread
        """
        base_size = 5  # contratos base
        
        # Ajuste por score de risco
        risk_adjustment = float(opportunity.risk_adjusted_score) * 3.0
        
        # Ajuste por magnitude do Z-score
        zscore_adjustment = min(4.0, float(abs(opportunity.spread.z_score)) / 1.5)
        
        position_size = int(base_size * risk_adjustment * zscore_adjustment)
        
        return min(position_size, self.spread_params['max_position_size'])
