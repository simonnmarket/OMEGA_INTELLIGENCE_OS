// Estruturas de dados sensoriais
struct MarketVision {
    double trendStrength;        // Força da tendência (0-1)
    double priceAction;          // Ação do preço (-1 a 1)
    double volumeProfile;        // Perfil de volume (0-1)
    double patternStrength;      // Força do padrão identificado (0-1)
    double marketStructure;      // Estrutura de mercado (0-1)
};

struct MarketHearing {
    double priceVelocity;       // Velocidade do preço (-1 a 1)
    double volumeIntensity;     // Intensidade do volume (0-1)
    double marketNoise;         // Ruído do mercado (0-1)
    double orderFlow;           // Fluxo de ordens (-1 a 1)
    double tickFrequency;       // Frequência de ticks (0-1)
};

struct MarketTouch {
    double marketMomentum;      // Momento do mercado (-1 a 1)
    double pressureLevel;       // Nível de pressão (0-1)
    double volatilityTouch;     // Toque da volatilidade (0-1)
    double supportResistance;   // Força do suporte/resistência (0-1)
    double priceStability;      // Estabilidade do preço (0-1)
};

struct MarketSmell {
    double riskLevel;           // Nível de risco (0-1)
    double marketSentiment;     // Sentimento do mercado (-1 a 1)
    double volatilitySmell;     // Cheiro da volatilidade (0-1)
    double trendExhaustion;     // Exaustão da tendência (0-1)
    double marketStress;        // Estresse do mercado (0-1)
};

struct MarketTaste {
    double profitPotential;     // Potencial de lucro (0-1)
    double riskReward;          // Relação risco/retorno (0-1)
    double marketQuality;       // Qualidade do mercado (0-1)
    double tradeOpportunity;    // Oportunidade de trade (0-1)
    double marketMaturity;      // Maturidade do mercado (0-1)
};

struct IntegratedSensoryData {
    MarketVision vision;
    MarketHearing hearing;
    MarketTouch touch;
    MarketSmell smell;
    MarketTaste taste;
    double sensoryCoherence;    // Coerência entre os sentidos (0-1)
    double overallPerception;   // Percepção geral do mercado (0-1)
};
