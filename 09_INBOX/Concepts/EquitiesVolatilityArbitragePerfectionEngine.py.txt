    def analyze_and_generate_signal(self, market_data: Dict, system_intentional_state: str) -> Optional[TradingSignalPerfeito]:
        """
        Analisa o mercado e gera um sinal de arbitragem de volatilidade.
        """
        # 1. Verificação de Intenção (Dr. Hale)
        if system_intentional_state != "SCAN_OPPORTUNITIES":
            return None

        options_data = market_data.get('options', {})
        stock_data = market_data.get('stocks', {})
        
        best_opportunity = None
        best_score = -1

        # 2. Itera sobre os ativos para encontrar oportunidades
        for stock in self.target_stocks:
            if stock not in options_data or stock not in stock_data:
                continue

            if not surface_opportunity:
                continue

            if not jump_opportunity:
                continue

            # 5. Filtro de Setup de Preço (Minervini)
            if not self._check_price_setup(stock, stock_data):
                continue
            
            # 6. Cálculo de Score Combinado
            opportunity_score = surface_opportunity['score'] * jump_opportunity['score']
            
            if opportunity_score > best_score:
                best_score = opportunity_score
                best_opportunity = {
                    'stock': stock,
                    'strategy_type': surface_opportunity['type'],
                    'surface_deviation': surface_opportunity['deviation'],
                    'jump_intensity': jump_opportunity['intensity'],
                    'legs': surface_opportunity['legs']
                }

        if not best_opportunity:
            return None

        # 7. Gestão de Risco e Geração do Sinal
        confidence = Decimal(str(min(best_score, 0.95)))
        risk_score = Decimal('0.4') # Risco inerente a volatilidade

            strategy_id=self.strategy_id,
            asset=f"VOL_{best_opportunity['stock']}_{best_opportunity['strategy_type']}",
            action='EXECUTE_VOL_ARBITRAGE',
            confidence=confidence,

            hale_intentional_state=system_intentional_state,
            # Metadados da perfeição

            # Metadados específicos da estratégia

                'strategy_type': best_opportunity['strategy_type'],
                'surface_deviation': best_opportunity['surface_deviation'],

    def _analyze_vol_surface(self, stock: str, options_data: Dict) -> Optional[Dict]:
        """Analisa anomalias na superfície de volatilidade (Rossi)."""
        # Em um sistema real, treinaríamos um GP com dados históricos de (S, K, T, tau, IV)
        # Aqui, simulamos a detecção de uma anomalia.
        # Suponha que a IV de uma opção ATM seja 35%, mas o modelo prevê 25%.
        current_iv = options_data[stock].get('implied_volatility', 0.25)
        model_predicted_iv = current_iv * 0.8 # Simulação de anomalia
        
        deviation = abs(current_iv - model_predicted_iv) / model_predicted_iv
        
        if deviation > self.min_surface_deviation:
            strategy_type = "SHORT_VOL" if current_iv > model_predicted_iv else "LONG_VOL"
            legs = self._construct_strategy_legs(stock, strategy_type, options_data)
            return {
                'type': strategy_type,
                'deviation': deviation,
                'score': deviation,
                'legs': legs
            }
        return None

    def _analyze_jump_diffusion(self, stock: str, options_data: Dict) -> Optional[Dict]:
        """Analisa a intensidade de saltos esperados (Petrov)."""
        # Em um sistema real, calibraríamos um modelo de Merton Jump-Diffusion
        # para estimar a intensidade (lambda) dos saltos.
        # Aqui, simulamos uma alta intensidade de salto perto de earnings.
        days_to_earnings = self._get_days_to_earnings(stock)
        
        if days_to_earnings < 10:
            jump_intensity = 0.5 # Alta intensidade
        else:
            jump_intensity = 0.1 # Baixa intensidade
            
        if jump_intensity > self.min_jump_intensity:
            return {
                'intensity': jump_intensity,
                'score': jump_intensity
            }
        return None
