import numpy as np
import pandas as pd
from scipy import stats, linalg, optimize
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

class OilTradingStrategyProven:
    """Estratégia de Oil usando apenas matemática/física comprovada"""
    
    def __init__(self):
        self.validated_signals = {}
        self.performance_metrics = {}
        
    def calculate_term_structure_alpha(self, futures_curve):
        """
        Alpha da estrutura a termo - Matemática de derivativos comprovada
        """
        # Calcular roll yield da curva de futuros
        if len(futures_curve) < 2:
            return 0
            
        # Contango/Backwardation: F1 - F2
        term_structure = futures_curve.iloc[0] - futures_curve.iloc[1]
        
        # Normalizar pelo preço spot
        normalized_structure = term_structure / futures_curve.iloc[0]
        
        # Sinal: backwardation positivo, contango negativo
        return float(normalized_structure * 100)  # Em percentual
    
    def inventory_cycle_detection(self, inventory_data, price_data):
        """
        Detecção de ciclos de inventário - Teoria de armazenamento
        Base: Working (1949) - Teoria do armazenamento comprovada
        """
        # Correlação inversa inventário-preço
        correlation = inventory_data.corr(price_data)
        
        # Derivada do inventário (aceleração)
        inventory_velocity = inventory_data.diff()
        inventory_acceleration = inventory_velocity.diff()
        
        # Ponto de inflexão do ciclo
        inflection_points = inventory_acceleration * inventory_velocity < 0
        
        cycle_phase = "ACCUMULATION" if inventory_velocity.iloc[-1] > 0 else "DRAWDOWN"
        
        return {
            'correlation': correlation,
            'cycle_phase': cycle_phase,
            'inflection_points': inflection_points.sum(),
            'velocity': inventory_velocity.iloc[-1]
        }
    
    def geopolitical_risk_premium(self, news_sentiment, volatility_index):
        """
        Prêmio de risco geopolítico - Matemática de insurance
        Base: Teoria do prêmio de risco (Arrow, 1963)
        """
        # Componentes do prêmio
        sentiment_component = news_sentiment * 0.3  # Peso do sentimento
        volatility_component = volatility_index * 0.7  # Peso da volatilidade
        
        # Prêmio total (basis points)
        risk_premium = (sentiment_component + volatility_component) * 100
        
        return max(0, risk_premium)
    
    def dollar_correlation_hedge(self, oil_prices, usd_index, window=30):
        """
        Hedge de correlação USD - Estatística multivariada comprovada
        """
        rolling_correlation = oil_prices.rolling(window).corr(usd_index)
        
        # Beta de hedge ótimo (MQO rolling)
        def calculate_hedge_beta(oil, usd):
            if len(oil) < 2:
                return 0
            return np.cov(oil, usd)[0,1] / np.var(usd)
        
        hedge_beta = oil_prices.rolling(window).apply(
            lambda x: calculate_hedge_beta(x, usd_index.loc[x.index])
        )
        
        return {
            'correlation': rolling_correlation.iloc[-1] if not rolling_correlation.empty else 0,
            'hedge_beta': hedge_beta.iloc[-1] if not hedge_beta.empty else 0,
            'hedge_ratio': abs(hedge_beta.iloc[-1]) if not hedge_beta.empty else 0
        }
    
    def seasonal_decomposition(self, price_series, period=252):
        """
        Decomposição sazonal - Análise de Fourier comprovada
        """
        from statsmodels.tsa.seasonal import seasonal_decompose
        
        try:
            decomposition = seasonal_decompose(price_series, period=period, extrapolate_trend=1)
            
            seasonal_strength = decomposition.seasonal.std() / price_series.std()
            trend_strength = decomposition.trend.std() / price_series.std()
            
            return {
                'seasonal_strength': seasonal_strength,
                'trend_strength': trend_strength,
                'seasonal_component': decomposition.seasonal,
                'current_seasonality': decomposition.seasonal.iloc[-1] if hasattr(decomposition.seasonal, 'iloc') else decomposition.seasonal[-1]
            }
        except:
            return {'seasonal_strength': 0, 'trend_strength': 0, 'current_seasonality': 0}
    
    def momentum_conservation_model(self, price_series, lookback=20):
        """
        Modelo de conservação de momento - Física Newtoniana
        Base: Leis do movimento (Newton, 1687)
        """
        price_velocity = price_series.diff()
        price_acceleration = price_velocity.diff()
        
        # Momento: p = mv (massa = volume negociado)
        momentum = price_velocity * lookback
        
        # Força: F = dp/dt
        force = momentum.diff()
        
        # Energia cinética: K = ½mv²
        kinetic_energy = 0.5 * lookback * (price_velocity ** 2)
        
        # Sinal baseado na conservação de energia
        energy_signal = kinetic_energy.rolling(5).mean()
        
        return {
            'momentum': momentum.iloc[-1] if not momentum.empty else 0,
            'force': force.iloc[-1] if not force.empty else 0,
            'energy_signal': energy_signal.iloc[-1] if not energy_signal.empty else 0,
            'acceleration': price_acceleration.iloc[-1] if not price_acceleration.empty else 0
        }
    
    def regime_identification(self, returns_series, n_regimes=3):
        """
        Identificação de regimes - Modelos de Markov comprovados
        Base: Cadeias de Markov (Andrey Markov, 1906)
        """
        from sklearn.mixture import GaussianMixture
        
        returns_reshaped = returns_series.dropna().values.reshape(-1, 1)
        
        if len(returns_reshaped) < n_regimes:
            return {'current_regime': 0, 'regime_confidence': 1.0}
        
        gmm = GaussianMixture(n_components=n_regimes, covariance_type='full')
        gmm.fit(returns_reshaped)
        
        current_regime = gmm.predict(returns_reshaped[-1].reshape(1, -1))[0]
        regime_probs = gmm.predict_proba(returns_reshaped[-1].reshape(1, -1))[0]
        
        return {
            'current_regime': current_regime,
            'regime_confidence': regime_probs[current_regime],
            'regime_volatility': np.sqrt(gmm.covariances_[current_regime][0,0]),
            'regime_return': gmm.means_[current_regime][0]
        }
    
    def volume_price_confirmation(self, price_series, volume_series, window=10):
        """
        Confirmação preço-volume - Teoria de Wyckoff comprovada
        Base: Análise técnica quantitativa (Richard Wyckoff, 1930)
        """
        price_trend = price_series.rolling(window).mean()
        volume_trend = volume_series.rolling(window).mean()
        
        price_direction = np.sign(price_trend.diff().iloc[-1]) if not price_trend.empty else 0
        volume_direction = np.sign(volume_trend.diff().iloc[-1]) if not volume_trend.empty else 0
        
        # Confirmação: mesma direção = confirmação
        confirmation_strength = 1 if price_direction == volume_direction else -1
        
        # Força do movimento
        price_magnitude = abs(price_series.pct_change().rolling(window).mean().iloc[-1] if not price_series.empty else 0)
        volume_magnitude = abs(volume_series.pct_change().rolling(window).mean().iloc[-1] if not volume_series.empty else 0)
        
        return {
            'confirmation': confirmation_strength,
            'strength': (price_magnitude + volume_magnitude) / 2,
            'price_trend': price_direction,
            'volume_trend': volume_direction
        }

class OilTradingExecution:
    """Sistema de execução para estratégia de Oil"""
    
    def __init__(self):
        self.strategy = OilTradingStrategyProven()
        
    def generate_trading_signals(self, market_data):
        """
        Gera sinais de trading baseados em todos os filtros
        """
        signals = {}
        
        # 1. Sinal da estrutura a termo
        if 'futures_curve' in market_data:
            signals['term_structure'] = self.strategy.calculate_term_structure_alpha(
                market_data['futures_curve']
            )
        
        # 2. Sinal de inventário
        if 'inventory' in market_data and 'prices' in market_data:
            inventory_signal = self.strategy.inventory_cycle_detection(
                market_data['inventory'], market_data['prices']
            )
            signals['inventory'] = inventory_signal
        
        # 3. Sinal geopolítico
        if 'news_sentiment' in market_data and 'volatility' in market_data:
            signals['geopolitical'] = self.strategy.geopolitical_risk_premium(
                market_data['news_sentiment'], market_data['volatility']
            )
        
        # 4. Hedge USD
        if 'prices' in market_data and 'usd_index' in market_data:
            usd_hedge = self.strategy.dollar_correlation_hedge(
                market_data['prices'], market_data['usd_index']
            )
            signals['usd_hedge'] = usd_hedge
        
        # 5. Sazonalidade
        if 'prices' in market_data:
            seasonal = self.strategy.seasonal_decomposition(market_data['prices'])
            signals['seasonal'] = seasonal
        
        # 6. Momento
        if 'prices' in market_data:
            momentum = self.strategy.momentum_conservation_model(market_data['prices'])
            signals['momentum'] = momentum
        
        # 7. Regime
        if 'prices' in market_data:
            returns = market_data['prices'].pct_change().dropna()
            regime = self.strategy.regime_identification(returns)
            signals['regime'] = regime
        
        # 8. Confirmação volume
        if 'prices' in market_data and 'volume' in market_data:
            volume_conf = self.strategy.volume_price_confirmation(
                market_data['prices'], market_data['volume']
            )
            signals['volume_confirmation'] = volume_conf
        
        return signals
    
    def calculate_position_size(self, signals, portfolio_value, risk_per_trade=0.02):
        """
        Calcula tamanho da posição baseado nos sinais
        Base: Kelly Criterion comprovado (John Kelly, 1956)
        """
        # Combinar sinais em score único
        total_score = 0
        signal_count = 0
        
        for signal_name, signal_value in signals.items():
            if isinstance(signal_value, dict):
                # Sinais complexos - extrair componente principal
                if 'confirmation' in signal_value:
                    total_score += signal_value['confirmation'] * signal_value['strength']
                    signal_count += 1
                elif 'correlation' in signal_value:
                    total_score += signal_value['correlation'] * 10  # Escalar
                    signal_count += 1
            else:
                # Sinais simples
                total_score += signal_value
                signal_count += 1
        
        if signal_count == 0:
            return 0
        
        average_score = total_score / signal_count
        
        # Kelly fraction simplificado
        kelly_fraction = average_score / 4  # Conservative Kelly
        
        # Limitar entre 0 e risk_per_trade
        position_fraction = max(0, min(risk_per_trade, kelly_fraction))
        
        return portfolio_value * position_fraction
    
    def execute_trading_decision(self, market_data, portfolio_value=100000):
        """
        Executa decisão de trading final
        """
        signals = self.generate_trading_signals(market_data)
        position_size = self.calculate_position_size(signals, portfolio_value)
        
        decision = {
            'timestamp': datetime.now(),
            'signals': signals,
            'position_size': position_size,
            'action': 'BUY' if position_size > 0 else 'HOLD',
            'confidence': self.calculate_confidence(signals)
        }
        
        return decision
    
    def calculate_confidence(self, signals):
        """
        Calcula confiança baseada na concordância dos sinais
        Base: Teoria da probabilidade (Bayes)
        """
        positive_signals = 0
        total_signals = 0
        
        for signal_name, signal_value in signals.items():
            if isinstance(signal_value, dict):
                if 'confirmation' in signal_value:
                    if signal_value['confirmation'] > 0:
                        positive_signals += 1
                    total_signals += 1
                elif 'correlation' in signal_value:
                    if signal_value['correlation'] > 0:
                        positive_signals += 1
                    total_signals += 1
            else:
                if signal_value > 0:
                    positive_signals += 1
                total_signals += 1
        
        if total_signals == 0:
            return 0.5
        
        return positive_signals / total_signals

# EXEMPLO DE USO COM DADOS SIMULADOS
def demonstrate_oil_strategy():
    """Demonstra a estratégia com dados simulados"""
    print("🛢️ ESTRATÉGIA OIL - DEMONSTRAÇÃO")
    print("=" * 50)
    
    # Gerar dados simulados
    dates = pd.date_range(start='2023-01-01', end='2024-01-20', freq='D')
    n_days = len(dates)
    
    # Preços do oil (WTI)
    np.random.seed(42)
    base_price = 75.0
    prices = [base_price]
    
    for i in range(1, n_days):
        # GARCH-like volatility
        vol = 0.02 + 0.1 * abs(prices[-1] - base_price) / base_price
        change = np.random.normal(0.001, vol)
        new_price = prices[-1] * (1 + change)
        prices.append(new_price)
    
    price_series = pd.Series(prices, index=dates, name='WTI_Price')
    
    # Dados auxiliares simulados
    inventory = pd.Series(np.random.normal(100, 10, n_days), index=dates)
    usd_index = pd.Series(np.random.normal(100, 2, n_days), index=dates)
    volume = pd.Series(np.random.normal(1000000, 200000, n_days), index=dates)
    news_sentiment = pd.Series(np.random.uniform(-1, 1, n_days), index=dates)
    volatility = pd.Series(np.random.uniform(0.1, 0.4, n_days), index=dates)
    
    # Curva de futuros simulada
    futures_curve = pd.DataFrame({
        'CL1': price_series,
        'CL2': price_series * 1.02  # Contango leve
    })
    
    market_data = {
        'prices': price_series,
        'inventory': inventory,
        'usd_index': usd_index,
        'volume': volume,
        'news_sentiment': news_sentiment,
        'volatility': volatility,
        'futures_curve': futures_curve.tail(2)  # Últimos 2 períodos
    }
    
    # Executar estratégia
    executor = OilTradingExecution()
    decision = executor.execute_trading_decision(market_data)
    
    print("📊 SINAIS GERADOS:")
    for signal_name, signal_value in decision['signals'].items():
        if isinstance(signal_value, dict):
            print(f"  {signal_name}:")
            for k, v in signal_value.items():
                if not isinstance(v, (pd.Series, pd.DataFrame)):
                    print(f"    {k}: {v:.4f}")
        else:
            print(f"  {signal_name}: {signal_value:.4f}")
    
    print(f"\n🎯 DECISÃO FINAL:")
    print(f"  Ação: {decision['action']}")
    print(f"  Tamanho Posição: ${decision['position_size']:,.2f}")
    print(f"  Confiança: {decision['confidence']:.1%}")
    
    return decision
