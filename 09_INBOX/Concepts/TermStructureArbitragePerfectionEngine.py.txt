    def _update_price_history(self, market_data: Dict):
        """Atualiza o histórico de preços para análise de cointegração."""
        for pair in self._generate_all_pairs():
            price = market_data.get(pair)
            if price:
                self.price_history[pair].append(float(price))

    def _detect_best_arbitrage_opportunity(self) -> Optional[ArbitrageOpportunity]:
        """Detecta a melhor oportunidade de arbitragem triangular."""
        best_opportunity = None
        best_score = -1.0

        for cycle in self.triangular_cycles:
            a, b, c = cycle
            pair_ab, pair_bc, pair_ca = f"{a}/{b}", f"{b}/{c}", f"{c}/{a}"
            
            # Verifica se temos dados suficientes
            if not all(pair in self.price_history and len(self.price_history[pair]) > 50 for pair in [pair_ab, pair_bc, pair_ca]):
                continue

            # Cálculo do Lucro Teórico
            price_ab = self.price_history[pair_ab][-1]
            price_bc = self.price_history[pair_bc][-1]
            price_ca = self.price_history[pair_ca][-1]
            
            # A arbitragem é: 1/A * 1/B * C
            profit_rate = (1.0 / price_ab) * (1.0 / price_bc) * price_ca
            profit_pips = (profit_rate - 1.0) * 10000
            
            if profit_pips < self.min_profit_pips:
                continue

            # Cálculo da Probabilidade de Execução (Petrov)
            execution_probability = self._calculate_execution_probability(cycle, profit_pips)
            
            # Cálculo do Risco Ajustado (Rossi)
            carry_return = self._calculate_carry_return(price_ab, price_ca)
            
            # Cálculo do Score Ajustado ao Risco
            risk_adjusted_score = profit_pips * execution_probability * (1 + carry_return)
            
            if risk_adjusted_score > best_score:
                best_score = risk_adjusted_score
                best_opportunity = ArbitrageOpportunity(
                    cycle=cycle,
                    theoretical_profit_pips=profit_pips,
                    execution_probability=execution_probability,
                    risk_adjusted_return=risk_adjusted_score,
                    direction='forward' if profit_rate > 1.0 else 'reverse',
                    execution_plan={'sequence': [pair_ab, pair_bc, pair_ca]}
                )

        return best_opportunity

    def _calculate_execution_probability(self, cycle: Tuple[str, str, str], profit_pips: float) -> float:
        """Calcula a probabilidade de execução (Petrov)."""
        # Modelo simplificado: a probabilidade diminui com o lucro (menos oportunidades são mais arriscadas)
        profit_factor = np.exp(-abs(profit_pips / 5.0) # Decaimento de 5 pips
        liquidity_factor = 1.0 # Placeholder para profundidade
        
        return max(0.1, min(0.95, profit_factor * liquidity_factor))

    def _calculate_carry_return(self, price_a: float, price_c: float) -> float:
        """Calcula carry return com precisão."""
        time_fraction = 1/365 # Assumindo 1 ano para simplificação
        return (price_c / price_a - 1.0) ** time_fraction

    def _calculate_risk_adjusted_return(self, profit_pips: float, prob: float, carry_return: float) -> float:
        """Calcula o retorno ajustado ao risco (Rossi)."""
        return profit_pips * prob * (1 + carry_return)
