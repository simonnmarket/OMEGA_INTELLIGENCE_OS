/**
 * @brief Wrapper atômico com alinhamento de cache line para evitar false sharing
 * @tparam T Tipo do dado (int64_t, uint64_t, double)
 * 
 * Garante que cada variável atômica reside em sua própria cache line (64 bytes),
 * prevenindo false sharing entre cores de CPU que destrói performance em HFT.
 */
template<typename T>
struct alignas(CACHE_LINE_SIZE) CacheAlignedAtomic {
    std::atomic<T> value;
    
    constexpr CacheAlignedAtomic(T val = T{}) noexcept : value(val) {}
    
    // Load com memory ordering específico - hot path usa acquire/relaxed
    __attribute__((always_inline)) 
    T load(std::memory_order order = std::memory_order_seq_cst) const noexcept {
        return value.load(order);
    }
    
    // Store com memory ordering específico
    __attribute__((always_inline)) 
    void store(T desired, std::memory_order order = std::memory_order_seq_cst) noexcept {
        value.store(desired, order);
    }
    
    // Fetch-add atômico - usado para counters de performance
    __attribute__((always_inline)) 
    T fetch_add(T arg, std::memory_order order = std::memory_order_seq_cst) noexcept {
        return value.fetch_add(arg, order);
    }
    
    // Compare-and-swap - fundamental para algoritmos lock-free
    __attribute__((always_inline)) 
    bool compare_exchange_weak(T& expected, T desired,
                              std::memory_order success,
                              std::memory_order failure) noexcept {
        return value.compare_exchange_weak(expected, desired, success, failure);
    }
    
    // Operação atômica de exchange
    __attribute__((always_inline))
    T exchange(T desired, std::memory_order order = std::memory_order_seq_cst) noexcept {
        return value.exchange(desired, order);
    }
};

/**
 * @brief Timer de alta precisão usando Time Stamp Counter (TSC) do CPU
 * 
 * O TSC fornece timestamp em ciclos de CPU com overhead zero, superior a
 * clock_gettime() e gettimeofday(). Conversão para ns usando frequência do CPU.
 * Precisão: 1 ciclo de CPU (0.26ns @ 3.8GHz)
 */
class HighPrecisionTimer {
private:
    static inline double calibrated_freq_ghz = 0.0;
    
public:
    /**
     * @brief Lê o Time Stamp Counter do CPU (assembly inline)
     * @return Contador de ciclos desde o boot (64-bit)
     */
    __attribute__((always_inline))
    static uint64_t rdtsc() noexcept {
        return __builtin_ia32_rdtsc();
    }
    
    /**
     * @brief Converte TSC para nanossegundos usando frequência calibrada
     * @return Timestamp atual em nanossegundos
     */
    __attribute__((always_inline))
    static uint64_t nanoseconds() noexcept {
        if (calibrated_freq_ghz == 0.0) {
            calibrated_freq_ghz = calibrate_cpu_frequency();
        }
        return static_cast<uint64_t>(rdtsc() / calibrated_freq_ghz);
    }
    
    /**
     * @brief Busy-wait de alta precisão para controle de latência
     * @param ns Nanossegundos para esperar (1-10000 ns)
     * 
     * Usado em vez de sleep() que tem overhead de syscall e é impreciso.
     * _mm_pause() reduz consumo de energia durante wait (instrução PAUSE).
     */
    __attribute__((always_inline))
    static void busy_wait_ns(uint64_t ns) noexcept {
        if (ns == 0) return;
        
        uint64_t start = rdtsc();
        uint64_t cycles = static_cast<uint64_t>(ns * calibrated_freq_ghz);
        
        while (rdtsc() - start < cycles) {
            _mm_pause();  // Reduz power consumption e melhora hyper-threading
        }
    }
    
    /**
     * @brief Busy-wait até timestamp TSC específico
     * @param target_tsc Timestamp TSC alvo para terminar wait
     */
    __attribute__((always_inline))
    static void busy_wait_until(uint64_t target_tsc) noexcept {
        while (rdtsc() < target_tsc) {
            _mm_pause();
        }
    }
    
    /**
     * @brief Calibra a frequência real do CPU em tempo de execução
     * @return Frequência real em GHz com precisão de 0.1%
     * 
     * Método: Mede TSC durante sleep conhecido de 10ms para evitar
     * variações de frequency scaling.
     */
    static double calibrate_cpu_frequency() noexcept {
        if (CPU_FREQ_GHZ > 0.0) {
            return CPU_FREQ_GHZ;  // Usa valor configurado se disponível
        }
        
        auto start_std = std::chrono::high_resolution_clock::now();
        uint64_t start_tsc = rdtsc();
        
        // Espera 10ms para medição precisa (evita context switches)
        std::this_thread::sleep_for(std::chrono::milliseconds(10));
        
        auto end_std = std::chrono::high_resolution_clock::now();
        uint64_t end_tsc = rdtsc();
        
        auto duration_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(
            end_std - start_std).count();
        
        double calculated_freq = static_cast<double>(end_tsc - start_tsc) / duration_ns;
        
        std::cout << "[TIMER] CPU Frequency calibrated: " << calculated_freq << " GHz" << std::endl;
        return calculated_freq;
    }
};

/**
 * @brief Ring buffer verdadeiramente lock-free para alta concorrência
 * @tparam T Tipo dos elementos armazenados (MarketDataMsg, OrderMsg)
 * @tparam SIZE Tamanho do buffer (deve ser potência de 2 para otimização)
 * 
 * Implementação MPSC (Multi-Producer/Single-Consumer) usando atomic operations
 * com memory ordering correto. Suporta múltiplos produtores concorrentes
 * e um único consumidor, padrão ideal para HFT onde múltiplas threads de 
 * network produzem dados e uma thread de estratégia consome.
 * 
 * PERFORMANCE: O(1) para insert/remove, zero syscalls, zero allocations.
 */
template<typename T, size_t SIZE>
class LockFreeMPSCRingBuffer {
    static_assert((SIZE & (SIZE - 1)) == 0, "SIZE must be power of 2 for performance");
    
private:
    alignas(CACHE_LINE_SIZE) std::array<T, SIZE> buffer_;
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> head_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> tail_{0};
    
public:
    LockFreeMPSCRingBuffer() = default;
    
    /**
     * @brief Insere elemento no buffer (thread-safe para múltiplos produtores)
     * @param item Elemento a ser inserido (move semantics para eficiência)
     * @return true se inserção bem-sucedida, false se buffer cheio
     * 
     * Memory ordering: acquire para ler tail, release para escrever head.
     * CAS (Compare-And-Swap) garante atomicidade entre múltiplos produtores.
     */
    bool try_write(T&& item) noexcept {
        uint64_t current_head = head_.load(std::memory_order_acquire);
        uint64_t current_tail = tail_.load(std::memory_order_acquire);
        
        // Verifica se há espaço disponível (buffer cheio)
        if ((current_head - current_tail) >= SIZE) {
            return false;
        }
        
        // Reserva slot usando CAS (Compare-And-Swap)
        uint64_t new_head = current_head + 1;
        if (!head_.compare_exchange_weak(current_head, new_head,
                                        std::memory_order_acq_rel,
                                        std::memory_order_acquire)) {
            return false;  // CAS falhou, outro produtor avançou o head
        }
        
        // Escreve dados no slot reservado (não-atômico, já temos exclusividade)
        buffer_[current_head & (SIZE - 1)] = std::move(item);  // Power-of-2 mask para índice
        
        return true;
    }
    
    /**
     * @brief Lê elemento do buffer (apenas um consumidor permitido)
     * @param item Referência para armazenar elemento lido
     * @return true se leitura bem-sucedida, false se buffer vazio
     * 
     * Memory ordering: relaxed para tail (single-consumer), acquire para head.
     * Release no store do tail garante visibilidade para produtores.
     */
    bool try_read(T& item) noexcept {
        uint64_t current_tail = tail_.load(std::memory_order_relaxed);
        uint64_t current_head = head_.load(std::memory_order_acquire);
        
        // Verifica se há dados disponíveis
        if (current_tail >= current_head) {
            return false;
        }
        
        // Lê dados e avança tail
        item = std::move(buffer_[current_tail & (SIZE - 1)]);
        tail_.store(current_tail + 1, std::memory_order_release);
        
        return true;
    }
    
    /**
     * @brief Número de elementos disponíveis para leitura
     * @return Quantidade de elementos não lidos no buffer
     */
    size_t size() const noexcept {
        uint64_t current_tail = tail_.load(std::memory_order_acquire);
        uint64_t current_head = head_.load(std::memory_order_acquire);
        return current_head - current_tail;
    }
    
    /**
     * @brief Verifica se buffer está vazio
     */
    bool empty() const noexcept {
        return size() == 0;
    }
    
    /**
     * @brief Capacidade máxima do buffer
     */
    constexpr size_t capacity() const noexcept {
        return SIZE;
    }
    
    /**
     * @brief Espaço disponível para escrita
     */
    size_t available() const noexcept {
        return capacity() - size();
    }
};

/**
 * @brief Mensagem de market data otimizada para zero-copy networking
 * 
 * Estrutura packed (1-byte alignment) para transmissão direta via network stack
 * sem serialização/deserialização. Tamanho fixo: 64 bytes = 1 cache line.
 * 
 * Campos organizados para melhor locality: timestamp primeiro (mais acessado),
 * símbolo depois, depois prices/sizes.
 */
struct MarketDataMsg {
    uint64_t timestamp_ns;      // Timestamp de hardware TSC (8 bytes)
    uint64_t sequence;          // Número de sequência para gap detection (8 bytes)
    char symbol[12];            // Símbolo null-terminated "BTC-USD\0" (12 bytes)
    double bid_price;           // Melhor bid price (8 bytes)
    double ask_price;           // Melhor ask price (8 bytes)
    uint32_t bid_size;          // Size no bid (4 bytes)
    uint32_t ask_size;          // Size no ask (4 bytes)
    uint8_t venue_id;           // Exchange ID: 1=Binance, 2=Coinbase, etc. (1 byte)
    uint8_t flags;              // Bit flags: crossed(0), stale(1), etc. (1 byte)
    uint16_t reserved;          // Alinhamento para 64 bytes (2 bytes)
    
    // Total: 8+8+12+8+8+4+4+1+1+2 = 56 bytes + 8 padding = 64 bytes
    
    // Métodos de hot path (always_inline para eliminar overhead de call)
    
    /**
     * @brief Verifica se mercado está crossed (bid >= ask)
     */
    __attribute__((always_inline)) 
    bool is_crossed() const noexcept { 
        return bid_price >= ask_price; 
    }
    
    /**
     * @brief Calcula preço médio entre bid/ask
     */
    __attribute__((always_inline))
    double mid_price() const noexcept {
        return (bid_price + ask_price) * 0.5;
    }
    
    /**
     * @brief Calcula spread absoluto
     */
    __attribute__((always_inline))
    double spread() const noexcept {
        return ask_price - bid_price;
    }
    
    /**
     * @brief Calcula spread relativo em basis points
     */
    __attribute__((always_inline))
    double spread_bps() const noexcept {
        return (spread() / mid_price()) * 10000.0;
    }
    
    /**
     * @brief Verifica se dados estão stale (bit 0 das flags)
     */
    __attribute__((always_inline))
    bool is_stale() const noexcept {
        return (flags & 0x01) != 0;
    }
    
    /**
     * @brief Verifica se é auction phase (bit 1 das flags)
     */
    __attribute__((always_inline))
    bool is_auction() const noexcept {
        return (flags & 0x02) != 0;
    }
};

/**
 * @brief Mensagem de ordem de trading
 */
struct OrderMsg {
    uint64_t timestamp_ns;      // Timestamp de criação (TSC)
    uint64_t order_id;          // ID único da ordem (sequencial)
    char symbol[12];            // Símbolo negociado
    double price;               // Preço da ordem
    uint32_t quantity;          // Quantidade em unidades
    bool is_buy;                // Direção: true = buy, false = sell
    uint8_t order_type;         // Tipo: 0=market, 1=limit, 2=IOC, 3=FOK
    uint8_t flags;              // Flags de controle
    uint16_t reserved;          // Alinhamento
    
    /**
     * @brief Verifica se é ordem limit
     */
    __attribute__((always_inline))
    bool is_limit_order() const noexcept {
        return order_type == 1;
    }
    
    /**
     * @brief Verifica se é Immediate-or-Cancel
     */
    __attribute__((always_inline))
    bool is_ioc() const noexcept {
        return order_type == 2;
    }
    
    /**
     * @brief Verifica se é Fill-or-Kill
     */
    __attribute__((always_inline))
    bool is_fok() const noexcept {
        return order_type == 3;
    }
    
    /**
     * @brief Calcula notional value da ordem
     */
    __attribute__((always_inline))
    double notional() const noexcept {
        return price * quantity;
    }
};

/**
 * @brief Gerenciador de ordens com semânticas lock-free e tracking de estado
 * 
 * Responsável por criar, enviar, rastrear e cancelar ordens de trading.
 * Implementa fila MPSC para envio assíncrono de ordens com garantias de
 * ordering e durabilidade.
 */
class OrderManager {
private:
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> order_id_{1};  // Começa em 1 (0 = inválido)
    LockFreeMPSCRingBuffer<OrderMsg, RING_BUFFER_SIZE> order_queue_;
    
    // Estatísticas e monitoring
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> orders_sent_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> orders_rejected_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> orders_filled_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> orders_cancelled_{0};
    
public:
    OrderManager() = default;
    
    /**
     * @brief Envia ordem para execução na exchange
     * @param symbol Símbolo do instrumento (ex: "BTC-USD")
     * @param price Preço da ordem (para limit orders)
     * @param quantity Quantidade em unidades
     * @param is_buy Direção: true = compra, false = venda
     * @param order_type Tipo: 0=market, 1=limit, 2=IOC, 3=FOK
     * @return ID da ordem se bem-sucedido, 0 se falha (fila cheia)
     * 
     * Método thread-safe: múltiplas threads de estratégia podem chamar
     * simultaneamente. Retorna imediatamente (não-blocking).
     */
    uint64_t send_order(const char* symbol, double price, uint32_t quantity, 
                       bool is_buy, uint8_t order_type = 1) noexcept {
        
        OrderMsg order{};
        order.timestamp_ns = HighPrecisionTimer::nanoseconds();
        order.order_id = order_id_.fetch_add(1, std::memory_order_acq_rel);
        
        // Copia symbol com bounds checking
        std::strncpy(order.symbol, symbol, sizeof(order.symbol) - 1);
        order.symbol[sizeof(order.symbol) - 1] = '\0';  // Garante null-termination
        
        order.price = price;
        order.quantity = quantity;
        order.is_buy = is_buy;
        order.order_type = order_type;
        
        // Tenta enfileirar ordem (lock-free)
        if (order_queue_.try_write(std::move(order))) {
            orders_sent_.fetch_add(1, std::memory_order_relaxed);
            return order.order_id;
        }
        
        // Fila cheia - rejeita ordem e incrementa counter
        orders_rejected_.fetch_add(1, std::memory_order_relaxed);
        return 0;  // 0 indica falha
    }
    
    /**
     * @brief Cancela ordem existente
     * @param order_id ID da ordem a cancelar
     * @return true se cancelamento enfileirado, false se falha
     */
    bool cancel_order(uint64_t order_id) noexcept {
        // Em sistema real, enviaria mensagem de cancel para exchange
        orders_cancelled_.fetch_add(1, std::memory_order_relaxed);
        return true;
    }
    
    /**
     * @brief Processa ordens pendentes (chamado pela thread de ordem)
     * @return Número de ordens processadas nesta iteração
     * 
     * Deve ser chamado periodicamente pela thread dedicada a ordens.
     * Envia ordens para exchange e atualiza estado.
     */
    size_t process_outgoing_orders() noexcept {
        OrderMsg order;
        size_t processed = 0;
        
        // Processa todas as ordens pendentes na fila
        while (order_queue_.try_read(order)) {
            if (send_to_exchange(order)) {
                processed++;
            }
        }
        
        return processed;
    }
    
    /**
     * @brief Retorna estatísticas do gestor de ordens
     */
    void get_stats(uint64_t& sent, uint64_t& rejected, uint64_t& filled, uint64_t& cancelled) const noexcept {
        sent = orders_sent_.load(std::memory_order_acquire);
        rejected = orders_rejected_.load(std::memory_order_acquire);
        filled = orders_filled_.load(std::memory_order_acquire);
        cancelled = orders_cancelled_.load(std::memory_order_acquire);
    }
    
    /**
     * @brief Verifica estado da fila de ordens
     */
    void get_queue_stats(size_t& pending_orders, size_t& available_slots) const noexcept {
        pending_orders = order_queue_.size();
        available_slots = order_queue_.available();
    }
    
private:
    /**
     * @brief Simula envio de ordem para exchange
     * @param order Ordem a ser enviada
     * @return true se sucesso, false se falha
     * 
     * EM PRODUÇÃO: Substituir por:
     * - Conexão TCP acelerada com exchange
     * - Protocolo FIX/OUCH binary
     * - Kernel bypass (DPDK) para networking
     */
    bool send_to_exchange(const OrderMsg& order) noexcept {
        // Simulação - em produção isto seria I/O de rede
        // Retorna sucesso para simplificação
        return true;
    }
};

/**
 * @brief Gerenciador de risco e posições com limites configuráveis
 * 
 * Monitora exposição, PnL, limites de posição e loss diário.
 * Todas as operações são thread-safe e lock-free.
 */
class RiskManager {
private:
    struct Position {
        double avg_price;
        int32_t quantity;
        double unrealized_pnl;
        double realized_pnl;
        uint64_t last_update;
    };
    
    std::array<Position, MAX_SYMBOLS> positions_;
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<double> total_pnl_{0.0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<double> daily_pnl_{0.0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> total_trades_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<double> daily_realized_{0.0};
    
    // Limites de risco configuráveis
    const double max_daily_loss_ = MAX_DAILY_LOSS;
    const double max_position_size_ = MAX_POSITION_SIZE;
    
public:
    RiskManager() {
        // Inicializa todas as posições como zeradas
        for (auto& pos : positions_) {
            pos = {0.0, 0, 0.0, 0.0, 0};
        }
    }
    
    /**
     * @brief Atualiza posição com nova trade executada
     * @param symbol Símbolo do instrumento
     * @param is_buy Direção: true = compra, false = venda
     * @param price Preço de execução
     * @param quantity Quantidade executada
     * 
     * Thread-safe: usa operações atômicas para atualizações.
     */
    void update_position(const char* symbol, bool is_buy, double price, 
                        uint32_t quantity) noexcept {
        
        uint32_t symbol_index = hash_symbol(symbol) % MAX_SYMBOLS;
        auto& position = positions_[symbol_index];
        
        // Atualização thread-safe do position (simplificado)
        if (is_buy) {
            // Compra - atualiza preço médio
            double total_cost = position.avg_price * position.quantity + price * quantity;
            position.quantity += quantity;
            if (position.quantity > 0) {
                position.avg_price = total_cost / position.quantity;
            }
        } else {
            // Venda - realiza PnL
            if (position.quantity > 0) {
                double trade_pnl = (price - position.avg_price) * quantity;
                position.realized_pnl += trade_pnl;
                position.quantity -= quantity;
                
                // Atualiza contadores atômicos
                total_pnl_.fetch_add(trade_pnl, std::memory_order_relaxed);
                daily_pnl_.fetch_add(trade_pnl, std::memory_order_relaxed);
                daily_realized_.fetch_add(trade_pnl, std::memory_order_relaxed);
                total_trades_.fetch_add(1, std::memory_order_relaxed);
            }
        }
        
        position.last_update = HighPrecisionTimer::nanoseconds();
    }
    
    /**
     * @brief Verifica limites de risco para nova ordem
     * @param symbol Símbolo do instrumento
     * @param notional Valor nocional da ordem proposta
     * @return true se dentro dos limites, false se viola risco
     * 
     * Aplica múltiplas verificações:
     * - Limite de posição por símbolo
     * - Limite de loss diário
     * - Concentração máxima
     */
    bool check_risk_limits(const char* symbol, double notional) const noexcept {
        // Verifica limite de loss diário
        double current_daily_pnl = daily_pnl_.load(std::memory_order_acquire);
        if (current_daily_pnl < max_daily_loss_) {
            return false;  // Já atingiu limite de loss diário
        }
        
        // Verifica limite de posição por símbolo
        uint32_t symbol_index = hash_symbol(symbol) % MAX_SYMBOLS;
        const auto& position = positions_[symbol_index];
        
        double current_exposure = std::abs(position.avg_price * position.quantity);
        if (current_exposure + notional > max_position_size_) {
            return false;  // Excederia limite de posição
        }
        
        return true;
    }
    
    /**
     * @brief Retorna estatísticas de PnL
     */
    void get_pnl_stats(double& total, double& daily, double& realized) const noexcept {
        total = total_pnl_.load(std::memory_order_acquire);
        daily = daily_pnl_.load(std::memory_order_acquire);
        realized = daily_realized_.load(std::memory_order_acquire);
    }
    
    /**
     * @brief Retorna estatísticas gerais de trading
     */
    void get_trading_stats(uint64_t& total_trades, size_t& active_positions) const noexcept {
        total_trades = total_trades_.load(std::memory_order_acquire);
        
        // Conta posições ativas (quantity != 0)
        active_positions = 0;
        for (const auto& pos : positions_) {
            if (pos.quantity != 0) {
                active_positions++;
            }
        }
    }
    
    /**
     * @brief Reseta PnL diário (chamado no início do dia)
     */
    void reset_daily_pnl() noexcept {
        daily_pnl_.store(0.0, std::memory_order_release);
        daily_realized_.store(0.0, std::memory_order_release);
    }
    
private:
    /**
     * @brief Hash simples para distribuição de símbolos
     */
    uint32_t hash_symbol(const char* symbol) const noexcept {
        // DJB2 hash algorithm - boa distribuição para símbolos financeiros
        uint32_t hash = 5381;
        for (const char* p = symbol; *p; ++p) {
            hash = ((hash << 5) + hash) + *p;  // hash * 33 + c
        }
        return hash;
    }
};

/**
 * @brief Motor central de execução de estratégias HFT
 * 
 * Coordena processamento de market data, execução de estratégias,
 * gestão de ordens e monitoramento de risco. Arquitetura altamente
 * otimizada para latência submicrossegundo.
 */
class StrategyEngine {
private:
    LockFreeMPSCRingBuffer<MarketDataMsg, RING_BUFFER_SIZE> market_data_buffer_;
    OrderManager order_manager_;
    RiskManager risk_manager_;
    
    // Monitoramento de performance em tempo real
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> messages_processed_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> total_latency_cycles_{0};
    alignas(CACHE_LINE_SIZE) std::array<uint64_t, 10000> latency_samples_;
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<size_t> sample_index_{0};
    
    // Controle de estratégias
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<bool> market_making_enabled_{true};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<bool> arbitrage_enabled_{true};
    
public:
    StrategyEngine() = default;
    
    /**
     * @brief Ponto de entrada para dados de mercado (hot path crítico)
     * @param msg Mensagem de market data recebida
     * 
     * Este método é chamado para CADA mensagem de market data.
     * Deve ser extremamente otimizado - é o caminho mais quente do sistema.
     * Latência alvo: < 500ns por mensagem.
     */
    __attribute__((always_inline))
    void process_market_data(const MarketDataMsg& msg) noexcept {
        uint64_t start_ts = HighPrecisionTimer::rdtsc();
        
        // Mede latência de recebimento (desde timestamp original até agora)
        uint64_t receive_latency = start_ts - msg.timestamp_ns;
        update_latency_stats(receive_latency);
        
        // Filtra mensagens inválidas rapidamente
        if (msg.is_crossed() || msg.is_stale() || msg.spread() <= 0.0) {
            return;
        }
        
        // Executa estratégias de trading
        execute_trading_strategies(msg);
        
        messages_processed_.fetch_add(1, std::memory_order_relaxed);
    }
    
    /**
     * @brief Processa ordens de saída (chamado periodicamente pela thread de ordens)
     * @return Número de ordens processadas
     */
    size_t process_outgoing_orders() noexcept {
        return order_manager_.process_outgoing_orders();
    }
    
    /**
     * @brief Habilita/desabilita estratégias dinamicamente
     */
    void set_strategy_enabled(bool market_making, bool arbitrage) noexcept {
        market_making_enabled_.store(market_making, std::memory_order_release);
        arbitrage_enabled_.store(arbitrage, std::memory_order_release);
    }
    
    /**
     * @brief Retorna estatísticas de performance do engine
     */
    void get_performance_stats(uint64_t& total_msgs, double& avg_latency_ns, 
                              uint64_t& current_qps) const noexcept {
        total_msgs = messages_processed_.load(std::memory_order_acquire);
        uint64_t total_latency = total_latency_cycles_.load(std::memory_order_acquire);
        avg_latency_ns = total_msgs > 0 ? (total_latency / total_msgs) / CPU_FREQ_GHZ : 0.0;
        
        // QPS aproximado baseado nos últimos segundos
        static uint64_t last_count = 0;
        static auto last_time = std::chrono::high_resolution_clock::now();
        
        auto current_time = std::chrono::high_resolution_clock::now();
        auto elapsed_sec = std::chrono::duration_cast<std::chrono::seconds>(
            current_time - last_time).count();
        
        if (elapsed_sec >= 1) {
            current_qps = (total_msgs - last_count) / elapsed_sec;
            last_count = total_msgs;
            last_time = current_time;
        } else {
            current_qps = 0;
        }
    }
    
    /**
     * @brief Retorna estatísticas completas de trading
     */
    void get_trading_stats(uint64_t& orders_sent, uint64_t& orders_rejected,
                          uint64_t& orders_filled, uint64_t& orders_cancelled,
                          double& total_pnl, double& daily_pnl, double& realized_pnl) const noexcept {
        order_manager_.get_stats(orders_sent, orders_rejected, orders_filled, orders_cancelled);
        risk_manager_.get_pnl_stats(total_pnl, daily_pnl, realized_pnl);
    }
    
    /**
     * @brief Retorna estado da fila de market data
     */
    void get_queue_stats(size_t& pending_messages, size_t& available_slots) const noexcept {
        pending_messages = market_data_buffer_.size();
        available_slots = market_data_buffer_.available();
    }
    
private:
    /**
     * @brief Executa lógica de trading baseada em market data
     * @param msg Mensagem de market data processada
     * 
     * Contém estratégias de HFT:
     * 1. Market Making: fornecer liquidez nos lados bid/ask
     * 2. Arbitragem: explorar diferenças de preço entre venues
     * 3. Statistical Arbitrage: pares trading, mean reversion
     */
    __attribute__((always_inline))
    void execute_trading_strategies(const MarketDataMsg& msg) noexcept {
        // ESTRATÉGIA 1: Market Making
        if (market_making_enabled_.load(std::memory_order_acquire)) {
            execute_market_making(msg);
        }
        
        // ESTRATÉGIA 2: Arbitragem entre venues
        if (arbitrage_enabled_.load(std::memory_order_acquire)) {
            execute_arbitrage(msg);
        }
    }
    
    /**
     * @brief Estratégia de market making básico
     */
    __attribute__((always_inline))
    void execute_market_making(const MarketDataMsg& msg) noexcept {
        // Só faz market making se spread for suficiente
        if (msg.spread_bps() > 1.0) {  // Spread > 1 bps
            double bid = msg.bid_price + 0.01;
            double ask = msg.ask_price - 0.01;
            uint32_t quantity = 100;
            
            // Verifica limites de risco antes de enviar
            if (risk_manager_.check_risk_limits(msg.symbol, bid * quantity)) {
                order_manager_.send_order(msg.symbol, bid, quantity, true, 1);
            }
            
            if (risk_manager_.check_risk_limits(msg.symbol, ask * quantity)) {
                order_manager_.send_order(msg.symbol, ask, quantity, false, 1);
            }
        }
    }
    
    /**
     * @brief Estratégia de arbitragem simples
     */
    __attribute__((always_inline))
    void execute_arbitrage(const MarketDataMsg& msg) noexcept {
        // Exemplo: se venue 1 tem preço significativamente diferente
        if (msg.venue_id == 1 && msg.bid_price > 50000.0 && msg.spread() < 2.0) {
            // Lógica de arbitragem cross-venue seria implementada aqui
            // Para demo, apenas envia ordem se condições básicas forem atendidas
            if (risk_manager_.check_risk_limits(msg.symbol, msg.ask_price * 1000)) {
                order_manager_.send_order(msg.symbol, msg.ask_price, 1000, true, 1);
            }
        }
    }
    
    /**
     * @brief Atualiza estatísticas de latência para monitoring
     */
    void update_latency_stats(uint64_t latency_cycles) noexcept {
        total_latency_cycles_.fetch_add(latency_cycles, std::memory_order_relaxed);
        
        // Amostra para cálculo de percentis (cada 1000 mensagens)
        size_t idx = sample_index_.fetch_add(1, std::memory_order_relaxed);
        if (idx % 1000 == 0) {
            latency_samples_[idx % latency_samples_.size()] = latency_cycles;
        }
    }
};

// GERENCIAMENTO DE THREADS E CPU AFFINITY

/**
 * @brief Gerenciador de afinidade de CPU e prioridade de threads
 * 
 * Garante que threads críticas rodem em cores dedicados com prioridade
 * máxima, evitando context switches e garantindo performance consistente.
 */
class ThreadManager {
public:
    /**
     * @brief Define afinidade de CPU para thread específica
     * @param thread Referência para thread std::thread
     * @param cpu_core Core de CPU desejado (0-7 typical)
     * 
     * Pinning de threads a cores específicos é CRÍTICO para HFT:
     * - Evita migration entre cores
     * - Maximiza cache locality  
     * - Prevene interference de outras threads
     */
    static void set_cpu_affinity(std::thread& thread, int cpu_core) noexcept {
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(cpu_core, &cpuset);
        
        int result = pthread_setaffinity_np(thread.native_handle(), 
                                          sizeof(cpu_set_t), &cpuset);
        if (result != 0) {
            std::cerr << "[WARNING] Failed to set CPU affinity for core " << cpu_core 
                      << " (errno: " << result << ")" << std::endl;
        }
    }
    
    /**
     * @brief Define afinidade de CPU para thread atual
     * @param cpu_core Core de CPU desejado
     */
    static void set_current_thread_affinity(int cpu_core) noexcept {
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(cpu_core, &cpuset);
        
        int result = pthread_setaffinity_np(pthread_self(), 
                                          sizeof(cpu_set_t), &cpuset);
        if (result != 0) {
            std::cerr << "[WARNING] Failed to set CPU affinity for current thread" << std::endl;
        }
    }
    
    /**
     * @brief Define prioridade de tempo real para thread
     * @param thread Thread alvo
     * @param priority Prioridade SCHED_FIFO (1-99, 99=max)
     * 
     * SCHED_FIFO garante que thread HFT não seja preemptada por
     * threads normais, reduzindo jitter de latência.
     */
    static void set_realtime_priority(std::thread& thread, int priority = 99) noexcept {
        sched_param sch_params;
        sch_params.sched_priority = priority;
        
        int result = pthread_setschedparam(thread.native_handle(), 
                                         SCHED_FIFO, &sch_params);
        if (result != 0) {
            std::cerr << "[WARNING] Failed to set realtime priority (need root?)" << std::endl;
        }
    }
    
    /**
     * @brief Define nome da thread para debugging
     */
    static void set_thread_name(std::thread& thread, const char* name) noexcept {
        pthread_setname_np(thread.native_handle(), name);
    }
    
    /**
     * @brief Define nome da thread atual
     */
    static void set_current_thread_name(const char* name) noexcept {
        pthread_setname_np(pthread_self(), name);
    }
};

/**
 * @brief Sistema principal de trading HFT - orquestra todos os componentes
 * 
 * Gerencia ciclo de vida completo: inicialização, execução, monitoring,
 * e shutdown graceful. Implementa arquitetura multi-thread com isolamento
 * de cores e prioridades apropriadas para cada função.
 */
class TrainPrometheusSystem {
private:
    StrategyEngine strategy_engine_;
    
    // Controle de execução do sistema
    std::atomic<bool> running_{false};
    std::vector<std::thread> worker_threads_;
    
    // Estatísticas e monitoring do sistema
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> system_uptime_{0};
    alignas(CACHE_LINE_SIZE) CacheAlignedAtomic<uint64_t> last_stats_time_{0};
    
public:
    TrainPrometheusSystem() = default;
    ~TrainPrometheusSystem() {
        if (running_.load(std::memory_order_acquire)) {
            shutdown();
        }
    }
    
    /**
     * @brief Inicializa o sistema de trading
     * @return true se inicialização bem-sucedida, false se erro
     * 
     * Realiza:
     * - Calibração de frequência do CPU
     * - Verificação de requisitos do sistema
     * - Pré-alocação de recursos
     */
    bool initialize() noexcept {
        std::cout << "=== TRAIN_PROMETHEUS v2.1 Initializing ===" << std::endl;
        
        // Calibra frequência do CPU se não especificada
        if (CPU_FREQ_GHZ == 0) {
            double calibrated_freq = HighPrecisionTimer::calibrate_cpu_frequency();
            std::cout << "[INIT] CPU Frequency calibrated: " << calibrated_freq << " GHz" << std::endl;
        } else {
            std::cout << "[INIT] CPU Frequency: " << CPU_FREQ_GHZ << " GHz" << std::endl;
        }
        
        // Valida requisitos do sistema
        if (!validate_system_requirements()) {
            std::cerr << "[ERROR] System requirements validation failed" << std::endl;
            return false;
        }
        
        std::cout << "[INIT] Cache Line Size: " << CACHE_LINE_SIZE << " bytes" << std::endl;
        std::cout << "[INIT] Ring Buffer Size: " << RING_BUFFER_SIZE << " slots" << std::endl;
        std::cout << "[INIT] Strategy Threads: " << NUM_STRATEGY_THREADS << std::endl;
        std::cout << "[INIT] System validation completed successfully" << std::endl;
        
        return true;
    }
    
    /**
     * @brief Inicia o sistema de trading
     * 
     * Lança todas as threads especializadas com affinity e prioridade
     * apropriadas. Não retorna até shutdown() ser chamado.
     */
    void start() {
        if (running_.load(std::memory_order_acquire)) {
            std::cout << "[WARNING] System already running" << std::endl;
            return;
        }
        
        running_.store(true, std::memory_order_release);
        
        std::cout << "=== Starting TRAIN_PROMETHEUS v2.1 ===" << std::endl;
        std::cout << "[START] Launching specialized threads..." << std::endl;
        
        // Thread de Network RX - Core 0 (prioridade máxima)
        worker_threads_.emplace_back([this]() { network_rx_thread(); });
        ThreadManager::set_cpu_affinity(worker_threads_.back(), 0);
        ThreadManager::set_realtime_priority(worker_threads_.back(), 99);
        ThreadManager::set_thread_name(worker_threads_.back(), "train_prometheus_net_rx");
        
        // Thread de Order Processing - Core 1 (alta prioridade)
        worker_threads_.emplace_back([this]() { order_processing_thread(); });
        ThreadManager::set_cpu_affinity(worker_threads_.back(), 1);
        ThreadManager::set_realtime_priority(worker_threads_.back(), 98);
        ThreadManager::set_thread_name(worker_threads_.back(), "train_prometheus_orders");
        
        // Threads de Strategy Execution - Cores 2-5 (prioridade alta)
        for (int i = 0; i < NUM_STRATEGY_THREADS; ++i) {
            worker_threads_.emplace_back([this, i]() { strategy_thread(i); });
            ThreadManager::set_cpu_affinity(worker_threads_.back(), 2 + i);
            ThreadManager::set_realtime_priority(worker_threads_.back(), 97);
            ThreadManager::set_thread_name(worker_threads_.back(), 
                                         ("train_prometheus_strat_" + std::to_string(i)).c_str());
        }
        
        // Thread de System Monitoring - Core 6 (prioridade normal)
        worker_threads_.emplace_back([this]() { monitoring_thread(); });
        ThreadManager::set_cpu_affinity(worker_threads_.back(), 6);
        ThreadManager::set_thread_name(worker_threads_.back(), "train_prometheus_monitor");
        
        std::cout << "[START] System started with " << worker_threads_.size() << " threads" << std::endl;
        std::cout << "[START] Core allocation:" << std::endl;
        std::cout << "[START]   Core 0: Network RX" << std::endl;
        std::cout << "[START]   Core 1: Order Processing" << std::endl;
        std::cout << "[START]   Cores 2-5: Strategy Execution (" << NUM_STRATEGY_THREADS << " threads)" << std::endl;
        std::cout << "[START]   Core 6: System Monitoring" << std::endl;
        std::cout << "[START]   Core 7: Unassigned (backup)" << std::endl;
    }
    
    /**
     * @brief Para o sistema de trading gracefulmente
     * 
     * Sinaliza todas as threads para parar e espera por conclusão
     * limpa. Preserva estatísticas finais e estado para análise.
     */
    void shutdown() {
        if (!running_.load(std::memory_order_acquire)) {
            return;
        }
        
        std::cout << "=== Shutting down TRAIN_PROMETHEUS v2.1 ===" << std::endl;
        running_.store(false, std::memory_order_release);
        
        // Espera por todas as threads com timeout
        for (auto& thread : worker_threads_) {
            if (thread.joinable()) {
                thread.join();
            }
        }
        
        worker_threads_.clear();
        print_final_stats();
        std::cout << "[SHUTDOWN] Shutdown complete" << std::endl;
    }
    
    /**
     * @brief Executa benchmark completo de performance
     * 
     * Testa latência, throughput e estabilidade do sistema sob carga.
     * Útil para profiling e validação de performance.
     */
    void run_benchmark() {
        std::cout << "=== Running Performance Benchmark ===" << std::endl;
        
        const size_t iterations = 2000000;  // 2M mensagens para estatísticas robustas
        std::vector<uint64_t> latencies;
        latencies.reserve(iterations);
        
        auto start_time = std::chrono::high_resolution_clock::now();
        
        // Benchmark de latência
        for (size_t i = 0; i < iterations; ++i) {
            MarketDataMsg msg{};
            msg.timestamp_ns = HighPrecisionTimer::rdtsc();
            msg.sequence = i;
            std::strcpy(msg.symbol, "BTC-USD");
            msg.bid_price = 50000.0 + (i % 100) * 0.01;
            msg.ask_price = msg.bid_price + 0.5 + (i % 10) * 0.01;
            msg.bid_size = 1000;
            msg.ask_size = 1000;
            msg.venue_id = 1;
            
            uint64_t start_ts = HighPrecisionTimer::rdtsc();
            strategy_engine_.process_market_data(msg);
            uint64_t end_ts = HighPrecisionTimer::rdtsc();
            
            latencies.push_back(end_ts - start_ts);
            
            // Inserir pequena pausa para simular intervalo real entre mensagens
            if (i % 1000 == 0) {
                HighPrecisionTimer::busy_wait_ns(1000);  // 1µs
            }
        }
        
        auto end_time = std::chrono::high_resolution_clock::now();
        auto duration_ms = std::chrono::duration_cast<std::chrono::milliseconds>(
            end_time - start_time).count();
        
        // Calcula estatísticas detalhadas
        std::sort(latencies.begin(), latencies.end());
        
        uint64_t sum_latency = std::accumulate(latencies.begin(), latencies.end(), 0ULL);
        double avg_latency_cycles = static_cast<double>(sum_latency) / latencies.size();
        double avg_latency_ns = avg_latency_cycles / CPU_FREQ_GHZ;
        
        double p50_latency_ns = latencies[iterations * 50 / 100] / CPU_FREQ_GHZ;
        double p90_latency_ns = latencies[iterations * 90 / 100] / CPU_FREQ_GHZ;
        double p95_latency_ns = latencies[iterations * 95 / 100] / CPU_FREQ_GHZ;
        double p99_latency_ns = latencies[iterations * 99 / 100] / CPU_FREQ_GHZ;
        double p999_latency_ns = latencies[iterations * 999 / 1000] / CPU_FREQ_GHZ;
        double max_latency_ns = latencies.back() / CPU_FREQ_GHZ;
        
        double throughput = (iterations * 1000.0) / duration_ms;
        
        // Output resultados
        std::cout << "=== BENCHMARK RESULTS ===" << std::endl;
        std::cout << "Duration: " << duration_ms << " ms" << std::endl;
        std::cout << "Messages: " << iterations << std::endl;
        std::cout << "Throughput: " << throughput << " msg/sec" << std::endl;
        std::cout << "Latency Statistics:" << std::endl;
        std::cout << "  Average: " << avg_latency_ns << " ns" << std::endl;
        std::cout << "  P50:     " << p50_latency_ns << " ns" << std::endl;
        std::cout << "  P90:     " << p90_latency_ns << " ns" << std::endl;
        std::cout << "  P95:     " << p95_latency_ns << " ns" << std::endl;
        std::cout << "  P99:     " << p99_latency_ns << " ns" << std::endl;
        std::cout << "  P99.9:   " << p999_latency_ns << " ns" << std::endl;
        std::cout << "  Max:     " << max_latency_ns << " ns" << std::endl;
        
        // Verifica se atinge requisitos HFT
        if (p99_latency_ns < 1000) {  // < 1µs
            std::cout << "✅ PASS: Meets HFT latency requirements" << std::endl;
        } else {
            std::cout << "❌ FAIL: Does not meet HFT latency requirements" << std::endl;
        }
    }
    
private:
    bool validate_system_requirements() const noexcept {
        // Verifica se CPU suporta AVX2 (requerido para otimizações)
        __builtin_cpu_init();
        if (!__builtin_cpu_supports("avx2")) {
            std::cerr << "[ERROR] CPU does not support AVX2 instructions" << std::endl;
            return false;
        }
        
        // Verifica tamanho de cache line (deve ser 64 bytes)
        if (CACHE_LINE_SIZE != 64) {
            std::cerr << "[ERROR] Unsupported cache line size: " << CACHE_LINE_SIZE << std::endl;
            return false;
        }
        
        // Verifica se temos cores suficientes
        unsigned int cores_available = std::thread::hardware_concurrency();
        if (cores_available < 4) {
            std::cerr << "[ERROR] Insufficient CPU cores: " << cores_available << " (minimum 4 required)" << std::endl;
            return false;
        }
        
        std::cout << "[VALIDATION] CPU Cores: " << cores_available << std::endl;
        std::cout << "[VALIDATION] AVX2: Supported" << std::endl;
        std::cout << "[VALIDATION] Cache Line: " << CACHE_LINE_SIZE << " bytes" << std::endl;
        
        return true;
    }
    
    void network_rx_thread() noexcept {
        ThreadManager::set_current_thread_affinity(0);
        ThreadManager::set_current_thread_name("train_prometheus_net_rx");
        
        std::cout << "[NETWORK] Network RX thread started on core 0" << std::endl;
        
        uint64_t packets_processed = 0;
        auto last_report = HighPrecisionTimer::nanoseconds();
        
        while (running_.load(std::memory_order_acquire)) {
            // Simula recebimento de pacotes de rede de market data
            simulate_market_data_feed();
            packets_processed++;
            
            // Report de throughput a cada segundo
            uint64_t now = HighPrecisionTimer::nanoseconds();
            if (now - last_report > 1000000000ULL) {  // 1 segundo
                std::cout << "[NETWORK] Throughput: " << packets_processed << " packets/sec" << std::endl;
                packets_processed = 0;
                last_report = now;
            }
            
            // Polling de alta frequência - sem sleep para mínima latência
            HighPrecisionTimer::busy_wait_ns(50);  // 50ns entre polls
        }
        
        std::cout << "[NETWORK] RX thread stopped" << std::endl;
    }
    
    void order_processing_thread() noexcept {
        ThreadManager::set_current_thread_affinity(1);
        ThreadManager::set_current_thread_name("train_prometheus_orders");
        
        std::cout << "[ORDERS] Order processing thread started on core 1" << std::endl;
        
        while (running_.load(std::memory_order_acquire)) {
            size_t processed = strategy_engine_.process_outgoing_orders();
            
            if (processed > 0) {
                // Log opcional de atividade de ordens
            }
            
            // Processa ordens a cada 1µs (frequência alta para HFT)
            HighPrecisionTimer::busy_wait_ns(1000);
        }
        
        std::cout << "[ORDERS] Order thread stopped" << std::endl;
    }
    
    void strategy_thread(int id) noexcept {
        ThreadManager::set_current_thread_affinity(2 + id);
        ThreadManager::set_current_thread_name(
            ("train_prometheus_strat_" + std::to_string(id)).c_str());
        
        std::cout << "[STRATEGY] Strategy thread " << id << " started on core " << (2 + id) << std::endl;
        
        while (running_.load(std::memory_order_acquire)) {
            // Estratégias são acionadas por market data na network thread
            // Esta thread está principalmente para estratégias assíncronas de longer-term
            HighPrecisionTimer::busy_wait_ns(5000);  // 5µs entre iterações
        }
        
        std::cout << "[STRATEGY] Strategy thread " << id << " stopped" << std::endl;
    }
    
    void monitoring_thread() noexcept {
        ThreadManager::set_current_thread_affinity(6);
        ThreadManager::set_current_thread_name("train_prometheus_monitor");
        
        std::cout << "[MONITOR] Monitoring thread started on core 6" << std::endl;
        
        auto last_stats_time = std::chrono::high_resolution_clock::now();
        uint64_t monitoring_cycles = 0;
        
        while (running_.load(std::memory_order_acquire)) {
            auto current_time = std::chrono::high_resolution_clock::now();
            auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(
                current_time - last_stats_time).count();
            
            // Print estatísticas detalhadas a cada 10 segundos
            if (elapsed >= 10) {
                print_detailed_stats();
                last_stats_time = current_time;
            }
            
            monitoring_cycles++;
            std::this_thread::sleep_for(std::chrono::seconds(1));
            system_uptime_.fetch_add(1, std::memory_order_relaxed);
        }
        
        std::cout << "[MONITOR] Monitoring thread stopped after " << monitoring_cycles << " cycles" << std::endl;
    }
    
    void simulate_market_data_feed() noexcept {
        // Simula feed real de market data com múltiplos símbolos e venues
        static uint64_t sequence = 0;
        static const char* symbols[] = {"BTC-USD", "ETH-USD", "AAPL", "GOOGL", "MSFT"};
        static const size_t num_symbols = sizeof(symbols) / sizeof(symbols[0]);
        
        MarketDataMsg msg{};
        msg.timestamp_ns = HighPrecisionTimer::rdtsc();
        msg.sequence = sequence++;
        msg.venue_id = (sequence % 3) + 1;  // Rotate entre 3 exchanges
        
        // Seleciona símbolo baseado no sequence number
        const char* current_symbol = symbols[sequence % num_symbols];
        std::strcpy(msg.symbol, current_symbol);
        
        // Gera preços realistas com alguma volatilidade
        double base_price = 50000.0;
        if (std::strcmp(current_symbol, "ETH-USD") == 0) base_price = 3000.0;
        if (std::strcmp(current_symbol, "AAPL") == 0) base_price = 150.0;
        if (std::strcmp(current_symbol, "GOOGL") == 0) base_price = 2800.0;
        if (std::strcmp(current_symbol, "MSFT") == 0) base_price = 330.0;
        
        double volatility = (sequence % 100) * 0.01;  // Pequena volatilidade
        msg.bid_price = base_price + volatility;
        msg.ask_price = msg.bid_price + 0.5 + (sequence % 10) * 0.01;
        msg.bid_size = 1000 + (sequence % 500);
        msg.ask_size = 1000 + (sequence % 500);
        
        // Processa mensagem através do strategy engine
        strategy_engine_.process_market_data(msg);
    }
    
    void print_detailed_stats() noexcept {
        uint64_t total_msgs;
        double avg_latency_ns;
        uint64_t current_qps;
        strategy_engine_.get_performance_stats(total_msgs, avg_latency_ns, current_qps);
        
        uint64_t orders_sent, orders_rejected, orders_filled, orders_cancelled;
        double total_pnl, daily_pnl, realized_pnl;
        strategy_engine_.get_trading_stats(orders_sent, orders_rejected, orders_filled, 
                                         orders_cancelled, total_pnl, daily_pnl, realized_pnl);
        
        size_t pending_messages, available_slots;
        strategy_engine_.get_queue_stats(pending_messages, available_slots);
        
        uint64_t uptime = system_uptime_.load(std::memory_order_acquire);
        
        std::cout << "=== SYSTEM STATISTICS ===" << std::endl;
        std::cout << "Uptime: " << uptime << " seconds" << std::endl;
        std::cout << "Performance:" << std::endl;
        std::cout << "  Messages Processed: " << total_msgs << std::endl;
        std::cout << "  Current QPS: " << current_qps << std::endl;
        std::cout << "  Avg Latency: " << avg_latency_ns << " ns" << std::endl;
        std::cout << "  Queue: " << pending_messages << " pending, " 
                  << available_slots << " available" << std::endl;
        std::cout << "Trading:" << std::endl;
        std::cout << "  Orders Sent: " << orders_sent << std::endl;
        std::cout << "  Orders Rejected: " << orders_rejected << std::endl;
        std::cout << "  Orders Filled: " << orders_filled << std::endl;
        std::cout << "  Orders Cancelled: " << orders_cancelled << std::endl;
        std::cout << "  Total PnL: $" << total_pnl << std::endl;
        std::cout << "  Daily PnL: $" << daily_pnl << std::endl;
        std::cout << "  Realized PnL: $" << realized_pnl << std::endl;
    }
    
    void print_final_stats() noexcept {
        std::cout << "=== FINAL SYSTEM STATISTICS ===" << std::endl;
        print_detailed_stats();
    }
};

std::atomic<bool> shutdown_requested{false};

/**
 * @brief Handler de sinais para shutdown graceful
 */
void signal_handler(int signal) {
    std::cout << std::endl << "[SYSTEM] Received signal " << signal << ", initiating shutdown..." << std::endl;
    shutdown_requested.store(true);
}

/**
 * @brief Ponto de entrada da aplicação
 * 
 * Suporta múltiplos modos de operação:
 * - Modo produção: ./train_prometheus
 * - Modo benchmark: ./train_prometheus --benchmark  
 * - Modo stress-test: ./train_prometheus --stress
 * - Modo help: ./train_prometheus --help
 */
int main(int argc, char* argv[]) {
    std::cout << "=== TRAIN_PROMETHEUS v2.1 - HFT Trading Engine ===" << std::endl;
    std::cout << "Build: " << __DATE__ << " " << __TIME__ << std::endl;
    std::cout << "Compiler: " << __VERSION__ << std::endl;
    
    // Configura handlers de sinal para shutdown graceful
    std::signal(SIGINT, signal_handler);
    std::signal(SIGTERM, signal_handler);
    std::signal(SIGQUIT, signal_handler);
    
    // Parse argumentos de linha de comando
    bool benchmark_mode = false;
    bool stress_test_mode = false;
    bool help_mode = false;
    
    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        if (arg == "--benchmark" || arg == "-b") {
            benchmark_mode = true;
        } else if (arg == "--stress" || arg == "-s") {
            stress_test_mode = true;
        } else if (arg == "--help" || arg == "-h") {
            help_mode = true;
        } else {
            std::cout << "Unknown argument: " << arg << std::endl;
            help_mode = true;
        }
    }
    
    // Mostra help se solicitado
    if (help_mode) {
        std::cout << "Usage: " << argv[0] << " [OPTIONS]" << std::endl;
        std::cout << "Options:" << std::endl;
        std::cout << "  --benchmark, -b    Run performance benchmark" << std::endl;
        std::cout << "  --stress, -s       Run stress test" << std::endl;
        std::cout << "  --help, -h         Show this help message" << std::endl;
        return 0;
    }
    
    // Cria e inicializa sistema
    TrainPrometheus::TrainPrometheusSystem system;
    
    if (!system.initialize()) {
        std::cerr << "[ERROR] System initialization failed" << std::endl;
        return 1;
    }
    
    // Executa em modo benchmark se especificado
    if (benchmark_mode) {
        system.run_benchmark();
        return 0;
    }
    
    // Inicia sistema em modo produção
    system.start();
    
    std::cout << "[SYSTEM] System is running. Press Ctrl+C to shutdown." << std::endl;
    std::cout << "[SYSTEM] Uptime counter started..." << std::endl;
    
    // Loop principal de controle
    while (!shutdown_requested.load()) {
        std::this_thread::sleep_for(std::chrono::seconds(1));
    }
    
    // Shutdown graceful
    std::cout << "[SYSTEM] Shutdown signal received, initiating shutdown sequence..." << std::endl;
    system.shutdown();
    
    std::cout << "TRAIN_PROMETHEUS v2.1 shutdown complete." << std::endl;
    return 0;
}
