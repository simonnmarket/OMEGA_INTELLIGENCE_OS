    async def _calculate_theoretical_profit_v7(self, path: Tuple[str, str, str], prices: Dict) -> Decimal:
        """
        CALCULA LUCRO TEÓRICO DA ARBITRAGEM TRIANGULAR
        """
        pair1, pair2, pair3 = path
        
        try:
            # Preços atuais
            price1 = prices[pair1]
            price2 = prices[pair2] 
            price3 = prices[pair3]
            
            # Cálculo da arbitragem triangular
            # Exemplo: BTC/USD -> USD/ETH -> ETH/BTC
            theoretical_rate = (Decimal('1') / price1) * (Decimal('1') / price2) * price3
            
            # Lucro teórico (deve ser > 1 para haver arbitragem)
            profit = theoretical_rate - Decimal('1')
            
            return max(profit, Decimal('0'))
            
        except Exception as e:
            logging.error(f"Erro cálculo lucro teórico {path}: {e}")
            return Decimal('0')

    async def _calculate_execution_probability_v7(self, path: Tuple[str, str, str], prices: Dict, volumes: Dict) -> Decimal:
        """
        CALCULA PROBABILIDADE DE EXECUÇÃO BEM-SUCEDIDA
        """
        probability_factors = []
        
        # Fator 1: Liquidez dos pares
        liquidity_prob = await self._liquidity_execution_probability(path, volumes)
        probability_factors.append(liquidity_prob)
        
        # Fator 2: Estabilidade de preços
        stability_prob = await self._price_stability_probability(path, prices)
        probability_factors.append(stability_prob)
        
        # Fator 3: Spread de mercado
        spread_prob = await self._spread_execution_probability(path, prices)
        probability_factors.append(spread_prob)
        
        # Fator 4: Velocidade de execução
        speed_prob = await self._execution_speed_probability(path)
        probability_factors.append(speed_prob)
        
        # Média ponderada dos fatores
        weights = [Decimal('0.3'), Decimal('0.3'), Decimal('0.2'), Decimal('0.2')]
        weighted_prob = sum(p * w for p, w in zip(probability_factors, weights))
        
        return min(weighted_prob, Decimal('1.0'))

    async def _filter_opportunities_v7(self, opportunities: List[ArbitrageOpportunityV7], 
                                     prices: Dict, volumes: Dict) -> List[ArbitrageOpportunityV7]:
        """
        FILTRA E RANQUEIA OPORTUNIDADES DE ARBITRAGEM
        """
        filtered = []
        
        for opportunity in opportunities:
            # Critérios de filtragem
            if (opportunity.theoretical_profit > self.arbitrage_params['min_profit_threshold'] and
                opportunity.execution_probability > Decimal('0.65') and
                opportunity.liquidity_score > Decimal('0.60') and
                opportunity.security_clearance):
                
                # Calcular score final ponderado

                    opportunity.theoretical_profit * Decimal('0.4') +
                    opportunity.execution_probability * Decimal('0.3') +

                    opportunity.risk_adjusted_score * Decimal('0.1')

                # Adicionar score ao objeto
                opportunity.risk_adjusted_score = final_score
                filtered.append(opportunity)
        
        # Ordenar por score
        filtered.sort(key=lambda x: x.risk_adjusted_score, reverse=True)
        return filtered

    def _calculate_arbitrage_position_size(self, opportunity: ArbitrageOpportunityV7, 
                                         allocated_capital: Decimal) -> Decimal:

        CALCULA TAMANHO DA POSIÇÃO PARA ARBITRAGEM

        min_size = Decimal('100')  # Mínimo €100
        
        return max(min(risk_adjusted_size, max_size), min_size)

    def _calculate_execution_window(self, opportunity: ArbitrageOpportunityV7) -> Decimal:

    # MÉTODOS DE GERAÇÃO DE CAMINHOS
    def _generate_triangular_paths(self) -> List[Tuple[str, str, str]]:
        """Gera todos os caminhos triangulares possíveis"""
        paths = []
        currencies = list(self.trading_pairs.keys())
        
        for base in currencies:
            for intermediate in currencies:
                for target in currencies:
                    if base != intermediate != target != base:
                        # Verificar se os pares existem
                        pair1 = f"{base}/{intermediate}"
                        pair2 = f"{intermediate}/{target}"
                        pair3 = f"{target}/{base}"
                        
                        if (self._pair_exists(pair1) and 
                            self._pair_exists(pair2) and 
                            self._pair_exists(pair3)):
                            paths.append((pair1, pair2, pair3))
        
        return paths

    def _pair_exists(self, pair: str) -> bool:
        """Verifica se um par existe no universo de trading"""
        base, quote = pair.split('/')
        return base in self.trading_pairs and quote in self.trading_pairs.get(base, [])

    # MÉTODOS DE ESTIMAÇÃO
    async def _estimate_actual_profit_v7(self, theoretical_profit: Decimal, path: Tuple[str, str, str], volumes: Dict) -> Decimal:
        """Estima lucro real considerando slippage e fees"""
        # Slippage estimada baseada na liquidez
        slippage = Decimal('0.001')  # 0.1% base
        
        # Ajustar slippage pela liquidez
        for pair in path:
            if volumes.get(pair, Decimal('0')) < Decimal('500000'):  # $500k
                slippage += Decimal('0.0005')
        
        # Fees estimadas (0.2% total para 3 trades)
        fees = Decimal('0.002')
        
        # Lucro real estimado
        actual_profit = theoretical_profit - slippage - fees
        
        return max(actual_profit, Decimal('0'))
