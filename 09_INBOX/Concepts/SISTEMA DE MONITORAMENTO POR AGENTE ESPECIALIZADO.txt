"""
Sistema de Monitoramento Prometheus para Arquitetura de Múltiplos Agentes
Métricas individuais por agente, filtro e camada de calibração
"""
from prometheus_client import start_http_server, Gauge, Counter, Histogram, Enum
import time
from typing import Dict, List
import threading
from datetime import datetime

class AgentMetrics:
    """Métricas individuais para cada agente especializado"""
    
    def __init__(self, agent_name: str, agent_type: str):
        self.agent_name = agent_name
        self.agent_type = agent_type
        
        # Métricas de performance individual
        self.processing_time = Histogram(
            f'{agent_name}_processing_time_seconds',
            f'Processing time for {agent_name}',
            ['agent_type', 'input_type'],
            buckets=[0.0001, 0.0005, 0.001, 0.005, 0.01, 0.05]
        )
        
        self.decision_confidence = Gauge(
            f'{agent_name}_decision_confidence',
            f'Decision confidence for {agent_name}',
            ['asset_class', 'timeframe']
        )
        
        self.error_count = Counter(
            f'{agent_name}_errors_total',
            f'Total errors for {agent_name}',
            ['error_type', 'severity']
        )

class FilterMetrics:
    """Métricas para cada filtro de processamento"""
    
    def __init__(self, filter_name: str, filter_type: str):
        self.filter_name = filter_name
        self.filter_type = filter_type
        
        self.efficiency = Gauge(
            f'{filter_name}_efficiency_ratio',
            f'Efficiency ratio for {filter_name} filter',
            ['data_type']
        )
        
        self.throughput = Counter(
            f'{filter_name}_processed_items_total',
            f'Total items processed by {filter_name}',
            ['item_type']
        )
        
        self.rejection_rate = Gauge(
            f'{filter_name}_rejection_rate',
            f'Rejection rate for {filter_name} filter',
            ['rejection_reason']
        )

class CalibrationMetrics:
    """Métricas para sistemas de calibração individual"""
    
    def __init__(self, calibration_name: str):
        self.calibration_name = calibration_name
        
        self.accuracy = Gauge(
            f'{calibration_name}_accuracy_score',
            f'Accuracy score for {calibration_name}',
            ['instrument_type']
        )
        
        self.calibration_time = Histogram(
            f'{calibration_name}_calibration_time_seconds',
            f'Time taken for calibration by {calibration_name}',
            ['complexity_level'],
            buckets=[0.001, 0.005, 0.01, 0.05, 0.1, 0.5]
        )
        
        self.drift_detected = Counter(
            f'{calibration_name}_drift_events_total',
            f'Calibration drift events for {calibration_name}',
            ['drift_severity']
        )

class MaestroCoreMetrics:
    """Métricas para o núcleo maestro coordenador"""
    
    def __init__(self):
        # Performance de orquestração
        self.orchestration_latency = Histogram(
            'maestro_orchestration_latency_seconds',
            'Latency of maestro orchestration decisions',
            ['decision_type'],
            buckets=[0.0001, 0.0005, 0.001, 0.005, 0.01]
        )
        
        self.agent_coordination = Gauge(
            'maestro_agent_coordination_score',
            'Coordination efficiency score across agents',
            ['orchestration_cycle']
        )
        
        self.resource_allocation = Gauge(
            'maestro_resource_allocation_ratio',
            'Resource allocation efficiency ratio',
            ['resource_type']
        )
        
        # Controle de qualidade da orquestra
        self.orchestra_harmony = Gauge(
            'orchestra_harmony_score',
            'Overall harmony score of the agent orchestra',
            ['time_window']
        )
        
        self.dissonance_events = Counter(
            'orchestra_dissonance_events_total',
            'Total dissonance events between agents',
            ['conflict_type', 'severity']
        )

class PrometheusOrchestraMonitor:
    """Sistema completo de monitoramento para arquitetura de orquestra"""
    
    def __init__(self, port=9091):
        self.port = port
        self.agents: Dict[str, AgentMetrics] = {}
        self.filters: Dict[str, FilterMetrics] = {}
        self.calibrations: Dict[str, CalibrationMetrics] = {}
        self.maestro = MaestroCoreMetrics()
        
    def register_agent(self, agent_name: str, agent_type: str):
        """Registra um novo agente no sistema de métricas"""
        self.agents[agent_name] = AgentMetrics(agent_name, agent_type)
        return self.agents[agent_name]
    
    def register_filter(self, filter_name: str, filter_type: str):
        """Registra um novo filtro no sistema de métricas"""
        self.filters[filter_name] = FilterMetrics(filter_name, filter_type)
        return self.filters[filter_name]
    
    def register_calibration(self, calibration_name: str):
        """Registra um novo sistema de calibração"""
        self.calibrations[calibration_name] = CalibrationMetrics(calibration_name)
        return self.calibrations[calibration_name]
    
    def start_monitoring_server(self):
        """Inicia servidor de métricas Prometheus"""
        start_http_server(self.port)
        print(f"🎼 Prometheus Orchestra Monitoring started on port {self.port}")
        
        # Thread para atualização contínua de métricas de harmonia
        self._start_harmony_monitor()
    
    def _start_harmony_monitor(self):
        """Inicia monitor contínuo de harmonia da orquestra"""
        def harmony_monitor():
            while True:
                harmony_score = self._calculate_orchestra_harmony()
                self.maestro.orchestra_harmony.labels(
                    time_window='realtime'
                ).set(harmony_score)
                time.sleep(1)
        
        thread = threading.Thread(target=harmony_monitor, daemon=True)
        thread.start()
    
    def _calculate_orchestra_harmony(self) -> float:
        """Calcula score de harmonia baseado no estado dos agentes"""
        # Implementação simplificada - na prática usaria ML
        active_agents = len(self.agents)
        if active_agents == 0:
            return 1.0
            
        error_rate = sum(
            1 for agent in self.agents.values() 
            if agent.error_count._value.get() > 0
        ) / active_agents
        
        return 1.0 - min(error_rate, 1.0)
