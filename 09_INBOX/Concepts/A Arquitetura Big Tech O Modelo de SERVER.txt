A Arquitetura "Big Tech": O Modelo de Serviços
O princípio fundamental é a desacoplamento. Cada componente do sistema é um serviço independente, com uma única responsabilidade, que se comunica através de uma API bem definida (no nosso caso, o socket TCP/IP).

O EA é um Cliente "Burro": Sua única função é conectar-se ao serviço central, receber sinais e executar ordens. Ele não deve ter lógica complexa nem iniciar outros processos.
O Cérebro Python é um Serviço Contínuo: O motor de análise (NumeiaTradingSystem) deve rodar 24/7 em segundo plano, independentemente de um EA estar conectado ou não. Ele analisa o mercado, gera sinais e os disponibiliza.
O Orquestrador é o Ponto Único de Entrada: Teremos um único script principal, um "daemon", que inicia e gerencia todos os serviços Python. Isso é o que você propôs como "servidor oficial".
Este modelo é robusto, escalável e profissional. O cérebro do sistema nunca para de trabalhar, mesmo que o EA seja desligado.

Por que esta estrutura é superior?

Clareza: Tudo que é "servidor" está na pasta Server/.
Manutenibilidade: Cada arquivo tem uma responsabilidade única.
Escalabilidade: Para adicionar um novo serviço (ex: um dashboard web), basta adicioná-lo à pasta Server/ e chamá-lo no main_server.py.

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
ORQUESTRADOR PRINCIPAL - SAMSUNG GLOBAL MARKET
Este é o único ponto de entrada para iniciar todos os serviços do sistema.
Arquitetura de Serviços Profissional.
"""

import threading
import time
import signal
import sys
import logging
from trading_engine import TradingEngine
from mt5_socket_service import MT5SocketService

# Configuração de logging unificada
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler("server.log"),
        logging.StreamHandler(sys.stdout)
    ]
)

class MainServer:
    def __init__(self):
        self.is_running = False
        self.services = {}

    def start_services(self):
        """Inicia todos os serviços do sistema."""
        logging.info("🚀 Iniciando o Samsung Global Market Server...")
        self.is_running = True

        # 1. Iniciar o Motor de Trading (Cérebro)
        try:
            self.trading_engine = TradingEngine()
            # O motor rodará em sua própria thread
            self.services['trading_engine'] = threading.Thread(
                target=self.trading_engine.run, 
                name="TradingEngine", 
                daemon=True
            )
            self.services['trading_engine'].start()
            logging.info("✅ Serviço 'TradingEngine' iniciado.")
        except Exception as e:
            logging.error(f"❌ Falha ao iniciar TradingEngine: {e}")
            self.stop_services()
            return

        # 2. Iniciar o Serviço de Socket para MT5 (Comunicação)
        try:
            self.socket_service = MT5SocketService()
            # O socket service precisa de uma referência ao motor para enviar sinais
            self.socket_service.set_trading_engine(self.trading_engine)
            self.services['socket_service'] = threading.Thread(
                target=self.socket_service.start, 
                name="SocketService", 
                daemon=True
            )
            self.services['socket_service'].start()
            logging.info("✅ Serviço 'SocketService' iniciado na porta 5555.")
        except Exception as e:
            logging.error(f"❌ Falha ao iniciar SocketService: {e}")
            self.stop_services()
            return

        logging.info("🎉 Todos os serviços foram iniciados com sucesso!")
        logging.info("📡 O EA 'SamsungGlobalMarket_EA' agora pode se conectar.")

    def stop_services(self):
        """Para todos os serviços de forma graciosa."""
        logging.info("🛑 Sinal de recebido. Parando todos os serviços...")
        self.is_running = False
        
        if 'socket_service' in self.services:
            self.socket_service.stop()
        
        if 'trading_engine' in self.services:
            self.trading_engine.stop()
            
        logging.info("✅ Todos os serviços foram parados.")

    def run(self):
        """Mantém o servidor principal ativo."""
        self.start_services()
        
        try:
            while self.is_running:
                time.sleep(1)
        except KeyboardInterrupt:
            self.stop_services()

# Graceful Shutdown
server = MainServer()
signal.signal(signal.SIGINT, lambda s, f: server.stop_services())

if __name__ == "__main__":
    server.run()



2. O Serviço de Socket: Server/mt5_socket_service.py



import socket
import json
import threading
import logging

class MT5SocketService:
    def __init__(self, host='127.0.0.1', port=5555):
        self.host = host
        self.port = port
        self.server_socket = None
        self.is_running = False
        self.clients = []
        self.trading_engine = None # Referência ao motor de trading
        self.logger = logging.getLogger("SocketService")

    def set_trading_engine(self, engine):
        """Recebe a referência do motor de trading."""
        self.trading_engine = engine

    def start(self):
        """Inicia o serviço de socket."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.is_running = True
        self.logger.info(f"Serviço de Socket iniciado em {self.host}:{self.port}")

        while self.is_running:
            try:
                client_socket, addr = self.server_socket.accept()
                self.clients.append(client_socket)
                self.logger.info(f"EA conectado de {addr}")
                # Thread para tratar cada cliente
                threading.Thread(target=self.handle_client, args=(client_socket, addr), daemon=True).start()
            except Exception as e:
                if self.is_running:
                    self.logger.error(f"Erro ao aceitar conexão: {e}")

    def handle_client(self, client_socket, addr):
        """Trata a comunicação com um cliente (EA)."""
        buffer = ""
        try:
            while self.is_running:
                data = client_socket.recv(1024).decode('utf-8')
                if not data:
                    break
                buffer += data
                while '\n' in buffer:
                    line, buffer = buffer.split('\n', 1)
                    if line:
                        self.process_message(json.loads(line.strip()), client_socket)
        except Exception as e:
            self.logger.error(f"Erro ao comunicar com {addr}: {e}")
        finally:
            if client_socket in self.clients:
                self.clients.remove(client_socket)
            client_socket.close()
            self.logger.info(f"EA de {addr} desconectado.")

    def process_message(self, message, client_socket):
        """Processa mensagens recebidas do EA."""
        msg_type = message.get("message_type")
        if msg_type == "HANDSHAKE":
            response = {"message_type": "HANDSHAKE_ACK", "server_name": "SGM Server", "version": "3.1", "status": "READY"}
            self.send_to_client(client_socket, response)
        # ... outros handlers (HEARTBEAT_ACK, etc.)

    def send_to_all_clients(self, message_dict):
        """Envia uma mensagem para todos os EAs conectados."""
        if not self.clients:
            return
        message = json.dumps(message_dict) + '\n'
        for client in self.clients:
            try:
                client.sendall(message.encode('utf-8'))
            except:
                self.clients.remove(client) # Remove cliente desconectado

    def stop(self):
        """Para o serviço de socket."""
        self.is_running = False
        if self.server_socket:
            self.server_socket.close()
        self.logger.info("Serviço de Socket parado.")
