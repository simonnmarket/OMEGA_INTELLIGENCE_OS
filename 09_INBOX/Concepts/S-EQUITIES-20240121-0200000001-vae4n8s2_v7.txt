    async def _find_vol_arbitrage_opportunities_v7(self, options_data: Dict, stock_data: Dict) -> List[VolatilityArbitrageOpportunity]:
        """
        IDENTIFICA OPORTUNIDADES DE VOLATILITY ARBITRAGE v7.0
        """
        opportunities = []
        
        for stock in self.target_stocks:
            try:
                # Verificar se temos dados de opções para o stock
                if stock not in options_data:
                    continue
                
                # Calcular métricas de volatilidade
                iv_skew = await self._calculate_iv_skew_v7(stock, options_data)
                iv_rank = await self._calculate_iv_rank_v7(stock, options_data)
                earnings_impact = await self._calculate_earnings_impact_v7(stock, options_data)
                term_structure = await self._calculate_term_structure_v7(stock, options_data)
                expected_move = await self._calculate_expected_move_v7(stock, options_data)
                
                # Determinar tipo de estratégia
                strategy_type = self._determine_vol_strategy_type(iv_skew, iv_rank, earnings_impact, term_structure)
                
                if strategy_type != "NO_TRADE":
                    # Calcular exposição vega
                    position_vega = await self._calculate_position_vega(stock, strategy_type, options_data)
                    
                    # Calcular probabilidade de execução
                    execution_probability = await self._calculate_vol_execution_probability_v7(
                        stock, strategy_type, iv_skew, iv_rank, earnings_impact
                    )
                    
                    # Verificar segurança
                    security_clearance = await self._security_check_vol_arbitrage_v7(stock, strategy_type, options_data)
                    
                    opportunity = VolatilityArbitrageOpportunity(
                        stock=stock,
                        strategy_type=strategy_type,
                        iv_skew=iv_skew,
                        iv_rank=iv_rank,
                        earnings_impact=earnings_impact,
                        term_structure=term_structure,
                        expected_move=expected_move,
                        position_vega=position_vega,
                        execution_probability=execution_probability,
                        security_clearance=security_clearance
                    )
                    
                    opportunities.append(opportunity)
                    
            except Exception as e:
                logging.warning(f"Erro análise volatility {stock}: {e}")
                continue
                
        return opportunities

    async def _calculate_iv_skew_v7(self, stock: str, options_data: Dict) -> Decimal:
        """
        CALCULA IV SKEW (diferença entre IV de puts e calls)
        """
        try:
            if 'options_chain' not in options_data[stock]:
                return Decimal('0')
            
            chain = options_data[stock]['options_chain']
            
            # Calcular IV média para calls e puts ATM
            calls_iv = self._extract_atm_iv(chain, 'call')
            puts_iv = self._extract_atm_iv(chain, 'put')
            
            if puts_iv > Decimal('0') and calls_iv > Decimal('0'):
                skew = (puts_iv - calls_iv) / calls_iv
                return skew
                
        except Exception as e:
            logging.error(f"Erro cálculo IV skew {stock}: {e}")
            
        return Decimal('0')

    async def _calculate_iv_rank_v7(self, stock: str, options_data: Dict) -> Decimal:
        """
        CALCULA IV RANK (percentil histórico da IV)
        """
        try:
            current_iv = options_data[stock].get('implied_volatility', Decimal('0'))
            iv_history = options_data[stock].get('iv_history', [])
            
            if not iv_history:
                return Decimal('0')
            
            # Calcular percentil atual
            iv_history_sorted = sorted(iv_history)
            rank = Decimal('0')
            
            for i, iv in enumerate(iv_history_sorted):
                if current_iv >= iv:
                    rank = Decimal(str(i)) / Decimal(str(len(iv_history_sorted)))
            
            return rank
            
        except Exception as e:
            logging.error(f"Erro cálculo IV rank {stock}: {e}")
            return Decimal('0')

    async def _calculate_earnings_impact_v7(self, stock: str, options_data: Dict) -> Decimal:
        """
        CALCULA IMPACTO ESPERADO DO EARNINGS NA VOLATILIDADE
        """
        try:
            days_to_earnings = self._get_days_to_earnings(stock)
            
            if days_to_earnings > self.vol_params['max_days_to_earnings']:
                return Decimal('0')
            
            # IV atual vs IV histórica pré-earnings
            current_iv = options_data[stock].get('implied_volatility', Decimal('0'))
            normal_iv = options_data[stock].get('normal_iv', Decimal('0'))
            
            if normal_iv > Decimal('0'):
                impact = (current_iv - normal_iv) / normal_iv
                return max(impact, Decimal('0'))
                
        except Exception as e:
            logging.error(f"Erro cálculo earnings impact {stock}: {e}")
            
        return Decimal('0')

    async def _calculate_term_structure_v7(self, stock: str, options_data: Dict) -> Decimal:
        """
        CALCULA TERM STRUCTURE (IV front-month vs back-month)
        """
        try:
            front_month_iv = options_data[stock].get('front_month_iv', Decimal('0'))
            back_month_iv = options_data[stock].get('back_month_iv', Decimal('0'))
            
            if front_month_iv > Decimal('0') and back_month_iv > Decimal('0'):
                structure = (front_month_iv - back_month_iv) / back_month_iv
                return structure
                
        except Exception as e:
            logging.error(f"Erro cálculo term structure {stock}: {e}")
            
        return Decimal('0')

    def _determine_vol_strategy_type(self, iv_skew: Decimal, iv_rank: Decimal, 
                                   earnings_impact: Decimal, term_structure: Decimal) -> str:
        """
        DETERMINA TIPO DE ESTRATÉGIA DE VOLATILITY ARBITRAGE
        """
        # Estratégia 1: IV Skew Arbitrage
        if abs(iv_skew) > self.vol_params['iv_skew_threshold']:
            if iv_skew > Decimal('0'):
                return "RISK_REVERSAL_SHORT"  # Skew positivo -> vender puts, comprar calls
            else:
                return "RISK_REVERSAL_LONG"   # Skew negativo -> comprar puts, vender calls
        
        # Estratégia 2: Earnings Volatility Play
        if (earnings_impact > self.vol_params['earnings_impact_min'] and 
            iv_rank > self.vol_params['iv_rank_threshold']):
            return "EARNINGS_STRADDLE"  # Vender straddle se IV alta
        
        # Estratégia 3: Term Structure Arbitrage
        if abs(term_structure) > self.vol_params['term_structure_threshold']:
            if term_structure > Decimal('0'):
                return "CALENDAR_SPREAD_SHORT"  # IV front > back -> vender front, comprar back
            else:
                return "CALENDAR_SPREAD_LONG"   # IV front < back -> comprar front, vender back
        
        return "NO_TRADE"

    async def _filter_vol_opportunities_v7(self, opportunities: List[VolatilityArbitrageOpportunity]) -> List[VolatilityArbitrageOpportunity]:
        """
        FILTRA OPORTUNIDADES DE VOLATILITY ARBITRAGE
        """
        filtered = []
        
        for opportunity in opportunities:
            # Critérios de filtragem por tipo de estratégia
            if opportunity.strategy_type == "RISK_REVERSAL_SHORT":
                if (abs(opportunity.iv_skew) > self.vol_params['iv_skew_threshold'] and
                    opportunity.execution_probability > Decimal('0.70')):
                    filtered.append(opportunity)
                    
            elif opportunity.strategy_type == "RISK_REVERSAL_LONG":
                if (abs(opportunity.iv_skew) > self.vol_params['iv_skew_threshold'] and
                    opportunity.execution_probability > Decimal('0.70')):
                    filtered.append(opportunity)
                    
            elif opportunity.strategy_type == "EARNINGS_STRADDLE":
                if (opportunity.earnings_impact > self.vol_params['earnings_impact_min'] and
                    opportunity.iv_rank > self.vol_params['iv_rank_threshold'] and
                    opportunity.execution_probability > Decimal('0.75')):
                    filtered.append(opportunity)
                    
            elif opportunity.strategy_type == "CALENDAR_SPREAD_SHORT":
                if (abs(opportunity.term_structure) > self.vol_params['term_structure_threshold'] and
                    opportunity.execution_probability > Decimal('0.65')):
                    filtered.append(opportunity)
                    
            elif opportunity.strategy_type == "CALENDAR_SPREAD_LONG":
                if (abs(opportunity.term_structure) > self.vol_params['term_structure_threshold'] and
                    opportunity.execution_probability > Decimal('0.65')):
                    filtered.append(opportunity)
        
        # Ordenar por probabilidade de execução
        filtered.sort(key=lambda x: x.execution_probability, reverse=True)
        return filtered

    def _calculate_vol_position_size(self, opportunity: VolatilityArbitrageOpportunity, allocated_capital: Decimal) -> Decimal:
        """
        CALCULA TAMANHO DA POSIÇÃO PARA VOLATILITY ARBITRAGE
        """
        # Base diferente por tipo de estratégia
        base_sizes = {
            "RISK_REVERSAL_SHORT": Decimal('0.04'),
            "RISK_REVERSAL_LONG": Decimal('0.03'),
            "EARNINGS_STRADDLE": Decimal('0.02'),
            "CALENDAR_SPREAD_SHORT": Decimal('0.05'),
            "CALENDAR_SPREAD_LONG": Decimal('0.04')
        }
        
        base_size = allocated_capital * base_sizes.get(opportunity.strategy_type, Decimal('0.03'))
        
        # Ajustar pela probabilidade de execução
        probability_adjustment = opportunity.execution_probability
        
        # Ajustar pelo vega exposure
        vega_adjustment = min(Decimal('1.0'), self.vol_params['max_vega_exposure'] / opportunity.position_vega)
        
        adjusted_size = base_size * probability_adjustment * vega_adjustment
        
        # Limites de segurança
        max_size = allocated_capital * self.vol_params['position_size_limit']
        min_size = Decimal('2000')  # Mínimo €2000 para options
        
        return max(min(adjusted_size, max_size), min_size)

    def _calculate_vol_risk_score(self, opportunity: VolatilityArbitrageOpportunity) -> Decimal:
        """
        CALCULA SCORE DE RISCO PARA VOLATILITY ARBITRAGE
        """
        # Risco variável por tipo de estratégia
        base_risks = {
            "RISK_REVERSAL_SHORT": Decimal('0.4'),
            "RISK_REVERSAL_LONG": Decimal('0.5'),
            "EARNINGS_STRADDLE": Decimal('0.6'),
            "CALENDAR_SPREAD_SHORT": Decimal('0.3'),
            "CALENDAR_SPREAD_LONG": Decimal('0.4')
        }
        
        base_risk = base_risks.get(opportunity.strategy_type, Decimal('0.5'))
        
        # Ajustes baseados nas métricas
        if opportunity.iv_rank > Decimal('0.8'):
            base_risk += Decimal('0.1')  # IV muito alta = mais risco
        if opportunity.earnings_impact > Decimal('0.15'):
            base_risk += Decimal('0.1')  # Earnings muito impactantes = mais risco
            
        return min(base_risk, Decimal('0.7'))

        """Calcula probabilidade de execução para volatility arbitrage"""
        factors = []
        
        if strategy_type in ["RISK_REVERSAL_SHORT", "RISK_REVERSAL_LONG"]:
            factors.append(min(Decimal('1.0'), abs(iv_skew) / Decimal('0.3')))
            factors.append(iv_rank)
        elif strategy_type == "EARNINGS_STRADDLE":

            factors.append(iv_rank)
        else:  # Calendar spreads
