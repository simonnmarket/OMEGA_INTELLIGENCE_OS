            // Implementação básica da cena 3D

            renderer.setSize(document.getElementById('visualization3D').clientWidth,

            document.getElementById('visualization3D').appendChild(renderer.domElement);

        function startAnalysis() {
            const symbols = document.getElementById('symbolsInput').value.split(',');
            const timeframes = Array.from(document.getElementById('timeframesSelect').selectedOptions)
                .map(option => option.value);
            
            fetch('/api/analyze', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({ symbols, timeframes })
            })
            .then(response => response.json())
            .then(data => {
                addLog('Analysis started with ID: ' + data.analysis_id);
            });
        }
        
        function addLog(message) {
            const logElement = document.getElementById('executionLog');
            const logEntry = document.createElement('div');
            logEntry.textContent = new Date().toLocaleTimeString() + ' - ' + message;
            logElement.appendChild(logEntry);
            logElement.scrollTop = logElement.scrollHeight;
        }
