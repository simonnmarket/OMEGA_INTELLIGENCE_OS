4. SISTEMA DE ALOCAÇÃO DINÂMICA
4.1 FÓRMULA KELLY OTIMIZADA:


def dynamic_kelly_allocation(asset, market_regime, current_performance):
    """Alocação dinâmica baseada em scores e performance"""
    
    # Parâmetros base
    p = 0.58  # Win rate bootstrap
    b = 1.3   # Profit/loss ratio
    q = 1 - p
    
    # Score do ativo
    asset_score = OFFICIAL_SCORES[asset]['total']
    
    # Multiplicador de regime
    regime_multiplier = 1.1 if market_regime == 'bull' else 0.9
    
    # Ajuste performance recente
    perf_adjustment = 1.0 + (current_performance - 0.5) * 0.2
    
    # Cálculo Kelly
    kelly_fraction = (p * b - q) / b
    dynamic_allocation = asset_score * regime_multiplier * perf_adjustment * kelly_fraction
    
    # Aplicar limites por Tier
    tier_limits = {'BTC': 0.05, 'ETH': 0.05, 'SOL': 0.008, 'ADA': 0.0, 'XRP': 0.0}
    
    return min(dynamic_allocation, tier_limits.get(asset, 0.0))
