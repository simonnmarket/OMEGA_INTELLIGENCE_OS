    def _calculate_arbitrage_position_size(self, opportunity: ArbitrageOpportunity) -> Decimal:
        """
        Calcula tamanho de posição para arbitragem de liquidez

        base_size = Decimal('0.05')  # 5% base

                       liquidity_adjustment * speed_adjustment)
        
        return min(position_size, Decimal('0.15'))  # Máximo 15%

    async def _execute_atomic_arbitrage(self, opportunity: ArbitrageOpportunity,
                                      position_size: Decimal) -> Dict:
        """
        Executa arbitragem atômica entre duas venues
        """
        try:
            # Ordem de compra na venue com melhor ask
            buy_order = await self._place_venue_order(
                venue=opportunity.buy_venue,
                pair=opportunity.pair,
                side='BUY',
                size=position_size
            )
            
            if not buy_order['success']:
                return {'success': False, 'error': 'Buy order failed'}
            
            # Ordem de venda na venue com melhor bid
            sell_order = await self._place_venue_order(
                venue=opportunity.sell_venue,
                pair=opportunity.pair, 
                side='SELL',
                size=position_size
            )
            
            if not sell_order['success']:
                # Rollback da ordem de compra
                await self._cancel_venue_order(opportunity.buy_venue, buy_order['order_id'])
                return {'success': False, 'error': 'Sell order failed'}
            
            # Calcular lucro real
            actual_profit = (sell_order['executed_price'] - 
                           buy_order['executed_price']) * position_size
            
            return {
                'success': True,
                'actual_profit': actual_profit,
                'buy_order_id': buy_order['order_id'],
                'sell_order_id': sell_order['order_id'],
                'execution_time': time.time() - buy_order['timestamp']
            }
            
        except Exception as e:
            logging.error(f"Atomic arbitrage execution error: {e}")
            return {'success': False, 'error': str(e)}
