4) Exemplo prático em Python — pipeline mínimo e backtest simples (momentum diário)

O código abaixo usa yfinance (dados de preço públicos) e pandas para um backtest simples de momentum (comprar se retorno passado positivo). É um ponto de partida que você pode adaptar para qualquer ativo OU (troque o ticker).

# Salve como momentum_backtest.py e rode com Python 3.10+
import pandas as pd
import numpy as np
import yfinance as yf

def fetch_data(ticker, start="2018-01-01", end=None):
    df = yf.download(ticker, start=start, end=end, progress=False)
    df = df[['Adj Close']].rename(columns={'Adj Close':'price'})
    df.dropna(inplace=True)
    return df

def generate_momentum_signal(df, lookback=20):
    # return over lookback days
    df['ret'] = df['price'].pct_change()
    df['mom'] = df['price'].pct_change(lookback)
    # simple long-only: 1 if mom>0 else 0
    df['signal'] = (df['mom'] > 0).astype(int)
    return df

def backtest(df, initial_capital=100000, fee=0.0005, slippage=0.0005):
    df = df.copy()
    df['position'] = df['signal'].shift(1).fillna(0)  # act on next day open (approx)
    # simulate returns
    df['strategy_ret'] = df['position'] * df['ret']
    # approximate fees on trades (when position changes)
    df['trade'] = df['position'].diff().abs()
    df['strategy_ret_after_costs'] = df['strategy_ret'] - df['trade'] * (fee + slippage)
    df['equity'] = initial_capital * (1 + df['strategy_ret_after_costs'].fillna(0)).cumprod()
    return df

def summary_stats(df):
    final = df['equity'].iloc[-1]
    total_return = final / df['equity'].iloc[0] - 1
    ann_return = (1 + total_return) ** (252 / len(df.index)) - 1
    # volatility annualized
    ann_vol = df['strategy_ret_after_costs'].std() * np.sqrt(252)
    sharpe = (ann_return) / ann_vol if ann_vol != 0 else np.nan
    drawdown = 1 - df['equity'] / df['equity'].cummax()
    max_dd = drawdown.max()
    return {
        "final_equity": final,
        "total_return": total_return,
        "annual_return_approx": ann_return,
        "annual_vol_approx": ann_vol,
        "sharpe_approx": sharpe,
        "max_drawdown": max_dd
    }

if __name__ == "__main__":
    ticker = "AAPL"   # substitua pelo ticker do ativo OU
    df = fetch_data(ticker, start="2018-01-01")
    df = generate_momentum_signal(df, lookback=20)
    df = backtest(df)
    stats = summary_stats(df)
    print("Stats:", stats)
    print(df[['price','mom','signal','position','equity']].tail())

Observações:

Esse backtest é simplificado (usa fechamento, não order book, não usa tamanhos por risco).

Para algo mais realista, considere usar backtrader, zipline, vectorbt ou escrever simulador com execução por candle, order book, slippage dinâmico e modelagem de liquidez.
