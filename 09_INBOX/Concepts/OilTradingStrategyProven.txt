# ==================== ESTRATÉGIA INTEGRADA: OIL TRADING PROVEN ====================

class OilStrategyIntegrated:
    """
    ESTRATÉGIA OIL TRADING PROVEN - VERSÃO SINTETIZADA PELO CONSELHO
    Wrapper que integra a estratégia comprovada à arquitetura Numeia.
    """
    def __init__(self, rossi_kelly, hale_ontology):
        self.strategy_id = "S-OIL-PROVEN-20240121"
        self.rossi_kelly = rossi_kelly
        self.hale_ontology = hale_ontology
        
        # Importar e instanciar a estratégia original fornecida pelo usuário
        # (Em um ambiente real, seria importado de um arquivo separado)
        self.oil_executor = OilTradingExecution()

    async def analyze(self, market_data: Dict) -> List[TradingSignalEnhanced]:
        """Ponto de entrada principal da estratégia de Oil."""
        signals = []
        
        # 1. Executar a lógica principal da estratégia de Oil
        decision = self.oil_executor.execute_trading_decision(
            market_data, portfolio_value=100000 # Valor de portfólio de referência
        )
        
        # 2. Validar a decisão através da ontologia do sistema
        is_rational_decision = self.hale_ontology.decide_on_action(decision['confidence'])
        
        if decision['action'] != 'HOLD' and is_rational_decision:
            # 3. Enriquecer o sinal com metadados do conselho
            signal = TradingSignalEnhanced(
                strategy_id=self.strategy_id,
                asset="OIL_WTI",
                action=decision['action'],
                confidence=Decimal(str(decision['confidence'])),
                risk_score=Decimal(str(1.0 - decision['confidence'])), # Risco inverso à confiança
                timestamp=int(time.time() * 1e6),
                
                # Metadados enriquecidos com a análise detalhada da estratégia
                metadata={
                    'position_size': decision['position_size'],
                    'detailed_signals': decision['signals'], # Preserva toda a riqueza da análise
                    
                    # Assinaturas do Conselho
                    'gauss_laplace_validation': 'term_structure_and_hedge_approved',
                    'newton_hamilton_validation': 'momentum_conservation_verified',
                    'bayes_shannon_validation': 'risk_premium_and_seasonality_processed',
                    'rosi_kelly_validation': 'position_sizing_optimized',
                    'market_masters_validation': 'volume_price_confirmed',
                    'hale_ontology_approval': is_rational_decision
                }
            )
            signals.append(signal)
            
        return signals

class NumeiaTradingSystem:

        logging.info("🚀 INICIANDO SISTEMA NUMEIA v2.0 - CONSELHO PLENO INTEGRADO")

        self.hale_ontology = HaleOntologyCore()
        self.petrov_engine = PetrovQuantumVolumeEngine()
        self.rossi_kelly = RossiKellyHFTEngine()
        self.tanaka_gnc = TanakaGNCPrecision()
        self.leblanc_crypto = LeblancCrypto3D()
        self.market_masters = MarketMastersStrategies()
        
        # Estratégias de Trading Integradas
        self.golden_strategy_futures = GoldenStrategyFutures(
            self.petrov_engine, self.rossi_kelly, self.tanaka_gnc, 
            self.leblanc_crypto, self.market_masters
        )
        self.oil_strategy_proven = OilStrategyIntegrated(
            self.rossi_kelly, self.hale_ontology
        )
        
        # Portfólio de estratégias ativo
        self.strategies = {
            'futures_calendar_spread': self.golden_strategy_futures,
            'oil_proven_fundamentals': self.oil_strategy_proven
        }

    async def run_trading_cycle(self, market_data: Dict):
        """Ciclo de trading principal orquestrando todas as estratégias."""
        logging.info("🔄 INICIANDO CICLO DE TRADING...")
        
        all_signals = []
        for strategy_name, strategy_instance in self.strategies.items():
            try:
                logging.info(f"📊 Executando estratégia: {strategy_name}")
                signals = await strategy_instance.analyze(market_data)
                all_signals.extend(signals)
                logging.info(f"✅ {len(signals)} sinais gerados por {strategy_name}")
            except Exception as e:
                logging.error(f"❌ Erro na estratégia {strategy_name}: {e}")
