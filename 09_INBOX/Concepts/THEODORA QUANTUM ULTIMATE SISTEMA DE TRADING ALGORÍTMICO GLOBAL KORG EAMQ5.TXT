//| Calcula ATR                                                      |
//+------------------------------------------------------------------+
double CalculateATR(string symbol, int period = 14)
{
   double high[], low[], close[];
   ArraySetAsSeries(high, true);
   ArraySetAsSeries(low, true);
   ArraySetAsSeries(close, true);
   CopyHigh(symbol, PERIOD_H1, 0, period + 1, high);
   CopyLow(symbol, PERIOD_H1, 0, period + 1, low);
   CopyClose(symbol, PERIOD_H1, 0, period + 1, close);
   double tr[];
   ArrayResize(tr, period);
   for(int i = 0; i < period; i++)
   {
      tr[i] = MathMax(high[i] - low[i], MathMax(MathAbs(high[i] - close[i+1]), MathAbs(low[i] - close[i+1])));
   }
   return iMAOnArray(tr, 0, period, 0, MODE_SMA, 0);
}

//+------------------------------------------------------------------+
//| Calcula Momentum                                                 |
//+------------------------------------------------------------------+
double CalculateMomentum(string symbol)
{
   double close[];
   ArraySetAsSeries(close, true);
   CopyClose(symbol, PERIOD_H1, 0, 10, close);
   double returns[];
   ArrayResize(returns, 9);
   for(int i = 0; i < 9; i++)
      returns[i] = (close[i] - close[i+1]) / close[i+1];
   return ArraySum(returns) / 9;
}

//+------------------------------------------------------------------+
//| Calcula Lote por Risco                                           |
//+------------------------------------------------------------------+
double ComputeLotByRisk(string symbol, double risk_pct, double sl_points)
{
   double balance = AccountBalance();
   double tick_size = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_SIZE);
   double tick_value = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_VALUE);
   double risk_amount = balance * risk_pct / 100.0;
   double lot = risk_amount / (sl_points * tick_value / tick_size);
   return NormalizeDouble(lot, 2);
}

//+------------------------------------------------------------------+
//| Normaliza Volume                                                 |
//+------------------------------------------------------------------+
double NormalizeVolume(string symbol, double lot)
{
   double min_lot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
   double max_lot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MAX);
   double step_lot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_STEP);
   lot = MathMax(min_lot, MathMin(max_lot, MathRound(lot / step_lot) * step_lot));
   return lot;
}

//| Atualiza Trailing Stop                                           |
//+------------------------------------------------------------------+
void UpdateTrailingStop(string symbol, ulong ticket, double entry_price, double atr, double entropy)
{
   if(!PositionSelectByTicket(ticket)) return;
   double current_price = SymbolInfoDouble(symbol, SYMBOL_BID);
   double sl = PositionGetDouble(POSITION_SL);
   double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
   if(point <= 0.0) point = _Point;
   double adjust_factor = 1.0 + entropy; // Ajusta trailing com base na entropia
   double new_sl;
   if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY)
   {
      new_sl = current_price - atr * TrailingFactor * adjust_factor * point;
      if(new_sl > sl && new_sl > entry_price)
         trade.PositionModify(ticket, new_sl, PositionGetDouble(POSITION_TP));
   }
   else
   {
      new_sl = current_price + atr * TrailingFactor * adjust_factor * point;
      if((sl == 0 || new_sl < sl) && new_sl < entry_price)
         trade.PositionModify(ticket, new_sl, PositionGetDouble(POSITION_TP));
   }
}

bool HttpGet(string path, string &result)
{
   char data[], res[];
   string headers = "Content-Type: application/json\r\n";
   int ret = WebRequest("GET", "http://127.0.0.1:5000" + path, headers, 5000, data, res, headers);
   if(ret != 200)
   {
      Print("[Theodora] HTTP falhou: ", ret, " Erro: ", GetLastError());
      return false;
   }
   result = CharArrayToString(res);
   return true;
}

//| Parseia Alocações da API                                         |
//+------------------------------------------------------------------+
bool ParseAllocations(string json, string &symbols[], double &weights[])
{
   int pos = StringFind(json, "\"selected\":[");
   if(pos == -1) return false;
   string selected = StringSubstr(json, pos + StringLen("\"selected\":["), StringFind(json, "]", pos) - pos - StringLen("\"selected\":["));
   string entries[];
   StringSplit(selected, '}', entries);
   ArrayResize(symbols, 0);
   ArrayResize(weights, 0);
   for(int i = 0; i < ArraySize(entries); i++)
   {
      string entry = entries[i];
      int sym_pos = StringFind(entry, "\"symbol\":\"") + StringLen("\"symbol\":\"");
      int sym_end = StringFind(entry, "\"", sym_pos);
      string sym = StringSubstr(entry, sym_pos, sym_end - sym_pos);
      int weight_pos = StringFind(entry, "\"pulse\":") + StringLen("\"pulse\":");
      int weight_end = StringFind(entry, ",", weight_pos);
      double weight = StringToDouble(StringSubstr(entry, weight_pos, weight_end - weight_pos));
      if(StringLen(sym) > 0)
      {
         ArrayResize(symbols, ArraySize(symbols) + 1);
         ArrayResize(weights, ArraySize(weights) + 1);
         symbols[ArraySize(symbols) - 1] = sym;
         weights[ArraySize(weights) - 1] = weight;
      }
   }
   return ArraySize(symbols) > 0;
}

//| Executa Ordem Escalonada                                         |
//+------------------------------------------------------------------+
bool ExecuteScaledOrder(string symbol, bool isBuy, double price, double sl_points, double tp_points, double max_dev, string missionId, double atr, double volatility, double entropy)
{
   if(!SymbolSelect(symbol, true))
   {
      Print("Erro: Símbolo não disponível: ", symbol);
      return false;
   }
   double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
   if(point <= 0.0) point = _Point;
   double momentum = CalculateMomentum(symbol);
   double scaling_adjust = 1.0 + volatility * 2;
   bool any_success = false;

   for(int i = 0; i < ArraySize(ScalingLevels); i++)
   {
      double lot = ComputeLotByRisk(symbol, TotalRiskPct / MaxSymbolsToTrade, sl_points);
      lot = NormalizeVolume(symbol, lot * ScalingLevels[i] * (1.0 + MathAbs(momentum)) * scaling_adjust);
      double sl = 0.0, tp = 0.0;
      if(sl_points > 0.0 && tp_points > 0.0)
      {
         if(isBuy)
         {
            sl = price - (sl_points + atr * SLFactor * (1 + entropy)) * point;
            tp = price + (tp_points + atr * TPFactor * ScalingLevels[i] * scaling_adjust) * point;
         }
         else
         {
            sl = price + (sl_points + atr * SLFactor * (1 + entropy)) * point;
            tp = price - (tp_points + atr * TPFactor * ScalingLevels[i] * scaling_adjust) * point;
         }
      }
      trade.SetDeviationInPoints((int)max_dev);
      trade.SetExpertMagicNumber(11041982);
      bool sent = false;
      if(isBuy)
         sent = trade.Buy(lot, symbol, price, sl, tp, missionId + "_Scaled_" + IntegerToString(i));
      else
         sent = trade.Sell(lot, symbol, price, sl, tp, missionId + "_Scaled_" + IntegerToString(i));
      if(sent)
         any_success = true;
      else
         Print("Falha ao enviar ordem escalonada em ", symbol, " code=", GetLastError());
   }
   return any_success;
}

//| Executa a partir de Decisão da API                               |
//+------------------------------------------------------------------+
bool ExecuteFromDecision(string pickedSymbol)
{
   string decisionBody;
   string decisionPath = "/decision?symbol=" + pickedSymbol;
   if(!HttpGet(decisionPath, decisionBody))
      return false;

   int pos = StringFind(decisionBody, "\"signal\":");
   if(pos == -1) return false;
   string signal = StringSubstr(decisionBody, pos + StringLen("\"signal\":\""), StringFind(decisionBody, "\"", pos + StringLen("\"signal\":\"")) - pos - StringLen("\"signal\":\""));
   if(signal != "BUY" && signal != "SELL")
      return false;

   double price = StringToDouble(StringSubstr(decisionBody, StringFind(decisionBody, "\"price\":") + StringLen("\"price\":")));
   double sl_points = StringToDouble(StringSubstr(decisionBody, StringFind(decisionBody, "\"sl_points\":") + StringLen("\"sl_points\":")));
   double tp_points = StringToDouble(StringSubstr(decisionBody, StringFind(decisionBody, "\"tp_points\":") + StringLen("\"tp_points\":")));
   double max_dev = StringToDouble(StringSubstr(decisionBody, StringFind(decisionBody, "\"max_slippage_points\":") + StringLen("\"max_slippage_points\":")));
   double volatility = StringToDouble(StringSubstr(decisionBody, StringFind(decisionBody, "\"volatility\":") + StringLen("\"volatility\":")));
   double entropy = StringToDouble(StringSubstr(decisionBody, StringFind(decisionBody, "\"entropy\":") + StringLen("\"entropy\":")));
   string missionId = StringSubstr(decisionBody, StringFind(decisionBody, "\"mission_id\":\"") + StringLen("\"mission_id\":\""), StringFind(decisionBody, "\"", StringFind(decisionBody, "\"mission_id\":\"")) - StringFind(decisionBody, "\"mission_id\":\"") - StringLen("\"mission_id\":\""));
   double atr = CalculateATR(pickedSymbol);

   return ExecuteScaledOrder(pickedSymbol, signal == "BUY", price, sl_points, tp_points, max_dev, missionId, atr, volatility, entropy);
}

//| Request AI Decision                                              |
//+------------------------------------------------------------------+
bool RequestAIDecision(string execution_symbol)
{
   string symsQuery = BuildSymbolsQuery();
   string allocPath = (StringLen(symsQuery) > 0 ? "/portfolio/allocate?symbols=" + symsQuery : "/portfolio/allocate");
   allocPath += ("&min_pulse=" + DoubleToString(ScanMinPulse, 1) + "&min_energy=" + DoubleToString(ScanMinEnergy, 1));
   string allocBody;
   if(HttpGet(allocPath, allocBody))
   {
      string asyms[]; double aweights[];
      if(ParseAllocations(allocBody, asyms, aweights))
      {
         int count = MathMin((int)ArraySize(asyms), MaxSymbolsToTrade);
         bool anyOk = false;
         for(int i = 0; i < count; i++)
            if(StringLen(asyms[i]) > 0 && ExecuteFromDecision(asyms[i]))
               anyOk = true;
         if(anyOk) return true;
      }
   }

   string scanBody;
   string scanPath = (StringLen(symsQuery) > 0 ? "/scan?symbols=" + symsQuery : "/scan");
   scanPath += ("&min_pulse=" + DoubleToString(ScanMinPulse, 1) + "&min_energy=" + DoubleToString(ScanMinEnergy, 1));
   if(!HttpGet(scanPath, scanBody))
      return false;

   int posSel = StringFind(scanBody, "\"selected\":[");
   bool anyOk = false;
   int executed = 0;
   if(posSel != -1)
   {
      int cur = posSel;
      while(true)
      {
         int ps = StringFind(scanBody, "\"symbol\":\"", cur);
         if(ps == -1) break;
         int s0 = ps + StringLen("\"symbol\":\"");
         int s1 = StringFind(scanBody, "\"", s0);
         if(s1 == -1) break;
         string s = StringSubstr(scanBody, s0, s1 - s0);
         cur = s1 + 1;
         if(StringLen(s) > 0)
         {
            if(ExecuteFromDecision(s)) { anyOk = true; executed++; }
            if(executed >= MaxSymbolsToTrade) break;
         }
      }
   }
   if(anyOk) return true;

   int posRes = StringFind(scanBody, "\"results\":[");
   if(posRes != -1)
   {
      int cur = posRes;
      while(true)
      {
         int okp = StringFind(scanBody, "\"thresholds_ok\":true", cur);
         if(okp == -1) break;
         int ps = StringFind(scanBody, "\"symbol\":\"", cur);
         if(ps == -1) break;
         int s0 = ps + StringLen("\"symbol\":\"");
         int s1 = StringFind(scanBody, "\"", s0);
         if(s1 == -1) break;
         string sym = StringSubstr(scanBody, s0, s1 - s0);
         if(StringLen(sym) > 0) return ExecuteFromDecision(sym);
         cur = s1 + 1;
      }
   }

   return ExecuteFromDecision(execution_symbol);
}

//| OnTimer                                                          |
//+------------------------------------------------------------------+
void OnTimer()
{
   if(GlobalPause)
   {
      Print("[Theodora] GlobalPause ativo - ciclo ignorado");
      return;
   }
   static bool closedOnce = false;
   if(EmergencyCloseAll && !closedOnce)
   {
      CloseAllOpenPositions();
      closedOnce = true;
   }
   if(g_isBusy)
      return;

   datetime now = TimeCurrent();
   if(g_fail_count > 0 && now < g_next_allowed)
      return;

   g_isBusy = true;
   bool ok = RequestAIDecision(_Symbol);
   if(ok)
   {
      g_fail_count = 0;
      g_backoff_sec = 0;
      g_next_allowed = 0;
   }
   else
   {
      g_fail_count++;
      int step = (int)MathPow(2, MathMin(g_fail_count, 10));
      g_backoff_sec = MathMin(step * TimerIntervalSec, BackoffMaxSec);
      g_next_allowed = now + g_backoff_sec;
   }
   g_isBusy = false;

   // Atualiza trailing stops
   for(int i = 0; i < PositionsTotal(); i++)
   {
      ulong ticket = PositionGetTicket(i);
      if(PositionSelectByTicket(ticket))
      {
         string symbol = PositionGetString(POSITION_SYMBOL);
         double atr = CalculateATR(symbol);
         double entropy = 0.5; // Placeholder, pode vir da API
         UpdateTrailingStop(symbol, ticket, PositionGetDouble(POSITION_PRICE_OPEN), atr, entropy);
      }
   }
}
