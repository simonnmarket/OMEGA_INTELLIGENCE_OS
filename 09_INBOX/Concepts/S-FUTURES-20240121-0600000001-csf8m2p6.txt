    async def _passes_quantum_quality_filters(self, spread: CalendarSpread, market_analysis: Dict) -> bool:
        """
        Aplica filtros quânticos multidimensionais

            abs(spread.roll_yield) >= self.spread_params['min_roll_yield'],
            spread.historical_percentile >= self.spread_params['historical_percentile_min'],
            spread.quantum_confidence >= self.spread_params['quantum_confidence_min'],
            spread.entropy_measure <= self.spread_params['entropy_threshold'],
            spread.liquidity_score >= self.spread_params['min_liquidity_score'],
            await self._is_quantum_liquid_spread(spread.front_contract, spread.back_contract),
            await self._validate_market_regime_compatibility(spread, market_analysis)
        ]
        
        return all(filters)

    async def _calculate_quantum_spread_price(self, contract: str, front: str, back: str, 
                                            market_intel: Dict) -> Decimal:
        """Calcula preço do spread com ajuste quântico"""
        futures_chain = await self._get_quantum_futures_chain(contract, market_intel)
        return futures_chain[back]['price'] - futures_chain[front]['price']

    async def _calculate_quantum_zscore(self, contract: str, front: str, back: str, 
                                      spread_price: Decimal, market_intel: Dict) -> Decimal:
        """Calcula Z-score com histórico quântico"""
        historical_data = await self._get_historical_spread_data(contract, front, back)
        if len(historical_data) < 10:
            return Decimal('0')
        
        mean_price = Decimal(str(np.mean(historical_data)))
        std_price = Decimal(str(np.std(historical_data)))
        
        if std_price == Decimal('0'):
            return Decimal('0')
            
        return (spread_price - mean_price) / std_price

    async def _calculate_quantum_roll_yield(self, contract: str, front: str, back: str, 
                                          market_intel: Dict) -> Decimal:
        """Calcula roll yield com modelo quântico"""
        futures_chain = await self._get_quantum_futures_chain(contract, market_intel)
        front_price, back_price = futures_chain[front]['price'], futures_chain[back]['price']
        
        if front_price == Decimal('0'):
            return Decimal('0')
            
        return (back_price / front_price - Decimal('1'))

    async def _calculate_quantum_historical_percentile(self, contract: str, front: str, back: str, 
                                                     spread_price: Decimal) -> Decimal:
        """Calcula percentil histórico quântico"""
        historical_data = await self._get_historical_spread_data(contract, front, back)
        if not historical_data:
            return Decimal('0.5')
            
        current_rank = sum(1 for x in historical_data if x <= float(spread_price))
        return Decimal(current_rank) / Decimal(len(historical_data))
