double CalculateHeisenbergUncertainty(double price_std, double volume_std)

//+------------------------------------------------------------------+
//| Calcula entropia quântica de Shannon-Boltzmann                   |
//+------------------------------------------------------------------+
double CalculateQuantumEntropy(const double &returns[], int bins=50)
{
    int size = ArraySize(returns);
    if(size < 2) 
        return 0.5;
    
    double min_val = returns[0];
    double max_val = returns[0];
    
    for(int i = 1; i < size; i++)
    {
        if(returns[i] < min_val) min_val = returns[i];
        if(returns[i] > max_val) max_val = returns[i];
    }
    
    double range = max_val - min_val;
    
    if(range == 0 || size < bins) 
        return 0.5;
    
    int histogram[];
    ArrayResize(histogram, bins);
    ArrayInitialize(histogram, 0);
    
    for(int i = 0; i < size; i++)
    {
        int bin = (int)((returns[i] - min_val) / range * (bins - 1));
        if(bin >= 0 && bin < bins)
            histogram[bin]++;
    }
    
    double entropy = 0.0;
    double total = (double)size;
    
    for(int i = 0; i < bins; i++)
    {
        if(histogram[i] > 0)
        {
            double p = histogram[i] / total;
            entropy -= p * MathLog(p);
        }
    }
    
    return entropy / MathLog(bins); // Normalizado
}

double CalculateQuantumATR(string symbol, int period=14, double entropy_adjustment=1.0)

//+------------------------------------------------------------------+
//| Calcula Momentum quântico                                        |
//+------------------------------------------------------------------+
double CalculateQuantumMomentum(string symbol, int window=10)
{
    double close[];
    ArraySetAsSeries(close, true);
    if(CopyClose(symbol, PERIOD_H1, 0, window+1, close) <= window)
        return 0.0;
    
    double returns[];
    ArrayResize(returns, window);
    
    for(int i = 0; i < window; i++)
        returns[i] = (close[i] - close[i+1]) / close[i+1];
    
    // Momentum quântico (média ponderada exponencialmente)
    double momentum = 0.0;
    double total_weight = 0.0;
    
    for(int i = 0; i < window; i++)
    {
        double weight = MathExp(i * 0.5); // Peso exponencial
        momentum += returns[i] * weight;
        total_weight += weight;
    }
    
    return momentum / total_weight;
}

//+------------------------------------------------------------------+
//| Calcula ATR                                                      |
//+------------------------------------------------------------------+
double CalculateATR(string symbol, int period=14)
{
    double high[], low[], close[];
    ArraySetAsSeries(high, true);
    ArraySetAsSeries(low, true);
    ArraySetAsSeries(close, true);
    
    int bars = period + 1;
    if(CopyHigh(symbol, PERIOD_H1, 0, bars, high) < bars ||
       CopyLow(symbol, PERIOD_H1, 0, bars, low) < bars ||
       CopyClose(symbol, PERIOD_H1, 0, bars, close) < bars)
        return 0.0;
    
    double tr[];
    ArrayResize(tr, period);
    
    for(int i = 0; i < period; i++)
    {
        double tr1 = high[i] - low[i];
        double tr2 = MathAbs(high[i] - close[i+1]);
        double tr3 = MathAbs(low[i] - close[i+1]);
        tr[i] = MathMax(tr1, MathMax(tr2, tr3));
    }
    
    // Calcular média simples manualmente
    double sum = 0;
    for(int i = 0; i < period; i++)
        sum += tr[i];
    
    return sum / period;
}

//+------------------------------------------------------------------+
//| Calcula lote por risco com ajuste quântico                       |
//+------------------------------------------------------------------+
double ComputeQuantumLotByRisk(string symbol, double risk_pct, double sl_points, double entropy_factor=1.0)
{
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double tick_size = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_SIZE);
    double tick_value = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_VALUE);
    
    if(tick_size == 0 || tick_value == 0)
        return MinLot;
    
    double risk_amount = balance * risk_pct / 100.0;
    double lot = risk_amount / (sl_points * tick_value / tick_size);
    
    // Ajuste quântico baseado na entropia
    lot *= (1.0 + 0.3 * (entropy_factor - 0.5));
    
    return NormalizeVolume(symbol, MathMax(MinLot, lot));
}

//+------------------------------------------------------------------+
//| Normaliza Volume                                                 |
//+------------------------------------------------------------------+
double NormalizeVolume(string symbol, double lot)
{
    double min_lot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
    double max_lot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MAX);
    double step_lot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_STEP);
    
    if(min_lot == 0 || max_lot == 0 || step_lot == 0)
        return MinLot;
    
    lot = MathMax(min_lot, MathMin(max_lot, lot));
    lot = MathRound(lot / step_lot) * step_lot;
    
    return lot;
}

//+------------------------------------------------------------------+
//| Atualiza Trailing Stop quântico                                  |
//+------------------------------------------------------------------+
void UpdateQuantumTrailingStop(string symbol, ulong ticket, double entry_price, double atr, double entropy)
{
    if(!PositionSelectByTicket(ticket)) 
        return;
    
    int position_type = (int)PositionGetInteger(POSITION_TYPE);
    double current_price = SymbolInfoDouble(symbol, position_type == POSITION_TYPE_BUY ? SYMBOL_BID : SYMBOL_ASK);
    double sl = PositionGetDouble(POSITION_SL);
    double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
    
    if(point <= 0.0) 
        point = _Point;
    
    double adjust_factor = 1.0 + entropy;
    double new_sl;
    
    if(position_type == POSITION_TYPE_BUY)
    {
        new_sl = current_price - atr * TrailingFactor * adjust_factor * point;
        if(new_sl > sl && new_sl > entry_price)
            trade.PositionModify(ticket, new_sl, PositionGetDouble(POSITION_TP));
    }
    else
    {
        new_sl = current_price + atr * TrailingFactor * adjust_factor * point;
        if((sl == 0 || new_sl < sl) && new_sl < entry_price)
            trade.PositionModify(ticket, new_sl, PositionGetDouble(POSITION_TP));
    }
}

//+------------------------------------------------------------------+
//| Parseia Alocações da API com tratamento de erro melhorado        |
//+------------------------------------------------------------------+
bool ParseQuantumAllocations(string json, string &symbols[], double &weights[])
{
    int pos = StringFind(json, "\"selected\":[");
    if(pos == -1) 
        return false;
    
    int end_pos = StringFind(json, "]", pos);
    if(end_pos == -1)
        return false;
    
    string selected = StringSubstr(json, pos + StringLen("\"selected\":["), end_pos - pos - StringLen("\"selected\":["));
    string entries[];
    int entry_count = StringSplit(selected, ',', entries);
    
    ArrayResize(symbols, 0);
    ArrayResize(weights, 0);
    
    for(int i = 0; i < entry_count; i++)
    {
        string entry = entries[i];
        int sym_pos = StringFind(entry, "\"symbol\":\"");
        if(sym_pos == -1) continue;
        
        sym_pos += StringLen("\"symbol\":\"");
        int sym_end = StringFind(entry, "\"", sym_pos);
        if(sym_end == -1) continue;
        
        string sym = StringSubstr(entry, sym_pos, sym_end - sym_pos);
        
        int pulse_pos = StringFind(entry, "\"pulse\":");
        if(pulse_pos == -1) continue;
        
        pulse_pos += StringLen("\"pulse\":");
        int pulse_end = StringFind(entry, ",", pulse_pos);
        if(pulse_end == -1) 
            pulse_end = StringFind(entry, "}", pulse_pos);
        
        if(pulse_end == -1) continue;
        
        string pulse_str = StringSubstr(entry, pulse_pos, pulse_end - pulse_pos);
        double pulse = StringToDouble(pulse_str);
        
        if(StringLen(sym) > 0)
        {
            int size = ArraySize(symbols);
            ArrayResize(symbols, size+1);
            ArrayResize(weights, size+1);
            symbols[size] = sym;
            weights[size] = pulse;
        }
    }
    
    return ArraySize(symbols) > 0;
}

//+------------------------------------------------------------------+
//| Executa Ordem Escalonada Quântica                                |
//+------------------------------------------------------------------+
bool ExecuteQuantumScaledOrder(string symbol, bool isBuy, double price, double sl_points, 
                              double tp_points, double max_dev, string missionId, 
                              double atr, double volatility, double entropy)
{
    if(!SymbolSelect(symbol, true))
    {
        Print("Erro: Símbolo não disponível: ", symbol);
        return false;
    }
    
    double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
    if(point <= 0.0) 
        point = _Point;
    
    // Garantir preço válido caso não venha da API
    if(price <= 0.0)
        price = (isBuy ? SymbolInfoDouble(symbol, SYMBOL_ASK) : SymbolInfoDouble(symbol, SYMBOL_BID));
    
    double momentum = CalculateQuantumMomentum(symbol);
    double quantum_adjust = 1.0 + volatility * 2 + entropy * 0.5;
    bool any_success = false;
    
    // Verificar número atual de posições
    int current_positions = PositionsTotal();
    if(current_positions >= MaxPositions)
    {
        Print("Limite máximo de posições atingido: ", current_positions);
        return false;
    }
    
    // Array de níveis de escalonamento
    double scaling_levels[4] = {ScalingLevels1, ScalingLevels2, ScalingLevels3, ScalingLevels4};
    
    for(int i = 0; i < 4; i++)
    {
        double lot = ComputeQuantumLotByRisk(symbol, TotalRiskPct / MaxSymbolsToTrade, sl_points, entropy);
        lot = NormalizeVolume(symbol, lot * scaling_levels[i] * (1.0 + MathAbs(momentum)) * quantum_adjust);
        
        // SL/TP respeitando StopLevel mínimo do símbolo
        int stops_points = (int)SymbolInfoInteger(symbol, SYMBOL_TRADE_STOPS_LEVEL);
        double min_stop = (stops_points > 0 ? stops_points * point : 0.0);
        double base_sl_dist = (sl_points + atr * SLFactor * (1 + entropy)) * point;
        double base_tp_dist = (tp_points + atr * TPFactor * scaling_levels[i] * quantum_adjust) * point;
        double sl = 0.0, tp = 0.0;
        if(sl_points > 0.0 && tp_points > 0.0)
        {
            if(isBuy)
            {
                sl = price - MathMax(base_sl_dist, min_stop);
                tp = price + MathMax(base_tp_dist, min_stop);
            }
            else
            {
                sl = price + MathMax(base_sl_dist, min_stop);
                tp = price - MathMax(base_tp_dist, min_stop);
            }
        }
        
        trade.SetDeviationInPoints((int)max_dev);
        trade.SetExpertMagicNumber(11041982);
        
        bool sent = false;
        if(isBuy)
            sent = trade.Buy(lot, symbol, price, sl, tp, missionId + "_Quantum_" + IntegerToString(i));
        else
            sent = trade.Sell(lot, symbol, price, sl, tp, missionId + "_Quantum_" + IntegerToString(i));
        
        if(sent)
        {
            any_success = true;
            Print("Ordem quântica executada: ", symbol, " ", isBuy ? "BUY" : "SELL", " ", lot, " lots");
        }
        else
        {
            Print("Falha ao enviar ordem escalonada em ", symbol, " code=", GetLastError());
        }
    }
    
    return any_success;
}

//+------------------------------------------------------------------+
//| Executa a partir de Decisão da API Quântica                      |
//+------------------------------------------------------------------+
bool ExecuteFromQuantumDecision(string pickedSymbol)
{
    string decisionBody;
    string decisionPath = "/quantum/decision?symbol=" + pickedSymbol +
                          "&entropy=" + DoubleToString(g_quantum_entropy, 4) +
                          "&bypass_cost_gate=1&fallback_on_hold=1&debug=1";
    
    if(!HttpGet(decisionPath, decisionBody, 3000))
        return false;
    
    int pos = StringFind(decisionBody, "\"signal\":");
    if(pos == -1) 
        return false;
    
    pos += StringLen("\"signal\":");
    int end_pos = StringFind(decisionBody, "\"", pos+1);
    if(end_pos == -1) 
        return false;
    
    string signal = StringSubstr(decisionBody, pos+1, end_pos-pos-1);
    if(signal != "BUY" && signal != "SELL")
    {
        Print("[Theodora][", pickedSymbol, "] decisão=HOLD (sem trade)");
        return true; // Considera sucesso para evitar backoff
    }
    
    // Parse dos parâmetros quânticos
    double price = 0, sl_points = 0, tp_points = 0, max_dev = 0, volatility = 0, entropy = 0;
    string missionId = "";
    
    pos = StringFind(decisionBody, "\"price\":");
    if(pos != -1) price = StringToDouble(StringSubstr(decisionBody, pos+StringLen("\"price\":")));
    
    pos = StringFind(decisionBody, "\"sl_points\":");
    if(pos != -1) sl_points = StringToDouble(StringSubstr(decisionBody, pos+StringLen("\"sl_points\":")));
    
    pos = StringFind(decisionBody, "\"tp_points\":");
    if(pos != -1) tp_points = StringToDouble(StringSubstr(decisionBody, pos+StringLen("\"tp_points\":")));
    
    pos = StringFind(decisionBody, "\"max_slippage_points\":");
    if(pos != -1) max_dev = StringToDouble(StringSubstr(decisionBody, pos+StringLen("\"max_slippage_points\":")));
    
    pos = StringFind(decisionBody, "\"volatility\":");
    if(pos != -1) volatility = StringToDouble(StringSubstr(decisionBody, pos+StringLen("\"volatility\":")));
    
    pos = StringFind(decisionBody, "\"entropy\":");
    if(pos != -1) entropy = StringToDouble(StringSubstr(decisionBody, pos+StringLen("\"entropy\":")));
    
    pos = StringFind(decisionBody, "\"mission_id\":\"");
    if(pos != -1)
    {
        pos += StringLen("\"mission_id\":\"");
        int end = StringFind(decisionBody, "\"", pos);
        if(end != -1) missionId = StringSubstr(decisionBody, pos, end-pos);
    }
    
    double atr = CalculateQuantumATR(pickedSymbol, 14, entropy);
    
    return ExecuteQuantumScaledOrder(pickedSymbol, signal == "BUY", price, sl_points, 
                                   tp_points, max_dev, missionId, atr, volatility, entropy);
}

//+------------------------------------------------------------------+
//| Request AI Quantum Decision                                      |
//+------------------------------------------------------------------+
bool RequestQuantumAIDecision(string execution_symbol)
{
    // Atualizar entropia quântica global
    UpdateGlobalQuantumEntropy();
    
    string symsQuery = BuildSymbolsQuery();
    string allocPath = (StringLen(symsQuery) > 0 ? 
                       "/quantum/portfolio/allocate?symbols=" + symsQuery : 
                       "/quantum/portfolio/allocate");
    
    allocPath += ("&min_pulse=" + DoubleToString(ScanMinPulse, 1) + 
                 "&min_energy=" + DoubleToString(ScanMinEnergy, 1) +
                 "&entropy=" + DoubleToString(g_quantum_entropy, 4));
    
    string allocBody;
    if(HttpGet(allocPath, allocBody, 3000)) // Timeout reduzido
    {
        string asyms[]; 
        double aweights[];
        
        if(ParseQuantumAllocations(allocBody, asyms, aweights))
        {
            int count = MathMin(ArraySize(asyms), MaxSymbolsToTrade);
            bool anyOk = false;
            
            for(int i = 0; i < count; i++)
            {
                if(StringLen(asyms[i]) > 0 && ExecuteFromQuantumDecision(asyms[i]))
                    anyOk = true;
                
                // Verificar limite de posições após cada execução
                if(PositionsTotal() >= MaxPositions)
                    break;
            }
            
            if(anyOk) 
                return true;
        }
    }
    
    // Fallback para scan quântico
    string scanBody;
    string scanPath = (StringLen(symsQuery) > 0 ? 
                      "/quantum/scan?symbols=" + symsQuery : 
                      "/quantum/scan");
    
    scanPath += ("&min_pulse=" + DoubleToString(ScanMinPulse, 1) + 
                "&min_energy=" + DoubleToString(ScanMinEnergy, 1) +
                "&entropy=" + DoubleToString(g_quantum_entropy, 4));
    
    if(!HttpGet(scanPath, scanBody, 3000))
        return false;
    
    // Processar resultados do scan quântico
    int posSel = StringFind(scanBody, "\"selected\":[");
    bool anyOk = false;
    int executed = 0;
    
    if(posSel != -1)
    {
        int cur = posSel;
        while(executed < MaxSymbolsToTrade && PositionsTotal() < MaxPositions)
        {
            int ps = StringFind(scanBody, "\"symbol\":\"", cur);
            if(ps == -1) break;
            
            int s0 = ps + StringLen("\"symbol\":\"");
            int s1 = StringFind(scanBody, "\"", s0);
            if(s1 == -1) break;
            
            string s = StringSubstr(scanBody, s0, s1 - s0);
            cur = s1 + 1;
            
            if(StringLen(s) > 0 && ExecuteFromQuantumDecision(s))
            {
                anyOk = true;
                executed++;
            }
        }
    }
    
    if(anyOk) 
        return true;
    
    // Último recurso: decisão para símbolo de execução
    if(PositionsTotal() < MaxPositions)
        return ExecuteFromQuantumDecision(execution_symbol);
    
    return false;
}

//+------------------------------------------------------------------+
//| Atualiza entropia quântica global                                |
//+------------------------------------------------------------------+
void UpdateGlobalQuantumEntropy()
{
    double returns[];
    int count = 0;
    
    // Coletar returns de todos os símbolos
    string symbols[];
    int sym_count = StringSplit(SymbolList, ',', symbols);
    
    for(int i = 0; i < sym_count; i++)
    {
        double close[];
        ArraySetAsSeries(close, true);
        int copied = CopyClose(symbols[i], PERIOD_H1, 0, 50, close);
        if(copied > 10)
        {
            int old_size = ArraySize(returns);
            ArrayResize(returns, old_size + copied - 1);
            
            for(int j = 0; j < copied-1; j++)
            {
                returns[old_size + j] = (close[j] - close[j+1]) / close[j+1];
            }
            
            count += copied - 1;
        }
    }
    
    if(count > 0)
    {
        g_quantum_entropy = CalculateQuantumEntropy(returns);
        Print("Entropia quântica global atualizada: ", g_quantum_entropy);
    }
}

    g_scaling_levels[2] = ScalingLevels3;
    g_scaling_levels[3] = ScalingLevels4;
    
    return(INIT_SUCCEEDED);
}

    EventKillTimer();
    Print("[Theodora] Finalizado. Motivo:", reason, " | Estado Quântico: ", g_quantum_state);
}

//+------------------------------------------------------------------+
//| OnTimer                                                          |
//+------------------------------------------------------------------+
void OnTimer()
{
    if(GlobalPause)
    {
        Print("[Theodora] GlobalPause ativo - ciclo ignorado");
        return;
    }
    
    static bool closedOnce = false;
    if(EmergencyCloseAll && !closedOnce)
    {
        CloseAllOpenPositions();
        closedOnce = true;
        return;
    }
    
    if(g_isBusy)
        return;
    
    datetime now = TimeCurrent();
    if(g_fail_count > 0 && now < g_next_allowed)
        return;
    
    g_isBusy = true;
    bool ok = RequestQuantumAIDecision(_Symbol);
    
    if(ok)
    {
        g_fail_count = 0;
        g_backoff_sec = 0;
        g_next_allowed = 0;
        Print("Decisão quântica executada com sucesso");
    }
    else
    {
        g_fail_count++;
        int step = (int)MathPow(2, MathMin(g_fail_count, 10));
        g_backoff_sec = MathMin(step * TimerIntervalSec, BackoffMaxSec);
        g_next_allowed = now + g_backoff_sec;
        Print("Falha na decisão quântica. Backoff: ", g_backoff_sec, "s");
    }
    
    g_isBusy = false;
    
    // Atualizar trailing stops quânticos
    UpdateAllQuantumTrailingStops();
}

//+------------------------------------------------------------------+
//| Atualiza todos os trailing stops quânticos                       |
//+------------------------------------------------------------------+
void UpdateAllQuantumTrailingStops()
{
    int total = PositionsTotal();
    for(int i = 0; i < total; i++)
    {
        ulong ticket = PositionGetTicket(i);
        if(PositionSelectByTicket(ticket))
        {
            string symbol = PositionGetString(POSITION_SYMBOL);
            double atr = CalculateQuantumATR(symbol, 14, g_quantum_entropy);
            double entry_price = PositionGetDouble(POSITION_PRICE_OPEN);
            
            UpdateQuantumTrailingStop(symbol, ticket, entry_price, atr, g_quantum_entropy);
        }
    }
}

//+------------------------------------------------------------------+
//| OnTick                                                           |
//+------------------------------------------------------------------+
void OnTick()
{
    // Execução principal em OnTimer, mas podemos adicionar
    // monitoramento de tick para operações de alta frequência
    static datetime last_tick = 0;
    datetime current_time = TimeCurrent();
    
    if(current_time > last_tick)
    {
        // Atualizar entropia a cada tick novo
        UpdateGlobalQuantumEntropy();
        last_tick = current_time;
    }
}
