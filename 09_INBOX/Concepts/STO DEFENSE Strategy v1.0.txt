//+------------------------------------------------------------------+
//|                    STO DEFENSE Strategy v1.0                   |
//|      Algoritmo baseado no conceito Sato's Defense                |
//+------------------------------------------------------------------+
#include <Trade/Trade.mqh>
CTrade trade;

// Parâmetros de entrada
input double LotSize = 0.1; // Tamanho da posição
input int LossPoints = 250; // Perda máxima em pontos
input int GainPoints = 100; // Ganho mínimo em pontos
input int SwingPeriod = 20; // Período para detectar swings

// Variáveis globais
double defenseTop = 0.0; // DEFENSE de topo
double defenseBottom = 0.0; // DEFENSE de fundo
bool positionOpen = false;

// Função de inicialização
int OnInit()
{
    Print("STO DEFENSE Strategy Iniciado!");
    return INIT_SUCCEEDED;
}

// Função principal de execução
void OnTick()
{
    if (!IsMarketConditionGood()) return;

    CalculateDefenseLevels(); // Calcula os níveis DEFENSE

    double price = iClose(_Symbol, PERIOD_M5, 0);

    if (price > defenseBottom && !positionOpen && CheckMargin(LotSize, price))
    {
        trade.Buy(LotSize, _Symbol, price, price - LossPoints * _Point, price + GainPoints * _Point, "STO DEFENSE Buy");
        Print(StringFormat("%s: Compra executada. Preço = %.5f", TimeToString(TimeCurrent(), TIME_DATE | TIME_SECONDS), price));
        positionOpen = true;
    }
    else if (price < defenseTop && !positionOpen && CheckMargin(LotSize, price))
    {
        trade.Sell(LotSize, _Symbol, price, price + LossPoints * _Point, price - GainPoints * _Point, "STO DEFENSE Sell");
        Print(StringFormat("%s: Venda executada. Preço = %.5f", TimeToString(TimeCurrent(), TIME_DATE | TIME_SECONDS), price));
        positionOpen = true;
    }

    ManagePositions(price); // Gerencia posições abertas
}

// Função de encerramento
void OnDeinit(const int reason)
{
    Print("STO DEFENSE Strategy Encerrado!");
}

// Calcula os níveis DEFENSE (topo e fundo)
void CalculateDefenseLevels()
{
    // Detecta o swing de topo (maior alta nos últimos 'SwingPeriod' candles)
    double high = iHigh(_Symbol, PERIOD_M5, 0);
    double low = iLow(_Symbol, PERIOD_M5, 0);
    double totalVolumeTop = 0.0;
    double weightedPriceTop = 0.0;

    double totalVolumeBottom = 0.0;
    double weightedPriceBottom = 0.0;

    for (int i = 0; i < SwingPeriod; i++)
    {
        double currentHigh = iHigh(_Symbol, PERIOD_M5, i);
        double currentLow = iLow(_Symbol, PERIOD_M5, i);
        double currentVolume = iVolume(_Symbol, PERIOD_M5, i);

        // Para o swing de topo
        if (currentHigh == high)
        {
            totalVolumeTop += currentVolume;
            weightedPriceTop += currentHigh * currentVolume;
        }

        // Para o swing de fundo
        if (currentLow == low)
        {
            totalVolumeBottom += currentVolume;
            weightedPriceBottom += currentLow * currentVolume;
        }
    }

    // Calcula o POC para topo e fundo
    defenseTop = NormalizeDouble(weightedPriceTop / totalVolumeTop, _Digits);
    defenseBottom = NormalizeDouble(weightedPriceBottom / totalVolumeBottom, _Digits);

    Print(StringFormat("DEFENSE Top = %.5f, DEFENSE Bottom = %.5f", defenseTop, defenseBottom));
}

// Verifica se há condições de mercado boas
bool IsMarketConditionGood()
{
    MqlDateTime timeStruct;
    TimeToStruct(TimeCurrent(), timeStruct);

    if (timeStruct.day_of_week >= 6) return false; // Não opera nos finais de semana

    double spread = SymbolInfoInteger(_Symbol, SYMBOL_SPREAD);
    double volume = iVolume(_Symbol, PERIOD_M5, 0);

    return (spread < 50 && volume > 100);
}

// Verifica margem suficiente
bool CheckMargin(double lotSize, double price)
{
    double marginRequired = 0.0;
    if (!OrderCalcMargin(ORDER_TYPE_BUY, _Symbol, lotSize, price, marginRequired))
    {
        Print("Erro ao calcular margem necessária.");
        return false;
    }

    double freeMargin = AccountInfoDouble(ACCOUNT_FREEMARGIN);
    if (marginRequired > freeMargin)
    {
        Print(StringFormat("Margem insuficiente (%.2f > %.2f)", marginRequired, freeMargin));
        return false;
    }

    return true;
}

// Gerencia posições abertas
void ManagePositions(double currentPrice)
{
    for (int i = PositionsTotal() - 1; i >= 0; i--)
    {
        ulong ticket = PositionGetTicket(i);
        if (PositionSelect(PositionGetString(ticket, POSITION_SYMBOL)) && PositionGetInteger(ticket, POSITION_MAGIC) == trade.RequestMagic())
        {
            double openPrice = PositionGetDouble(ticket, POSITION_PRICE_OPEN);
            int type = PositionGetInteger(ticket, POSITION_TYPE);

            if (type == ORDER_TYPE_BUY)
            {
                if ((currentPrice - openPrice) / _Point >= GainPoints)
                {
                    trade.PositionClose(ticket);
                    Print(StringFormat("Compra fechada por ganho (%.5f)", currentPrice));
                }
                if ((openPrice - currentPrice) / _Point >= LossPoints)
                {
                    trade.PositionClose(ticket);
                    Print(StringFormat("Compra fechada por perda (%.5f)", currentPrice));
                }
            }
            else if (type == ORDER_TYPE_SELL)
            {
                if ((openPrice - currentPrice) / _Point >= GainPoints)
                {
                    trade.PositionClose(ticket);
                    Print(StringFormat("Venda fechada por ganho (%.5f)", currentPrice));
                }
                if ((currentPrice - openPrice) / _Point >= LossPoints)
                {
                    trade.PositionClose(ticket);
                    Print(StringFormat("Venda fechada por perda (%.5f)", currentPrice));
                }
            }
        }
    }
}
