#!/bin/bash
# deploy-orchestra-monitoring.sh

echo "🎼 Deploying Prometheus Monitoring for Agent Orchestra Architecture"

# Create directory structure
mkdir -p orchestra-prometheus orchestra-alerts orchestra-dashboards

# Create orchestra-specific Prometheus config
cat > orchestra-prometheus/prometheus.yml << 'EOF'
global:
  scrape_interval: 500ms
  evaluation_interval: 500ms

scrape_configs:
  - job_name: 'orchestra_metrics'
    static_configs:
      - targets: ['maestro-core:9091']
    metrics_path: '/metrics'
    scrape_interval: 100ms
EOF

# Create orchestra alerts
cat > orchestra-alerts/orchestra-alerts.yml << 'EOF'
groups:
- name: orchestra_alerts
  rules:
  - alert: AgentPerformanceIssue
    expr: rate(agent_processing_time_seconds_sum[5m]) / rate(agent_processing_time_seconds_count[5m]) > 0.005
    for: 3m
    labels:
      severity: warning
    annotations:
      summary: "Agent performance issue detected"
      description: "Agent {{ $labels.agent_name }} showing high processing time"
EOF

# Deploy the orchestra monitoring stack
docker-compose -f docker-compose-orchestra.yml up -d

echo "✅ Orchestra monitoring deployed successfully"
echo "📊 Access Grafana at http://localhost:3000"
echo "📈 Access Prometheus at http://localhost:9090"
echo "🎼 Maestro core metrics at http://localhost:9091/metrics"
