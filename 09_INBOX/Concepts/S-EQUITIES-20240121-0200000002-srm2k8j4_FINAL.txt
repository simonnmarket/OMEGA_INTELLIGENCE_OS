    async def _calculate_momentum_score_v7(self, sector: Sector, sector_data: Dict) -> Decimal:
        """
        CALCULA SCORE DE MOMENTUM PARA SETOR
        """
        try:
            etf_symbol = self.sector_etfs[sector]
            
            if etf_symbol not in sector_data:
                return Decimal('0.5')  # Score neutro
            
            sector_info = sector_data[etf_symbol]
            
            # Fatores de momentum
            price_momentum = await self._calculate_price_momentum(sector_info)
            relative_strength = await self._calculate_relative_strength(sector_info)
            trend_strength = await self._calculate_trend_strength(sector_info)
            
            # Score combinado
            momentum_factors = [price_momentum, relative_strength, trend_strength]
            momentum_score = sum(momentum_factors) / Decimal(str(len(momentum_factors)))
            
            return max(Decimal('0'), min(momentum_score, Decimal('1')))
            
        except Exception as e:
            logging.error(f"Erro cálculo momentum {sector.value}: {e}")
            return Decimal('0.5')

    async def _determine_optimal_allocations_v7(self, opportunities: List[SectorRotationOpportunity]) -> Dict[Sector, Decimal]:
        """
        DETERMINA ALOÇÕES ÓTIMAS BASEADAS NOS SCORES
        """
        # Filtrar setores qualificados
        qualified_opportunities = [
            opp for opp in opportunities 
            if (opp.composite_score >= self.rotation_params['min_composite_score'] and 
                opp.security_clearance and
                opp.execution_probability > Decimal('0.65'))
        ]
        
        # Ordenar por score composto
        qualified_opportunities.sort(key=lambda x: x.composite_score, reverse=True)
        
        # Limitar número de setores ativos
        active_sectors = qualified_opportunities[:int(self.rotation_params['max_sectors_active'])]
        
        if not active_sectors:
            return {}
        
        # Calcular alocações proporcionais aos scores
        total_score = sum(opp.composite_score for opp in active_sectors)
        allocations = {}
        
        for opportunity in active_sectors:
            # Alocação base proporcional ao score
            base_allocation = opportunity.composite_score / total_score
            
            # Ajustar para limites mínimos e máximos
            adjusted_allocation = max(
                self.rotation_params['min_allocation'],
                min(base_allocation, self.rotation_params['max_allocation'])
            )
            
            allocations[opportunity.sector] = adjusted_allocation
            opportunity.target_allocation = adjusted_allocation
        
        # Normalizar para soma 100%
        total_allocated = sum(allocations.values())
        if total_allocated > Decimal('1'):
            allocations = {sector: alloc / total_allocated for sector, alloc in allocations.items()}
        
        return allocations

    async def _generate_rebalance_signals_v7(self, target_allocations: Dict[Sector, Decimal], allocated_capital: Decimal) -> List[TradingSignal]:
        """
        GERA SINAIS DE REBALANCEAMENTO BASEADO NAS ALOÇÕES ALVO
        """
        signals = []
        
        for sector, target_allocation in target_allocations.items():
            try:
                current_allocation = self.current_allocations[sector]
                allocation_diff = target_allocation - current_allocation
                
                # Verificar se precisa rebalancear (diferença > threshold)
                if abs(allocation_diff) > self.rotation_params['rebalance_threshold']:
                    etf_symbol = self.sector_etfs[sector]
                    position_size = allocated_capital * target_allocation
                    
                    action = 'BUY' if allocation_diff > Decimal('0') else 'SELL'
                    
                    signal = TradingSignal(
                        strategy_id=self.strategy_id,
                        asset=f"SECTOR_{etf_symbol}",
                        action=action,
                        confidence=Decimal('0.85'),  # Alta confiança para rebalanceamento
                        risk_score=Decimal('0.2'),   # Baixo risco para ETFs setoriais
                        timestamp=int(time.time() * 1e6),
                        metadata={
                            'sector': sector.value,
                            'etf_symbol': etf_symbol,
                            'current_allocation': float(current_allocation),
                            'target_allocation': float(target_allocation),
                            'allocation_change': float(allocation_diff),
                            'position_size': float(position_size),
                            'rebalance_type': 'SECTOR_ROTATION',
                            'composite_score': float(next((opp.composite_score for opp in opportunities if opp.sector == sector), Decimal('0'))),
                            'momentum_score': float(next((opp.momentum_score for opp in opportunities if opp.sector == sector), Decimal('0'))),
                            'value_score': float(next((opp.value_score for opp in opportunities if opp.sector == sector), Decimal('0'))),
                            'growth_score': float(next((opp.growth_score for opp in opportunities if opp.sector == sector), Decimal('0'))),
                            'quality_score': float(next((opp.quality_score for opp in opportunities if opp.sector == sector), Decimal('0'))),
                            'macro_score': float(next((opp.macro_score for opp in opportunities if opp.sector == sector), Decimal('0'))),
                            'security_checks_passed': True
                        }
                    )
                    signals.append(signal)
                    
                    # Atualizar alocação atual
                    self.current_allocations[sector] = target_allocation
                    
            except Exception as e:
                logging.error(f"Erro geração sinal rebalanceamento {sector.value}: {e}")
                continue
                
        return signals

    def _normalize_value_factor(self, value: Decimal, lower_better: bool = True) -> Decimal:
        """Normaliza fatores de value"""
        if lower_better:
            # Valores mais baixos são melhores (PE, PB)
            return max(Decimal('0'), min(Decimal('1'), Decimal('1') - (value / Decimal('50'))))
        else:
            # Valores mais altos são melhores (dividend yield)
            return max(Decimal('0'), min(Decimal('1'), value / Decimal('0.1')))

    def _normalize_quality_factor(self, value: Decimal, higher_better: bool = True) -> Decimal:
        """Normaliza fatores de quality"""
        if higher_better:
            # Valores mais altos são melhores (ROE, profit margin)
            return max(Decimal('0'), min(Decimal('1'), value / Decimal('0.3')))
        else:
            # Valores mais baixos são melhores (debt/equity)
            return max(Decimal('0'), min(Decimal('1'), Decimal('1') - (value / Decimal('2'))))
