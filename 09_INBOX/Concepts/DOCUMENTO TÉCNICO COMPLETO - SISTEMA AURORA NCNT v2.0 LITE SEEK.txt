# ==================== ENUMS E DATA CLASSES ================================== #

class ModuleMode(Enum):
    """Modos de operação do módulo"""
    TEST = "TEST"           # Compliance apenas loga, não bloqueia
    LIMITED_PROD = "LIMITED_PROD"  # Compliance básico, bloqueia apenas risco crítico
    FULL_PROD = "FULL_PROD"        # Compliance completo pode bloquear
    
class ComplianceLevel(Enum):
    """Níveis de compliance progressivo"""
    BASIC = "BASIC"         # Apenas integridade e logging
    ESSENTIAL = "ESSENTIAL" # Risk controls básicos
    STANDARD = "STANDARD"   # Regulatório básico
    INSTITUTIONAL = "INSTITUTIONAL" # Completo

@dataclass
class ModuleVitals:
    """Métricas vitais do módulo"""
    module_name: str
    module_version: str
    status: str = "DOWN"  # UP, DOWN, DEGRADED
    mode: ModuleMode = ModuleMode.TEST
    compliance_level: ComplianceLevel = ComplianceLevel.BASIC
    
    # Métricas de performance
    latency_ms: float = 0.0
    error_rate: float = 0.0
    total_operations: int = 0
    successful_operations: int = 0
    
    # Métricas de trading (quando aplicável)
    pnl: float = 0.0
    sharpe_ratio: float = 0.0
    max_drawdown: float = 0.0
    win_rate: float = 0.0
    
    # Compliance metrics
    compliance_score: float = 100.0
    last_compliance_check: Optional[datetime] = None
    violations: List[str] = field(default_factory=list)
    
    # Timestamps
    created_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    last_heartbeat: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    
    def to_dict(self) -> Dict:
        """Converte para dicionário para logging/serialização"""
        return {
            "module_name": self.module_name,
            "module_version": self.module_version,
            "status": self.status,
            "mode": self.mode.value,
            "compliance_level": self.compliance_level.value,
            "latency_ms": round(self.latency_ms, 2),
            "error_rate": round(self.error_rate, 4),
            "total_operations": self.total_operations,
            "successful_operations": self.successful_operations,
            "pnl": round(self.pnl, 2),
            "sharpe_ratio": round(self.sharpe_ratio, 2),
            "max_drawdown": round(self.max_drawdown, 4),
            "win_rate": round(self.win_rate, 4),
            "compliance_score": round(self.compliance_score, 2),
            "last_compliance_check": self.last_compliance_check.isoformat() if self.last_compliance_check else None,
            "violations": self.violations,
            "last_heartbeat": self.last_heartbeat.isoformat()
        }

# ==================== CLASSE PRINCIPAL DO TEMPLATE ========================== #

class NCNTModuleLite:
    """
    Template NCNT Lite v2.0 - Foco em trading com compliance progressivo
    
    PRINCÍPIOS:
    1. Simplicidade primeiro (~300 linhas vs ~1100 do v2.0 original)
    2. Compliance progressivo baseado em capital
    3. Thread-safe para produção
    4. Métricas de trading integradas
    5. Upgrade path para v3.0
    """
    
    # Configurações padrão baseadas em capital
    CAPITAL_THRESHOLDS = {
        1000: ComplianceLevel.BASIC,       # < €1,000
        10000: ComplianceLevel.ESSENTIAL,  # €1,000 - €10,000
        50000: ComplianceLevel.STANDARD,   # €10,000 - €50,000
        float('inf'): ComplianceLevel.INSTITUTIONAL  # > €50,000
    }
    
    def __init__(
        self,
        module_name: str,
        module_version: str = "1.0.0",
        capital_allocation: float = 3500.0,  # Capital alocado ao módulo
        config: Optional[Dict] = None,
        mode: ModuleMode = ModuleMode.TEST
    ):
        """
        Inicializa módulo NCNT Lite
        
        Args:
            module_name: Nome único do módulo
            module_version: Versão do módulo
            capital_allocation: Capital em EUR alocado para este módulo
            config: Configuração específica do módulo
            mode: Modo de operação (TEST, LIMITED_PROD, FULL_PROD)
        """
        self.module_name = module_name
        self.module_version = module_version
        self.capital_allocation = capital_allocation
        self.config = config or {}
        self.mode = mode
        
        # Determinar nível de compliance baseado no capital
        self.compliance_level = self._determine_compliance_level(capital_allocation)
        
        # Thread safety
        self._lock = threading.RLock()
        
        # Sistema de integridade
        self.module_checksum = self._calculate_module_checksum()
        self.instance_checksum = self._calculate_instance_checksum()
        
        # Métricas e vitalidade
        self.vitals = ModuleVitals(
            module_name=module_name,
            module_version=module_version,
            mode=mode,
            compliance_level=self.compliance_level
        )
        
        # Conexões neurais (simplificadas)
        self.neural_connections: Dict[str, Dict] = {}
        
        # Dependências
        self.required_dependencies: List[str] = []
        self.optional_dependencies: List[str] = []
        
        # Inicialização
        self._initialize_module()
        
        logger.info(
            f"Module initialized",
            extra={
                "module": self.module_name,
                "version": self.module_version,
                "capital": self.capital_allocation,
                "compliance_level": self.compliance_level.value,
                "mode": self.mode.value,
                "checksum": self.instance_checksum[:16]
            }
        )
    
    # ==================== MÉTODOS DE INTEGRIDADE ============================ #
    
    def _calculate_module_checksum(self) -> str:
        """Calcula checksum do código-fonte do módulo (SHA3-256)"""
        try:
            # Para simplificação, usamos hash da classe
            source = f"{self.__class__.__name__}{self.module_version}"
            return hashlib.sha3_256(source.encode()).hexdigest()
        except Exception as e:
            logger.warning(f"Failed to calculate module checksum: {e}")
            return "unknown"
    
    def _calculate_instance_checksum(self) -> str:
        """Calcula checksum único para esta instância"""
        with self._lock:
            data = {
                "module_name": self.module_name,
                "module_version": self.module_version,
                "capital": self.capital_allocation,
                "config": self.config,
                "timestamp": datetime.now(timezone.utc).isoformat(),
                "random": os.urandom(16).hex() if 'os' in globals() else "static"
            }
            json_str = json.dumps(data, sort_keys=True)
            return hashlib.sha3_256(json_str.encode()).hexdigest()
    
    def validate_integrity(self) -> bool:
        """Valida integridade do módulo"""
        current_checksum = self._calculate_instance_checksum()
        return current_checksum == self.instance_checksum
    
    # ==================== COMPLIANCE PROGRESSIVO ============================ #
    
    def _determine_compliance_level(self, capital: float) -> ComplianceLevel:
        """Determina nível de compliance baseado no capital"""
        for threshold, level in sorted(self.CAPITAL_THRESHOLDS.items()):
            if capital <= threshold:
                return level
        return ComplianceLevel.INSTITUTIONAL
    
    def _check_basic_compliance(self, operation: str, data: Dict) -> Dict:
        """Verificação básica de compliance (nível BASIC)"""
        violations = []
        warnings = []
        
        # Regras básicas independentes de capital
        if self.mode == ModuleMode.FULL_PROD:
            # Em FULL_PROD, verifica integridade
            if not self.validate_integrity():
                violations.append("INTEGRITY_CHECK_FAILED")
        
        # Verificação de tamanho de posição (básico)
        if "position_size" in data:
            max_position = self.capital_allocation * 0.02  # 2% do capital
            if data["position_size"] > max_position:
                if self.mode in [ModuleMode.LIMITED_PROD, ModuleMode.FULL_PROD]:
                    violations.append("POSITION_SIZE_EXCEEDED")
                else:
                    warnings.append("Position size exceeds 2% of capital (would block in PROD)")
        
        return {
            "compliant": len(violations) == 0,
            "violations": violations,
            "warnings": warnings,
            "compliance_level": self.compliance_level.value,
            "timestamp": datetime.now(timezone.utc).isoformat()
        }
    
    def _check_essential_compliance(self, operation: str, data: Dict) -> Dict:
        """Verificação essencial de compliance (nível ESSENTIAL)"""
        basic_check = self._check_basic_compliance(operation, data)
        
        if not basic_check["compliant"]:
            return basic_check
        
        violations = basic_check["violations"]
        warnings = basic_check["warnings"]
        
        # Regras adicionais para ESSENTIAL
        if "risk_score" in data:
            if data["risk_score"] > 0.8:  # Risk score alto
                if self.mode == ModuleMode.FULL_PROD:
                    violations.append("RISK_SCORE_TOO_HIGH")
                else:
                    warnings.append("High risk score detected")
        
        # Verificação de drawdown
        if hasattr(self, 'current_drawdown'):
            if self.current_drawdown > 0.1:  # 10% drawdown
                warnings.append("Drawdown exceeds 10%")
        
        return {
            "compliant": len(violations) == 0,
            "violations": violations,
            "warnings": warnings,
            "compliance_level": self.compliance_level.value,
            "timestamp": datetime.now(timezone.utc).isoformat()
        }
    
    def check_compliance(self, operation: str, data: Dict) -> Dict:
        """
        Verificação de compliance progressiva baseada no nível
        
        AIC NOTA: Este método escala automaticamente com capital_allocation
        """
        with self._lock:
            self.vitals.last_compliance_check = datetime.now(timezone.utc)
            
            if self.compliance_level == ComplianceLevel.BASIC:
                result = self._check_basic_compliance(operation, data)
            elif self.compliance_level == ComplianceLevel.ESSENTIAL:
                result = self._check_essential_compliance(operation, data)
            else:
                # Para STANDARD e INSTITUTIONAL, usa ESSENTIAL por enquanto
                # (será estendido quando capital aumentar)
                result = self._check_essential_compliance(operation, data)
            
            # Atualizar métricas
            if not result["compliant"]:
                self.vitals.violations.extend(result["violations"])
                self.vitals.compliance_score = max(0, self.vitals.compliance_score - 5)
            
            # Logging estruturado
            logger.info(
                "Compliance check completed",
                extra={
                    "module": self.module_name,
                    "operation": operation,
                    "compliant": result["compliant"],
                    "violations": len(result["violations"]),
                    "warnings": len(result["warnings"]),
                    "compliance_level": self.compliance_level.value
                }
            )
            
            return result
    
    # ==================== MÉTODOS DE OPERAÇÃO =============================== #
    
    def execute(self, operation: str, data: Dict) -> Dict:
        """
        Método principal de execução - DEVE SER SOBRESCRITO por módulos filhos
        
        Args:
            operation: Tipo de operação (ex: "PLACE_TRADE", "UPDATE_POSITION")
            data: Dados da operação
            
        Returns:
            Resultado da operação com metadados
        """
        start_time = datetime.now(timezone.utc)
        
        try:
            with self._lock:
                # 1. Verificação de compliance pré-execução
                compliance_result = self.check_compliance(operation, data)
                
                # 2. Determinar se bloqueia baseado no modo
                should_block = (
                    (self.mode == ModuleMode.FULL_PROD and not compliance_result["compliant"]) or
                    (self.mode == ModuleMode.LIMITED_PROD and any(
                        v in ["INTEGRITY_CHECK_FAILED", "POSITION_SIZE_EXCEEDED"] 
                        for v in compliance_result["violations"]
                    ))
                )
                
                if should_block:
                    end_time = datetime.now(timezone.utc)
                    latency = (end_time - start_time).total_seconds() * 1000
                    
                    # Atualizar métricas
                    self._update_metrics(operation, False, latency)
                    
                    return {
                        "success": False,
                        "operation": operation,
                        "reason": "BLOCKED_BY_COMPLIANCE",
                        "violations": compliance_result["violations"],
                        "warnings": compliance_result["warnings"],
                        "latency_ms": round(latency, 2),
                        "timestamp": end_time.isoformat()
                    }
                
                # 3. Executar operação (método abstrato - implementado por filhos)
                operation_result = self._execute_operation(operation, data)
                
                end_time = datetime.now(timezone.utc)
                latency = (end_time - start_time).total_seconds() * 1000
                
                # 4. Atualizar métricas
                self._update_metrics(operation, True, latency)
                
                # 5. Retornar resultado
                return {
                    "success": True,
                    "operation": operation,
                    "result": operation_result,
                    "compliance": compliance_result,
                    "latency_ms": round(latency, 2),
                    "timestamp": end_time.isoformat()
                }
                
        except Exception as e:
            end_time = datetime.now(timezone.utc)
            latency = (end_time - start_time).total_seconds() * 1000
            
            logger.error(
                f"Execution failed: {str(e)}",
                extra={
                    "module": self.module_name,
                    "operation": operation,
                    "error": str(e),
                    "latency_ms": round(latency, 2)
                },
                exc_info=True
            )
            
            self._update_metrics(operation, False, latency)
            
            return {
                "success": False,
                "operation": operation,
                "error": str(e),
                "latency_ms": round(latency, 2),
                "timestamp": end_time.isoformat()
            }
    
    def _execute_operation(self, operation: str, data: Dict) -> Any:
        """
        Método interno de execução - DEVE SER IMPLEMENTADO por módulos específicos
        
        AIC: Sobrescrever este método em módulos filhos para lógica específica
        """
        raise NotImplementedError(
            f"Module {self.module_name} must implement _execute_operation method"
        )
    
    # ==================== MÉTODOS DE GERENCIAMENTO ========================== #
    
    def _initialize_module(self):
        """Inicialização do módulo"""
        with self._lock:
            self.vitals.status = "UP"
            self.vitals.last_heartbeat = datetime.now(timezone.utc)
            
            # Emitir sinal de inicialização
            self._emit_completion_signal()
    
    def _update_metrics(self, operation: str, success: bool, latency_ms: float):
        """Atualiza métricas do módulo"""
        with self._lock:
            self.vitals.total_operations += 1
            if success:
                self.vitals.successful_operations += 1
            
            # Calcular error rate (janela móvel)
            if self.vitals.total_operations > 0:
                error_count = self.vitals.total_operations - self.vitals.successful_operations
                self.vitals.error_rate = error_count / self.vitals.total_operations
            
            # Atualizar latência (média móvel)
            if self.vitals.latency_ms == 0:
                self.vitals.latency_ms = latency_ms
            else:
                # EMA com fator 0.1
                self.vitals.latency_ms = 0.9 * self.vitals.latency_ms + 0.1 * latency_ms
            
            self.vitals.last_heartbeat = datetime.now(timezone.utc)
    
    def _emit_completion_signal(self):
        """Emite sinal de conclusão (simplificado - sem emojis para Windows)"""
        print(f"[NCNT] Module {self.module_name} v{self.module_version} initialized")
        print(f"[NCNT] Compliance Level: {self.compliance_level.value}")
        print(f"[NCNT] Mode: {self.mode.value}")
        print(f"[NCNT] Checksum: {self.instance_checksum[:16]}...")
        print(f"[NCNT] Capital Allocation: €{self.capital_allocation:,.2f}")
        print("-" * 60)
    
    def get_vitals(self) -> Dict:
        """Retorna métricas vitais do módulo"""
        with self._lock:
            return self.vitals.to_dict()
    
    def heartbeat(self):
        """Atualiza heartbeat do módulo"""
        with self._lock:
            self.vitals.last_heartbeat = datetime.now(timezone.utc)
            return True
    
    def shutdown(self):
        """Desligamento gracioso do módulo"""
        with self._lock:
            self.vitals.status = "DOWN"
            logger.info(f"Module {self.module_name} shutdown")
            return True
    
    # ==================== MÉTODOS DE CONEXÃO NEURAL ======================== #
    
    def add_neural_connection(self, target_module: str, connection_type: str = "DATA"):
        """Adiciona conexão neural simplificada"""
        with self._lock:
            self.neural_connections[target_module] = {
                "type": connection_type,
                "established_at": datetime.now(timezone.utc).isoformat(),
                "last_activity": datetime.now(timezone.utc).isoformat(),
                "active": True
            }
            logger.info(f"Neural connection added: {target_module}")
    
    def update_neural_activity(self, target_module: str):
        """Atualiza atividade de conexão neural"""
        with self._lock:
            if target_module in self.neural_connections:
                self.neural_connections[target_module]["last_activity"] = \
                    datetime.now(timezone.utc).isoformat()
    
    # ==================== MÉTODOS DE DEPENDÊNCIA =========================== #
    
    def add_dependency(self, module_name: str, required: bool = True):
        """Adiciona dependência do módulo"""
        with self._lock:
            if required:
                self.required_dependencies.append(module_name)
            else:
                self.optional_dependencies.append(module_name)
    
    def check_dependencies(self, available_modules: List[str]) -> Dict:
        """Verifica se dependências estão disponíveis"""
        with self._lock:
            missing_required = [
                dep for dep in self.required_dependencies 
                if dep not in available_modules
            ]
            
            missing_optional = [
                dep for dep in self.optional_dependencies 
                if dep not in available_modules
            ]
            
            return {
                "all_dependencies_met": len(missing_required) == 0,
                "missing_required": missing_required,
                "missing_optional": missing_optional,
                "required_count": len(self.required_dependencies),
                "optional_count": len(self.optional_dependencies)
            }

"""

# INÍCIO DO ARQUIVO trading_core.py

import numpy as np
from typing import Dict, List, Optional, Tuple
from datetime import datetime, timedelta

class TradingCoreModule(NCNTModuleLite):
    """
    Módulo core para estratégias de trading
    
    Extende NCNTModuleLite com funcionalidades específicas de trading:
    - Gestão de posições
    - Cálculo de P&L
    - Risk management básico
    - Signal generation
    """
    
    def __init__(
        self,
        module_name: str,
        strategy_config: Dict,
        capital_allocation: float = 3500.0,
        risk_per_trade: float = 0.02,  # 2% por trade
        max_drawdown: float = 0.15,    # 15% máximo
        **kwargs
    ):
        """
        Inicializa módulo de trading
        
        Args:
            strategy_config: Configuração específica da estratégia
            risk_per_trade: Risco máximo por trade (0.01 = 1%)
            max_drawdown: Drawdown máximo permitido
        """
        # Configuração específica de trading
        trading_config = {
            "risk_per_trade": risk_per_trade,
            "max_drawdown": max_drawdown,
            "position_sizing_method": "kelly" if capital_allocation > 10000 else "fixed",
            **strategy_config
        }
        
        # Inicializar módulo pai
        super().__init__(
            module_name=module_name,
            capital_allocation=capital_allocation,
            config=trading_config,
            **kwargs
        )
        
        # Estado de trading
        self.positions: Dict[str, Dict] = {}
        self.trade_history: List[Dict] = []
        self.current_pnl: float = 0.0
        self.current_drawdown: float = 0.0
        self.peak_capital: float = capital_allocation
        
        # Métricas de performance
        self.sharpe_ratio: float = 0.0
        self.win_rate: float = 0.0
        self.total_trades: int = 0
        self.winning_trades: int = 0
        
        # Adicionar dependências padrão
        self.add_dependency("MarketData", required=True)
        self.add_dependency("RiskManager", required=False)
        
        logger.info(
            f"Trading module initialized",
            extra={
                "module": self.module_name,
                "risk_per_trade": risk_per_trade,
                "max_drawdown": max_drawdown,
                "position_sizing": trading_config["position_sizing_method"]
            }
        )
    
    def _execute_operation(self, operation: str, data: Dict) -> Any:
        """
        Implementação específica para operações de trading
        """
        if operation == "ANALYZE_MARKET":
            return self._analyze_market(data)
        elif operation == "GENERATE_SIGNAL":
            return self._generate_signal(data)
        elif operation == "PLACE_TRADE":
            return self._place_trade(data)
        elif operation == "UPDATE_POSITION":
            return self._update_position(data)
        elif operation == "CLOSE_POSITION":
            return self._close_position(data)
        elif operation == "CALCULATE_PNL":
            return self._calculate_pnl(data)
        else:
            raise ValueError(f"Unknown operation: {operation}")

    def _place_trade(self, trade_data: Dict) -> Dict:
        """
        Coloca um trade com risk management
        """
        # Validar dados do trade
        required_fields = ["symbol", "direction", "price", "size"]
        for field in required_fields:
            if field not in trade_data:
                raise ValueError(f"Missing required field: {field}")
        
        # Calcular tamanho da posição baseado no risco
        position_size = self._calculate_position_size(
            trade_data["price"],
            trade_data.get("stop_loss"),
            trade_data.get("take_profit")
        )
        
        # Verificar se está dentro dos limites
        if not self._validate_position_size(position_size):
            return {
                "status": "REJECTED",
                "reason": "POSITION_SIZE_EXCEEDS_LIMIT",
                "requested_size": trade_data["size"],
                "allowed_size": position_size
            }
        
        # Criar posição
        position_id = f"{trade_data['symbol']}_{datetime.now(timezone.utc).timestamp()}"
        
        position = {
            "position_id": position_id,
            "symbol": trade_data["symbol"],
            "direction": trade_data["direction"],  # LONG or SHORT
            "entry_price": trade_data["price"],
            "size": position_size,
            "entry_time": datetime.now(timezone.utc).isoformat(),
            "stop_loss": trade_data.get("stop_loss"),
            "take_profit": trade_data.get("take_profit"),
            "status": "OPEN"
        }
        
        # Adicionar à lista de posições
        self.positions[position_id] = position
        
        # Registrar no histórico
        trade_record = {
            **position,
            "operation": "OPEN",
            "capital_at_open": self.capital_allocation
        }
        self.trade_history.append(trade_record)
        
        # Atualizar métricas
        self.total_trades += 1
        
        logger.info(
            "Trade placed",
            extra={
                "module": self.module_name,
                "position_id": position_id,
                "symbol": trade_data["symbol"],
                "direction": trade_data["direction"],
                "size": position_size,
                "price": trade_data["price"]
            }
        )
        
        return {
            "status": "EXECUTED",
            "position_id": position_id,
            "size": position_size,
            "entry_price": trade_data["price"],
            "timestamp": datetime.now(timezone.utc).isoformat()
        }
    
    def _update_position(self, update_data: Dict) -> Dict:
        """Atualiza uma posição existente"""
        position_id = update_data.get("position_id")
        
        if position_id not in self.positions:
            return {"status": "ERROR", "reason": "POSITION_NOT_FOUND"}
        
        # Atualizar posição
        for key, value in update_data.items():
            if key != "position_id" and key in self.positions[position_id]:
                self.positions[position_id][key] = value
        
        # Registrar atualização
        update_record = {
            "position_id": position_id,
            "update": update_data,
            "timestamp": datetime.now(timezone.utc).isoformat()
        }
        self.trade_history.append(update_record)
        
        return {"status": "UPDATED", "position_id": position_id}
    
    def _close_position(self, close_data: Dict) -> Dict:
        """Fecha uma posição e calcula P&L"""
        position_id = close_data.get("position_id")
        exit_price = close_data.get("exit_price")
        
        if position_id not in self.positions:
            return {"status": "ERROR", "reason": "POSITION_NOT_FOUND"}
        
        position = self.positions[position_id]
        
        # Calcular P&L
        pnl = self._calculate_single_pnl(position, exit_price)
        
        # Atualizar métricas
        self.current_pnl += pnl
        self.capital_allocation += pnl
        
        if pnl > 0:
            self.winning_trades += 1
        
        # Atualizar win rate
        if self.total_trades > 0:
            self.win_rate = self.winning_trades / self.total_trades
        
        # Atualizar drawdown
        if self.capital_allocation > self.peak_capital:
            self.peak_capital = self.capital_allocation
            self.current_drawdown = 0.0
        else:
            self.current_drawdown = (self.peak_capital - self.capital_allocation) / self.peak_capital
        
        # Fechar posição
        self.positions[position_id]["status"] = "CLOSED"
        self.positions[position_id]["exit_price"] = exit_price
        self.positions[position_id]["exit_time"] = datetime.now(timezone.utc).isoformat()
        self.positions[position_id]["pnl"] = pnl
        
        # Registrar fechamento
        close_record = {
            "position_id": position_id,
            "operation": "CLOSE",
            "exit_price": exit_price,
            "pnl": pnl,
            "capital_at_close": self.capital_allocation,
            "timestamp": datetime.now(timezone.utc).isoformat()
        }
        self.trade_history.append(close_record)
        
        logger.info(
            "Position closed",
            extra={
                "module": self.module_name,
                "position_id": position_id,
                "symbol": position["symbol"],
                "pnl": pnl,
                "total_pnl": self.current_pnl
            }
        )
        
        # Atualizar vitals com métricas de trading
        self.vitals.pnl = self.current_pnl
        self.vitals.win_rate = self.win_rate
        self.vitals.max_drawdown = self.current_drawdown
        
        return {
            "status": "CLOSED",
            "position_id": position_id,
            "pnl": pnl,
            "total_pnl": self.current_pnl,
            "current_capital": self.capital_allocation
        }
    
    # ==================== MÉTODOS DE CÁLCULO ================================ #
    
    def _calculate_position_size(self, price: float, stop_loss: Optional[float] = None, 
                                take_profit: Optional[float] = None) -> float:
        """Calcula tamanho da posição baseado no risco"""
        risk_amount = self.capital_allocation * self.config["risk_per_trade"]
        
        if stop_loss is not None:
            # Baseado em stop loss
            risk_per_unit = abs(price - stop_loss)
            if risk_per_unit > 0:
                position_size = risk_amount / risk_per_unit
            else:
                position_size = risk_amount / (price * 0.01)  # Fallback: 1% do preço
        else:
            # Posição fixa baseada no capital
            if self.config["position_sizing_method"] == "kelly" and self.win_rate > 0:
                # Kelly Criterion simplificado
                kelly_fraction = self.win_rate - ((1 - self.win_rate) / 1.0)  # Assume payoff 1:1
                position_size = (self.capital_allocation * kelly_fraction * 0.5) / price  # Half Kelly
            else:
                # Posição fixa: 2% do capital
                position_size = risk_amount / price
        
        # Arredondar para lote padrão
        return self._round_to_lot(position_size)
    
    def _calculate_single_pnl(self, position: Dict, exit_price: float) -> float:
        """Calcula P&L para uma única posição"""
        if position["direction"] == "LONG":
            pnl = (exit_price - position["entry_price"]) * position["size"]
        else:  # SHORT
            pnl = (position["entry_price"] - exit_price) * position["size"]
        
        return pnl
    
    def _calculate_pnl(self, data: Dict) -> Dict:
        """Calcula P&L total e métricas"""
        total_pnl = 0.0
        open_pnl = 0.0
        closed_pnl = 0.0
        
        # P&L fechado
        for trade in self.trade_history:
            if trade.get("pnl"):
                closed_pnl += trade["pnl"]
        
        # P&L aberto (mark-to-market)
        for position_id, position in self.positions.items():
            if position["status"] == "OPEN" and "current_price" in data.get(position["symbol"], {}):
                current_price = data[position["symbol"]]["current_price"]
                open_pnl += self._calculate_single_pnl(position, current_price)
        
        total_pnl = closed_pnl + open_pnl
        
        # Calcular Sharpe ratio (simplificado)
        if len(self.trade_history) > 1:
            pnl_series = [t.get("pnl", 0) for t in self.trade_history if t.get("pnl")]
            if len(pnl_series) > 1:
                returns = np.array(pnl_series) / self.capital_allocation
                self.sharpe_ratio = np.mean(returns) / np.std(returns) * np.sqrt(252) if np.std(returns) > 0 else 0.0
                self.vitals.sharpe_ratio = self.sharpe_ratio
        
        return {
            "total_pnl": total_pnl,
            "closed_pnl": closed_pnl,
            "open_pnl": open_pnl,
            "sharpe_ratio": self.sharpe_ratio,
            "win_rate": self.win_rate,
            "current_drawdown": self.current_drawdown,
            "positions_open": len([p for p in self.positions.values() if p["status"] == "OPEN"]),
            "positions_closed": len([p for p in self.positions.values() if p["status"] == "CLOSED"])
        }
    
    # ==================== MÉTODOS AUXILIARES ================================ #
    
    def _validate_position_size(self, size: float) -> bool:
        """Valida se o tamanho da posição está dentro dos limites"""
        max_position_value = size * self._get_current_price()
        max_allowed = self.capital_allocation * 0.1  # Máximo 10% do capital em uma posição
        
        return max_position_value <= max_allowed
    
    def _round_to_lot(self, size: float, lot_size: float = 0.01) -> float:
        """Arredonda tamanho da posição para lote padrão"""
        return round(size / lot_size) * lot_size
    
    def _get_current_price(self) -> float:
        """Obtém preço atual (simulado - em produção viria de MarketData)"""
        return 1.0  # Placeholder
    
    def get_trading_metrics(self) -> Dict:
        """Retorna métricas específicas de trading"""
        return {
            "total_trades": self.total_trades,
            "winning_trades": self.winning_trades,
            "win_rate": self.win_rate,
            "total_pnl": self.current_pnl,
            "sharpe_ratio": self.sharpe_ratio,
            "current_drawdown": self.current_drawdown,
            "positions_open": len([p for p in self.positions.values() if p["status"] == "OPEN"]),
            "capital_allocation": self.capital_allocation,
            "peak_capital": self.peak_capital
        }

# INÍCIO DO ARQUIVO integration_gate.py

import hashlib
import inspect
from typing import Dict, List, Any

class IntegrationGate:
    """
    Integration Gate simplificado - Certificação de módulos NCNT
    
    Verifica:
    1. Integridade (checksums)
    2. Interface mínima (métodos obrigatórios)
    3. Dependências
    4. Compliance básico
    """
    
    def __init__(self):
        self.certified_modules: Dict[str, Dict] = {}
        self.certification_history: List[Dict] = []
    
    def certify_module(self, module_instance: Any) -> Dict:
        """
        Certifica um módulo NCNT
        
        Returns:
            Dict com status e detalhes da certificação
        """
        module_name = getattr(module_instance, 'module_name', 'unknown')
        
        certification = {
            "module_name": module_name,
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "checks_passed": [],
            "checks_failed": [],
            "warnings": [],
            "certified": False
        }
        
        # 1. Verificar integridade
        if hasattr(module_instance, 'validate_integrity'):
            try:
                integrity_ok = module_instance.validate_integrity()
                if integrity_ok:
                    certification["checks_passed"].append("INTEGRITY_CHECK")
                else:
                    certification["checks_failed"].append("INTEGRITY_CHECK")
            except Exception as e:
                certification["checks_failed"].append(f"INTEGRITY_CHECK_ERROR: {str(e)}")
        else:
            certification["warnings"].append("NO_INTEGRITY_CHECK_METHOD")
        
        # 2. Verificar métodos obrigatórios
        required_methods = ['execute', 'get_vitals', 'heartbeat', 'shutdown']
        for method in required_methods:
            if hasattr(module_instance, method) and callable(getattr(module_instance, method)):
                certification["checks_passed"].append(f"HAS_METHOD_{method.upper()}")
            else:
                certification["checks_failed"].append(f"MISSING_METHOD_{method.upper()}")
        
        # 3. Verificar vitals
        if hasattr(module_instance, 'get_vitals'):
            try:
                vitals = module_instance.get_vitals()
                if isinstance(vitals, dict):
                    certification["checks_passed"].append("VITALS_AVAILABLE")
                    certification["vitals_summary"] = {
                        "status": vitals.get("status", "unknown"),
                        "mode": vitals.get("mode", "unknown"),
                        "compliance_level": vitals.get("compliance_level", "unknown")
                    }
                else:
                    certification["warnings"].append("VITALS_NOT_DICT")
            except Exception as e:
                certification["warnings"].append(f"VITALS_ERROR: {str(e)}")
        
        # 4. Verificar dependências
        if hasattr(module_instance, 'check_dependencies'):
            try:
                # Lista simulada de módulos disponíveis
                available_modules = ["MarketData", "RiskManager", "ExecutionEngine"]
                dep_check = module_instance.check_dependencies(available_modules)
                
                if dep_check.get("all_dependencies_met", False):
                    certification["checks_passed"].append("DEPENDENCIES_MET")
                else:
                    missing_required = dep_check.get("missing_required", [])
                    if missing_required:
                        certification["checks_failed"].append(f"MISSING_DEPENDENCIES: {missing_required}")
                    else:
                        certification["checks_passed"].append("DEPENDENCIES_MET")
            except Exception as e:
                certification["warnings"].append(f"DEPENDENCY_CHECK_ERROR: {str(e)}")
        
        # 5. Verificar compliance básico
        if hasattr(module_instance, 'check_compliance'):
            try:
                # Teste básico de compliance
                test_operation = "TEST_OPERATION"
                test_data = {"test": True}
                compliance_result = module_instance.check_compliance(test_operation, test_data)
                
                if isinstance(compliance_result, dict):
                    certification["checks_passed"].append("COMPLIANCE_CHECK_WORKS")
                    certification["compliance_test"] = {
                        "compliant": compliance_result.get("compliant", False),
                        "level": compliance_result.get("compliance_level", "unknown")
                    }
                else:
                    certification["warnings"].append("COMPLIANCE_RESULT_NOT_DICT")
            except Exception as e:
                certification["warnings"].append(f"COMPLIANCE_CHECK_ERROR: {str(e)}")
        
        # Determinar se certifica
        failed_critical = any("MISSING_METHOD_" in check for check in certification["checks_failed"])
        failed_integrity = "INTEGRITY_CHECK" in certification["checks_failed"]
        
        certification["certified"] = not (failed_critical or failed_integrity)
        
        if certification["certified"]:
            certification["certification_id"] = self._generate_certification_id(module_name)
            self.certified_modules[module_name] = certification
            print(f"[INTEGRATION GATE] ✅ Module {module_name} CERTIFIED")
        else:
            print(f"[INTEGRATION GATE] ❌ Module {module_name} FAILED certification")
        
        # Registrar no histórico
        self.certification_history.append(certification)
        
        # Manter histórico limitado
        if len(self.certification_history) > 100:
            self.certification_history = self.certification_history[-100:]
        
        return certification
    
    def _generate_certification_id(self, module_name: str) -> str:
        """Gera ID único de certificação"""
        timestamp = datetime.now(timezone.utc).isoformat()
        data = f"{module_name}_{timestamp}_{hashlib.sha256(os.urandom(16)).hexdigest()[:8]}"
        return hashlib.sha3_256(data.encode()).hexdigest()[:16]
    
    def get_certification_status(self, module_name: str) -> Optional[Dict]:
        """Obtém status de certificação de um módulo"""
        return self.certified_modules.get(module_name)
    
    def list_certified_modules(self) -> List[str]:
        """Lista todos os módulos certificados"""
        return list(self.certified_modules.keys())

# INÍCIO DO ARQUIVO genesis_includes.py

from typing import Dict, Any, Optional

class GenesisIncludes:
    """
    Container IoC simplificado para o sistema Aurora
    
    Gerencia:
    - Registro de módulos
    - Resolução de dependências
    - Ciclo de vida
    """
    
    def __init__(self):
        self.registry: Dict[str, Any] = {}
        self.dependency_graph: Dict[str, List[str]] = {}
    
    def register(self, name: str, dependency: Any) -> None:
        """
        Registra um módulo ou dependência
        
        Args:
            name: Nome único do componente
            dependency: Instância do componente
        """
        if name in self.registry:
            print(f"[GENESIS] ⚠️ Component {name} already registered, overwriting")
        
        self.registry[name] = dependency
        print(f"[GENESIS] ✅ Registered: {name}")
    
    def resolve(self, name: str) -> Any:
        """
        Resolve uma dependência pelo nome
        
        Args:
            name: Nome da dependência
            
        Returns:
            Instância da dependência
            
        Raises:
            KeyError: Se dependência não encontrada
        """
        if name not in self.registry:
            raise KeyError(f"Dependency '{name}' not found in registry")
        
        return self.registry[name]
    
    def register_module(self, module_instance: Any, alias: Optional[str] = None) -> str:
        """
        Registra um módulo NCNT (método conveniente)
        
        Args:
            module_instance: Instância do módulo
            alias: Alias opcional (usa module_name por padrão)
            
        Returns:
            Nome registrado
        """
        if not hasattr(module_instance, 'module_name'):
            raise ValueError("Module instance must have 'module_name' attribute")
        
        name = alias or module_instance.module_name
        
        # Registrar módulo
        self.register(name, module_instance)
        
        # Registrar dependências se disponível
        if hasattr(module_instance, 'required_dependencies'):
            self.dependency_graph[name] = module_instance.required_dependencies
        
        return name
    
    def get_all_modules(self) -> Dict[str, Any]:
        """Retorna todos os módulos registrados"""
        return {k: v for k, v in self.registry.items() 
                if hasattr(v, 'module_name')}
    
    def get_module(self, module_name: str) -> Optional[Any]:
        """Obtém um módulo específico"""
        modules = self.get_all_modules()
        return modules.get(module_name)
    
    def check_dependency_health(self) -> Dict[str, Any]:
        """Verifica saúde das dependências"""
        results = {}
        
        for name, module in self.get_all_modules().items():
            if hasattr(module, 'check_dependencies'):
                available = list(self.get_all_modules().keys())
                dep_check = module.check_dependencies(available)
                results[name] = dep_check
        
        return results
    
    def shutdown_all(self):
        """Desliga todos os módulos graciosamente"""
        print("[GENESIS] Shutting down all modules...")
        
        for name, module in self.get_all_modules().items():
            if hasattr(module, 'shutdown'):
                try:
                    module.shutdown()
                    print(f"[GENESIS] ✅ {name} shutdown complete")
                except Exception as e:
                    print(f"[GENESIS] ❌ {name} shutdown failed: {e}")

        )
        
        results = []
        errors = []
        
        def worker(worker_id):
            try:
                for i in range(100):
                    vitals = module.get_vitals()
                    module.heartbeat()
                    time.sleep(0.001)
                results.append(worker_id)
            except Exception as e:
                errors.append(str(e))
        
        # Criar múltiplas threads
        threads = []
        for i in range(10):
            t = threading.Thread(target=worker, args=(i,))
            threads.append(t)
            t.start()
        
        # Aguardar todas
        for t in threads:
            t.join()
        
        # Verificar resultados
        self.assertEqual(len(errors), 0, f"Errors occurred: {errors}")
        self.assertEqual(len(results), 10, "Not all threads completed")
        
        # Verificar integridade após concorrência
        self.assertTrue(module.validate_integrity())

def generate_checksums():
    """Gera checksums de todos os arquivos importantes"""
    print("\n" + "="*70)
    print("GERANDO CHECKSUMS")
    print("="*70)
    
    import hashlib
    
    checksums = {}
    
    # Arquivos para checksum
    files_to_hash = [
        "Aurora/modules/ncnt_module_lite.py",
        "Aurora/modules/trading_core.py",
        "Aurora/governance/integration_gate.py",
        "Aurora/governance/genesis_includes.py",
        "Aurora/tests/test_ncnt_lite.py"
    ]
    
    for file_path in files_to_hash:
        if os.path.exists(file_path):
            with open(file_path, 'rb') as f:
                content = f.read()
                checksum = hashlib.sha3_256(content).hexdigest()
                checksums[file_path] = checksum
                print(f"📦 {os.path.basename(file_path)}: {checksum[:16]}...")
        else:
            print(f"⚠️  Arquivo não encontrado: {file_path}")
    
    # Salvar checksums
    checksum_file = "Aurora/config/checksums.json"
    os.makedirs(os.path.dirname(checksum_file), exist_ok=True)
    
    with open(checksum_file, 'w') as f:
        json.dump({
            "generated_at": datetime.now().isoformat(),
            "checksums": checksums
        }, f, indent=2)
    
    print(f"✅ Checksums salvos em: {checksum_file}")
    return True
