    async def _calculate_kelly_size_robust_v7(self, asset: str, zscore: Decimal,

        KELLY CRITERION v7.0 - INTEGRADO COM GESTÃO DE CAPITAL

        # Calcula tamanho em capital

        return max(position_capital, Decimal('100'))  # Mínimo €100

    def _zscore_to_win_rate(self, zscore: Decimal) -> Decimal:
        """Converte z-score para probabilidade de acerto"""
        z_abs = abs(zscore)
        if z_abs < Decimal('1.0'):
            return Decimal('0.55')
        elif z_abs < Decimal('2.0'):
            return Decimal('0.65')
        elif z_abs < Decimal('2.5'):
            return Decimal('0.72')
        else:
            return Decimal('0.68')  # Reversion em extremos
