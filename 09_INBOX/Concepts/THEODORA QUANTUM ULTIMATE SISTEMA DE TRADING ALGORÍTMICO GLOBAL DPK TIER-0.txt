    def create_genesis_block(self):
        genesis_block = {
            'index': 0,
            'timestamp': datetime.now().isoformat(),
            'data': 'Genesis Block - Quantum Trading System',
            'previous_hash': '0' * 64,
            'hash': self.calculate_hash(0, '0' * 64, 'Genesis Block')
        }
        self.blockchain.append(genesis_block)
    
    def calculate_hash(self, index, previous_hash, data):
        value = f"{index}{previous_hash}{data}".encode()
        return hashlib.sha256(value).hexdigest()
    
    def add_block(self, data):
        previous_block = self.blockchain[-1]
        index = previous_block['index'] + 1
        previous_hash = previous_block['hash']
        hash = self.calculate_hash(index, previous_hash, data)
        
        block = {
            'index': index,
            'timestamp': datetime.now().isoformat(),
            'data': data,
            'previous_hash': previous_hash,
            'hash': hash
        }
        self.blockchain.append(block)
        return block

class AdvancedQuantumMarketMechanics:
    def __init__(self):
        # Constantes fundamentais
        self.hbar = 1.054571817e-34  # Constante de Planck reduzida
        self.kB = 1.380649e-23       # Constante de Boltzmann
        self.c = 299792458           # Velocidade da luz
        
        # Estados quânticos do mercado
        self.market_states = {
            'bull': np.array([1, 0], dtype=complex),
            'bear': np.array([0, 1], dtype=complex),
            'superposition': np.array([1/np.sqrt(2), 1/np.sqrt(2)], dtype=complex)
        }
    
    @njit(parallel=True)
    def heisenberg_uncertainty_matrix(self, price_std: np.ndarray, volume_std: np.ndarray, time_std: np.ndarray) -> np.ndarray:
        """Matriz de incerteza de Heisenberg para múltiplas dimensões"""
        n = len(price_std)
        uncertainty_matrix = np.zeros((n, n))
        
        for i in numba.prange(n):
            for j in numba.prange(n):
                if i == j:
                    # Princípio da incerteza para a mesma dimensão
                    uncertainty_matrix[i, j] = self.hbar / (2 * price_std[i] * volume_std[i] + 1e-15)
                else:
                    # Correlações quânticas entre dimensões diferentes
                    uncertainty_matrix[i, j] = self.hbar / (4 * np.sqrt(price_std[i] * volume_std[j] * time_std[i] * time_std[j]) + 1e-15)
        
        return uncertainty_matrix
    
    @njit
    def quantum_entropy_density(self, returns: np.ndarray, window: int = 20) -> np.ndarray:
        """Entropia quântica de densidade com janela deslizante"""
        n = len(returns)
        entropy_values = np.zeros(n)
        
        for i in range(window, n):
            window_returns = returns[i-window:i]
            hist, _ = np.histogram(window_returns, bins=50, density=True)
            hist = hist[hist > 0]
            entropy_values[i] = -np.sum(hist * np.log(hist + 1e-15))
        
        return entropy_values
    
    @njit
    def navier_stokes_market_dynamics(self, prices: np.ndarray, volumes: np.ndarray, 
                                     viscosity: float = 0.01, density: float = 1.0) -> np.ndarray:
        """Equações de Navier-Stokes adaptadas para dinâmica de mercado"""
        n = len(prices)
        velocity = np.gradient(prices)
        acceleration = np.gradient(velocity)
        
        # Pressão do mercado (derivada segunda dos preços)
        pressure = np.gradient(np.gradient(prices))
        
        # Termo de viscosidade (atrito do mercado)
        viscosity_term = viscosity * np.gradient(np.gradient(velocity))
        
        # Termo de conveção
        convection_term = density * velocity * np.gradient(velocity)
        
        # Equação de Navier-Stokes simplificada para mercados
        market_force = convection_term - viscosity_term + pressure
        
        return market_force
    
    def quantum_wavefunction_collapse(self, probability_amplitude: complex, 
                                     observation_strength: float = 1.0) -> str:
        """Colapso da função de onda para tomada de decisão"""
        # Probabilidade de colapso para estado |0⟩ (bull)
        p_bull = min(1.0, max(0.0, (abs(probability_amplitude) ** 2) * observation_strength))
        
        # Decisão baseada no colapso da função de onda
        if np.random.random() < p_bull:
            return "BUY"
        else:
            return "SELL"
    
    def create_quantum_circuit(self, qubits: int = 4) -> cirq.Circuit:
        """Cria circuito quântico para análise de mercado"""
        # Qubits representando diferentes aspectos do mercado
        qreg = [cirq.GridQubit(0, i) for i in range(qubits)]
        circuit = cirq.Circuit()
        
        # Superposição inicial - mercado em todos os estados possíveis
        for i in range(qubits):
            circuit.append(cirq.H(qreg[i]))
        
        # Entrelaçamento quântico entre diferentes aspectos do mercado
        for i in range(qubits-1):
            circuit.append(cirq.CNOT(qreg[i], qreg[i+1]))
        
        # Medição final
        circuit.append(cirq.measure(*qreg, key='market_measurement'))
        
        return circuit

    @njit
    def _calculate_quantum_momentum(self, prices: np.ndarray, window: int = 10) -> float:
        """Calcula momentum com ajuste quântico"""
        if len(prices) < window + 1:
            return 0.0
        
        returns = np.diff(prices) / prices[:-1]
        recent_returns = returns[-window:] if len(returns) >= window else returns
        
        # Momentum quântico (média ponderada exponencialmente)
        weights = np.exp(np.linspace(0, 1, len(recent_returns)))
        weights /= np.sum(weights)
        
        return np.sum(recent_returns * weights)
    
    @njit
    def _calculate_quantum_volatility(self, highs: np.ndarray, lows: np.ndarray, 
                                     closes: np.ndarray, window: int = 24) -> float:
        """Calcula volatilidade com ajuste quântico"""
        if len(highs) < window or len(lows) < window:
            return 0.01
        
        # True Range quântico
        tr1 = highs[-window:] - lows[-window:]
        tr2 = np.abs(highs[-window:] - closes[-window:])
        tr3 = np.abs(lows[-window:] - closes[-window:])
        true_range = np.maximum(np.maximum(tr1, tr2), tr3)
        
        # Volatilidade quântica (entropia do true range)
        hist, _ = np.histogram(true_range, bins=10, density=True)
        hist = hist[hist > 0]
        volatility_entropy = -np.sum(hist * np.log(hist + 1e-10))
        
        return max(0.01, volatility_entropy)
    
    def calculate_position_size(self, symbol: str, price: float, stop_loss: float, 
                               account_balance: float, risk_pct: float) -> float:
        """Calcula tamanho da posição com gestão de risco quântica"""
        # Calcular risco por trade
        risk_amount = account_balance * risk_pct
        
        # Calcular pontos de risco
        points_risk = abs(price - stop_loss)
        
        # Obter informações do símbolo
        symbol_info = mt5.symbol_info(symbol)
        if symbol_info is None:
            return 0.01  # Tamanho mínimo fallback
        
        # Calcular valor do ponto
        point_value = 1.0  # Placeholder - implementar cálculo real
        
        # Calcular tamanho do lote
        lot_size = risk_amount / (points_risk * point_value)
        
        # Ajustar para tamanhos permitidos
        min_lot = symbol_info.volume_min
        max_lot = symbol_info.volume_max
        lot_step = symbol_info.volume_step
        
        lot_size = max(min_lot, min(max_lot, round(lot_size / lot_step) * lot_step))
        
        return lot_size
    
    def execute_quantum_trade(self, symbol: str, signal: str, price: float, 
                             atr: float, volatility: float, entropy: float, 
                             mission_id: str) -> List[Dict]:
        """Executa trade com escalonamento quântico"""
        results = []
        account_balance = mt5.account_info().balance
        
        # Obter dados recentes para cálculo de momentum
        rates = mt5.copy_rates_from_pos(symbol, mt5.TIMEFRAME_H1, 0, 50)
        if rates is None:
            print(f"Erro ao obter dados para {symbol}")
            return results
        
        closes = np.array([r['close'] for r in rates])
        highs = np.array([r['high'] for r in rates])
        lows = np.array([r['low'] for r in rates])
        
        # Calcular métricas quânticas
        quantum_momentum = self._calculate_quantum_momentum(closes)
        quantum_volatility = self._calculate_quantum_volatility(highs, lows, closes)
        
        # Fator de ajuste quântico
        quantum_adjustment = 1.0 + quantum_volatility * 2 + entropy * 0.5
        
        for i, level in enumerate(self.scaling_levels):
            # Calcular stop loss e take profit
            if signal == 'BUY':
                sl = price - (atr * self.sl_factor * quantum_adjustment)
                tp = price + (atr * self.tp_factor * level * quantum_adjustment)
            else:  # SELL
                sl = price + (atr * self.sl_factor * quantum_adjustment)
                tp = price - (atr * self.tp_factor * level * quantum_adjustment)
            
            # Calcular tamanho da posição
            position_size = self.calculate_position_size(
                symbol, price, sl, account_balance, self.risk_factor * level
            )
            
            # Ajustar pelo momentum quântico
            position_size *= (1 + abs(quantum_momentum) * 0.5)
            
            # Executar ordem
            request = {
                "action": mt5.TRADE_ACTION_DEAL,
                "symbol": symbol,
                "volume": position_size,
                "type": mt5.ORDER_TYPE_BUY if signal == 'BUY' else mt5.ORDER_TYPE_SELL,
                "price": price,
                "sl": sl,
                "tp": tp,
                "deviation": 20,

    def update_quantum_trailing_stops(self):
        """Atualiza trailing stops com mecânica quântica"""
        positions = mt5.positions_get()
        if positions is None:
            return
        
        for position in positions:
            symbol = position.symbol
            ticket = position.ticket
            position_type = position.type
            entry_price = position.price_open
            current_sl = position.sl
            current_tp = position.tp
            
            # Obter dados atuais
            symbol_info = mt5.symbol_info(symbol)
            if symbol_info is None:
                continue
            
            current_price = symbol_info.bid if position_type == mt5.ORDER_TYPE_BUY else symbol_info.ask
            
            # Calcular ATR recente
            rates = mt5.copy_rates_from_pos(symbol, mt5.TIMEFRAME_H1, 0, 20)
            if rates is None:
                continue
            
            highs = np.array([r['high'] for r in rates])
            lows = np.array([r['low'] for r in rates])
            closes = np.array([r['close'] for r in rates])
            
            # Calcular ATR
            tr = np.maximum(highs - lows, 
                           np.maximum(np.abs(highs - np.roll(closes, 1)), 
                                     np.abs(lows - np.roll(closes, 1))))
            atr = np.mean(tr[-14:]) if len(tr) >= 14 else np.mean(tr) if len(tr) > 0 else 0.0
            
            # Calcular entropia recente para ajuste quântico
            returns = np.diff(closes) / closes[:-1]
            if len(returns) > 10:
                hist, _ = np.histogram(returns[-10:], bins=5, density=True)
                hist = hist[hist > 0]
                recent_entropy = -np.sum(hist * np.log(hist + 1e-10))
            else:
                recent_entropy = 0.5
            
            # Calcular novo trailing stop
            adjustment = 1.0 + recent_entropy * 0.3
            point = symbol_info.point
            
            if position_type == mt5.ORDER_TYPE_BUY:
                new_sl = current_price - atr * self.trailing_factor * adjustment * point
                if new_sl > current_sl and new_sl > entry_price:
                    # Atualizar stop loss
                    request = {
                        "action": mt5.TRADE_ACTION_SLTP,
                        "position": ticket,
                        "sl": new_sl,
                        "tp": current_tp
                    }
                    mt5.order_send(request)
            else:  # SELL
                new_sl = current_price + atr * self.trailing_factor * adjustment * point
                if (current_sl == 0 or new_sl < current_sl) and new_sl < entry_price:
                    # Atualizar stop loss
                    request = {
                        "action": mt5.TRADE_ACTION_SLTP,
                        "position": ticket,
                        "sl": new_sl,
                        "tp": current_tp
                    }
                    mt5.order_send(request)

class QuantumRiskManager:
    def __init__(self, initial_capital: float = 100000.0, total_risk_pct: float = 1.0):
        self.initial_capital = initial_capital
        self.current_capital = initial_capital
        self.equity_high = initial_capital
        self.total_risk_pct = total_risk_pct
        
        # Histórico de trades com entrelaçamento quântico
        self.trade_history = []
        self.risk_metrics = {}
        
        # Limites de risco quânticos
        self.max_drawdown = 0.05  # 5% max drawdown
        self.var_confidence = 0.95  # 95% VaR confidence
        self.cvar_confidence = 0.99  # 99% CVaR confidence
        
        # Estados quânticos de risco
        self.risk_states = {
            'low': 0.2,
            'medium': 0.5,
            'high': 0.8,
            'extreme': 0.95
        }
    
    @njit
    def calculate_quantum_var(self, returns: np.ndarray, alpha: float = 0.95) -> float:
        """Value at Risk com ajuste quântico"""
        if len(returns) == 0:
            return 0.0
        
        # VaR tradicional
        traditional_var = np.percentile(returns, 100 * (1 - alpha))
        
        # Ajuste quântico baseado na entropia
        hist, _ = np.histogram(returns, bins=50, density=True)
        hist = hist[hist > 0]
        entropy = -np.sum(hist * np.log(hist + 1e-10))
        
        # VaR quântico (ajustado pela entropia)
        quantum_var = traditional_var * (1 + 0.2 * entropy)
        
        return quantum_var
    
    @njit
    def calculate_quantum_cvar(self, returns: np.ndarray, alpha: float = 0.99) -> float:
        """Conditional Value at Risk com ajuste quântico"""
        if len(returns) == 0:
            return 0.0
        
        # Calcular VaR primeiro
        var = np.percentile(returns, 100 * (1 - alpha))
        
        # CVaR tradicional (média das perdas beyond VaR)
        losses_beyond_var = returns[returns <= var]
        if len(losses_beyond_var) == 0:
            traditional_cvar = var
        else:
            traditional_cvar = np.mean(losses_beyond_var)
        
        # Ajuste quântico baseado na assimetria da cauda
        tail_skewness = stats.skew(returns) if len(returns) > 2 else 0.0
        quantum_cvar = traditional_cvar * (1 + 0.1 * tail_skewness)
        
        return quantum_cvar
    
    def update_risk_metrics(self, current_capital: float, new_trades: List[Dict] = None) -> Dict[str, Any]:
        """Atualiza métricas de risco com análise quântica"""
        self.current_capital = current_capital
        
        # Atualizar drawdown
        self.equity_high = max(self.equity_high, current_capital)
        drawdown = (self.equity_high - current_capital) / self.equity_high
        
        # Adicionar novos trades ao histórico
        if new_trades:
            self.trade_history.extend(new_trades)
        
        # Calcular returns dos trades
        trade_returns = []
        for trade in self.trade_history:
            if 'profit' in trade and 'capital_used' in trade:
                if trade['capital_used'] > 0:
                    trade_returns.append(trade['profit'] / trade['capital_used'])
        
        trade_returns = np.array(trade_returns) if trade_returns else np.array([0.0])
        
        # Calcular métricas de risco quânticas
        quantum_var = self.calculate_quantum_var(trade_returns, self.var_confidence)
        quantum_cvar = self.calculate_quantum_cvar(trade_returns, self.cvar_confidence)
        
        # Calcular Sharpe ratio quântico
        if len(trade_returns) > 1:
            sharpe_ratio = np.mean(trade_returns) / np.std(trade_returns) * np.sqrt(252)
        else:
            sharpe_ratio = 0.0
        
        # Determinar estado de risco quântico
        if drawdown > self.max_drawdown * 1.5:
            risk_state = 'extreme'
        elif drawdown > self.max_drawdown:
            risk_state = 'high'
        elif abs(quantum_cvar) > 0.05:
            risk_state = 'medium'
        else:
            risk_state = 'low'
        
        # Atualizar métricas
        self.risk_metrics = {
            'current_capital': current_capital,
            'equity_high': self.equity_high,
            'drawdown': drawdown,
            'quantum_var': quantum_var,
            'quantum_cvar': quantum_cvar,
            'sharpe_ratio': sharpe_ratio,
            'risk_state': risk_state,
            'risk_multiplier': self.risk_states[risk_state],
            'trade_count': len(self.trade_history),
            'win_rate': np.mean(np.array(trade_returns) > 0) if len(trade_returns) > 0 else 0.0
        }
        
        return self.risk_metrics
    
    def should_enter_trade(self, symbol: str, signal_strength: float, volatility: float, 
                          current_risk: Dict) -> Tuple[bool, float]:
        """Decide se deve entrar em um trade baseado no risco quântico"""
        risk_multiplier = current_risk['risk_multiplier']
        
        # Fatores de decisão quântica
        volatility_factor = max(0.1, min(1.0, 1.0 / (volatility * 10 + 1e-10)))
        signal_factor = min(1.0, abs(signal_strength))
        
        # Probabilidade quântica de entrada
        entry_probability = signal_factor * volatility_factor * risk_multiplier
        
        # Colapso da função de onda para decisão binária
        should_enter = np.random.random() < entry_probability
        
        return should_enter, entry_probability
    
    def get_position_sizing(self, current_risk: Dict, volatility: float) -> float:
        """Calcula tamanho da posição baseado no risco quântico"""
        base_size = self.total_risk_pct
        
        # Ajustar pelo estado de risco
        risk_adjusted = base_size * current_risk['risk_multiplier']
        
        # Ajustar pela volatilidade
        volatility_adjusted = risk_adjusted * min(1.0, 0.1 / (volatility + 1e-10))
        
        return max(0.01, min(0.1, volatility_adjusted))

    def _calculate_atr(self, symbol: str, period: int = 14) -> float:
        """Calcula Average True Range"""
        rates = mt5.copy_rates_from_pos(symbol, mt5.TIMEFRAME_H1, 0, period * 2)
        if rates is None or len(rates) < period:
            return 0.005  # Fallback
        
        highs = np.array([r['high'] for r in rates])
        lows = np.array([r['low'] for r in rates])
        closes = np.array([r['close'] for r in rates])
        
        # Calcular True Range
        tr1 = highs[1:] - lows[1:]
        tr2 = np.abs(highs[1:] - closes[:-1])
        tr3 = np.abs(lows[1:] - closes[:-1])
        true_range = np.maximum(np.maximum(tr1, tr2), tr3)
        
        # Calcular ATR
        atr = np.mean(true_range[-period:]) if len(true_range) >= period else np.mean(true_range)
        
        return atr

        self.quantum_logger.log('INFO', 'Iniciando modo de trading quântico contínuo')
        
        cycle_count = 0
        max_cycles = 1000  # Prevenir loop infinito
        
        while self.mission_status['active'] and cycle_count < max_cycles:
            try:
                # Executar ciclo de trading
                await self.run_quantum_trading_cycle()
                
                # Esperar entre ciclos (ajustável baseado na volatilidade)
                await asyncio.sleep(300)  # 5 minutos entre ciclos
                
                cycle_count += 1
                
            except Exception as e:
                self.quantum_logger.log('ERROR', 'Erro no trading contínuo', {'error': str(e)})
                await asyncio.sleep(60)  # Esperar 1 minuto antes de continuar
        
        self.quantum_logger.log('INFO', 'Modo de trading quântico contínuo finalizado', 
                               {'cycles_completed': cycle_count})
    
    def get_performance_report(self) -> Dict[str, Any]:
        """Gera relatório de performance quântica"""
        risk_metrics = self.risk_manager.risk_metrics
        
        return {
            'mission_status': self.mission_status,
            'risk_metrics': risk_metrics,
            'quantum_state': self.quantum_state,
            'market_memory_entropy': {sym: data.get('entropy', 0.5) for sym, data in self.market_memory.items()},
            'execution_stats': {
                'total_trades': len(self.risk_manager.trade_history),
                'active_positions': len(mt5.positions_get() or []),
                'quantum_cache_size': len(self.data_oracle.quantum_cache)
            },
            'timestamp': datetime.now().isoformat(),
            'quantum_hash': hashlib.sha256(str(self.quantum_state).encode()).hexdigest()[:16]
        }

        }

@app.get("/performance")
async def get_performance():
    """Retorna relatório de performance completo"""
    if theodora_system:
        return theodora_system.get_performance_report()
    return {"error": "System not initialized"}

@app.post("/control/{command}")
async def control_system(command: str, background_tasks: BackgroundTasks):
    """Controla o sistema Theodora"""
    if not theodora_system:
        return {"error": "System not initialized"}
    
    if command == "pause":
        theodora_system.mission_status['active'] = False
        return {"status": "paused"}
    
    elif command == "resume":
        theodora_system.mission_status['active'] = True
        background_tasks.add_task(theodora_system.run_continuous_quantum_trading)
        return {"status": "resumed"}
    
    elif command == "shutdown":
        theodora_system.execution_engine.disconnect_mt5()
        return {"status": "shutdown"}
    
    return {"error": "Unknown command"}

@app.get("/quantum/state")
async def get_quantum_state():
    """Retorna o estado quântico atual do sistema"""
    if theodora_system:
        return {
            "quantum_state": theodora_system.quantum_state,
            "entanglement": theodora_system.config['quantum']['entanglement'],
            "superposition": theodora_system.config['quantum']['superposition']
        }
    return {"error": "System not initialized"}
