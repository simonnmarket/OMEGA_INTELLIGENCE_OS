class AutonomousAssetAgent:
    """Agente especializado por ativo (AASA Architecture)"""
    
    def __init__(self, asset: str, config: Dict[str, Any]):
        self.asset = asset.upper()
        self.name = f"Agent_{self.asset}"
        self.config = config
        self.position = 0.0
        self.performance_history = []
        self.learning_enabled = True
        self.is_active = True
        self.last_heartbeat = datetime.now(timezone.utc)
        
        # Configurações específicas
        self.execution_threshold = config.get("execution_threshold", 0.7)
        self.max_position = config.get("max_position", 100000.0)
        self.order_size = config.get("order_size", 1.0)
        self.cycle_interval = config.get("cycle_interval_sec", 30)
        
        # Circuit breaker
        self.failure_count = 0
        self.circuit_tripped = False
        self.circuit_trip_time = None
        
        logger.info(f"🤖 Agente criado: {self.name}", config=self.config)
    
    async def analyze(self, market_data: MarketData) -> AgentVote:
        """Analisar dados de mercado e retornar voto"""
        try:
            self.last_heartbeat = datetime.now(timezone.utc)
            
            # Verificar circuit breaker
            if self.circuit_tripped:
                if self._should_reset_circuit():
                    self._reset_circuit()
                else:
                    return AgentVote(
                        agent_name=self.name,
                        decision="NO_VOTE",
                        confidence=0.0,
                        reason="Circuit breaker ativo",
                        technical_error=True
                    )
            
            # Análise específica do ativo
            analysis = await self._perform_analysis(market_data)
            
            # Aplicar regras de trading
            decision = self._apply_trading_rules(analysis, market_data)
            
            # Registrar performance
            self._record_analysis(analysis, decision)
            
            return decision
            
        except Exception as e:
            logger.error(f"❌ {self.name} análise falhou", error=str(e))
            self.failure_count += 1
            
            if self.failure_count >= 3:
                self._trip_circuit()
            
            return AgentVote(
                agent_name=self.name,
                decision="NO_VOTE",
                confidence=0.0,
                reason=f"Erro de análise: {str(e)}",
                technical_error=True
            )
    
    async def _perform_analysis(self, market_data: MarketData) -> Dict[str, Any]:
        """Realizar análise específica do ativo"""
        # Base analysis - sobrescrever em classes especializadas
        analysis = {
            "trend": market_data.ma_signal,
            "strength": abs(market_data.ma5 - market_data.ma20) if market_data.ma5 and market_data.ma20 else 0.0,
            "volatility": market_data.volatility_24h,
            "spread_ok": market_data.spread < 10.0,
            "volume_ok": market_data.volume > 1000
        }
        
        # Adicionar análise específica por ativo
        if "XAU" in self.asset or "GOLD" in self.asset:
            analysis["safe_haven"] = True
            analysis["inflation_hedge"] = True
        elif "BTC" in self.asset or "ETH" in self.asset:
            analysis["crypto_volatility"] = market_data.volatility_24h > 2.0
            analysis["market_cap_trend"] = "UP"  # Placeholder
        
        return analysis
    
    def _apply_trading_rules(self, analysis: Dict[str, Any], market_data: MarketData) -> AgentVote:
        """Aplicar regras de trading para gerar decisão"""
        
        # REGRA 1: Spread alto = NO_VOTE
        if not analysis.get("spread_ok", True):
            return AgentVote(
                agent_name=self.name,
                decision="NO_VOTE",
                confidence=0.3,
                reason=f"Spread alto: {market_data.spread:.1f} pips"
            )
        
        # REGRA 2: Volume baixo = HOLD
        if not analysis.get("volume_ok", True):
            return AgentVote(
                agent_name=self.name,
                decision="HOLD",
                confidence=0.5,
                reason="Volume insuficiente"
            )
        
        # REGRA 3: Sinal técnico claro = BUY/SELL decisivo
        trend = analysis.get("trend", "NEUTRAL")
        strength = analysis.get("strength", 0.0)
        
        if trend == "BULLISH" and strength > 0.0005:
            confidence = min(0.8 + (strength * 100), 0.95)
            return AgentVote(
                agent_name=self.name,
                decision="BUY",
                confidence=confidence,
                reason=f"Trend bullish forte (força: {strength:.4f})",
                weight=self.config.get("weight", 1.0)
            )
        
        elif trend == "BEARISH" and strength > 0.0005:
            confidence = min(0.8 + (strength * 100), 0.95)
            return AgentVote(
                agent_name=self.name,
                decision="SELL",
                confidence=confidence,
                reason=f"Trend bearish forte (força: {strength:.4f})",
                weight=self.config.get("weight", 1.0)
            )
        
        # REGRA 4: Sem sinal claro = HOLD
        return AgentVote(
            agent_name=self.name,
            decision="HOLD",
            confidence=0.6,
            reason="Mercado lateral/sem sinal claro",
            weight=self.config.get("weight", 1.0)
        )
    
    def _trip_circuit(self):
        """Ativar circuit breaker"""
        self.circuit_tripped = True
        self.circuit_trip_time = datetime.now(timezone.utc)
        logger.warning(f"🔌 Circuit breaker ativado para {self.name}")
    
    def _should_reset_circuit(self) -> bool:
        """Verificar se circuit breaker deve ser resetado"""
        if not self.circuit_tripped or not self.circuit_trip_time:
            return True
        
        recovery_time = self.config.get("circuit_recovery_sec", 300)
        time_elapsed = (datetime.now(timezone.utc) - self.circuit_trip_time).total_seconds()
        
        return time_elapsed >= recovery_time
    
    def _reset_circuit(self):
        """Resetar circuit breaker"""
        self.circuit_tripped = False
        self.circuit_trip_time = None
        self.failure_count = 0
        logger.info(f"🔄 Circuit breaker resetado para {self.name}")
    
    def _record_analysis(self, analysis: Dict[str, Any], vote: AgentVote):
        """Registrar análise para aprendizado futuro"""
        record = {
            "timestamp": datetime.now(timezone.utc),
            "analysis": analysis,
            "vote": vote.model_dump(),
            "asset": self.asset
        }
        self.performance_history.append(record)
        
        # Manter histórico limitado
        if len(self.performance_history) > 1000:
            self.performance_history = self.performance_history[-1000:]
    
    def get_performance_metrics(self) -> Dict[str, Any]:
        """Obter métricas de performance do agente"""
        if not self.performance_history:
            return {"accuracy": 0.5, "total_analyses": 0}
        
        # Calcular métricas básicas
        decisions = [record["vote"]["decision"] for record in self.performance_history]
        buy_count = decisions.count("BUY")
        sell_count = decisions.count("SELL")
        hold_count = decisions.count("HOLD")
        total = len(decisions)
        
        return {
            "total_analyses": total,
            "buy_percentage": buy_count / total if total > 0 else 0,
            "sell_percentage": sell_count / total if total > 0 else 0,
            "hold_percentage": hold_count / total if total > 0 else 0,
            "avg_confidence": np.mean([record["vote"]["confidence"] for record in self.performance_history]),
            "active": self.is_active,
            "failures": self.failure_count,
            "circuit_tripped": self.circuit_tripped
        }

        vote = super()._apply_trading_rules(analysis, market_data)
        
        # Aumentar confiança se múltiplos fatores positivos

            analysis.get("safe_haven_demand", False),
            analysis.get("inflation_hedge_strength", 0) > 0.5,

            vote.confidence = min(vote.confidence * 1.2, 0.95)
            vote.reason += " | Múltiplos fatores macro positivos"
        
        return vote
