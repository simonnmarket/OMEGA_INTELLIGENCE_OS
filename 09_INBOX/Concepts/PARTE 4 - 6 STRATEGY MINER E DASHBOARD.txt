class StrategyMiner:
    """Minerador de estratégias do GitHub com RAG"""
    
    def __init__(self, github_token: Optional[str] = None):
        self.github = Github(github_token) if github_token else Github()
        self.model = SentenceTransformer("all-MiniLM-L6-v2")
        self.strategies_db = []
        self.vector_index = None
        self.db_path = Path("data/strategies")
        self.db_path.mkdir(parents=True, exist_ok=True)
        
        # Carregar DB existente
        self._load_db()
        
        logger.info("⛏️ Strategy Miner inicializado")
    
    async def mine_strategies(self, query: str = "trading strategy filetype:yaml"):
        """Minera estratégias do GitHub"""
        logger.info(f"Iniciando mineração: {query}")
        
        try:
            # Buscar no GitHub
            results = self.github.search_code(
                query=query,
                sort="stars",
                order="desc"
            )
            
            new_strategies = []
            count = 0
            
            for code_file in results[:20]:  # Limitar a 20 resultados
                try:
                    if count >= 10:  # Coletar 10 estratégias
                        break
                    
                    strategy = await self._parse_strategy_file(code_file)
                    if strategy:
                        new_strategies.append(strategy)
                        count += 1
                        
                        logger.debug(f"Estratégia minerada: {strategy['title'][:50]}...")
                        
                except Exception as e:
                    logger.debug(f"Erro ao processar arquivo: {e}")
                    continue
            
            # Adicionar ao banco de dados
            for strategy in new_strategies:
                if not any(s["id"] == strategy["id"] for s in self.strategies_db):
                    self.strategies_db.append(strategy)
                    logger.info(f"✅ Nova estratégia: {strategy['title']}")
            
            # Atualizar índice vetorial
            await self._update_vector_index()
            
            # Salvar banco de dados
            await self._save_db()
            
            logger.info(f"Mineração completada: {len(new_strategies)} novas estratégias")
            return new_strategies
            
        except Exception as e:
            logger.error(f"❌ Mineração falhou: {e}")
            return []
    
    async def _parse_strategy_file(self, code_file) -> Optional[Dict]:
        """Analisar arquivo de estratégia"""
        try:
            # Obter conteúdo bruto
            raw_url = code_file.download_url
            response = requests.get(raw_url, timeout=10)
            response.raise_for_status()
            content = response.text
            
            # Determinar formato
            if code_file.name.endswith(('.yaml', '.yml')):
                data = yaml.safe_load(content)
            elif code_file.name.endswith('.json'):
                data = json.loads(content)
            else:
                # Tentar extrair de Markdown ou código
                data = self._extract_from_text(content)
            
            if not data:
                return None
            
            # Extrair informações
            title = self._extract_title(data, code_file.name)
            description = self._extract_description(data, content)
            symbols = self._extract_symbols(content, data)
            asset_classes = self._infer_asset_classes(symbols)
            backtest_stats = self._extract_backtest_stats(data, content)
            
            # Criar embedding
            text_for_embedding = f"{title} {description} {' '.join(symbols)}"
            embedding = self.model.encode(text_for_embedding).tolist()
            
            strategy = {
                "id": f"{code_file.repository.full_name}/{code_file.path}",
                "title": title,
                "description": description,
                "symbols": symbols,
                "asset_classes": asset_classes,
                "timeframe": data.get("timeframe", "H1"),
                "logic_summary": self._summarize_logic(data),
                "backtest": backtest_stats,
                "source_url": f"https://github.com/{code_file.repository.full_name}/blob/master/{code_file.path}",
                "stars": code_file.repository.stargazers_count,
                "updated_at": code_file.repository.updated_at.isoformat() if code_file.repository.updated_at else datetime.now(timezone.utc).isoformat(),
                "embedding": embedding,
                "mined_at": datetime.now(timezone.utc).isoformat()
            }
            
            return strategy
            
        except Exception as e:
            logger.debug(f"Parse falhou: {e}")
            return None
    
    def _extract_title(self, data: Dict, filename: str) -> str:
        """Extrair título da estratégia"""
        candidates = [
            data.get("strategy", {}).get("name"),
            data.get("name"),
            data.get("title"),
            filename.replace(".yaml", "").replace(".json", "").replace("_", " ").title()
        ]
        
        for candidate in candidates:
            if candidate and isinstance(candidate, str) and len(candidate) > 3:
                return candidate[:100]
        
        return f"Strategy {filename}"
    
    def _extract_description(self, data: Dict, content: str) -> str:
        """Extrair descrição"""
        # Tentar do YAML/JSON primeiro
        if isinstance(data, dict):
            desc = data.get("description") or data.get("desc") or data.get("about", "")
            if desc and isinstance(desc, str) and len(desc) > 10:
                return desc[:200]
        
        # Extrair do Markdown
        import re
        md_desc = re.search(r'#+\s*[Dd]escription\s*\n+(.*?)(\n#+|\n---|\Z)', content, re.DOTALL)
        if md_desc:
            return md_desc.group(1).strip()[:200]
        
        # Primeiro parágrafo
        lines = [l.strip() for l in content.split('\n') if l.strip() and not l.strip().startswith('#')]
        if lines:
            return lines[0][:200]
        
        return "No description available"
    
    def _extract_symbols(self, content: str, data: Dict) -> List[str]:
        """Extrair símbolos mencionados"""
        symbols = set()
        content_upper = content.upper()
        
        # Pares forex comuns
        forex_pairs = ["EURUSD", "GBPUSD", "USDJPY", "USDCHF", "AUDUSD", "USDCAD", "NZDUSD"]
        for pair in forex_pairs:
            if pair in content_upper:
                symbols.add(pair)
        
        # Metais
        if "XAU" in content_upper or "GOLD" in content_upper:
            symbols.add("XAUUSD")
        if "XAG" in content_upper or "SILVER" in content_upper:
            symbols.add("XAGUSD")
        
        # Crypto
        if "BTC" in content_upper:
            symbols.add("BTCUSD")
        if "ETH" in content_upper:
            symbols.add("ETHUSD")
        
        # Índices
        indices = ["US500", "US30", "NAS100", "UK100"]
        for idx in indices:
            if idx in content_upper:
                symbols.add(idx)
        
        # Verificar em data keys
        for key in ["symbols", "pairs", "assets", "instrument"]:
            if key in data:
                value = data[key]
                if isinstance(value, list):
                    for item in value:
                        if isinstance(item, str) and len(item) in (6, 7):
                            symbols.add(item.upper())
                elif isinstance(value, str):
                    symbols.add(value.upper())
        
        return sorted(list(symbols))
    
    def _infer_asset_classes(self, symbols: List[str]) -> List[str]:
        """Inferir classes de ativo dos símbolos"""
        classes = set()
        
        for symbol in symbols:
            if "XAU" in symbol or "XAG" in symbol:
                classes.add("metals")
            elif any(crypto in symbol for crypto in ["BTC", "ETH", "XRP"]):
                classes.add("crypto")
            elif any(idx in symbol for idx in ["US500", "US30", "NAS100", "UK100"]):
                classes.add("indices")
            elif len(symbol) == 6 and symbol.isalpha():
                classes.add("forex")
        
        return sorted(list(classes)) or ["multi"]
    
    def _extract_backtest_stats(self, data: Dict, content: str) -> Dict:
        """Extrair estatísticas de backtest"""
        stats = {}
        
        # Procurar em data
        if isinstance(data, dict):
            for key, value in data.items():
                if isinstance(value, dict) and any(k in str(value).lower() for k in ["sharpe", "profit", "drawdown", "win"]):
                    stats = value
                    break
        
        # Procurar em texto
        import re
        
        sharpe_match = re.search(r'sharpe[^0-9]*([0-9]*\.?[0-9]+)', content, re.IGNORECASE)
        if sharpe_match:
            stats["sharpe"] = float(sharpe_match.group(1))
        
        winrate_match = re.search(r'win.*rate[^0-9]*([0-9]*\.?[0-9]+)%', content, re.IGNORECASE)
        if winrate_match:
            stats["win_rate"] = float(winrate_match.group(1))
        
        dd_match = re.search(r'drawdown[^0-9]*([0-9]*\.?[0-9]+)%', content, re.IGNORECASE)
        if dd_match:
            stats["max_drawdown"] = f"{dd_match.group(1)}%"
        
        return stats
    
    def _summarize_logic(self, data: Dict) -> str:
        """Resumir lógica da estratégia"""
        logic_keys = ["logic", "rules", "conditions", "strategy"]
        
        for key in logic_keys:
            if key in data:
                logic = data[key]
                if isinstance(logic, str):
                    return logic[:150]
                elif isinstance(logic, dict):
                    return json.dumps(logic)[:150]
        
        return "Logic not specified"
    
    def _extract_from_text(self, content: str) -> Dict:
        """Extrair dados de texto não estruturado"""
        data = {}
        
        # Tentar encontrar blocos YAML/JSON
        import re
        
        yaml_match = re.search(r'```(?:yaml|yml)\n(.*?)\n```', content, re.DOTALL)
        if yaml_match:
            try:
                data = yaml.safe_load(yaml_match.group(1))
                return data
            except:
                pass
        
        json_match = re.search(r'```json\n(.*?)\n```', content, re.DOTALL)
        if json_match:
            try:
                data = json.loads(json_match.group(1))
                return data
            except:
                pass
        
        # Extrair pares chave-valor simples
        lines = content.split('\n')
        for line in lines:
            if ':' in line:
                parts = line.split(':', 1)
                if len(parts) == 2:
                    key = parts[0].strip()
                    value = parts[1].strip()
                    if key and value:
                        data[key] = value
        
        return data
    
    async def search_strategies(self, query: str, top_k: int = 5) -> List[Dict]:
        """Buscar estratégias semanticamente similares"""
        if not self.strategies_db or not self.vector_index:
            return []
        
        # Gerar embedding da query
        query_embedding = self.model.encode(query).reshape(1, -1)
        
        # Buscar no índice
        distances, indices = self.vector_index.search(query_embedding.astype("float32"), top_k)
        
        # Retornar estratégias
        results = []
        for idx, distance in zip(indices[0], distances[0]):
            if idx < len(self.strategies_db):
                strategy = self.strategies_db[idx].copy()
                strategy["similarity_score"] = 1.0 / (1.0 + distance)  # Converter distância para similaridade
                results.append(strategy)
        
        return results
    
    async def get_strategies_for_symbol(self, symbol: str) -> List[Dict]:
        """Obter estratégias relevantes para um símbolo"""
        relevant = []
        
        for strategy in self.strategies_db:
            if symbol in strategy["symbols"]:
                relevant.append(strategy)
        
        # Ordenar por qualidade (stars + backtest)
        relevant.sort(key=lambda x: (
            x.get("stars", 0),
            float(x.get("backtest", {}).get("sharpe", 0))
        ), reverse=True)
        
        return relevant[:10]  # Top 10
    
    async def _update_vector_index(self):
        """Atualizar índice vetorial FAISS"""
        if not self.strategies_db:
            return
        
        embeddings = []
        valid_strategies = []
        
        for strategy in self.strategies_db:
            if "embedding" in strategy and strategy["embedding"]:
                embeddings.append(strategy["embedding"])
                valid_strategies.append(strategy)
        
        if not embeddings:
            return
        
        embeddings_array = np.array(embeddings).astype("float32")
        
        # Criar índice
        dimension = embeddings_array.shape[1]
        self.vector_index = faiss.IndexFlatL2(dimension)
        self.vector_index.add(embeddings_array)
        
        logger.debug(f"Índice vetorial atualizado: {len(embeddings)} estratégias")
    
    def _load_db(self):
        """Carregar banco de dados salvo"""
        db_file = self.db_path / "strategies.json"
        
        if db_file.exists():
            try:
                with open(db_file, 'r') as f:
                    data = json.load(f)
                
                self.strategies_db = data
                logger.info(f"✅ DB carregado: {len(self.strategies_db)} estratégias")
                
                # Reconstruir índice
                asyncio.create_task(self._update_vector_index())
                
            except Exception as e:
                logger.error(f"❌ Erro ao carregar DB: {e}")
                self.strategies_db = []
    
    async def _save_db(self):
        """Salvar banco de dados"""
        db_file = self.db_path / "strategies.json"
        
        # Remover embeddings para economizar espaço
        strategies_to_save = []
        for strategy in self.strategies_db:
            s_copy = strategy.copy()
            if "embedding" in s_copy:
                del s_copy["embedding"]
            strategies_to_save.append(s_copy)
        
        try:
            with open(db_file, 'w') as f:
                json.dump(strategies_to_save, f, indent=2, default=str)
            
            logger.info(f"💾 DB salvo: {len(strategies_to_save)} estratégias")
            
        except Exception as e:
            logger.error(f"❌ Erro ao salvar DB: {e}")
