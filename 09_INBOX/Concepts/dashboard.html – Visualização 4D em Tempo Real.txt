<head>
    <title>NumeiaEA Quantum Grid Network</title>
    <script src="https://d3js.org/d3.v7.min.js"></script>
    <style>
        body { font-family: 'Consolas', monospace; margin: 0; background: #121212; color: #e0e0e0; }
        .node { stroke: #fff; stroke-width: 1.5px; cursor: pointer; }
        .link { stroke: #999; stroke-opacity: 0.6; }
        .error { stroke: #f44336; stroke-width: 2.5px; }
        .warning { stroke: #ff9800; stroke-width: 2px; }
        .success { stroke: #4caf50; stroke-width: 1.5px; }
        .tooltip { position: absolute; background: #1e1e1e; color: #fff; padding: 8px; 
                  border-radius: 4px; box-shadow: 0 0 10px rgba(0,0,0,0.5); pointer-events: none; }
        #controls { position: absolute; top: 10px; left: 10px; z-index: 100; background: #1e1e1e; 
                  padding: 10px; border-radius: 5px; }
        #status { position: absolute; top: 10px; right: 10px; z-index: 100; font-weight: bold; }
        .legend { position: absolute; bottom: 10px; left: 10px; z-index: 100; display: flex; gap: 15px; }
        .legend-item { display: flex; align-items: center; font-size: 12px; }
        .legend-color { width: 12px; height: 12px; margin-right: 5px; }
    </style>
</head>
<body>
    <div id="controls">
        <button onclick="refreshData()">Atualizar</button>
        <button onclick="autoRefresh()">Auto Atualizar (5s)</button>
    </div>
    <div id="status">Status: Pronto</div>
    <div class="legend">
        <div class="legend-item"><div class="legend-color" style="background: #f44336;"></div> Erro</div>
        <div class="legend-item"><div class="legend-color" style="background: #ff9800;"></div> Aviso</div>
        <div class="legend-item"><div class="legend-color" style="background: #4caf50;"></div> Conectado</div>
        <div class="legend-item"><div class="legend-color" style="background: #2196f3;"></div> CORE</div>
        <div class="legend-item"><div class="legend-color" style="background: #9c27b0;"></div> SECURITY</div>
    </div>
    <script>
        const width = window.innerWidth;
        const height = window.innerHeight;
        const svg = d3.select("body").append("svg")
            .attr("width", width)
            .attr("height", height);

        const tooltip = d3.select("body").append("div")
            .attr("class", "tooltip")
            .style("opacity", 0);

        const simulation = d3.forceSimulation()
            .force("link", d3.forceLink().id(d => d.id).distance(150))
            .force("charge", d3.forceManyBody().strength(-1000))
            .force("center", d3.forceCenter(width / 2, height / 2))
            .force("collision", d3.forceCollide().radius(30));

        let autoInterval;

        function updateGraph(data) {
            svg.selectAll("*").remove();

            const link = svg.append("g")
                .selectAll("line")
                .data(data.links)
                .enter().append("line")
                .attr("class", d => `link ${d.status.toLowerCase()}`)
                .attr("stroke-width", 2);

            const node = svg.append("g")
                .selectAll("circle")
                .data(data.nodes)
                .enter().append("g");

            node.append("circle")
                .attr("r", 10)
                .attr("class", "node")
                .attr("fill", d => {
                    const type = d.type;
                    if (type === "CORE") return "#2196f3";
                    if (type === "SECURITY") return "#9c27b0";
                    if (type === "UTILS") return "#009688";
                    if (type === "INTELLIGENCE") return "#ff5722";
                    return "#607d8b";
                })
                .on("mouseover", function(event, d) {
                    tooltip.transition().duration(200).style("opacity", .9);
                    tooltip.html(`Arquivo: ${d.id}<br>Tipo: ${d.type}`)
                        .style("left", (event.pageX + 10) + "px")
                        .style("top", (event.pageY - 20) + "px");
                })
                .on("mouseout", function() {
                    tooltip.transition().duration(500).style("opacity", 0);
                });

            node.append("text")
                .attr("dx", 12)
                .attr("dy", 4)
                .text(d => d.id)
                .attr("font-size", "10px")
                .attr("fill", "#e0e0e0");

            simulation
                .nodes(data.nodes)
                .on("tick", () => {
                    link.attr("x1", d => d.source.x)
                        .attr("y1", d => d.source.y)
                        .attr("x2", d => d.target.x)
                        .attr("y2", d => d.target.y);

                    node.attr("transform", d => `translate(${d.x},${d.y})`);
                });

            simulation.force("link").links(data.links);
            simulation.alpha(1).restart();
        }

        function refreshData() {
            d3.json("dependencies.json").then(data => {
                updateGraph(data);
                document.getElementById("status").textContent = `Atualizado: ${new Date().toLocaleTimeString()}`;
            }).catch(err => {
                document.getElementById("status").textContent = "Erro ao carregar dados";
            });
        }

        function autoRefresh() {
            if (autoInterval) {
                clearInterval(autoInterval);
                autoInterval = null;
                document.querySelector("#controls button:nth-child(2)").textContent = "Auto Atualizar (5s)";
            } else {
                autoInterval = setInterval(refreshData, 5000);
                document.querySelector("#controls button:nth-child(2)").textContent = "Parar Auto";
            }
        }

        // Carrega ao iniciar
        refreshData();
    </script>
</body>
