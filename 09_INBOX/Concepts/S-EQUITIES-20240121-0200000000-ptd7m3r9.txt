        # Atualizar histórico

        # Manter apenas window_size mais recentes

        if len(self.price_ratios_history[pair]) > window_size:

        if len(self.price_ratios_history[pair]) < 30:  # Mínimo para cálculo

        mean_ratio = sum(ratios) / Decimal(str(len(ratios)))
