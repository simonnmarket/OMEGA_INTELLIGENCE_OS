    def _calculate_macro_index(self, macro_data: Dict) -> MacroIndex:
        """Calcula o índice de força macro para o ouro."""
        # Dados simulados, na prática viriam de APIs econômicas
        usd_strength = macro_data.get('dxy', np.random.normal(0, 0.01))
        real_rates = macro_data.get('real_rates', np.random.normal(-0.01, 0.005))
        inflation_exp = macro_data.get('inflation', np.random.normal(0.025, 0.005))
        geo_risk = macro_data.get('geo_risk', np.random.beta(2, 5))
        
        # Pesos de Williams
        composite_score = (usd_strength * -0.4) + (real_rates * -0.6) + (inflation_exp * 0.3) + (geo_risk * 0.2)
        
        return MacroIndex(
            usd_strength=usd_strength,
            real_rates=real_rates,
            inflation_exp=inflation_exp,
            geo_risk=geo_risk,
            composite_score=composite_score
        )

    def _calculate_quantum_state(self, prices: List[float], macro_index: MacroIndex) -> QuantumState:
        """Modela o ouro como um sistema de dois níveis de energia."""
        # A volatilidade é a energia do sistema
        returns = np.diff(np.log(prices[-50:]))
        energy = np.std(returns)
        
        # O Índice Macro é a barreira de potencial
        potential_barrier = abs(macro_index.composite_score)
        
        # Probabilidade de tunelamento (fórmula de WKB simplificada)
        # P ~ exp(-2 * integral(sqrt(2m(V-E)))dx)
        # Simplificado: P ~ exp(-k * barrier / energy)
        if energy == 0: return QuantumState(0.9, 0.1, 0.0)
        
        tunneling_prob = np.exp(-5 * potential_barrier / energy)
        
        ground_state_prob = 1 - tunneling_prob
        
        return QuantumState(
            ground_state_prob=ground_state_prob,
            excited_state_prob=tunneling_prob,
            tunneling_prob=tunneling_prob
        )
        
    def _predict_fourier_inflection(self, prices: List[float]) -> int:
        """Usa Fourier para prever o próximo ponto de inflexão cíclica."""
        if len(prices) < 128: return len(prices) + 30 # Default
        
        # Remove a tendência
        detrended = np.diff(prices)
        
        # Análise de Fourier
        yf = fft(detrended)
        xf = fftfreq(len(detrended))
        
        # Encontra as frequências com maior potência
        power_spectrum = np.abs(yf)
        main_freq_idx = np.argmax(power_spectrum[1:len(power_spectrum)//2]) + 1
        main_freq = xf[main_freq_idx]
        
        if main_freq == 0: return len(prices) + 30
        
        # Calcula o período do ciclo dominante
        period_days = int(1 / abs(main_freq))
        
        # Prevê o próximo ponto de inflexão (meio ciclo à frente)
        next_inflection = len(prices) + (period_days // 2)
        
        return next_inflection

    def _get_dynamic_factor_weights(self, prices: List[float]) -> List[float]:
        """Simula a extração de pesos de fatores latentes (Prof. Rossi)."""
        # Na prática, aqui rodaríamos PCA nos retornos de 3 sub-estratégias
        # e usaríamos um Filtro de Kalman para estimar os pesos dinâmicos.
        # Para este exemplo, retornamos pesos fixos como placeholder.
        return [0.4, 0.3, 0.3] # [Reversão, Momentum, Fuga]
