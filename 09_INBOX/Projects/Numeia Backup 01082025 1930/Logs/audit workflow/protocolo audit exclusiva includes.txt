   - Dois ou mais arquivos se incluem mutuamente, causando conflitos.

bool AuditIncludes(string file_path) {
   string content = read_file(file_path);
   string includes[];
   int include_count = ExtractIncludes(content, includes);
   
   for(int i = 0; i < include_count; i++) {
      string include_path = NormalizePath(includes[i]);
      string expected_path = MapToOfficialStructure(include_path);
      
      if(!IsValidPath(expected_path)) {
         m_logger.log_error("Caminho inválido: " + include_path + " em " + file_path);
         return false;
      }
      
      if(!FileIsExist(expected_path)) {
         m_logger.log_error("Arquivo ausente: " + expected_path + " em " + file_path);
         m_logger.log_warning("Solicitar aprovação TIER-0 para verificar/criar: " + expected_path);
         return false;
      }
   }
   m_logger.log_info("Auditoria de includes concluída para: " + file_path);
   return true;
}

string NormalizePath(string path) {
   StringReplace(path, "\\", "/");
   StringReplace(path, "<", "");
   StringReplace(path, ">", "");
   return path;
}

string MapToOfficialStructure(string include_path) {
   // Mapeia caminhos para a estrutura oficial
   if(StringFind(include_path, "include/risk/") >= 0)
      return "risk/" + StringSubstr(include_path, StringFind(include_path, "risk/") + 5);
   if(StringFind(include_path, "include/visuals/") >= 0)
      return "visuals/" + StringSubstr(include_path, StringFind(include_path, "visuals/") + 8);
   if(StringFind(include_path, "include/intelligence/") >= 0)
      return "intelligence/" + StringSubstr(include_path, StringFind(include_path, "intelligence/") + 12);
   if(StringFind(include_path, "include/analysis/") >= 0)
      return "analysis/" + StringSubstr(include_path, StringFind(include_path, "analysis/") + 9);
   if(StringFind(include_path, "include/executionlogic/") >= 0)
      return "executionlogic/" + StringSubstr(include_path, StringFind(include_path, "executionlogic/") + 14);
   if(StringFind(include_path, "include/decisionengine/") >= 0)
      return "decisionengine/" + StringSubstr(include_path, StringFind(include_path, "decisionengine/") + 14);
   if(StringFind(include_path, "include/integration/") >= 0)
      return "integration/" + StringSubstr(include_path, StringFind(include_path, "integration/") + 11);
   if(StringFind(include_path, "include/modules/") >= 0)
      return "modules/" + StringSubstr(include_path, StringFind(include_path, "modules/") + 8);
   if(StringFind(include_path, "include/tools/") >= 0)
      return "tools/" + StringSubstr(include_path, StringFind(include_path, "tools/") + 6);
   if(StringFind(include_path, "include/types/") >= 0)
      return "types/" + StringSubstr(include_path, StringFind(include_path, "types/") + 6);
   return include_path;
}

bool IsValidPath(string path) {
   // Verifica se o caminho está na estrutura oficial
   string valid_folders[] = {
      "core/",
      "agents/",
      "utils/",
      "config/",
      "risk/",
      "analysis/",
      "executionlogic/",
      "decisionengine/",
      "intelligence/",
      "integration/",
      "visuals/",
      "modules/",
      "tools/",
      "types/",
      "auditor/",
      "expert/",
      "scripts/",
      "logs/",
      "docs/"
   };
   for(int i = 0; i < ArraySize(valid_folders); i++) {
      if(StringFind(path, valid_folders[i]) == 0)
         return true;
   }
   return false;
}

int ExtractIncludes(string content, string &includes[]) {
   int count = 0;
   int pos = 0;
   while((pos = StringFind(content, "#include", pos)) >= 0) {
      int start = StringFind(content, "<", pos);
      int end = StringFind(content, ">", start);
      if(start >= 0 && end > start) {
         ArrayResize(includes, count + 1);
         includes[count] = StringSubstr(content, start + 1, end - start - 1);
         count++;
      }
      pos = end + 1;
   }
   return count;
}
