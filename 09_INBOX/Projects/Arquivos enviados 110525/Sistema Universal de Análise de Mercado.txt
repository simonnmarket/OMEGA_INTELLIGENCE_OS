// OnInit (atualizado)
int OnInit() {
   // Inicialização existente
   ma200_handle = iMA(_Symbol, PERIOD_CURRENT, 200, 0, MODE_SMA, PRICE_CLOSE);
   ma50_handle = iMA(_Symbol, PERIOD_CURRENT, 50, 0, MODE_SMA, PRICE_CLOSE);
   rsi_handle = iRSI(_Symbol, PERIOD_CURRENT, 14, PRICE_CLOSE);
   macd_handle = iMACD(_Symbol, PERIOD_CURRENT, 12, 26, 9, PRICE_CLOSE);
   atr_handle = iATR(_Symbol, PERIOD_CURRENT, 14);
   bbands_handle = iBands(_Symbol, PERIOD_CURRENT, 20, 0, 2, PRICE_CLOSE);
   volume_handle = iVolumes(_Symbol, PERIOD_CURRENT, VOLUME_TICK);
   
   if(ma200_handle == INVALID_HANDLE || ma50_handle == INVALID_HANDLE || 
      rsi_handle == INVALID_HANDLE || macd_handle == INVALID_HANDLE || 
      atr_handle == INVALID_HANDLE || bbands_handle == INVALID_HANDLE ||
      volume_handle == INVALID_HANDLE) {
      Print("Erro ao inicializar indicadores!");
      return INIT_FAILED;
   }
   
   // Inicialização do Volume Profile
   if(!volumeProfile.Init(_Symbol, PERIOD_CURRENT, VolumeProfilePeriod)) {
      Print("Erro ao inicializar Volume Profile!");
      return INIT_FAILED;
   }
   
   // Inicialização do calendário de notícias
   if(!newsCalendar.Init()) {
      Print("Erro ao inicializar NewsCalendar!");
      return INIT_FAILED;
   }
   
   currentState.Init();
   volumeState.Init();
   gravityState.Init();
   
   trade.SetExpertMagicNumber(123456);
   trade.SetMarginMode();
   trade.SetTypeFillingBySymbol(_Symbol);
   trade.SetDeviationInPoints(10);
   
   if(!symbolInfo.Name(_Symbol) || !symbolInfo.RefreshRates()) {
      Print("Erro ao configurar símbolo!");
      return INIT_FAILED;
   }
   
   Print("Apollo11 Quantum EA 2.0 inicializado!");
   return INIT_SUCCEEDED;
}

// CheckTradingConditions (otimizado)
bool CheckTradingConditions() {
   // 1. Filtro de horário
   if(UseTimeFilter) {
      datetime currentTime = TimeCurrent();
      string currentTimeStr = TimeToString(currentTime, TIME_MINUTES);
      if(currentTimeStr < StartTime || currentTimeStr > EndTime) {
         Print("Fora do horário de trading");
         return false;
      }
      if(UseSessionFilter && !IsHighLiquiditySession(currentTime)) {
         Print("Sessão de baixa liquidez");
         return false;
      }
   }
   
   // 2. Filtro de notícias
   if(UseNewsFilter && IsNewsTime()) {
      Print("Período de notícias de alto impacto");
      return false;
   }
   
   // 3. Filtro de volume
   if(UseVolumeFilter) {
      if(volumeState.force < MinVolumeForce) {
         Print("Força de volume insuficiente: ", volumeState.force);
         return false;
      }
      if(UseSessionFilter && GetSessionVolume() < MinSessionVolume) {
         Print("Volume de sessão insuficiente");
         return false;
      }
   }
   
   // 4. Filtro de tendência
   if(UseTrendFilter && !IsTrendValid()) {
      Print("Tendência inválida");
      return false;
   }
   
   return true;
}

// IsNewsTime (implementado)
bool IsNewsTime() {
   if(!UseNewsFilter) return false;
   datetime currentTime = TimeCurrent();
   if(newsCalendar.IsHighImpactEvent(currentTime, NewsMinutesBefore, NewsMinutesAfter)) {
      return true;
   }
   // Fallback: detectar entropia extrema (buraco negro)
   if(currentState.entropy > EntropyThreshold * 1.5) {
      Print("Entropia extrema detectada: possível evento de mercado");
      return true;
   }
   return false;
}

// UpdateGravitationalState (evoluído)
void UpdateGravitationalState() {
   double currentPrice;
   if(!SymbolInfoDouble(_Symbol, SYMBOL_BID, currentPrice)) {
      Print("Erro ao obter SYMBOL_BID: ", GetLastError());
      gravityState.Init();
      return;
   }
   
   double atr[];
   ArraySetAsSeries(atr, true);
   if(CopyBuffer(atr_handle, 0, 0, 1, atr) <= 0) {
      Print("Erro ao obter ATR: ", GetLastError());
      gravityState.Init();
      return;
   }
   
   // Calcula o VWAP e o POC do Volume Profile
   double vwap = CalculateVWAP();
   gravityState.poc_level = volumeProfile.GetPOCLevel();
   if(vwap == 0 || gravityState.poc_level == 0) {
      Print("Erro: VWAP ou POC inválido");
      gravityState.Init();
      return;
   }
   
   // Cálculo da massa nodal (incorpora skewness)
   gravityState.nodal_mass = volumeState.force * atr[0] * (1 + MathAbs(volumeState.skewness));
   
   // Cálculo da distância ao nó (usa POC como nó primário)
   gravityState.nodal_distance = MathAbs(currentPrice - gravityState.poc_level);
   if(gravityState.nodal_distance < 1e-10) {
      gravityState.nodal_distance = 1e-10;
   }
   
   // Cálculo do campo gravitacional
   gravityState.gravitational_field = GMarketConstant * gravityState.nodal_mass / 
                                     MathPow(gravityState.nodal_distance, 2);
   
   // Cálculo da velocidade de escape
   gravityState.escape_velocity = MathSqrt(2 * GMarketConstant * gravityState.nodal_mass / 
                                          gravityState.nodal_distance);
   
   // Cálculo da energia potencial
   gravityState.potential_energy = -GMarketConstant * gravityState.nodal_mass * 
                                  volumeState.force / gravityState.nodal_distance;
   
   // Cálculo da força gravitacional
   gravityState.gravitational_pull = gravityState.gravitational_field * volumeState.smart_money_flow;
}

// CalculateTransitionProbability (parametrizado)
double CalculateTransitionProbability() {
   double probability = 0;
   probability += WeightProbCoherence * currentState.coherence;
   probability += WeightProbEntanglement * currentState.entanglement;
   probability += WeightProbEnergy * (currentState.energy * 1e30);
   probability += WeightProbVolume * volumeState.force;
   probability += WeightProbGravity * (1 - gravityState.gravitational_field / 
                                      (GMarketConstant * gravityState.nodal_mass));
   // Ajuste por incerteza
   probability *= (1 - currentState.uncertainty);
   return MathMin(probability, 1.0);
}

// CalculateSignal (refinado)
ENUM_APOLLO11_SIGNAL CalculateSignal() {
   double ma200[], ma50[], rsi[], macd_main[], macd_signal[], bbands_upper[], bbands_lower[];
   ArraySetAsSeries(ma200, true);
   ArraySetAsSeries(ma50, true);
   ArraySetAsSeries(rsi, true);
   ArraySetAsSeries(macd_main, true);
   ArraySetAsSeries(macd_signal, true);
   ArraySetAsSeries(bbands_upper, true);
   ArraySetAsSeries(bbands_lower, true);
   
   if(CopyBuffer(ma200_handle, 0, 0, 2, ma200) <= 0 ||
      CopyBuffer(ma50_handle, 0, 0, 2, ma50) <= 0 ||
      CopyBuffer(rsi_handle, 0, 0, 2, rsi) <= 0 ||
      CopyBuffer(macd_handle, 0, 0, 2, macd_main) <= 0 ||
      CopyBuffer(macd_handle, 1, 0, 2, macd_signal) <= 0 ||
      CopyBuffer(bbands_handle, 1, 0, 2, bbands_upper) <= 0 ||
      CopyBuffer(bbands_handle, 2, 0, 2, bbands_lower) <= 0) {
      Print("Erro ao obter dados dos indicadores");
      return SIGNAL_NONE;
   }
   
   double currentPrice;
   if(!SymbolInfoDouble(_Symbol, SYMBOL_BID, currentPrice)) {
      Print("Erro ao obter SYMBOL_BID: ", GetLastError());
      return SIGNAL_NONE;
   }
   
   // Superposição quântica: calcula probabilidades para compra e venda
   double buyProbability = 0, sellProbability = 0;
   if(ma50[0] > ma200[0] && rsi[0] < 70 && macd_main[0] > macd_signal[0]) {
      buyProbability += 0.3;
   }
   if(ma50[0] < ma200[0] && rsi[0] > 30 && macd_main[0] < macd_signal[0]) {
      sellProbability += 0.3;
   }
   if(currentPrice < bbands_upper[0]) buyProbability += 0.2;
   if(currentPrice > bbands_lower[0]) sellProbability += 0.2;
   if(volumeState.force > MinVolumeForce) {
      buyProbability += 0.1;
      sellProbability += 0.1;
   }
   if(volumeState.imbalance > ImbalanceThreshold) buyProbability += 0.15;
   if(volumeState.imbalance < -ImbalanceThreshold) sellProbability += 0.15;
   if(currentState.coherence > CoherenceThreshold) {
      buyProbability += 0.1;
      sellProbability += 0.1;
   }
   if(currentState.probability > MinProbability) {
      buyProbability += 0.1;
      sellProbability += 0.1;
   }
   if(volumeState.momentum > 0) buyProbability += 0.05;
   if(volumeState.momentum < 0) sellProbability += 0.05;
   if(gravityState.escape_velocity > MinEscapeVelocity) {
      buyProbability += 0.1;
      sellProbability += 0.1;
   }
   if(gravityState.gravitational_pull < GravitationalPullThreshold) {
      buyProbability += 0.1;
      sellProbability += 0.1;
   }
   if(currentState.uncertainty < UncertaintyThreshold) {
      buyProbability += 0.1;
      sellProbability += 0.1;
   }
   
   // Colapso do estado quântico
   bool bullish = buyProbability > sellProbability &&
                  currentState.energy > 0 &&
                  currentState.entropy < EntropyThreshold &&
                  volumeState.entropy_gradient > 0;
   bool bearish = sellProbability > buyProbability &&
                  currentState.energy > 0 &&
                  currentState.entropy < EntropyThreshold &&
                  volumeState.entropy_gradient < 0;
   
   if(bullish) return SIGNAL_BUY;
   if(bearish) return SIGNAL_SELL;
   return SIGNAL_NONE;
}
