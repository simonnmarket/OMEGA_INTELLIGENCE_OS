import numpy as np
import pandas as pd

class QuantumVPOC:
    def __init__(self, data, period=100):
        """
        Inicializa o indicador Quantum VPOC.
        
        :param data: DataFrame com colunas ['open', 'high', 'low', 'close', 'volume']
        :param period: Número de barras para calcular o perfil de volume
        """
        self.data = data
        self.period = period

    def calculate_volume_profile(self):
        """
        Calcula o perfil de volume para o período especificado.
        
        :return: Dicionário com o volume acumulado por nível de preço
        """
        # Extrai os dados do período
        recent_data = self.data.tail(self.period)
        
        # Define os níveis de preço (por exemplo, intervalos de $0.10)
        price_bins = np.arange(recent_data['low'].min(), recent_data['high'].max(), 0.1)
        
        # Calcula o volume acumulado por nível de preço
        volume_profile = {}
        for i in range(len(price_bins) - 1):
            low_bin = price_bins[i]
            high_bin = price_bins[i + 1]
            
            # Filtra os candles que tocaram este intervalo de preço
            mask = (recent_data['low'] <= high_bin) & (recent_data['high'] >= low_bin)
            volume_in_bin = recent_data.loc[mask, 'volume'].sum()
            
            # Armazena o volume acumulado no nível médio do bin
            midpoint = (low_bin + high_bin) / 2
            volume_profile[midpoint] = volume_in_bin
        
        return volume_profile

    def find_vpoc(self, volume_profile):
        """
        Encontra o Ponto de Controle de Volume (VPOC).
        
        :param volume_profile: Dicionário com o volume acumulado por nível de preço
        :return: Preço correspondente ao maior volume acumulado
        """
        vpoc_price = max(volume_profile, key=volume_profile.get)
        return vpoc_price

    def identify_hvn_lvn(self, volume_profile, threshold=0.7):
        """
        Identifica os nós de alto e baixo volume (HVN e LVN).
        
        :param volume_profile: Dicionário com o volume acumulado por nível de preço
        :param threshold: Limiar para definir HVN e LVN (ex.: 70% do volume máximo)
        :return: Dicionários separados para HVN e LVN
        """
        max_volume = max(volume_profile.values())
        hvn = {price: vol for price, vol in volume_profile.items() if vol >= threshold * max_volume}
        lvn = {price: vol for price, vol in volume_profile.items() if vol < threshold * max_volume}
        return hvn, lvn

    def run(self):
        """
        Executa o cálculo completo do Quantum VPOC.
        
        :return: VPOC, HVN, LVN
        """
        volume_profile = self.calculate_volume_profile()
        vpoc = self.find_vpoc(volume_profile)
        hvn, lvn = self.identify_hvn_lvn(volume_profile)
        return vpoc, hvn, lvn

// Parâmetros
input int Periodos = 100; // Período de análise
input double Gravidade = 0.5; // Fator de gravidade (0-1)

double POCBuffer[];

int OnInit() {
    SetIndexBuffer(0, POCBuffer, INDICATOR_DATA);
    return INIT_SUCCEEDED;
}

int OnCalculate(const int rates_total, ...) {
    for(int i = rates_total - Periodos - 1; i >= 0; i--) {
        double volumeMax = 0.0;
        double poc = 0.0;
        
        // Encontra o preço com maior volume (POC clássico)
        for(int j = i; j < i + Periodos; j++) {
            double volume = iVolume(_Symbol, PERIOD_CURRENT, j);
            if(volume > volumeMax) {
                volumeMax = volume;
                poc = iClose(_Symbol, PERIOD_CURRENT, j);
            }
        }
        
        // Suaviza com "gravidade" (MME)
        POCBuffer[i] = (Gravidade * poc) + ((1 - Gravidade) * POCBuffer[i+1]);
    }
    return rates_total;
}
