    CONDITION_TRENDING,
    CONDITION_RANGING,
    CONDITION_VOLATILE,
    CONDITION_UNDEFINED
};

//+------------------------------------------------------------------+
//| Estruturas Principais                                             |
//+------------------------------------------------------------------+
struct MarketData {
    // Preços
    double open;
    double high;
    double low;
    double close;
    
    // Volume
    double volume;
    double volumeMA;
    double volumeDelta;
    
    // Tendência
    double trend;
    double momentum;
    double volatility;
    
    // Market Profile
    double valueAreaHigh;
    double valueAreaLow;
    double poc;              // Point of Control
    
    // Order Flow
    double buyingPressure;
    double sellingPressure;
    double deltaFlow;
    
    // Estado
    ENUM_MARKET_CONDITION condition;
    bool isValid;
    datetime lastUpdate;
};

struct TradeSetup {
    ENUM_TRADE_SIGNAL signal;
    double entry;
    double stopLoss;
    double takeProfit;
    double volume;
    double risk;
    double reward;
    double confidence;
    string reason;
    bool isValid;
};
