#+------------------------------------------------------------------+
#| CLASSE: QuantumCache - Cache Quântico com bloqueio                |
#+------------------------------------------------------------------+
class QuantumCache:
    def __init__(self):
        self._cache = {}
        self._lock = threading.Lock()
    
    def get(self, key):
        with self._lock:
            return self._cache.get(key)
    
    def set(self, key, value):
        with self._lock:
            self._cache[key] = value

# Instância global de cache
FILE_CACHE = QuantumCache()

#+------------------------------------------------------------------+
#| CLASSE: AgentAlpha - Supervisor TIER-0                            |
#+------------------------------------------------------------------+
class AgentAlpha:
    def __init__(self, root_dir="MQL5/", output_dir="HTML/", include_dir="Include/"):
        self.root_dir = os.path.normpath(root_dir)
        self.output_dir = os.path.normpath(output_dir)
        self.include_dir = os.path.normpath(include_dir)
        self.graph = None
        self.running = False
        self.lock = threading.Lock()

    def sanitize_path(self, path):
        """Sanitiza caminho para prevenir injeção"""
        dangerous = ["..\\", "../", "|", "&", "$", "`", "(", ")"]
        for seq in dangerous:
            path = path.replace(seq, "")
        
        try:
            norm_path = os.path.normpath(path)
            abs_path = os.path.abspath(norm_path)
            root_abs = os.path.abspath(self.root_dir)
            
            if os.path.commonpath([root_abs]) != os.path.commonpath([root_abs, abs_path]):
                return None
            return norm_path
        except:
            return None

    def calculate_quantum_sha3(self, file_path):
        """Calcula SHA3-256 com bloqueio"""
        try:
            with open(file_path, 'rb') as f:
                data = f.read()
                return hashlib.sha3_256(data).hexdigest()
        except:
            return "ERROR"

    def check_file_integrity(self, file_path):
        """Verifica integridade com cache"""
        cache_key = f"integrity_{file_path}"
        cached = FILE_CACHE.get(cache_key)
        if cached:
            return cached
        
        exists = os.path.exists(file_path)
        sha3 = self.calculate_quantum_sha3(file_path) if exists else "NOT_FOUND"
        integrity = 1.0 if exists and sha3 != "ERROR" else 0.0
        
        result = {
            "exists": exists,
            "integrity": integrity,
            "sha3": sha3,
            "last_modified": datetime.fromtimestamp(os.path.getmtime(file_path)).isoformat() if exists else None
        }
        
        FILE_CACHE.set(cache_key, result)
        return result

    def resolve_dependency(self, dep, root_dir):
        """Resolução inteligente de caminhos com fallback"""
        # 1. Tentativa direta
        if os.path.exists(dep):
            return dep
        
        # 2. Busca case-insensitive
        base = os.path.basename(dep)
        for root, _, files in os.walk(root_dir):
            for f in files:
                if f.lower() == base.lower():
                    return os.path.join(root, f)
        
        # 3. Verificar includes padrão
        std_path = os.path.join(root_dir, 'Include', base)
        if os.path.exists(std_path):
            return std_path
        
        # 4. Busca por similaridade
        all_files = [f for _, _, files in os.walk(root_dir) for f in files]
        matches = get_close_matches(base, all_files, n=1, cutoff=0.8)
        if matches:
            return os.path.join(root_dir, matches[0])
        
        return None

    def extract_hyper_links(self, file_path):
        """Extrai #include com regex e normaliza caminhos"""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
                matches = re.findall(r'#include\s+[<"](.*?)[">]', content)
                
                normalized = []
                for m in matches:
                    norm = self.sanitize_path(m)
                    if norm:
                        resolved = self.resolve_dependency(norm, self.root_dir)
                        if resolved:
                            normalized.append(resolved)
                return normalized
        except:
            return []

    def process_file(self, file_path):
        """Processa um arquivo individualmente"""
        try:
            rel_path = os.path.relpath(file_path, self.root_dir)
            integrity = self.check_file_integrity(file_path)
            criticality = 1.0 if "logger_institutional" in file_path.lower() else 0.8 if "quantum" in file_path.lower() else 0.5
            
            return {
                "file_path": file_path,
                "rel_id": rel_path,
                "type": self.classify_tier(file_path),
                "status": "GREEN" if integrity["exists"] else "RED",
                "integrity": integrity["integrity"],
                "sha3": integrity["sha3"],
                "criticality": criticality,
                "last_modified": integrity["last_modified"]
            }
        except:
            return None

    def process_edges(self, file_path):
        """Processa arestas de um arquivo"""
        edges = []
        try:
            deps = self.extract_hyper_links(file_path)
            for dep in deps:
                exists = os.path.exists(dep)
                edges.append({
                    "source": file_path,
                    "target": dep,
                    "status": "GREEN" if exists else "RED",
                    "error_type": "INCLUDE_NOT_FOUND" if not exists else ""
                })
        except:
            pass
        return edges

    def classify_tier(self, file_path):
        """Classificação hierárquica"""
        if "CORE" in file_path.upper():
            return "CORE"
        elif "AUDITOR" in file_path.upper():
            return "AUDITOR"
        elif "SECURITY" in file_path.upper():
            return "SECURITY"
        elif "INTELLIGENCE" in file_path.upper():
            return "INTELLIGENCE"
        elif "INCLUDE" in file_path.upper():
            return "INCLUDE"
        else:
            return "UTILS"

    def parallel_scan(self):
        """Escaneamento paralelo"""
        files = []
        with ThreadPoolExecutor(max_workers=os.cpu_count() * 4) as executor:
            futures = []
            for folder, _, filenames in os.walk(self.root_dir):
                for file in filenames:
                    if file.endswith(('.mqh', '.mq5')):
                        file_path = os.path.join(folder, file)
                        futures.append(executor.submit(self.process_file, file_path))
            
            for future in as_completed(futures):
                result = future.result()
                if result:
                    files.append(result["file_path"])
        return files

    def build_dependency_graph(self):
        """Constrói o grafo de dependências"""
        self.graph = nx.DiGraph()
        files = self.parallel_scan()
        
        # Processamento paralelo de nós
        node_data = []
        with ThreadPoolExecutor(max_workers=os.cpu_count() * 2) as executor:
            future_to_file = {executor.submit(self.process_file, file): file for file in files}
            for future in as_completed(future_to_file):
                result = future.result()
                if result:
                    node_data.append(result)
        
        for data in node_data:
            self.graph.add_node(
                data["file_path"],
                id=data["rel_id"],
                file_path=data["file_path"],
                type=data["type"],
                status=data["status"],
                integrity=data["integrity"],
                sha3=data["sha3"],
                criticality=data["criticality"],
                last_modified=data["last_modified"]
            )
        
        # Processamento paralelo de arestas
        edge_data = []
        with ThreadPoolExecutor(max_workers=os.cpu_count() * 2) as executor:
            future_to_file = {executor.submit(self.process_edges, file): file for file in files}
            for future in as_completed(future_to_file):
                result = future.result()
                edge_data.extend(result)
        
        for data in edge_data:
            self.graph.add_edge(
                data["source"], data["target"],
                status=data["status"],
                error_type=data["error_type"]
            )
        
        return self.graph

    def export_to_json(self):
        """Exporta para JSON"""
        data = {
            "metadata": {
                "generated_at": datetime.now().isoformat(),
                "total_files": len(self.graph.nodes),
                "total_dependencies": len(self.graph.edges),
                "system_status": "APPROVED" if all(self.graph.nodes[n]["status"] == "GREEN" for n in self.graph.nodes) else "CRITICAL"
            },
            "nodes": [],
            "links": []
        }
        
        for n in self.graph.nodes:
            rel_id = os.path.relpath(n, self.root_dir)
            data["nodes"].append({
                "id": rel_id,
                "type": self.graph.nodes[n]["type"],
                "status": self.graph.nodes[n]["status"],
                "integrity": self.graph.nodes[n]["integrity"],
                "criticality": self.graph.nodes[n]["criticality"],
                "sha3": self.graph.nodes[n]["sha3"],
                "last_modified": self.graph.nodes[n]["last_modified"]
            })
        
        for u, v in self.graph.edges:
            data["links"].append({
                "source": os.path.relpath(u, self.root_dir),
                "target": os.path.relpath(v, self.root_dir),
                "status": self.graph[u][v]["status"],
                "error_type": self.graph[u][v].get("error_type", "")
            })
        
        output_path = os.path.join(self.root_dir, self.output_dir, "dependencies.json")
        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump(data, f, indent=2)
        print(f"[AGENT_ALPHA] dependencies.json gerado em {output_path}")

    def enhanced_audit_log(self):
        """Geração de relatório institucional"""
        report = {
            "timestamp": datetime.now().isoformat(),
            "system": "MQL5 Dependency Analyzer",
            "version": "TIER-0++",
            "metrics": {
                "total_files": len(self.graph.nodes),
                "broken_links": sum(1 for _, _, d in self.graph.edges(data=True) if d['status'] != 'GREEN'),
                "core_files": sum(1 for n in self.graph.nodes if self.graph.nodes[n]['type'] == 'CORE'),
                "avg_integrity": sum(self.graph.nodes[n]['integrity'] for n in self.graph.nodes) / len(self.graph.nodes)
            },
            "critical_issues": [
                {"file": os.path.relpath(u, self.root_dir), "dependency": os.path.relpath(v, self.root_dir), "type": "broken_include"}
                for u, v, d in self.graph.edges(data=True) if d['status'] != 'GREEN'
            ]
        }
        
        audit_path = os.path.join(self.root_dir, "Audit", f"audit_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json")
        os.makedirs(os.path.dirname(audit_path), exist_ok=True)
        with open(audit_path, 'w') as f:
            json.dump(report, f, indent=2)
        
        print(f"[AGENT_ALPHA] Relatório de auditoria gerado em {audit_path}")
        return audit_path

    def run(self):
        """Executa o escaneamento completo"""
        print(f"[AGENT_ALPHA] Iniciando escaneamento em {self.root_dir}...")
        self.build_dependency_graph()
        self.export_to_json()
        self.enhanced_audit_log()
        print(f"[AGENT_ALPHA] Escaneamento concluído. {len(self.graph.nodes)} arquivos, {len(self.graph.edges)} dependências")
