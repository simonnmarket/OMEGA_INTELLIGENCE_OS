#ifndef __TIER0_AUTOFIXER_MQH__
#define __TIER0_AUTOFIXER_MQH__

#include <include/utils/logger_institutional.mqh>
#include <include/security/QuantumBlockchain.mqh>

class Tier0AutoFixer
{
private:
   logger_institutional &m_logger;
   QuantumBlockchain    &m_blockchain;
   string               m_symbol;

   //+--------------------------------------------------------------+
   //| Valida contexto antes da execução                             |
   //+--------------------------------------------------------------+
   bool is_valid_context()
   {
      if(!TerminalInfoInteger(TERMINAL_CONNECTED))
      {
         m_logger.log_error("[FIXER] Sem conexão com o servidor");
         return false;
      }

      if(!m_logger.is_initialized())
      {
         m_logger.log_error("[FIXER] Logger não inicializado");
         return false;
      }

      if(!m_blockchain.IsReady())
      {
         m_logger.log_warning("[FIXER] Blockchain não está pronto");
         return false;
      }

      return true;
   }

   //+--------------------------------------------------------------+
   //| Corrige caminho de include                                    |
   //+--------------------------------------------------------------+
   string FixIncludePath(string include_path)
   {
      // Remove prefixos incorretos
      StringReplace(include_path, "include/", "");
      StringReplace(include_path, "include\\", "");
      StringReplace(include_path, "Include/", "");
      StringReplace(include_path, "Include\\", "");
      
      // Retorna caminho corrigido
      return include_path;
   }

   //+--------------------------------------------------------------+
   //| Aplica correção no conteúdo do arquivo                        |
   //+--------------------------------------------------------------+
   bool ApplyFix(string &content, string old_include, string new_include)
   {
      string pattern1 = "#include <" + old_include + ">";
      string pattern2 = "#include \"" + old_include + "\"";
      string replacement = "#include <" + new_include + ">";
      
      int pos1 = StringFind(content, pattern1);
      int pos2 = StringFind(content, pattern2);
      
      if(pos1 >= 0)
      {
         StringReplace(content, pattern1, replacement);
         return true;
      }
      else if(pos2 >= 0)
      {
         StringReplace(content, pattern2, replacement);
         return true;
      }
      
      return false;
   }

   //+--------------------------------------------------------------+
   //| Registra correção no blockchain                               |
   //+--------------------------------------------------------------+
   void LogFix(string file, string old_path, string new_path, bool success)
   {
      string data = StringFormat("FIX=INCLUDE_PATH|FILE=%s|OLD=%s|NEW=%s|SUCCESS=%s|TIME=%s",
                               file,
                               old_path,
                               new_path,
                               success ? "YES" : "NO",
                               TimeToString(TimeCurrent(), TIME_DATE|TIME_SECONDS));
      m_blockchain.RecordTransaction(data, "AUTO_FIX");
      
      if(success)
         m_logger.log_info("[FIXER] Include corrigido: " + old_path + " → " + new_path);
      else
         m_logger.log_error("[FIXER] Falha ao corrigir include: " + old_path);
   }

public:
   //+--------------------------------------------------------------+
   //| CONSTRUTOR                                                   |
   //+--------------------------------------------------------------+
   Tier0AutoFixer(logger_institutional &logger,
                 QuantumBlockchain &qb,
                 string symbol = _Symbol) :
      m_logger(logger),
      m_blockchain(qb),
      m_symbol(symbol)
   {
      m_logger.log_info("[FIXER] Corretor Automático TIER-0 v2.0 ativado");
   }

   //+--------------------------------------------------------------+
   //| Executa correção automática                                  |
   //+--------------------------------------------------------------+
   bool RunAutoFix()
   {
      if(!is_valid_context()) return false;

      m_logger.log_info("[FIXER] Iniciando correção automática...");
      
      // Lista de arquivos críticos
      string critical_files[] = {
         "NumeiaEA.mq5",
         "core_brain_manager.mqh",
         "quantum_processor.mqh",
         "risk_profile.mqh",
         "trade_executor.mqh"
      };
      
      for(int i = 0; i < ArraySize(critical_files); i++)
      {
         string file_path = "MQL5/" + critical_files[i];
         string content;
         
         if(FileReadToString(file_path, content))
         {
            // Procurar includes com caminhos incorretos
            string patterns[] = {
               "include/utils/logger_institutional.mqh",
               "include/security/QuantumBlockchain.mqh",
               "include/quantum/quantum_processor.mqh",
               "include/risk/risk_profile.mqh",
               "include/executionlogic/trade_executor.mqh"
            };
            
            for(int j = 0; j < ArraySize(patterns); j++)
            {
               string new_path = FixIncludePath(patterns[j]);
               if(ApplyFix(content, patterns[j], new_path))
               {
                  // Salvar arquivo corrigido
                  if(FileWriteFromString(file_path, content))
                  {
                     LogFix(critical_files[i], patterns[j], new_path, true);
                     m_logger.log_success("[FIXER] Correção aplicada: " + patterns[j] + " → " + new_path);
                  }
                  else
                  {
                     LogFix(critical_files[i], patterns[j], new_path, false);
                  }
               }
            }
         }
      }
      
      m_logger.log_success("[FIXER] Correção automática concluída");
      return true;
   }
};

#endif // __TIER0_AUTOFIXER_MQH__
