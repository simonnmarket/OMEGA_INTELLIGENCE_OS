class QuantumFileLock:
    """Bloqueio quântico para evitar race condition"""
    def __init__(self):
        self._locks = {}
        self._lock = threading.Lock()
    
    @contextmanager
    def lock_for_file(self, file_path):
        file_path = os.path.normpath(file_path)
        with self._lock:
            if file_path not in self._locks:
                self._locks[file_path] = threading.Lock()
            file_lock = self._locks[file_path]
        
        with file_lock:
            yield

# Instância global de bloqueio
file_locker = QuantumFileLock()

class AgentAlpha:
    """
    Supervisor TIER-0 de dependências com escaneamento quântico,
    classificação neural e autocura automática.
    """
    def __init__(self, root_dir="MQL5/", output_dir="HTML/", include_dir="Include/"):
        self.root_dir = os.path.normpath(root_dir)
        self.output_dir = os.path.normpath(output_dir)
        self.include_dir = os.path.normpath(include_dir)
        self.graph = None
        self.running = False
        self.lock = threading.Lock()
        
        # Cache de integridade
        self.integrity_cache = {}
        self.cache_lock = threading.Lock()

    def sanitize_path(self, path):
        """Sanitiza caminho para prevenir injeção"""
        # Remove sequências perigosas
        dangerous = ["..\\", "../", "|", "&", "$", "`", "(", ")"]
        for seq in dangerous:
            path = path.replace(seq, "")
        
        # Normaliza e valida
        try:
            norm_path = os.path.normpath(path)
            abs_path = os.path.abspath(norm_path)
            root_abs = os.path.abspath(self.root_dir)
            
            # Garante que o caminho está dentro do diretório do projeto
            if os.path.commonpath([root_abs]) != os.path.commonpath([root_abs, abs_path]):
                return None
                
            return norm_path
        except:
            return None

    def calculate_quantum_sha3(self, file_path):
        """Calcula SHA3-256 com bloqueio para evitar race condition"""
        with file_locker.lock_for_file(file_path):
            try:
                with open(file_path, 'rb') as f:
                    data = f.read()
                    return hashlib.sha3_256(data).hexdigest()
            except Exception as e:
                print(f"[AGENT_ALPHA] Erro ao ler {file_path}: {e}")
                return "ERROR"

    def check_file_integrity(self, file_path):
        """Verifica integridade com cache e bloqueio"""
        with self.cache_lock:
            if file_path in self.integrity_cache:
                return self.integrity_cache[file_path]
        
        exists = os.path.exists(file_path)
        sha3 = self.calculate_quantum_sha3(file_path) if exists else "NOT_FOUND"
        integrity = 1.0 if exists and sha3 != "ERROR" else 0.0
        
        result = {
            "exists": exists,
            "integrity": integrity,
            "sha3": sha3,
            "last_modified": datetime.fromtimestamp(os.path.getmtime(file_path)).isoformat() if exists else None
        }
        
        with self.cache_lock:
            self.integrity_cache[file_path] = result
            
        return result

    def quantum_file_analysis(self, file_path):
        """Análise paralelizada com checksum quântico"""
        sanitized = self.sanitize_path(file_path)
        if not sanitized:
            return None
            
        integrity = self.check_file_integrity(sanitized)
        return {
            "path": sanitized,
            "valid": integrity["exists"] and integrity["integrity"] > 0.0,
            "quantum_hash": integrity["sha3"]
        }

    def quantum_scan(self):
        """Escaneamento paralelo usando princípios quânticos"""
        files = []
        
        with ThreadPoolExecutor(max_workers=os.cpu_count() * 4) as executor:
            futures = []
            for folder, _, filenames in os.walk(self.root_dir):
                for file in filenames:
                    if file.endswith(('.mqh', '.mq5')):
                        file_path = os.path.join(folder, file)
                        futures.append(
                            executor.submit(self.quantum_file_analysis, file_path)
                        )
            
            for future in as_completed(futures):
                result = future.result()
                if result and result["valid"]:
                    files.append(result["path"])
        
        return files

    def classify_tier(self, file_path):
        """Classificação neural baseada em importância"""
        # Características do arquivo
        features = self.extract_features(file_path)
        
        # Simulação de modelo neural (em produção, carregue modelo real)
        weights = np.array([0.3, 0.4, 0.2, 0.1])  # Pesos simulados
        score = np.dot(features, weights)
        
        # Classificação hierárquica
        if "CORE" in file_path.upper():
            return "CORE"
        elif "AUDITOR" in file_path.upper():
            return "AUDITOR"
        elif "SECURITY" in file_path.upper() or "BLOCKCHAIN" in file_path.upper():
            return "SECURITY"
        elif "QUANTUM" in file_path.upper() or "INTELLIGENCE" in file_path.upper():
            return "INTELLIGENCE"
        elif "INCLUDE" in file_path.upper():
            return "INCLUDE"
        elif "UTILS" in file_path.upper():
            return "UTILS"
        else:
            return "OTHER"

    def extract_features(self, file_path):
        """Extrai características para classificação neural"""
        try:
            # Tamanho do arquivo
            size = os.path.getsize(file_path) / 1024  # KB
            
            # Número de linhas
            with open(file_path, 'r', encoding='utf-8') as f:
                lines = len(f.readlines())
                
            # Número de dependências
            deps = self.extract_hyper_links(file_path)
            dep_count = len(deps)
            
            # Modificado recentemente? (últimas 24h)
            modified_recently = 1 if (datetime.now().timestamp() - os.path.getmtime(file_path)) < 86400 else 0
            
            return [size, lines, dep_count, modified_recently]
        except:
            return [0, 0, 0, 0]

    def normalize_include_path(self, include_path):
        """Normaliza o caminho do include para o padrão MetaQuotes"""
        # Sanitiza
        sanitized = self.sanitize_path(include_path)
        if not sanitized:
            return None
            
        # Remove prefixos comuns de erro
        normalized = re.sub(r'^[\.\\/]*include[\\/]', '', sanitized, flags=re.IGNORECASE)
        normalized = re.sub(r'^[\.\\/]*Include[\\/]', '', normalized, flags=re.IGNORECASE)
        
        # Reconstrói caminho físico
        physical = os.path.normpath(os.path.join(self.root_dir, "Include", normalized))
        return physical

    def extract_hyper_links(self, file_path):
        """Extrai #include com regex e normaliza caminhos"""
        try:
            with file_locker.lock_for_file(file_path):
                with open(file_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                    matches = re.findall(r'#include\s+[<"](.*?)[">]', content)
                    
                    # Normaliza cada caminho
                    normalized = []
                    for m in matches:
                        norm = self.normalize_include_path(m)
                        if norm:
                            normalized.append(norm)
                    return normalized
        except Exception as e:
            print(f"[AGENT_ALPHA] Erro ao ler {file_path}: {e}")
            return []

    def suggest_fixes(self, broken_path):
        """Sugere correções usando lógica genética simulada"""
        suggestions = []
        filename = os.path.basename(broken_path).lower()
        
        # Possíveis localizações
        possible_dirs = [
            "Include/utils",
            "Include/security",
            "Include/quantum",
            "Include/analysis",
            "Include/intelligence",
            "Include/executionlogic",
            "Include/risk",
            "Include/decisionengine",
            "Include/integration",
            "Include/visuals",
            "CORE"
        ]
        
        for dir_path in possible_dirs:
            candidate = os.path.normpath(os.path.join(self.root_dir, dir_path, os.path.basename(broken_path)))
            if os.path.exists(candidate):
                suggestions.append({
                    "path": candidate,
                    "confidence": 0.9,
                    "reason": f"Arquivo encontrado em {dir_path}"
                })
        
        # Se não encontrou, tenta por nome similar
        if not suggestions:
            for root, _, files in os.walk(os.path.join(self.root_dir, "Include")):
                for file in files:
                    if filename in file.lower() or file.lower() in filename:
                        candidate = os.path.normpath(os.path.join(root, file))
                        suggestions.append({
                            "path": candidate,
                            "confidence": 0.6,
                            "reason": f"Nome similar encontrado: {file}"
                        })
        
        # Ordena por confiança
        suggestions.sort(key=lambda x: x["confidence"], reverse=True)
        return suggestions[:3]  # Retorna até 3 sugestões

    def apply_fix(self, source_file, fix_suggestion):
        """Aplica correção no arquivo fonte"""
        try:
            with file_locker.lock_for_file(source_file):
                with open(source_file, 'r', encoding='utf-8') as f:
                    content = f.read()
                
                # Encontra o include quebrado
                broken_include = None
                for line in content.split('\n'):
                    if '#include' in line:
                        match = re.search(r'#include\s+[<"](.*?)[">]', line)
                        if match:
                            include_path = match.group(1)
                            if os.path.basename(include_path).lower() in os.path.basename(fix_suggestion["path"]).lower():
                                broken_include = include_path
                                break
                
                if broken_include:
                    # Normaliza o caminho de destino
                    rel_path = os.path.relpath(fix_suggestion["path"], os.path.join(self.root_dir, "Include"))
                    new_include = rel_path.replace('\\', '/')
                    
                    # Substitui no conteúdo
                    pattern1 = f'#include <{broken_include}>'
                    pattern2 = f'#include "{broken_include}"'
                    replacement = f'#include <{new_include}>'
                    
                    if pattern1 in content:
                        content = content.replace(pattern1, replacement)
                    elif pattern2 in content:
                        content = content.replace(pattern2, replacement)
                    else:
                        return False
                    
                    # Salva arquivo corrigido
                    with open(source_file, 'w', encoding='utf-8') as f:
                        f.write(content)
                    
                    print(f"[AGENT_ALPHA] Correção aplicada: {broken_include} → {new_include} em {source_file}")
                    return True
        except Exception as e:
            print(f"[AGENT_ALPHA] Falha ao aplicar correção: {e}")
            return False

    def auto_healing_system(self):
        """Sistema que corrige automaticamente includes quebrados"""
        if not self.graph:
            return False
            
        broken_links = [(u, v) for u, v in self.graph.edges if self.graph[u][v]["status"] == "RED"]
        
        corrections_made = 0
        for source, target in broken_links:
            possible_fixes = self.suggest_fixes(target)
            if possible_fixes:
                if self.apply_fix(source, possible_fixes[0]):
                    corrections_made += 1
                    # Atualiza o grafo
                    self.graph[source][target]["status"] = "YELLOW"
                    # Reconecta ao novo caminho
                    self.graph.add_edge(source, possible_fixes[0]["path"], status="GREEN", auto_healed=True)
        
        print(f"[AGENT_ALPHA] Sistema de autocura concluído. {corrections_made} correções aplicadas.")
        return corrections_made > 0

    def build_dependency_graph(self):
        """Constrói o grafo de dependências com metadados"""
        self.graph = nx.DiGraph()
        files = self.quantum_scan()
        
        # Processamento paralelo de nós
        node_data = []
        with ThreadPoolExecutor(max_workers=os.cpu_count() * 2) as executor:
            future_to_file = {
                executor.submit(self.process_node, file): file for file in files
            }
            for future in as_completed(future_to_file):
                result = future.result()
                if result:
                    node_data.append(result)
        
        # Adiciona nós ao grafo
        for data in node_data:
            self.graph.add_node(
                data["file_path"],
                id=data["rel_id"],
                file_path=data["file_path"],
                type=data["type"],
                status=data["status"],
                integrity=data["integrity"],
                sha3=data["sha3"],
                criticality=data["criticality"],
                timeResolution=100,
                last_modified=data["last_modified"]
            )
        
        # Processamento paralelo de arestas
        edge_data = []
        with ThreadPoolExecutor(max_workers=os.cpu_count() * 2) as executor:
            future_to_file = {
                executor.submit(self.process_edges, file): file for file in files
            }
            for future in as_completed(future_to_file):
                result = future.result()
                edge_data.extend(result)
        
        # Adiciona arestas ao grafo
        for data in edge_data:
            self.graph.add_edge(
                data["source"], data["target"],
                status=data["status"],
                error_type=data["error_type"]
            )
        
        return self.graph

    def process_node(self, file_path):
        """Processa um nó individualmente"""
        try:
            rel_path = os.path.relpath(file_path, self.root_dir)
            integrity = self.check_file_integrity(file_path)
            criticality = 1.0 if "logger_institutional" in file_path.lower() else 0.8 if "quantum" in file_path.lower() else 0.5
            
            return {
                "file_path": file_path,
                "rel_id": rel_path,
                "type": self.classify_tier(file_path),
                "status": "GREEN" if integrity["exists"] else "RED",
                "integrity": integrity["integrity"],
                "sha3": integrity["sha3"],
                "criticality": criticality,
                "last_modified": integrity["last_modified"]
            }
        except:
            return None

    def process_edges(self, file_path):
        """Processa arestas de um arquivo"""
        edges = []
        try:
            deps = self.extract_hyper_links(file_path)
            for dep in deps:
                exists = os.path.exists(dep)
                edges.append({
                    "source": file_path,
                    "target": dep,
                    "status": "GREEN" if exists else "RED",
                    "error_type": "INCLUDE_NOT_FOUND" if not exists else ""
                })
        except:
            pass
        return edges

    def export_to_json(self):
        """Exporta para JSON com gerenciamento de memória"""
        # Força coleta de lixo
        gc.collect()
        
        data = {
            "metadata": {
                "generated_at": datetime.now().isoformat(),
                "total_files": len(self.graph.nodes),
                "total_dependencies": len(self.graph.edges),
                "system_status": "APPROVED" if all(self.graph.nodes[n]["status"] == "GREEN" for n in self.graph.nodes) else "CRITICAL"
            },
            "nodes": [],
            "links": []
        }
        
        # Mapeia nós para IDs relativos
        node_map = {}
        for n in self.graph.nodes:
            rel_id = os.path.relpath(n, self.root_dir)
            node_map[n] = rel_id
            data["nodes"].append({
                "id": rel_id,
                "type": self.graph.nodes[n]["type"],
                "status": self.graph.nodes[n]["status"],
                "integrity": self.graph.nodes[n]["integrity"],
                "criticality": self.graph.nodes[n]["criticality"],
                "sha3": self.graph.nodes[n]["sha3"],
                "last_modified": self.graph.nodes[n]["last_modified"]
            })
        
        for u, v in self.graph.edges:
            data["links"].append({
                "source": os.path.relpath(u, self.root_dir),
                "target": os.path.relpath(v, self.root_dir),
                "status": self.graph[u][v]["status"],
                "error_type": self.graph[u][v].get("error_type", "")
            })
        
        output_path = os.path.join(self.root_dir, self.output_dir, "dependencies.json")
        try:
            with open(output_path, 'w', encoding='utf-8') as f:
                json.dump(data, f, indent=2)
            print(f"[AGENT_ALPHA] dependencies.json gerado em {output_path}")
            return True
        except Exception as e:
            print(f"[AGENT_ALPHA] Falha ao salvar JSON: {e}")
            return False

    def export_to_gexf(self):
        """Exporta para GEXF para análise externa"""
        output_path = os.path.join(self.root_dir, self.output_dir, "dependencies.gexf")
        try:
            nx.write_gexf(self.graph, output_path)
            print(f"[AGENT_ALPHA] dependencies.gexf gerado em {output_path}")
            return True
        except Exception as e:
            print(f"[AGENT_ALPHA] Falha ao salvar GEXF: {e}")
            return False

    def run(self):
        """Executa o escaneamento completo"""
        print(f"[AGENT_ALPHA] Iniciando escaneamento em {self.root_dir}...")
        
        # 1. Constrói grafo
        self.build_dependency_graph()
        
        # 2. Aplica autocura
        if self.auto_healing_system():
            # Reconstrói grafo após correções
            self.build_dependency_graph()
        
        # 3. Exporta dados
        self.export_to_json()
        self.export_to_gexf()
        
        print(f"[AGENT_ALPHA] Escaneamento concluído. {len(self.graph.nodes)} arquivos, {len(self.graph.edges)} dependências")
        return self.graph
