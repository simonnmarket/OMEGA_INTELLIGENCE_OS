        // Configurações de Teste
        struct TestConfig {
            string symbol;
            ENUM_TIMEFRAMES timeframe;
            datetime start_time;
            datetime end_time;
            double initial_deposit;
        };
        
    public:
        bool RunAutomatedTests() {
