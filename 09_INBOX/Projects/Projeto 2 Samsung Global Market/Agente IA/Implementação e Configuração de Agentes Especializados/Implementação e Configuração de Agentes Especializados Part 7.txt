        // Constantes e Enumerações
        enum TRADING_MODE {
            MODE_AUTOMATIC,
            MODE_SEMI_AUTOMATIC,
            MODE_MANUAL
        };
        
        // Estruturas de Dados
        struct TradeSetup {
            datetime time;
            double entry_price;
            double stop_loss;
            double take_profit;
            ENUM_ORDER_TYPE type;
        };
        
    public:
        bool Initialize() {
            if(!LoadIndicators()) {
                Print("Failed to load indicators");
                return false;
            }
            return true;
        }
};
