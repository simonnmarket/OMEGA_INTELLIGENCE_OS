2. IMPLEMENTAÇÃO GRANGER CAUSALITY
class GrangerAnalysis:
    def __init__(self):
        self.analysis_parameters = {
            'stationarity_test': {
                'method': 'augmented_dickey_fuller',
                'significance': 0.05
            },
            'lag_selection': {
                'max_lags': 10,
                'criterion': 'aic'
            },
            'causality_test': {
                'hypothesis_test': 'f_test',
                'confidence_level': 0.95
            }
        }
        
    def test_causality(self, series_x, series_y):
        """
        Testa causalidade de Granger entre duas séries
        """
        if self.check_stationarity(series_x) and self.check_stationarity(series_y):
            optimal_lag = self.find_optimal_lag(series_x, series_y)
            return self.run_granger_test(series_x, series_y, optimal_lag)
        return None
