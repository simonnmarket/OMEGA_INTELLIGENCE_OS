// Exemplo de componente React para exibir posições
import React, { useEffect, useState } from 'react';

function PositionsDashboard({ positions }) {
    return (
        <div>
            <h1>Posições Abertas</h1>
            <table>
                <thead>
                    <tr>
                        <th>Símbolo</th>
                        <th>Tipo</th>
                        <th>Preço</th>
                        <th>Lucro</th>
                    </tr>
                </thead>
                <tbody>
                    {positions.map(pos => (
                        <tr key={pos.id}>
                            <td>{pos.symbol}</td>
                            <td>{pos.type}</td>
                            <td>{pos.price.toFixed(5)}</td>
                            <td>{pos.profit.toFixed(2)}</td>
                        </tr>
                    ))}
                </tbody>
            </table>
        </div>
    );
}
