input ENUM_TIMEFRAMES Timeframe = PERIOD_H1;     // Timeframe para análise
input string          SymbolToTrade = "EURUSD";   // Símbolo a ser operado
input double          Lots = 0.1;                 // Tamanho do lote
input int             Slippage = 5;               // Slippage máximo
input int             StopLoss = 200;             // Stop Loss (em pontos)
input int             TakeProfit = 400;           // Take Profit (em pontos)
input int             MovingAveragePeriod = 20;   // Período da média móvel

int ma_handle;  // Handle da média móvel

//+------------------------------------------------------------------+
//| Função de inicialização                                          |
//+------------------------------------------------------------------+
int OnInit()
  {
   if(Symbol() != SymbolToTrade)
     {
      Print("Este EA só deve ser executado no gráfico de ", SymbolToTrade);
      return(INIT_FAILED);
     }

   ma_handle = iMA(SymbolToTrade, Timeframe, MovingAveragePeriod, 0, MODE_SMA, PRICE_CLOSE);
   if(ma_handle == INVALID_HANDLE)
     {
      Print("Erro ao criar handle da média móvel.");
      return(INIT_FAILED);
     }

   Print("SkylerBotV1.5 inicializado com sucesso.");
   return(INIT_SUCCEEDED);
  }

//+------------------------------------------------------------------+
//| Função principal de execução                                     |
//+------------------------------------------------------------------+
void OnTick()
  {
   if(Symbol() != SymbolToTrade) return;

   static datetime last_time = 0;
   datetime current_time = iTime(SymbolToTrade, Timeframe, 0);
   if(current_time == last_time) return;
   last_time = current_time;

   // Verificar posição atual
   bool hasBuy = PositionSelect(SymbolToTrade) && PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY;
   bool hasSell = PositionSelect(SymbolToTrade) && PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_SELL;

   // Obter preço atual
   double price = SymbolInfoDouble(SymbolToTrade, SYMBOL_BID);
   if(price == 0) return;

   // Obter valor da média
   double ma_value[];
   if(CopyBuffer(ma_handle, 0, 0, 1, ma_value) < 1)
     {
      Print("Erro ao obter valor da média.");
      return;
     }

   if(!hasBuy && !hasSell)
     {
      if(price > ma_value[0])
        OpenBuy();
      else if(price < ma_value[0])
        OpenSell();
     }
  }

//+------------------------------------------------------------------+
//| Função para abrir compra                                         |
//+------------------------------------------------------------------+
void OpenBuy()
  {
   double ask = SymbolInfoDouble(SymbolToTrade, SYMBOL_ASK);
   double sl = ask - StopLoss * _Point;
   double tp = ask + TakeProfit * _Point;

   MqlTradeRequest request = {};
   MqlTradeResult result = {};

   request.action = TRADE_ACTION_DEAL;
   request.symbol = SymbolToTrade;
   request.volume = NormalizeDouble(Lots, _Digits);
   request.type = ORDER_TYPE_BUY;
   request.price = ask;
   request.sl = NormalizeDouble(sl, _Digits);
   request.tp = NormalizeDouble(tp, _Digits);
   request.deviation = Slippage;
   request.magic = 20250511;
   request.comment = "SkylerBot Buy";
   request.type_filling = ORDER_FILLING_IOC;

   if(!OrderSend(request, result))
      Print("Erro ao enviar ordem de compra: ", result.retcode);
   else
      Print("Ordem de compra enviada com sucesso. Ticket: ", result.order);
  }

//+------------------------------------------------------------------+
//| Função para abrir venda                                          |
//+------------------------------------------------------------------+
void OpenSell()
  {
   double bid = SymbolInfoDouble(SymbolToTrade, SYMBOL_BID);
   double sl = bid + StopLoss * _Point;
   double tp = bid - TakeProfit * _Point;

   MqlTradeRequest request = {};
   MqlTradeResult result = {};

   request.action = TRADE_ACTION_DEAL;
   request.symbol = SymbolToTrade;
   request.volume = NormalizeDouble(Lots, _Digits);
   request.type = ORDER_TYPE_SELL;
   request.price = bid;
   request.sl = NormalizeDouble(sl, _Digits);
   request.tp = NormalizeDouble(tp, _Digits);
   request.deviation = Slippage;
   request.magic = 20250511;
   request.comment = "SkylerBot Sell";
   request.type_filling = ORDER_FILLING_IOC;

   if(!OrderSend(request, result))
      Print("Erro ao enviar ordem de venda: ", result.retcode);
   else
      Print("Ordem de venda enviada com sucesso. Ticket: ", result.order);
  }
