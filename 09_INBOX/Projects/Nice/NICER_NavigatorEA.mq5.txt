double confidence_score = 0.0;

    confidence_score += 0.15;

if(CCI > 100 || UltimateOscillator > 65)
    confidence_score += 0.10;

if(SupportResistance.IsKeyZoneTested(0.9445) && ve > VOLUME_FORCE_THRESHOLD)
    confidence_score += 0.10;

// BONUS: Confluência de Cenários
if(ve > VE_PEAK && WyckoffDetector.IsSpring(0.9204))
    confidence_score += 0.15;

if(SupportResistance.IsKeyZoneTested(0.9336) && ve > 1.3 && TrendModule.PriceAboveMA(200))
    confidence_score += 0.10;

    confidence_score += 0.10;

// Decisão final
if(confidence_score >= MIN_CONFIDENCE_THRESHOLD_INIT)
{
    if(confidence_score >= MIN_CONFIDENCE_THRESHOLD_STRONG)
        ExecuteEntry(ORDER_TYPE_HIGH_CONFIDENCE);
    else
        ExecuteEntry(ORDER_TYPE_STANDARD);
}

✅ 2. Manter linhas no gráfico após término do teste

No MetaTrader 5, objetos gráficos são automaticamente apagados ao fim do teste visual a menos que você use OBJPROP_SELECTABLE, salve os objetos ou gere relatórios externos. Uma das soluções mais práticas é usar o indicador customizado para visualização separada, ou então utilizar o ChartRedraw() com OBJPROP_TIMEFRAMES e flags como OBJ_NO_PERIODS.

✅ 3. Marcação gráfica persistente

Adicionaremos objetos como OBJ_TREND, OBJ_ARROW, OBJ_RECTANGLE, e OBJ_TEXT com as seguintes propriedades:

ObjectCreate(0, "ApolloLine", OBJ_TREND, 0, time1, price1, time2, price2);
ObjectSetInteger(0, "ApolloLine", OBJPROP_COLOR, clrLime);
ObjectSetInteger(0, "ApolloLine", OBJPROP_WIDTH, 2);
ObjectSetInteger(0, "ApolloLine", OBJPROP_RAY_RIGHT, true); // projeta para o futuro
✅ 4. Marcação geométrica tridimensional (futurística)

Como o MT5 é um ambiente 2D, não suporta gráficos 3D nativamente, mas podemos simular uma “projeção quântica” ou representação tridimensional abstrata com uso de múltiplas formas e códigos de cor.

Exemplo:

X (tempo): eixo horizontal

Y (preço): eixo vertical

Z (profundidade / força): representado com cor ou tamanho

ObjectCreate(0, "QuantumSphere", OBJ_CIRCLE, 0, TimeCurrent() + 3600, price);
ObjectSetInteger(0, "QuantumSphere", OBJPROP_WIDTH, 2);
ObjectSetInteger(0, "QuantumSphere", OBJPROP_COLOR, clrBlue);
ObjectSetInteger(0, "QuantumSphere", OBJPROP_STYLE, STYLE_SOLID);

Também podemos usar objetos OBJ_ARROW, OBJ_RECTANGLE_LABEL com dados codificados no nome para exportação futura.
